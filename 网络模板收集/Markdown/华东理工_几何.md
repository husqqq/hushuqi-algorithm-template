699   inline void sort_coplanar_points_ccw(const Plane & P, vector<Point3> & pts) {
700          // 将共面点按 CCW 顺序排序（相对质心的极角）
701          if (pts.size() <= 2) return;

703          vector<pair<T, T>> q(pts.size()); T cx = 0, cy = 0;
704          for (size_t i = 0; i < pts.size(); ++i) { q[i] = to2D(B, pts[i]); cx +=
      q[i].first; cy += q[i].second; }
705          cx /= (T)pts.size(); cy /= (T)pts.size();
706          auto ang = [&](const pair<T, T> & a) { return atan2l(a.second - cy, a.first -
      cx); };
707          vector<int> id(pts.size()); std::iota(id.begin(), id.end(), 0);
708          std::sort(id.begin(), id.end(), [&](int i, int j) {

      (q[i].second - cy);
712             const T dj = (q[j].first - cx) * (q[j].first - cx) + (q[j].second - cy) *
      (q[j].second - cy);
713             return di < dj;
714       });

      T merge_eps = 1e-9L) {

793       struct Key { long long x, y, z; };

      long>()(k.x * 1315423911LL ^ k.y * 97531LL ^ k.z * 19260817LL); } };

      llroundl(p.y / merge_eps), llroundl(p.z / merge_eps) }; };

      merge_eps * 2; };

802            const Key K = key(p);
803            auto it = id.find(K);



                                              40 / 59
804            if (it != id.end()) { if (eqp(pts[it->second], p)) return it->second; }
805            const int idx = (int)pts.size(); id[K] = idx; pts.push_back(p);
      adj.push_back({}); return idx;
806       };

808       for (const auto & s : segs) {
809            const int u = get(s.p1), v = get(s.p2);
810            if (u == v) continue;
811            adj[u].push_back(v);

993         };
