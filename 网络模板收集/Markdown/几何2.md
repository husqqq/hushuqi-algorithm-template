struct Point
{
    T x = 0, y = 0;
    Point(T _x = 0, T _y = 0) : x(_x), y(_y) {}
 
    // 向量代数运算
    friend bool operator==(const Point &x, const Point &y) { return (abs(x.x - y.x) <= EPS && abs(x.y - y.y) <= EPS); }
    bool operator<(const Point &a) const
    {
        if (abs(x - a.x) <= EPS)
            return y < a.y - EPS;
        return x < a.x - EPS;
    }
    bool operator>(const Point &a) const { return !(*this < a || *this == a); }
    friend Point operator+(const Point &x, const Point &y) { return {x.x + y.x, x.y + y.y}; }
    friend Point operator-(const Point &x, const Point &y) { return {x.x - y.x, x.y - y.y}; }
    friend Point operator*(const Point &x, const T k) { return {k * x.x, k * x.y}; }
    friend Point operator*(const T k, const Point &x) { return {k * x.x, k * x.y}; }
    friend Point operator/(const Point &x, const T k) { return {x.x / k, x.y / k}; }
    friend T operator*(const Point &x, const Point &y) { return x.x * y.x + x.y * y.y; }
    friend T operator^(const Point &x, const Point &y) { return x.x * y.y - x.y * y.x; }
    Point operator-() const { return {-x, -y}; }
    Point &operator+=(const Point &x) { return *this = *this + x; }
    Point &operator-=(const Point &x) { return *this = *this - x; }
    Point &operator*=(T k) { return *this = *this * k; }
    // 长度和旋转
    T len2() const { return (*this) * (*this); }                                                            // 模长平方
    T len2(Point a) const { return ((*this) - a).len2(); }                                                  // 两点距离平方
    T len() const { return (T)sqrtl(len2()); }                                                              // 模长
    T len(Point a) const { return ((*this) - a).len(); }                                                    // 两点距离
    T ang(const Point &a) const { return acosl(max(-1.0l, min(1.0l, ((*this) * a) / (len() * a.len())))); } // 向量夹角
    Point rot90() const { return {-y, x}; }
    Point rot(const T rad) const { return {x * cos(rad) - y * sin(rad), x * sin(rad) + y * cos(rad)}; } // 逆时针旋转（给定角度）
    Point rot(const T cosr, const T sinr) const { return {x * cosr - y * sinr, x * sinr + y * cosr}; }  // 逆时针旋转（给定角度的正弦与余弦）
    // to_left测试(向量传入向量)：1表示在左侧，0表示共线，-1表示右侧
    int toleft(Point p) const
    {
        T ans = (*this) ^ p;
        return (ans > EPS) - (ans < -EPS);
    }
    // 象限判断 0:原点 1:x轴正 2:第一象限 3:y轴正 4:第二象限 5:x轴负 6:第三象限 7:y轴负 8:第四象限
    int quad() const
    {
        if (abs(x) <= EPS && abs(y) <= EPS)
            return 0;
        if (abs(y) <= EPS)
            return x > EPS ? 1 : 5;
        if (abs(x) <= EPS)
            return y > EPS ? 3 : 7;
        return y > EPS ? (x > EPS ? 2 : 4) : (x > EPS ? 8 : 6);
    }
};
// 半平面极角排序 bool argcmp(const Point &a,const Point &b) { return (a^b)>EPS; }
 
// 全平面极角排序
struct Argcmp
{
    Point o;
    Argcmp(Point p = {0, 0}) : o(p) {}
    bool operator()(const Point &a, const Point &b) const
    {
        const int qa = (a - o).quad(), qb = (b - o).quad();
        if (qa != qb)
            return qa < qb;
        const auto t = (a - o) ^ (b - o);
        // if (abs(t) <= EPS) return (a - o) * (a - o) < (b - o) * (b - o) - EPS;// 不同长度的向量需要分开
        return t > EPS;
    }
};
 
struct Line
{
    Point p, v;
    Line() {}
    Line(Point a, Point b) { p = a, v = b; }
 
    bool operator<(const Line &a) const // 半平面交算法定义的排序
    {
        if (abs(v ^ a.v) <= EPS && v * a.v >= -EPS)
            return toleft(a.p) == -1;
        return Argcmp()(v, a.v);
    }
 
    bool is_connect(const Line &a) const { return abs(v ^ a.v) > EPS; } // 两直线是否有严格交点（非平行和重合）
    int toleft(const Point &a) const { return v.toleft(a - p); } // to-left测试(线段传入点)
    // 直线相关计算
    T dis(const Point &a) const { return abs(v ^ (a - p)) / v.len(); }             // 点到直线距离
    Point proj(const Point &a) const { return p + v * ((v * (a - p)) / (v * v)); } // 点在直线上的投影
    Point inter(const Line &a) const                                               // 直线交点
    {
        assert(is_connect(a));
        return p + v * ((a.v ^ (p - a.p)) / (v ^ a.v));
    }
};
 
struct Segment
{
    Point a, b;
    Segment() {}
    Segment(Point _a, Point _b) : a(_a), b(_b) {}
 
    // 判定相关
    //  判断点是否在线段上
    //  -1 点在线段端点 | 0 点不在线段上 | 1 点严格在线段上
    int is_on(Point p) const
    {
        if (p == a || p == b)
            return -1;
        return (b - a).toleft(p - a) == 0 && (p - b) * (p - a) < -EPS;
    }
    // 判断线段直线是否相交
    // -1 直线经过线段端点 | 0 线段和直线不相交 | 1 线段和直线严格相交
    int is_inter(const Line &l) const
    {
        if (l.toleft(a) == 0 || l.toleft(b) == 0)
            return -1;
        return l.toleft(a) != l.toleft(b);
    }
    // 判断两线段是否相交
    // -1 在某一线段端点处相交 | 0 两线段不相交 | 1 两线段严格相交
    int is_inter(const Segment &s) const
    {
        if (is_on(s.a) || is_on(s.b) || s.is_on(a) || s.is_on(b))
            return -1;
        const Line l{a, b - a}, ls{s.a, s.b - s.a};
        return l.toleft(s.a) * l.toleft(s.b) == -1 && ls.toleft(a) * ls.toleft(b) == -1;
    }
 
    // 距离相关
    //  点到线段距离（必须用浮点数）
    T dis(const Point &p) const
    {
        if ((p - a) * (b - a) < -EPS || (p - b) * (a - b) < -EPS)
            return min(a.len(p), b.len(p));
        const Line l(a, b - a);
        return l.dis(p);
    }
    // 两线段间距离（必须用浮点数）
    T dis(const Segment &s) const
    {
        if (is_inter(s))
            return 0;
        return min({dis(s.a), dis(s.b), s.dis(a), s.dis(b)});
    }
};
 
struct Polygon
{
    vector<Point> point;
    Polygon() {}
    Polygon(vector<Point> p) : point(p) {}
 
    // 获取下一个和上一个下标
    int nxt(const int i) const { return i == point.size() - 1 ? 0 : i + 1; }
    int pre(const int i) const { return i == 0 ? point.size() - 1 : i - 1; }
    // 求转数：判断点是否在一般多边形上或内外部：bool范围是否在边上，int计数回转数
    // https://ac.nowcoder.com/acm/contest/27249/B
    pair<bool, int> winding(const Point &a) const
    {
        int cnt = 0;
        for (int i = 0; i < point.size(); ++i)
        {
            const Point u = point[i], v = point[nxt(i)];
            if (Segment(u, v).is_on(a))
                return {true, 0};
            if (abs(u.y - v.y) <= EPS)
                continue;
            const Line uv = {u, v - u};
            if (u.y < v.y - EPS && uv.toleft(a) <= 0)
                continue;
            if (u.y > v.y + EPS && uv.toleft(a) >= 0)
                continue;
            if (u.y < a.y - EPS && v.y >= a.y - EPS)
                cnt++;
            if (u.y >= a.y - EPS && v.y < a.y - EPS)
                cnt--;
        }
        return {false, cnt};
    }
    // 多边形面积的两倍：可用于判断点的存储顺序是顺时针或逆时针
    T area() const
    {
        T sum = 0;
        for (int i = 0; i < point.size(); ++i)
            sum += point[i] ^ point[nxt(i)];
        return sum;
    }
    // 多边形的周长
    T circ() const
    {
        T sum = 0;
        for (int i = 0; i < point.size(); ++i)
            sum += point[i].len(point[nxt(i)]);
        return sum;
    }
};
struct Convex : Polygon
{
    Convex() {}
    Convex(vector<Point> p) : Polygon(p) {}
 
    // 闵可夫斯基和
    Convex operator+(const Convex &c) const
    {
        vector<Segment> e1(point.size()), e2(c.point.size()), edge(point.size() + c.point.size());
        vector<Point> res;
        res.reserve(point.size() + c.point.size());
        const auto cmp = [](const Segment &u, const Segment &v)
        { return Argcmp()(u.b - u.a, v.b - v.a); };
        for (int i = 0; i < point.size(); ++i)
            e1[i] = Segment(point[i], point[nxt(i)]);
        for (int i = 0; i < c.point.size(); ++i)
            e2[i] = Segment(c.point[i], c.point[c.nxt(i)]);
        rotate(e1.begin(), min_element(e1.begin(), e1.end(), cmp), e1.end());
        rotate(e2.begin(), min_element(e2.begin(), e2.end(), cmp), e2.end());
        merge(e1.begin(), e1.end(), e2.begin(), e2.end(), edge.begin(), cmp);
        const auto check = [](const vector<Point> &res, const Point &u)
        {
            const auto back1 = res.back(), back2 = *prev(res.end(), 2);
            return (back1 - back2).toleft(u - back1) == 0 && (back1 - back2) * (u - back1) >= -EPS;
        };
        auto u = e1[0].a + e2[0].a;
        for (const auto &v : edge)
        {
            while (res.size() > 1 && check(res, u))
                res.pop_back();
            res.push_back(u);
            u = u + v.b - v.a;
        }
        if (res.size() > 1 && check(res, res[0]))
            res.pop_back();
        return res;
    }
    Convex operator-(const Convex &c) const
    {
        Convex nc = c;
        for (auto &p : nc.point)
            p = -p;
        return (*this) + nc;
    }
 
    // 旋转卡壳：O(n)求凸包直径平方
    // https://ac.nowcoder.com/acm/contest/28513/B
    // https://www.luogu.com.cn/problem/A452
    T rotcaliper() const
    {
        if (point.size() <= 1)
            return 0;
        if (point.size() == 2)
            return point[0].len2(point[1]);
        T ret = 0;
        const auto area = [](const Point &u, const Point &v, const Point &w)
        { return (w - u) ^ (w - v); };
        int n = point.size(), r = 1;
        for (int i = 0; i < n; ++i)
        {
            ret = max({ret, point[i].len2(point[r]), point[nxt(i)].len2(point[r])});
            while (area(point[i], point[nxt(i)], point[r]) <= area(point[i], point[nxt(i)], point[nxt(r)]))
            {
                r = nxt(r);
                ret = max({ret, point[i].len2(point[r]), point[nxt(i)].len2(point[r])});
            }
        }
        return ret;
    }
 
    // 判断点是否在凸多边形内，复杂度 O(logn)
    // https://ac.nowcoder.com/acm/contest/28770/A
    // -1 点在多边形边上 | 0 点在多边形外 | 1 点在多边形内
    int is_in(const Point &p) const
    {
        int n = point.size();
        if (n == 1)
            return p == point[0] ? -1 : 0;
        if (n == 2)
            return Segment(point[0], point[1]).is_on(p) ? -1 : 0;
        if (p == point[0])
            return -1;
        if ((point[1] - point[0]).toleft(p - point[0]) == -1 || (point.back() - point[0]).toleft(p - point[0]) == 1)
            return 0;
        const auto cmp = [&](const Point &u, const Point &v)
        { return (u - point[0]).toleft(v - point[0]) == 1; };
        const int i = lower_bound(point.begin() + 1, point.end(), p, cmp) - point.begin();
        if (i == 1)
            return Segment(point[0], point[i]).is_on(p) ? -1 : 0;
        if (i == n - 1 && Segment(point[0], point[i]).is_on(p))
            return -1;
        if (Segment(point[i - 1], point[i]).is_on(p))
            return -1;
        return (point[i] - point[i - 1]).toleft(p - point[i - 1]) > 0;
    }
 
    // 凸多边形关于某一方向的极点
    // 复杂度 O(logn)
    // 参考资料：https://codeforces.com/blog/entry/48868
    template <typename F>
    int extreme(const F &dir) const
    {
        const auto check = [&](const int i)
        { return dir(point[i]).toleft(point[this->nxt(i)] - point[i]) >= 0; };
        const auto dir0 = dir(point[0]);
        const auto check0 = check(0);
        if (!check0 && check(point.size() - 1))
            return 0;
        const auto cmp = [&](const Point &v)
        {
            const int vi = &v - point.data();
            if (vi == 0)
                return 1;
            const auto checkv = check(vi);
            const auto t = dir0.toleft(v - point[0]);
            if (vi == 1 && checkv == check0 && t == 0)
                return 1;
            return checkv ^ (checkv == check0 && t <= 0);
        };
        return partition_point(point.begin(), point.end(), cmp) - point.begin();
    }
 
    // 过凸多边形外一点求凸多边形的切线，返回切点下标
    // 复杂度 O(logn)
    // 必须保证点在多边形外
    pair<int, int> tangent(const Point &a) const
    {
        if (point.size() == 1)
            return {0, 0};
        if (point.size() == 2)
            return {0, 1};
        const int u = extreme([&](const Point &u)
            { return u - a; });
        const int v = extreme([&](const Point &u)
            { return a - u; });
        return {u, v};
    }
 
    // 求平行于给定直线的凸多边形的切线，返回切点下标
    // 复杂度 O(logn)
    pair<int, int> tangent(const Line &a) const
    {
        const int u = extreme([&](...)
            { return a.v; });
        const int v = extreme([&](...)
            { return -a.v; });
        return {u, v};
    }
};
 
// 点集的凸包: Andrew 算法，复杂度 O(nlogn)，保证结果是逆时针
// https://ac.nowcoder.com/acm/contest/28513/A
// https://www.luogu.com.cn/problem/P2742
Convex convexhull(vector<Point> p)
{
    vector<Point> st;
    if (p.empty())
        return Convex(st);
    if (p.size() == 1)
        return Convex(p);
    sort(p.begin(), p.end());
    const auto check = [](const vector<Point> &st, const Point &u)
    {
        const auto back1 = st.back(), back2 = *prev(st.end(), 2);
        return (back1 - back2).toleft(u - back1) <= 0;
    };
    for (const Point &u : p)
    {
        while (st.size() > 1 && check(st, u))
            st.pop_back();
        st.push_back(u);
    }
    int k = st.size();
    p.pop_back();
    reverse(p.begin(), p.end());
    for (const Point &u : p)
    {
        while (st.size() > k && check(st, u))
            st.pop_back();
        st.push_back(u);
    }
    st.pop_back();
    return Convex(st);
}
 
// 点集中面积最大最小的三角形，注意返回的面积两倍
// 这里使用极角排序和扫描线，O(n^2logn)
// https://ac.nowcoder.com/acm/contest/28513/C
pair<T, T> mnmxTriangle(const vector<Point> &ps)
{
    if (ps.size() <= 2)
        return {0, 0};
    vector<pii> dir;
    T mx = 0, mn = INF;
    for (int i = 0; i < ps.size(); ++i)
    {
        for (int j = 0; j < ps.size(); ++j)
        {
            if (i == j)
                continue;
            if (ps[i] == ps[j])
                mn = 0;
            else if (i != j)
                dir.push_back({i, j});
        }
    }
    sort(dir.begin(), dir.end(), [&](const pii &x, const pii &y)
    {
        Point u = ps[x.first] - ps[x.second];
        Point v = ps[y.first] - ps[y.second];
        return Argcmp()(u, v); });
    vector<int> vx(ps.size()), ivx(ps.size());
    for (int i = 0; i < ps.size(); ++i)
        vx[i] = i;
    sort(vx.begin(), vx.end(), [&](const int &x, const int &y)
         { return ps[x].y > ps[y].y; });
    for (int i = 0; i < ps.size(); ++i)
        ivx[vx[i]] = i;
    for (const auto &[u, v] : dir)
    {
        int x = ivx[u], y = ivx[v];
        int l = min(x, y), r = max(x, y);
        if (l > 0)
            mn = min(mn, abs((ps[vx[l - 1]] - ps[u]) ^ (ps[vx[l - 1]] - ps[v])));
        else if (r < ps.size() - 1)
            mn = min(mn, abs((ps[vx[r + 1]] - ps[u]) ^ (ps[vx[r + 1]] - ps[v])));
        mx = max({mx, abs((ps[vx.front()] - ps[u]) ^ (ps[vx.front()] - ps[v])), abs((ps[vx.back()] - ps[u]) ^ (ps[vx.back()] - ps[v]))});
        swap(vx[x], vx[y]);
        ivx[u] = y, ivx[v] = x;
    }
    return {mn, mx};
}
// 仅获取点集的面积最大三角形
// 这里使用凸包和旋转卡壳，O(n^2)
T mxTriangle(const vector<Point> &ps)
{
    Convex plg = convexhull(ps);
    vector<Point> nps = plg.point;
    if (nps.size() <= 2)
        return 0;
    const auto area = [](const Point &u, const Point &v, const Point &w)
    { return abs((w - u) ^ (w - v)); };
    int n = nps.size();
    T ret = 0;
    for (int i = 0; i < n; ++i)
    {
        int r = (i + 1) % n;
        for (int j = (i + 1) % n; j != i; j = (j + 1) % n)
        {
            ret = max(ret, area(nps[i], nps[j], nps[r]));
            while (area(nps[i], nps[j], nps[r]) <= area(nps[i], nps[j], nps[(r + 1) % n]))
            {
                r = (r + 1) % n;
                ret = max(ret, area(nps[i], nps[j], nps[r]));
            }
        }
    }
    return ret;
}
 
// 半平面交 排序增量法，复杂度 O(nlogn)
// 输入与返回值都是用直线表示的半平面集合
// https://www.luogu.com.cn/problem/A452
// https://ac.nowcoder.com/acm/contest/29764/A
vector<Line> halfinter(vector<Line> l, const T lim = 1e18)
{
    const auto check = [](const Line &a, const Line &b, const Line &c)
    { return a.toleft(b.inter(c)) < 0; };
    /* 无精度误差的方法，但注意取值范围会扩大到三次方
    const auto check=[](const Line &a,const Line &b,const Line &c)
    {
        const Point p=a.v*(b.v^c.v),q=b.p*(b.v^c.v)+b.v*(c.v^(b.p-c.p))-a.p*(b.v^c.v);
        return p.toleft(q)<0;
    };
    */
    l.push_back({{-lim, 0}, {0, -1}}), l.push_back({{0, -lim}, {1, 0}});
    l.push_back({{lim, 0}, {0, 1}}), l.push_back({{0, lim}, {-1, 0}});
    sort(l.begin(), l.end());
    deque<Line> q;
    for (int i = 0; i < l.size(); ++i)
    {
        if (i > 0 && l[i - 1].v.toleft(l[i].v) == 0 && l[i - 1].v * l[i].v > EPS)
            continue;
        while (q.size() > 1 && check(l[i], q.back(), q[q.size() - 2]))
            q.pop_back();
        while (q.size() > 1 && check(l[i], q[0], q[1]))
            q.pop_front();
        if (!q.empty() && q.back().v.toleft(l[i].v) <= 0)
            return vector<Line>();
        q.push_back(l[i]);
    }
    while (q.size() > 1 && check(q[0], q.back(), q[q.size() - 2]))
        q.pop_back();
    while (q.size() > 1 && check(q.back(), q[0], q[1]))
        q.pop_front();
    return vector<Line>(q.begin(), q.end());
}
 
// 平面最近点对 扫描线 复杂度 O(nlogn)
// https://ac.nowcoder.com/acm/contest/52826/C
// https://www.luogu.com.cn/problem/P7883
T closest_pair(vector<Point> points)
{
    sort(points.begin(), points.end());
    const auto cmpy = [](const Point &a, const Point &b)
    {
        if (abs(a.y - b.y) <= EPS)
            return a.x < b.x - EPS;
        return a.y < b.y - EPS;
    };
    multiset<Point, decltype(cmpy)> s{cmpy};
    T ans = INF;
    for (int i = 0, l = 0; i < points.size(); ++i)
    {
        const T sqans = sqrtl(ans) + 1;
        while (l < i && points[i].x - points[l].x >= sqans)
            s.erase(s.find(points[l++]));
        for (auto it = s.lower_bound(Point{-INF, points[i].y - sqans});
             it != s.end() && it->y - points[i].y <= sqans; ++it)
            ans = min(ans, points[i].len2(*it));
        s.insert(points[i]);
    }
    return ans;
}
 
struct Circle
{
    Point o;
    T r;
 
    Circle() {}
    Circle(Point p, T _r) : o(p), r(_r) {}
 
    bool operator==(const Circle &a) const
    {
        return o == a.o && abs(r - a.r) <= EPS;
    }
    T circ() const { return 2 * PI * r; } // 周长
    T area() const { return PI * r * r; } // 面积
 
    // 点与圆的关系
    // -1 圆上 | 0 圆外 | 1 圆内
    int is_in(const Point &p) const
    {
        const T d = p.len2(o);
        return abs(d - r * r) <= EPS ? -1 : d < r * r - EPS;
    }
 
    // 直线与圆关系
    // 0 相离 | 1 相切 | 2 相交
    int relation(const Line &l) const
    {
        const T d = l.dis(o);
        if (d > r + EPS)
            return 0;
        if (abs(d - r) <= EPS)
            return 1;
        return 2;
    }
 
    // 圆与圆关系
    // -1 相同 | 0 相离 | 1 外切 | 2 相交 | 3 内切 | 4 内含
    int relation(const Circle &a) const
    {
        if (*this == a)
            return -1;
        const T d = o.len(a.o);
        if (d > r + a.r + EPS)
            return 0;
        if (abs(d - r - a.r) <= EPS)
            return 1;
        if (abs(d - abs(r - a.r)) <= EPS)
            return 3;
        if (d < abs(r - a.r) - EPS)
            return 4;
        return 2;
    }
 
    // 直线与圆的交点
    vector<Point> inter(const Line &l) const
    {
        const T d = l.dis(o);
        const Point p = l.proj(o);
        const int t = relation(l);
        if (t == 0)
            return vector<Point>();
        if (t == 1)
            return vector<Point>{p};
        const T k = sqrt(r * r - d * d);
        return vector<Point>{p - (l.v / l.v.len()) * k, p + (l.v / l.v.len()) * k};
    }
 
    // 圆与圆交点
    vector<Point> inter(const Circle &a) const
    {
        const T d = o.len(a.o);
        const int t = relation(a);
        if (t == -1 || t == 0 || t == 4)
            return vector<Point>();
        Point e = a.o - o;
        e = e / e.len() * r;
        if (t == 1 || t == 3)
        {
            if (r * r + d * d - a.r * a.r >= -EPS)
                return vector<Point>{o + e};
            return vector<Point>{o - e};
        }
        const T costh = (r * r + d * d - a.r * a.r) / (2 * r * d), sinth = sqrt(1 - costh * costh);
        return vector<Point>{o + e.rot(costh, -sinth), o + e.rot(costh, sinth)};
    }
 
    // 圆与圆交面积
    T interArea(const Circle &a) const
    {
        const T d = o.len(a.o);
        const int t = relation(a);
        if (t == -1)
            return area();
        if (t < 2)
            return 0;
        if (t > 2)
            return min(area(), a.area());
        const T costh1 = (r * r + d * d - a.r * a.r) / (2 * r * d);
        const T costh2 = (a.r * a.r + d * d - r * r) / (2 * a.r * d);
        const T sinth1 = sqrt(1 - costh1 * costh1), sinth2 = sqrt(1 - costh2 * costh2);
        const T th1 = acos(costh1), th2 = acos(costh2);
        return r * r * (th1 - costh1 * sinth1) + a.r * a.r * (th2 - costh2 * sinth2);
    }
 
    // 圆与多边形面积交
    T interArea(const Polygon &plg)
    {
        const auto cal = [](const Circle &circ, const Point &a, const Point &b)
        {
            if ((a - circ.o).toleft(b - circ.o) == 0)
                return 0.0l;
            const auto ina = circ.is_in(a), inb = circ.is_in(b);
            const Line ab = {a, b - a};
            if (ina && inb)
                return ((a - circ.o) ^ (b - circ.o)) / 2;
            if (ina && !inb)
            {
                const auto t = circ.inter(ab);
                const Point p = t.size() == 1 ? t[0] : t[1];
                const T ans = ((a - circ.o) ^ (p - circ.o)) / 2;
                const T th = (p - circ.o).ang(b - circ.o);
                const T d = circ.r * circ.r * th / 2;
                if ((a - circ.o).toleft(b - circ.o) == 1)
                    return ans + d;
                return ans - d;
            }
            if (!ina && inb)
            {
                const Point p = circ.inter(ab)[0];
                const T ans = ((p - circ.o) ^ (b - circ.o)) / 2;
                const T th = (a - circ.o).ang(p - circ.o);
                const T d = circ.r * circ.r * th / 2;
                if ((a - circ.o).toleft(b - circ.o) == 1)
                    return ans + d;
                return ans - d;
            }
            const auto p = circ.inter(ab);
            if (p.size() == 2 && Segment(a, b).dis(circ.o) <= circ.r + EPS)
            {
                const T ans = ((p[0] - circ.o) ^ (p[1] - circ.o)) / 2;
                const T th1 = (a - circ.o).ang(p[0] - circ.o), th2 = (b - circ.o).ang(p[1] - circ.o);
                const T d1 = circ.r * circ.r * th1 / 2, d2 = circ.r * circ.r * th2 / 2;
                if ((a - circ.o).toleft(b - circ.o) == 1)
                    return ans + d1 + d2;
                return ans - d1 - d2;
            }
            const T th = (a - circ.o).ang(b - circ.o);
            if ((a - circ.o).toleft(b - circ.o) == 1)
                return circ.r * circ.r * th / 2;
            return -circ.r * circ.r * th / 2;
        };
 
        T ans = 0;
        for (int i = 0; i < plg.point.size(); ++i)
        {
            const Point a = plg.point[i], b = plg.point[plg.nxt(i)];
            ans += cal(*this, a, b);
        }
        return ans;
    }
 
    // 过圆外一点圆的切线
    vector<Line> tangent(const Point &a) const
    {
        const int t = is_in(a);
        if (t == 1)
            return vector<Line>();
        if (t == -1)
        {
            const Point v = {-(a - o).y, (a - o).x};
            return vector<Line>{{a, v}};
        }
        Point e = a - o;
        e = e / e.len() * r;
        const T costh = r / o.len(a), sinth = sqrt(1 - costh * costh);
        const Point t1 = o + e.rot(costh, -sinth), t2 = o + e.rot(costh, sinth);
        return vector<Line>{{a, t1 - a}, {a, t2 - a}};
    }
 
    // 两圆的公切线
    vector<Line> tangent(const Circle &a) const
    {
        const int t = relation(a);
        vector<Line> lines;
        if (t == -1 || t == 4)
            return lines;
        // t=1 (外切): 这条线是过切点的"内公切线"，垂直于圆心连线。
        //             它把两个圆隔开，不在凸包边界上。-> 必须去掉
        // t=3 (内切): 虽然是外围的线，但在圆凸包算法中，被包含的圆通常不贡献边界。
        //             且该点容易导致数值不稳定。-> 去掉通常无影响
        if (t == 1 || t == 3)
        {
            const Point p = inter(a)[0], v = {-(a.o - o).y, (a.o - o).x};
            lines.push_back({p, v});
        }
        const T d = o.len(a.o);
        const Point e = (a.o - o) / (a.o - o).len();
        if (t <= 2) // 外公切线
        {
            const T costh = (r - a.r) / d, sinth = sqrt(1 - costh * costh);
            const Point d1 = e.rot(costh, -sinth), d2 = e.rot(costh, sinth);
            const Point u1 = o + d1 * r, u2 = o + d2 * r, v1 = a.o + d1 * a.r, v2 = a.o + d2 * a.r;
            lines.push_back({u1, v1 - u1}), lines.push_back({u2, v2 - u2});
        }
        if (t == 0) // 内公切线
        {
            const T costh = (r + a.r) / d, sinth = sqrt(1 - costh * costh);
            const Point d1 = e.rot(costh, -sinth), d2 = e.rot(costh, sinth);
            const Point u1 = o + d1 * r, u2 = o + d2 * r, v1 = a.o - d1 * a.r, v2 = a.o - d2 * a.r;
            lines.push_back({u1, v1 - u1}), lines.push_back({u2, v2 - u2});
        }
        return lines;
    }
};
 
// 给定三个点求外接圆，注意自定义共线情况
Circle CircumCircle(Point A, Point B, Point C)
{
    Line L1((A + B) / 2, (B - A).rot90());
    Line L2((A + C) / 2, (C - A).rot90());
    if (!L1.is_connect(L2))
    {
        // 三点共线，理论上不应进入此分支，除非数据有误或逻辑需要退化处理
        // 在SEC算法中，这里可以 assert(0) 或者返回两点直径圆
        // 值得注意的是如果有点重合出现零向量方向向量的直线也会判成平行
        // assert(1);
        return {{0, 0}, INF};
    }
    Point center = L1.inter(L2);
    return Circle(center, center.len(A));
}
 
// 给定三个点求内切圆
Circle InCircle(Point A, Point B, Point C)
{
    T a = B.len(C), b = A.len(C), c = A.len(B);
    T area = abs((B - A) ^ (C - A));
    T sum = a + b + c;
    Point center = (A * a + B * b + C * c) / (a + b + c);
    T r = area / (a + b + c);
    return Circle(center, r);
}
 
// 最小圆覆盖 最坏O(n^3)，随机化下认为是O(n)
// https://ac.nowcoder.com/acm/contest/52826/D
// https://www.luogu.com.cn/problem/A742
Circle min_cover_circle(vector<Point> ps)
{
    int n = ps.size();
    random_shuffle(ps.begin(), ps.end());
    Circle cir(ps[0], 0);
    for (int i = 0; i < n; i++)
    {
        if (cir.is_in(ps[i]))
            continue;
        cir.o = ps[i], cir.r = 0;
        for (int j = 0; j < i; j++)
        {
            if (cir.is_in(ps[j]))
                continue;
            cir.o = (ps[i] + ps[j]) / 2;
            cir.r = ps[i].len(ps[j]) / 2;
            for (int k = 0; k < j; k++)
            {
                if (cir.is_in(ps[k]))
                    continue;
                cir = CircumCircle(ps[i], ps[j], ps[k]);
            }
        }
    }
    return cir;
}
 
// 构建阿波罗尼斯圆: 集合 P 满足 dist(P, B) <= k * dist(P, A) 的边界圆
//  k: 比率 (必须 k != 1, 且 k > 0)
// 如果 k < 1，圆包裹点 B。
// 如果 k > 1，圆包裹点 A。
// 如果 k = 1，这是垂直平分线
Circle getApolloniusCircle(Point A, Point B, T k)
{
    assert(abs(k - 1.0) > EPS && k > EPS); // 处理特殊情况 k=1 (垂直平分线，不是圆)
    T k2 = k * k;
    Point o = (B - A * k2) / (1.0 - k2);
    T dist_AB = (A - B).len();
    T r = (k * dist_AB) / abs(1.0 - k2);
    return {o, r};
}
 
struct Triangle
{
    int a = 0, b = 0, c = 0;
    Triangle(int _a = 0, int _b = 0, int _c = 0) : a(_a), b(_b), c(_c) {}
};
T triangle_area2(const Point &a, const Point &b, const Point &c)
{
    return (b - a) ^ (c - a);
}
bool in_triangle_closed(const Point &p, const Point &a, const Point &b, const Point &c)
{
    return triangle_area2(a, b, p) >= -EPS &&
           triangle_area2(b, c, p) >= -EPS &&
           triangle_area2(c, a, p) >= -EPS;
}
vector<Triangle> polygon_triangulation(const Polygon &plg)
{
    vector<Triangle> ret;
    const int n = plg.point.size();
    if (n < 3)
        return ret;
 
    vector<int> v;
    v.reserve(n);
    for (int i = 0; i < n; ++i)
    {
        if (v.empty() || !(plg.point[i] == plg.point[v.back()]))
            v.push_back(i);
    }
    if (v.size() > 1 && plg.point[v.front()] == plg.point[v.back()])
        v.pop_back();
    if (v.size() < 3)
        return ret;
 
    auto area_of = [&](const vector<int> &ids)
    {
        T ans = 0;
        for (int i = 0; i < ids.size(); ++i)
            ans += plg.point[ids[i]] ^ plg.point[ids[(i + 1) % ids.size()]];
        return ans;
    };
 
    T area = area_of(v);
    if (abs(area) <= EPS)
        return ret;
    if (area < 0)
        reverse(v.begin(), v.end());
 
    bool changed = true;
    while (changed && v.size() >= 3)
    {
        changed = false;
        const int m = v.size();
        for (int i = 0; i < m; ++i)
        {
            const Point &a = plg.point[v[(i + m - 1) % m]];
            const Point &b = plg.point[v[i]];
            const Point &c = plg.point[v[(i + 1) % m]];
            if (abs(triangle_area2(a, b, c)) <= EPS && (b - a) * (c - b) >= -EPS)
            {
                v.erase(v.begin() + i);
                changed = true;
                break;
            }
        }
    }
    const int m = v.size();
    if (m < 3 || abs(area_of(v)) <= EPS)
        return ret;
 
    vector<int> pre(m), nxt(m);
    vector<char> alive(m, true);
    iota(pre.begin(), pre.end(), 0);
    iota(nxt.begin(), nxt.end(), 0);
    for (int i = 0; i < m; ++i)
    {
        pre[i] = (i + m - 1) % m;
        nxt[i] = (i + 1) % m;
    }
 
    auto is_ear = [&](int i)
    {
        if (!alive[i])
            return false;
        const int l = pre[i], r = nxt[i];
        const Point &a = plg.point[v[l]];
        const Point &b = plg.point[v[i]];
        const Point &c = plg.point[v[r]];
        if (triangle_area2(a, b, c) <= EPS)
            return false;
        for (int j = 0; j < m; ++j)
        {
            if (!alive[j] || j == l || j == i || j == r)
                continue;
            if (in_triangle_closed(plg.point[v[j]], a, b, c))
                return false;
        }
        return true;
    };
 
    set<int> ears;
    for (int i = 0; i < m; ++i)
        if (is_ear(i))
            ears.insert(i);
 
    int left = m;
    while (left > 3)
    {
        if (ears.empty())
            return vector<Triangle>();
        const int i = *ears.begin();
        ears.erase(ears.begin());
        if (!alive[i] || !is_ear(i))
            continue;
 
        const int l = pre[i], r = nxt[i];
        ret.push_back({v[l], v[i], v[r]});
        alive[i] = false;
        nxt[l] = r;
        pre[r] = l;
        --left;
        ears.erase(l);
        ears.erase(r);
        if (is_ear(l))
            ears.insert(l);
        if (is_ear(r))
            ears.insert(r);
    }
 
    int a = -1, b = -1, c = -1;
    for (int i = 0; i < m; ++i)
    {
        if (alive[i])
        {
            if (a == -1)
                a = i;
            else if (b == -1)
                b = i;
            else
                c = i;
        }
    }
    if (a == -1 || triangle_area2(plg.point[v[a]], plg.point[v[b]], plg.point[v[c]]) <= EPS)
        return vector<Triangle>();
    ret.push_back({v[a], v[b], v[c]});
    return ret;
}
// ==================== 凸多边形面积并（轮廓线积分法） ====================
T poly_union_area(const vector<Convex> &polys)
{
    T total_area = 0;
    int n = polys.size();
 
    for (int i = 0; i < n; ++i)
    {
        // 如果多边形退化，则跳过
        if (polys[i].point.size() < 3)
            continue;
 
        for (int j = 0; j < polys[i].point.size(); ++j)
        {
            Point A = polys[i].point[j];
            Point B = polys[i].point[polys[i].nxt(j)];
 
            vector<pair<T, T>> covered;
            for (int k = 0; k < n; ++k)
            {
                if (i == k)
                    continue;
                if (polys[k].point.size() < 3)
                    continue;
 
                T t_in = 0.0, t_out = 1.0;
                for (int e = 0; e < polys[k].point.size(); ++e)
                {
                    Point C = polys[k].point[e];
                    Point D = polys[k].point[polys[k].nxt(e)];
 
                    T cp1 = (D - C) ^ (A - C);
                    T cp2 = (D - C) ^ (B - A);
 
                    if (abs(cp2) < EPS)
                    { // 平行或共线
                        if (cp1 < -EPS)
                        {
                            t_in = 1.0;
                            t_out = -1.0;
                            break;
                        }
                        if (abs(cp1) <= EPS)
                        {
                            if ((D - C) * (B - A) > 0)
                            {
                                if (k < i)
                                {
                                    t_in = 1.0;
                                    t_out = -1.0;
                                    break;
                                }
                            }
                            else
                            {
                                t_in = 1.0;
                                t_out = -1.0;
                                break;
                            }
                        }
                    }
                    else
                    { // 相交
                        T t = -cp1 / cp2;
                        if (cp2 > 0)
                            t_in = max(t_in, t);
                        else
                            t_out = min(t_out, t);
                    }
                }
                if (t_in < t_out + EPS)
                {
                    covered.push_back({t_in, t_out});
                }
            }
 
            sort(covered.begin(), covered.end());
            T last = 0.0;
            for (auto &seg : covered)
            {
                if (seg.first > last)
                {
                    Point P1 = A + (B - A) * last;
                    Point P2 = A + (B - A) * seg.first;
                    total_area += (P1 ^ P2);
                }
                last = max(last, seg.second);
            }
            if (last < 1.0)
            {
                Point P1 = A + (B - A) * last;
                Point P2 = B;
                total_area += (P1 ^ P2);
            }
        }
    }
    return total_area / 2.0;
}
