```cpp

using namespace std;
#define int long long

constexpr int inf = 1E9;
constexpr long long INF = numeric_limits<long long>::max() / 4;
constexpr long double eps = 1E-12L;

template <class T> bool chmin(T &a, const T &b)
{
    // a 是当前值，b 是候选值；若 b 更小则更新 a 并返回 true。
    if (b < a)
    {
        a = b;
        return true;
    }
    return false;
}

template <class T> bool chmax(T &a, const T &b)
{
    // a 是当前值，b 是候选值；若 b 更大则更新 a 并返回 true。
    if (a < b)
    {
        a = b;
        return true;
    }
    return false;
}

int combMulMod(int a, int b, int mod)
{
    // mod>0 且 a、b 均为 [0,mod)；小模数直接乘，大模数才使用局部宽乘法。
    constexpr int limit = 3037000499LL;
    if (mod <= limit)

    }
    return (int)((__int128)a * b % mod);
```

`combMulMod(a,b,mod)` 要求 `0<=a,b<mod` 且 `mod>0`。模数不超过 `3037000499` 时两个余数的乘积仍在 `long long` 范围内；只有更大的运行期模数才对这一条乘法局部使用 `__int128`。

## 8.1 组合数

### 8.1.001 阶乘、逆阶乘与组合数

预处理阶乘表 `fac` 与逆阶乘表 `ifac` 后，$\binom{n}{k}=\frac{n!}{k!(n-k)!}$、$A(n,k)=\frac{n!}{(n-k)!}$ 都退化为常数次乘法。逆阶乘不必逐个求逆：先用一次除法得到最大的 $(n!)^{-1}$，再借 $((i-1)!)^{-1}=(i!)^{-1}\cdot i$ 反向递推，全表只用一次求逆。`init` 支持增量扩表，`fac.size()` 已够时直接返回，扩容时只补算新增前缀，所以系数类型 `T` 必须支持除法（模素数或有理数），且所有用到的 $n!$ 在该类型下可逆——否则某个 $i!\equiv 0$ 会让 `T(1)/fac[n]` 失去意义。

```cpp
template <class T> class Comb
{
    // fac[i]、ifac[i] 分别保存 i! 与其乘法逆元。
    vector<T> fac{1}, ifac{1};

  public:
    void init(int n)
    {
        // n 是要支持的最大上标；把阶乘表扩展到 n，无返回值。
        assert(n >= 0); // 调试检查，可删。
        if ((int)fac.size() > n)
        {
            return;
        }
        int old = fac.size();
        fac.resize(n + 1);
        ifac.resize(n + 1);
        for (int i = old; i <= n; i++)
        {
            fac[i] = fac[i - 1] * T(i);
        }
        ifac[n] = T(1) / fac[n];
        for (int i = n; i >= old; i--)
        {
            ifac[i - 1] = ifac[i] * T(i);
        }
    }

    T C(int n, int k)
    {
        // n 是对象数，k 是选择数；返回组合数，k 不在 [0,n] 时返回 0。
        if (k < 0 || k > n)
        {
            return T(0);
        }
        init(n);
        return fac[n] * ifac[k] * ifac[n - k];
    }

    T A(int n, int k)
    {
        // n 是对象数，k 是有序选取数；返回排列数，k 不在 [0,n] 时返回 0。
        if (k < 0 || k > n)
        {
            return T(0);
        }
        init(n);
        return fac[n] * ifac[n - k];
    }
};
```

以 `Comb<Z> c` 实例化，`Z` 需支持整数构造与四则运算，`c.C(n,k)`、`c.A(n,k)` 在 $k\notin[0,n]$ 时返回 $0$；`Z` 为模整数时模数必须大于所有用到的 $n$。扩表到 $n$ 为 $O(n)$ 时间与空间，之后每次查询 $O(1)$。这是素数模且 $n$ 小于模数时的首选；一旦 $n$ 超过素模 $p$，某个 $p!\equiv 0$ 破坏可逆性，须改用 [8.1.003 Lucas 定理](#81003) 把上下标按 $p$ 进制拆开，合数模则只能走 [8.1.004 扩展 Lucas](#81004)。分母不可逆或只需要整行组合数时换 [8.1.002 杨辉三角](#81002) 的纯加法递推。

准备 [1.1.008 静态取模类](01-基础与工程.md#11008) 后，可用同一个较大的组合数同时检查扩表、逆阶乘和三次模乘；下面两个结果分别是 $\binom{132}{66}$ 在最常用的两个模数下的已知值：

```cpp
signed main()
{
    // 无参数；核对两个常见模数下的组合数与边界，返回 0。
    using Z998 = MInt<998244353>;
    using Z1e9 = MInt<1000000007>;
    Comb<Z998> a;
    Comb<Z1e9> b;
    assert(a.C(132, 66).val() == 241200029);
    assert(b.C(132, 66).val() == 598375978);
    assert(a.C(0, 0).val() == 1);
    assert(a.C(10, 11).val() == 0);
    return 0;
}
```

### 8.1.002 杨辉三角

对是否选中某个指定元素分类，就得到 Pascal 恒等式 $\binom{i}{j}=\binom{i-1}{j-1}+\binom{i-1}{j}$，每行两端恒为 $1$。整张表只用加法递推，不涉及除法，因此模数不保证分母可逆、或需要一整行组合数时它比 [8.1.001 阶乘、逆阶乘与组合数](#81001) 更合适；代价是行数不能太大。

```cpp
template <class T> vector<vector<T>> pascalTri(int n)
{
    // n 是最大行号；返回第 0..n 行杨辉三角，其中 a[i][j]=C(i,j)。
    assert(n >= 0); // 调试检查，可删。
    vector<vector<T>> a(n + 1);
    for (int i = 0; i <= n; i++)
    {
        a[i].assign(i + 1, T(1));
        for (int j = 1; j < i; j++)
        {
            a[i][j] = a[i - 1][j - 1] + a[i - 1][j];
        }
    }
    return a;
}
```

`pascalTri<T>(n)` 返回第 $0..n$ 行，`T` 只需支持加法和从 $1$ 构造，`a[i][j]` 即 $\binom{i}{j}$。$n=5$ 时第 $5$ 行为 `1 5 10 10 5 1`。整表 $O(n^2)$ 时间与空间，只求单个组合数时不该建整表：那种场景用 [8.1.001 阶乘、逆阶乘与组合数](#81001) 的 $O(1)$ 查询，或结果不取模、能放进 `__int128` 时用 [8.1.010 球盒模型与鸽巢原理](#81010) 里的逐项约分 `comb`。

### 8.1.003 Lucas 定理

Lucas 定理来自把 $(1+x)^n$ 在模素数 $p$ 下按 $p$ 进制分块：$(1+x)^p\equiv 1+x^p\pmod p$，于是 $n=\sum n_i p^i$ 时 $(1+x)^n\equiv\prod_i(1+x^{p^i})^{n_i}$，比较 $x^k$ 系数即得 $\binom{n}{k}\equiv\prod_i\binom{n_i}{k_i}\pmod p$，任一位 $k_i>n_i$ 则整体为 $0$。它只处理素数模，把远大于 $p$ 的上标拆成 $O(\log_p n)$ 位、每位组合数在长度 $p$ 的阶乘表上 $O(1)$ 取得。

```cpp
int powPrime(int a, int b, int p)
{
    // a 是底数，b 是非负指数，p 是素数；返回 a^b mod p。
    assert(b >= 0 && p >= 2); // 调试检查，可删。
    a %= p;
    if (a < 0)
    {
        a += p;
    }
    int ans = 1 % p;
    while (b)
    {
        if (b & 1)
        {
            ans = combMulMod(ans, a, p);
        }
        a = combMulMod(a, a, p);
        b >>= 1;
    }
    return ans;
}

int lucas(int n, int k, int p)
{
    // n、k 是组合数参数，p 是素数；返回 C(n,k) mod p。
    assert(n >= 0 && p >= 2); // 调试检查，可删。
    if (k < 0 || k > n)
    {
        return 0;
    }
    vector<int> fac(p, 1), ifac(p, 1);
    for (int i = 1; i < p; i++)
    {
        fac[i] = combMulMod(fac[i - 1], i, p);
    }
    ifac[p - 1] = powPrime(fac[p - 1], p - 2, p);
    for (int i = p - 1; i; i--)
    {
        ifac[i - 1] = combMulMod(ifac[i], i, p);
    }
    int ans = 1;
    while (n || k)
    {
        int a = n % p, b = k % p;
        if (b > a)
        {
            return 0;
        }
        ans = combMulMod(combMulMod(ans, fac[a], p), ifac[b], p);
        ans = combMulMod(ans, ifac[a - b], p);
        n /= p;
        k /= p;
    }
    return ans;
}
```

`lucas(n,k,p)` 要求 `n>=0`，且 `p>=2` 是能完整建 $p$ 长阶乘表的素数；实测 `lucas(100,20,7)=0`、`lucas(100,30,7)=0`、`lucas(100,50,7)=4`。每次 $O(p+\log_p n)$ 时间、$O(p)$ 空间，同一 `p` 反复查询时应把 `fac`/`ifac` 提到外层复用。若 $n<p$，可直接用 [8.1.001 阶乘、逆阶乘与组合数](#81001)；若 $n\ge p$ 且 $p$ 大到不能建表，普通阶乘表同样会在 $p!$ 处失去逆元，必须改为逐位按较小一侧乘除、按需缓存阶乘或采用大素模阶乘算法。模数为合数时改用 [8.1.004 扩展 Lucas](#81004) 按质数幂分别处理再 CRT 合并。

### 8.1.004 扩展 Lucas

合数模 $m=\prod p_i^{e_i}$ 下 $\binom{n}{k}$ 的困难在于阶乘含 $p$ 因子、逆元不存在，扩展 Lucas 把它拆到每个 $p^e$ 上单独算再 CRT 合并。关键是把 $n!$ 分成 $p$ 的幂部分与「单位部分」：$n!=p^{\,v_p(n!)}\cdot u(n)$，其中 $v_p(n!)=\sum_{i\ge1}\lfloor n/p^i\rfloor$（`exponent`），$u(n)$ 是删去所有 $p$ 因子后的乘积模 $p^e$（`split` 的第二项，靠 $[1,p^e]$ 内非 $p$ 倍数之积的周期性快速幂加余项递归求得）。于是 $\binom{n}{k}$ 在 $p^e$ 下的 $p$ 指数为 $v_p(n!)-v_p(k!)-v_p((n-k)!)$，$\ge e$ 则该分量为 $0$，否则单位部分可逆，乘上 $p^{\text{指数}}$ 即得；各 $p_i^{e_i}$ 两两互质，`crt` 用 $\sum \text{get}\cdot\frac{m}{m_i}\cdot(\frac{m}{m_i})^{-1}$ 合并。本条依赖 [7.1.001 快速幂](07-数论.md#71001) 的 `mulMod`、`powerMod`、`invMod`，并复用 [7.2.009 阶乘模合数](07-数论.md#72009) 的 `PPFactorial::split`、`fact` 与 `modulus`；只新增组合数与 CRT 层，不能另写同义的质数幂单位阶乘。

```cpp
struct BinomMod
{
    struct PP
    {
        // p 是质因子，e 是 p 在总模数中的指数。
        int p, e;
        // fac 维护模 p^e 的单位阶乘与 p-adic 指数。
        PPFactorial fac;

        PP(int p, int e) : p(p), e(e), fac(p, e)
        {
            // p 是质因子，e 是正指数；建立对应的质数幂阶乘表。
        }

        int modulus() const
        {
            // 无参数；返回本分量的质数幂模数 p^e。
            return fac.modulus();
        }

        int comb(int n, int k) const
        {
            // n、k 是组合数参数；返回 C(n,k) mod p^e，k 不在 [0,n] 时返回 0。
            assert(n >= 0); // 调试检查，可删。
            if (k < 0 || k > n)
            {
                return 0;
            }
            auto [en, un] = fac.split(n);
            auto [ek, uk] = fac.split(k);
            auto [er, ur] = fac.split(n - k);
            int q = en - ek - er;
            if (q >= e)
            {
                return 0;
            }
            int mod = modulus();
            int den = mulMod(uk, ur, mod);
            int z = mulMod(un, *invMod(den, mod), mod);
            return mulMod(z, powerMod(p, q, mod), mod);
        }

        int fact(int n) const
        {
            // n 是非负整数；返回 n! mod p^e。
            return fac.fact(n);
        }
    };

    // mod 是查询所用的正模数。
    int mod;
    // a 保存 mod 的全部质数幂分量。
    vector<PP> a;

    explicit BinomMod(int mod) : mod(mod)
    {
        // mod 是正模数且每个质数幂分量不超过 5e6；分解 mod 并建立单位阶乘表。
        assert(mod > 0); // 调试检查，可删。
        int n = mod;
        for (int p = 2; p <= n / p; p++)
        {
            if (n % p == 0)
            {
                int e = 0;
                while (n % p == 0)
                {
                    n /= p;
                    e++;
                }
                a.emplace_back(p, e);
            }
        }
        if (n > 1)
        {
            a.emplace_back(n, 1);
        }
    }

    template <class F> long long crt(F get) const
    {
        // get(q) 返回模质数幂 q.mod 的余数；返回 CRT 合并后的模 mod 最小非负余数。
        if (mod == 1)
        {
            return 0;
        }
        long long ans = 0;
        for (const auto &q : a)
        {
            int qmod = q.modulus();
            int m = mod / qmod;
            int x = mulMod(get(q), m, mod);
            x = mulMod(x, *invMod(m, qmod), mod);
            ans = ans >= mod - x ? ans - (mod - x) : ans + x;
        }
        return ans;
    }

    int C(int n, int k) const
    {
        // n、k 是组合数参数；返回 C(n,k) mod mod，k 不在 [0,n] 时返回 0。
        assert(n >= 0); // 调试检查，可删。
        return crt(
            [&](const auto &q)
            {
                return q.comb(n, k);
            });
    }

    int fact(int n) const
    {
        // n 是非负整数；返回 n! mod mod。
        assert(n >= 0); // 调试检查，可删。
        return crt(
            [&](const auto &q)
            {
                return q.fact(n);
            });
    }
};
```

构造 `BinomMod q(mod)` 后调用 `q.C(n,k)` 或 `q.fact(n)`。构造时分解 `mod`、为每个质数幂 $p^e$ 建长度 $p^e$ 的 `f` 表；当前实现用调试断言要求每个 $p^e\le5\times10^6$，避免误建无法承受的表。构造为 $O(\sqrt m+\sum p^e)$ 时间与 $O(\sum p^e)$ 空间，单次查询 $O(\sum\log_p n)$。素数模且 $n$ 小于模数时直接用 [8.1.001 阶乘、逆阶乘与组合数](#81001)，素数模且 $n$ 超模数时用 [8.1.003 Lucas 定理](#81003)；合数模才使用本条。需要按 `n,k,mod` 一次性调用而不持有对象时见 [8.1.005 任意模数组合数](#81005) 的薄封装。

公开的 `crt(get)` 可合并调用者按各质数幂计算的余数，例如重建 $23\bmod12$：

```cpp
BinomMod q(12);
int x = q.crt([](const auto &part)
{
    return 23 % part.modulus();
});
assert(x == 11);
```

下面固定覆盖模 $1$、质数幂、混合质数幂和越界排名：

```cpp
signed main()
{
    assert(BinomMod(1).C(20, 7) == 0);
    assert(BinomMod(8).C(10, 3) == 0);
    assert(BinomMod(72).C(20, 6) == 24);
    BinomMod q(60);
    assert(q.C(40, 41) == 0);
}
```

### 8.1.005 任意模数组合数

任意正模数下的组合数与扩展 Lucas 是同一数学算法，这里不重复底层类，只依赖 [8.1.004 扩展 Lucas](#81004) 的 `BinomMod`，包一层按 `n,k,mod` 一次性调用的入口；区别只在于是否预先持有模数对象。

```cpp
int binomAnyMod(int n, int k, int mod)
{
    // n、k 是组合数参数，mod 是正模数且每个质数幂分量不超过 5e6；返回 C(n,k) mod mod。
    return BinomMod(mod).C(n, k);
}
```

`binomAnyMod(n,k,mod)` 每次都新建 `BinomMod`，因此同样要求 `mod` 的每个质数幂分量不超过 $5\times10^6$；实测 `binomAnyMod(10,3,12)=0`、`binomAnyMod(10,4,1000)=210`。单次构造加查询的开销与 [8.1.004 扩展 Lucas](#81004) 相同，只适合零散查询；同一 `mod` 反复调用时应持有一个 `BinomMod` 复用预处理。

### 8.1.006 Vandermonde 卷积

从 $m$ 个第一类元素与 $n$ 个第二类元素中共选 $r$ 个，按第一类恰选了 $k$ 个分类，各类互斥且并起来是全部方案，于是有 Vandermonde 卷积 $\sum_k\binom{m}{k}\binom{n}{r-k}=\binom{m+n}{r}$。下面这段既是它的直接实现，也可当作核对组合数接口的验证：把左侧逐项求和与右侧 $\binom{m+n}{r}$ 比对。求和下标已收紧到 $k\in[\max(0,r-n),\min(m,r)]$，落在此外的组合数必为 $0$。

```cpp
template <class C> auto vandermonde(int m, int n, int r, C comb)
{
    // m、n 是两组大小，r 是总选择数，comb(x,y) 返回 C(x,y)；返回 Vandermonde 左侧求和。
    assert(m >= 0 && n >= 0); // 调试检查，可删。
    using T = decltype(comb(0, 0));
    T ans{};
    for (int k = max<int>(0, r - n); k <= min(m, r); k++)
    {
        ans += comb(m, k) * comb(n, r - k);
    }
    return ans;
}
```

把任一组合数接口作为回调传给 `vandermonde(m,n,r,comb)`，`comb(x,y)` 对合法参数返回同一系数类型，返回值类型由 `decltype(comb(0,0))` 推断；右侧直接写 `comb(m+n,r)` 即可，固定查验值为 `vandermonde(2,3,2,comb)==10`。共 $O(\min(m,r)-\max(0,r-n)+1)$ 次组合数调用、$O(1)$ 额外空间。`comb` 通常取 [8.1.001 阶乘、逆阶乘与组合数](#81001) 的 `Comb<Z>::C`（素数模）或 [8.1.004 扩展 Lucas](#81004) 的 `BinomMod::C`（合数模）；本条本身只做求和，不预处理阶乘。

固定素数模下可把 `Comb<Z>::C` 包成 Lambda：

```cpp
Comb<Z> c;
auto comb = [&](int n, int k)
{
    return c.C(n, k);
};
assert(vandermonde(2, 3, 2, comb) == Z(10));
```

### 8.1.007 容斥原理

容斥把并集计数写成所有非空交集按集合个数奇加偶减：$|\bigcup A_i|=\sum_{\varnothing\ne S}(-1)^{|S|+1}\,|\bigcap_{i\in S}A_i|$，前提是各交集计数针对同一全集。这里的示例把它落到「$[1,n]$ 中至少被某个 $d_i$ 整除的整数个数」：子集 $S$ 对应除数的最小公倍数 $\operatorname{lcm}$，其倍数个数为 $\lfloor n/\operatorname{lcm}\rfloor$，按 `popcount(s)` 的奇偶加减。累乘 `lcm` 时用 `l > n / (d[i]/g)` 提前判溢出并跳过该子集（此时 $\operatorname{lcm}>n$，贡献本就为 $0$），避免乘积炸掉 `int`。

```cpp
namespace incl_excl
{
int countMult(int n, vector<int> d)
{
    // n 是闭区间上界，d 是正除数表；返回 [1,n] 中至少被一个 d[i] 整除的整数个数。
    assert(n >= 0 && d.size() < 63); // 调试检查，可删。
    for (int x : d)
    {
        assert(x > 0); // 调试检查，可删。
    }
    int ans = 0;
    for (unsigned long long s = 1; s < (1ULL << d.size()); s++)
    {
        int l = 1;
        bool ok = true;
        for (int i = 0; i < (int)d.size(); i++)
        {
            if (!(s >> i & 1))
            {
                continue;
            }
            int g = gcd(l, d[i]);
            if (l > n / (d[i] / g))
            {
                ok = false;
                break;
            }
            l *= d[i] / g;
        }
        if (ok)
        {
            ans += (popcount(s) & 1 ? 1 : -1) * (n / l);
        }
    }
    assert(0 <= ans && ans <= n); // 调试检查，可删。
    return ans;
}
}
```

`countMult(n, d)` 要求 `n>=0`、每个除数为正且个数 $k<63$（否则子集掩码溢出），返回 $[1,n]$ 内至少被一个 $d_i$ 整除的整数个数；还须保证容斥过程的带符号部分和能由 `int` 表示。枚举共 $O(2^k k)$ 时间、$O(1)$ 额外空间。除数很多时，先删掉被更小除数整除的冗余项；值域不大可筛出所有倍数，具有额外数论结构时再用 Möbius 或倍数 DP，不能用 [8.1.009 单位根反演](#81009) 直接替代任意除数集合的并集。第 $k$ 大与子集最小值和之间的转换见 [8.1.008 扩展 min-max 容斥与 KMAX-MIN 反演](#81008)。

### 8.1.008 扩展 min-max 容斥与 KMAX-MIN 反演

普通 min-max 容斥是 $\max(S)=\sum_{\varnothing\ne T\subseteq S}(-1)^{|T|+1}\min(T)$；扩展版把「第 $k$ 大」写成各子集最小值的带二项式系数组合：$\operatorname{kthmax}(S)=\sum_{T}(-1)^{|T|-k}\binom{|T|-1}{k-1}\min(T)$。按大小聚合后只依赖每个尺寸 $j$ 的子集最小值之和 $mn_j$，得 $\text{第 }k\text{ 大}=\sum_{j\ge k}(-1)^{j-k}\binom{j-1}{k-1}mn_j$。为避免巨大交错部分和，代码分别在三个互素大素数下递推二项式系数，再用 CRT 恢复唯一的 64 位有符号答案；三模乘积远大于整个 64 位值域。

```cpp
namespace minmax_inv
{
int kthMaxMinSum(const vector<int> &mn, int k)
{
    // mn[j] 是所有大小为 j 的子集最小值之和且 mn[0] 不用，k 从 1 开始；返回第 k 大元素。
    int n = (int)mn.size() - 1;
    assert(1 <= k && k <= n); // 调试检查，可删。
    constexpr array<long long, 3> ps{998244353, 1000000007, 1000000009};
    assert(n < ps[0]); // 调试检查，可删。
    array<long long, 3> rem{};
    for (int z = 0; z < 3; z++)
    {
        long long mod = ps[z];
        vector<long long> inv(n - k + 1);
        if (n > k)
        {
            inv[1] = 1;
        }
        for (int i = 2; i <= n - k; i++)
        {
            inv[i] = mod - (mod / i) * inv[mod % i] % mod;
        }
        long long c = 1;
        long long ans = 0;
        for (int j = k; j <= n; j++)
        {
            if (j > k)
            {
                c = c * (j - 1) % mod * inv[j - k] % mod;
            }
            long long x = mn[j] % mod;
            if (x < 0)
            {
                x += mod;
            }
            long long v = c * x % mod;
            if ((j - k) & 1)
            {
                ans -= v;
                if (ans < 0)
                {
                    ans += mod;
                }
            }
            else
            {
                ans += v;
                if (ans >= mod)
                {
                    ans -= mod;
                }
            }
        }
        rem[z] = ans;
    }
    __int128 ans = rem[0];
    __int128 m = ps[0];
    for (int z = 1; z < 3; z++)
    {
        long long mod = ps[z];
        long long a = rem[z] - (long long)(ans % mod);
        if (a < 0)
        {
            a += mod;
        }
        long long b = m % mod;
        long long e = mod - 2;
        long long iv = 1;
        while (e)
        {
            if (e & 1)
            {
                iv = iv * b % mod;
            }
            b = b * b % mod;
            e >>= 1;
        }
        ans += m * (a * iv % mod);
        m *= mod;
    }
    if (ans > m / 2)
    {
        ans -= m;
    }
    assert(numeric_limits<int>::min() <= ans && ans <= numeric_limits<int>::max()); // 调试检查，可删。
    return (int)ans;
}
}
```

`kthMaxMinSum(mn, k)` 中 `mn[j]` 是所有大小为 $j$ 的子集最小值之和（`mn[0]` 不用），每个和与最终答案都必须能放进 `int`；`k` 从 $1$ 计，数组长度小于三个模数。三次线性递推与常数次 CRT 共 $O(n)$ 时间、$O(n)$ 额外空间，不会因最终抵消前的部分和而溢出。这套反演用于「各尺寸子集最小值和已知、要还原第 $k$ 名」；只需并集大小时用 [8.1.007 容斥原理](#81007)。

### 8.1.009 单位根反演

单位根的正交性给出指示函数 $[m\mid t]=\frac1m\sum_{j=0}^{m-1}\omega^{jt}$（$\omega$ 是 $m$ 次本原单位根）。把它套到生成函数 $A(x)=\sum_k a_k x^k$ 上，取 $t=k-r$ 就能滤出指数模 $m$ 余 $r$ 的那些项：$\sum_{k\equiv r\,(m)}a_k=\frac1m\sum_{j=0}^{m-1}\omega^{-jr}A(\omega^j)$，非目标指数在求和中相位抵消。这里在模 $998244353$ 的有限域里做，$\omega=3^{(p-1)/m}$ 要求 $m\mid p-1$；`rootFilter` 对每个 $\omega^j$ 用 Horner（`a.rbegin()` 起）求 $A(\omega^j)$，乘上 $\omega^{-jr}$ 累加，最后乘 $m^{-1}$。

```cpp
namespace roots_unity
{
constexpr int mod = 998244353;

int qpow(int a, int n)
{
    // a 是底数，n 是非负指数；返回模 mod 幂。
    assert(n >= 0); // 调试检查，可删。
    a %= mod;
    if (a < 0)
    {
        a += mod;
    }
    int r = 1;
    while (n)
    {
        if (n & 1)
        {
            r = r * a % mod;
        }
        a = a * a % mod;
        n >>= 1;
    }
    return r;
}

int rootFilter(const vector<int> &a, int m, int rem)
{
    // a 是多项式系数，m 是整除 mod-1 的正整数，rem 是余数；返回次数同余 rem 的系数和。
    assert(m > 0 && (mod - 1) % m == 0); // 调试检查，可删。
    rem %= m;
    if (rem < 0)
    {
        rem += m;
    }
    int w = qpow(3, (mod - 1) / m);
    int ans = 0;
    for (int j = 0; j < m; j++)
    {
        int x = qpow(w, j);
        int y = 0;
        for (auto it = a.rbegin(); it != a.rend(); ++it)
        {
            int c = *it % mod;
            if (c < 0)
            {
                c += mod;
            }
            y = (y * x + c) % mod;
        }
        int e = ((m - rem) % m) * j % (mod - 1);
        ans = (ans + y * qpow(w, e)) % mod;
    }
    return ans * qpow(m, mod - 2) % mod;
}
}
```

`rootFilter(a, m, rem)` 中 `a` 是多项式系数（`a[i]` 为 $x^i$ 系数），`m` 必须整除 $p-1$，`rem` 是目标余数（内部先规约到 $[0,m)$）；返回次数同余 `rem` 的系数和。直接求值 $O(mn)$ 时间、$O(1)$ 额外空间（$n$ 为多项式次数）。当 $m$ 很小而需要频繁分组，或题目本身要卷积时，用第 10 章的 NTT 把多点求值降到 $O(n\log n)$ 更划算；这里的朴素版胜在实现短、无需搬多项式全家桶。它与 [8.1.007 容斥原理](#81007) 都在做「按同余类筛选」，但单位根反演处理的是幂级数系数而非集合并。

### 8.1.010 球盒模型与鸽巢原理

球盒计数先分清球、盒是否有标号，能否为空，有无容量限制。本条实现最常用的隔板法：无标号球放进有标号盒，允许空盒时为 $\binom{b+m-1}{m-1}$，要求非空时为 $\binom{b-1}{m-1}$。`comb` 每轮先约分再做一次受检乘法，默认用 64 位；需要更宽结果时由调用者显式选择类型。鸽巢下界用商与余数计算，不形成可能溢出的 `objects+boxes-1`。

```cpp
namespace balls_boxes
{
template <class T = long long>
T comb(T n, int k)
{
    // n 是非负上标，k 是下标；返回二项式系数。
    static_assert(numeric_limits<T>::is_integer && numeric_limits<T>::is_signed &&
                  sizeof(T) >= sizeof(long long));
    assert(n >= 0); // 调试检查，可删。
    if (k < 0 || (T)k > n)
    {
        return T(0);
    }
    if ((T)k > n - k)
    {
        k = (int)(n - k);
    }
    T ans = 1;
    for (int i = 1; i <= k; i++)
    {
        T x = n - k + i;
        int y = i;
        int g = gcd((int)(x % y), y);
        x /= g;
        y /= g;
        assert(ans % y == 0); // 调试检查，可删。
        ans /= y;
        ans *= x;
    }
    return ans;
}

template <class T = long long>
T unlabBoxDist(int balls, int boxes, bool nonempty)
{
    // balls 是球数，boxes 是盒数，nonempty 表示每盒非空；返回方案数。
    static_assert(numeric_limits<T>::is_integer && numeric_limits<T>::is_signed &&
                  sizeof(T) >= sizeof(long long));
    assert(balls >= 0 && boxes >= 0); // 调试检查，可删。
    if (boxes == 0)
    {
        return T(balls == 0);
    }
    if (nonempty)
    {
        return balls < boxes ? T(0) : comb<T>((T)balls - 1, boxes - 1);
    }
    T b = balls, m = boxes;
    return comb<T>(b + m - 1, boxes - 1);
}

int pigeonBound(int objects, int boxes)
{
    // objects 是对象数，boxes 是正盒数；返回必有一盒至少包含的对象数。
    assert(objects >= 0 && boxes > 0); // 调试检查，可删。
    return objects / boxes + (objects % boxes != 0);
}
}
```

两个计数函数默认返回 `long long`；调用者须保证上标和答案能由 `T` 表示，需要 128 位精确值时调用 `comb<__int128>` 或 `unlabBoxDist<__int128>`。`comb` 为 $O(\min(k,n-k))$ 时间、$O(1)$ 空间。超过 128 位用 [14.1.001 高精度整数（BigInt）](#141001)，只求模意义答案则用 [8.1.001 阶乘、逆阶乘与组合数](#81001) 或 [8.1.004 扩展 Lucas](#81004)。

### 8.1.011 q-二项式与 q-Lucas

Gaussian 二项式由 $\genfrac{[}{]}{0pt}{}{n}{k}_q=\prod_{i=1}^{k}\frac{[n-k+i]_q}{[i]_q}$ 定义，其中 $[i]_q=1+q+\cdots+q^{i-1}$。模素数 $p$ 时，直接除法会在某个 $[i]_q=0$ 后失效；令 $d$ 为首个正的零 q-整数，若 $q=0$ 或遍历到 $p-1$ 仍无零则取 $d=p$。对 $0\le n,k<p$ 有 q-Lucas 分解
$\genfrac{[}{]}{0pt}{}{n}{k}_q=\binom{\lfloor n/d\rfloor}{\lfloor k/d\rfloor}\genfrac{[}{]}{0pt}{}{n\bmod d}{k\bmod d}_q\pmod p$。
低位的 q-阶乘均非零，高位是普通组合数；$q=0$ 时合法组合数恒为 $1$，$q=1$ 时退化为普通组合数，二者都无需另写特判。

```cpp
class QBinomPrime
{
    // p 是素数模，q 是规范到 [0,p) 的参数，d 是首个零 q-整数的位置；无零时 d=p。
    int p, q, d;
    // fac/ifac 是 0..d-1 的 q-阶乘及逆元，cfac/cifac 是高位普通阶乘及逆元。
    vector<int> fac, ifac, cfac, cifac;

    int mul(int a, int b) const
    {
        // a、b 是模 p 乘数；返回 ab mod p。
        return combMulMod(a, b, p);
    }

    int power(int a, int n) const
    {
        // a 是模 p 底数，n 是非负指数；返回 a^n mod p。
        int r = 1;
        while (n)
        {
            if (n & 1)
            {
                r = mul(r, a);
            }
            a = mul(a, a);
            n >>= 1;
        }
        return r;
    }

    int small(int n, int k) const
    {
        // 0<=n<d，k 是下标；返回对应低位 q-二项式，k 不在 [0,n] 时返回 0。
        if (k < 0 || k > n)
        {
            return 0;
        }
        return mul(mul(fac[n], ifac[k]), ifac[n - k]);
    }

    int ordinary(int n, int k) const
    {
        // n 是高位上标，k 是高位下标；返回普通 C(n,k) mod p。
        if (k < 0 || k > n)
        {
            return 0;
        }
        return mul(mul(cfac[n], cifac[k]), cifac[n - k]);
    }

  public:
    QBinomPrime(int p, int q) : p(p), q(q)
    {
        // p 是可承担 O(p) 预处理的素数，q 是 q-二项式参数；建立 q-Lucas 查询表。
        assert(p >= 2); // 调试检查，可删。
        this->q %= p;
        if (this->q < 0)
        {
            this->q += p;
        }
        fac = {1};
        vector<int> num;
        int x = 1;
        for (int i = 1; i < p; i++)
        {
            if (x == 0)
            {
                break;
            }
            num.push_back(x);
            fac.push_back(mul(fac.back(), x));
            x = (mul(this->q, x) + 1) % p;
        }
        d = fac.size();
        ifac.resize(d);
        ifac[d - 1] = power(fac[d - 1], p - 2);
        for (int i = d - 1; i >= 1; i--)
        {
            ifac[i - 1] = mul(ifac[i], num[i - 1]);
        }

        int m = (p - 1) / d;
        cfac.assign(m + 1, 1);
        cifac.assign(m + 1, 1);
        for (int i = 1; i <= m; i++)
        {
            cfac[i] = mul(cfac[i - 1], i);
        }
        cifac[m] = power(cfac[m], p - 2);
        for (int i = m; i >= 1; i--)
        {
            cifac[i - 1] = mul(cifac[i], i);
        }
    }

    int C(int n, int k) const
    {
        // n、k 是 q-二项式参数且 0<=n<p；返回 [n choose k]_q mod p，k 越界时返回 0。
        assert(0 <= n && n < p); // 调试检查，可删。
        if (k < 0 || k > n)
        {
            return 0;
        }
        return mul(ordinary(n / d, k / d), small(n % d, k % d));
    }
};
```

构造 `QBinomPrime cb(p,q)` 后用 `cb.C(n,k)`，要求 `p` 为素数、$0\le n<p$，并且 $p$ 小到能承受线性预处理。固定查验值为 `QBinomPrime(7,2).C(6,3)==2`。设 q-整数周期为 $d$，构造 $O(d+p/d)$ 时间与空间，单次查询 $O(1)$；同一 `p,q` 的多组询问必须复用对象。若 $n\ge p$，还要继续按 $p$ 进制处理普通组合数高位，不能直接套本接口；普通二项式见 [8.1.003 Lucas 定理](#81003)，多项式形式的快速运算见第 10 章。

### 8.1.012 二项式恒等式与反演

约定 $n\ge0$ 且 $\binom nk=0$ 对所有 $k\notin[0,n]$ 成立。二项式定理及其系数形式为

$$
\begin{aligned}
(x+y)^n&=\sum_{k=0}^{n}\binom nk x^{n-k}y^k,\\
[x^k](1+x)^n&=\binom nk.
\end{aligned}
$$

分别代入 $x=1$ 与 $x=-1$，得到

$$
\sum_{k=0}^{n}\binom nk=2^n,
\qquad
\sum_{k=0}^{n}(-1)^k\binom nk=[n=0].
$$

对 $(1+x)^n$ 求导后乘 $x$，再令 $x=1$，可依次得到加权和。下式要求 $n\ge2$；$n=0,1$ 直接代入求和式。

$$
\begin{aligned}
\sum_k k\binom nk&=n2^{n-1},\\
\sum_k k(k-1)\binom nk&=n(n-1)2^{n-2},\\
\sum_k k^2\binom nk&=n(n+1)2^{n-2}.
\end{aligned}
$$

Vandermonde 卷积见 [8.1.006 Vandermonde 卷积](#81006)。令其中一侧退化，可得 hockey-stick 恒等式；其条件为 $n\ge r\ge0$：

$$
\sum_{k=r}^{n}\binom kr=\binom{n+1}{r+1}.
$$

二项式反演必须成对使用同一符号约定。对任意序列，若

$$
b_n=\sum_{k=0}^{n}\binom nk a_k,
$$

则恰有

$$
a_n=\sum_{k=0}^{n}(-1)^{n-k}\binom nk b_k.
$$

这些等式是整数恒等式，取模时可直接逐项取模，不涉及除法；按 $(-1)^k$ 书写的等价反演式需要同时调整两边符号，不能只替换一个求和式。

## 8.2 经典数列与计数

### 8.2.001 Fibonacci / Lucas 数

矩阵快速幂 $\begin{bmatrix}1&1\\1&0\end{bmatrix}^n$ 展开后可消去冗余乘法，得到倍增恒等式 $F_{2n}=F_n(2F_{n+1}-F_n)$、$F_{2n+1}=F_n^2+F_{n+1}^2$。`fibPair` 每次递归把下标折半、同时维护相邻对 $\{F_n,F_{n+1}\}$，比通用 $2\times2$ 矩阵幂常数更小；Lucas 数则由 $L_n=2F_{n+1}-F_n$ 直接换算。模乘调用 `combMulMod`，常见小模数直接使用 64 位乘法，接近 `long long` 上界时才局部使用 `__int128`；$n=0$ 时返回 $\{0,1\bmod mod\}$，兼容 `mod=1`。

```cpp
array<int, 2> fibPair(unsigned long long n, int mod)
{
    // n 是下标，mod 是正模数；返回 {F_n,F_{n+1}} mod mod。
    assert(mod > 0); // 调试检查，可删。
    if (n == 0)
    {
        return {0, 1 % mod};
    }
    auto [a, b] = fibPair(n >> 1, mod);
    int t = b >= mod - b ? b - (mod - b) : b + b;
    t -= a;
    if (t < 0)
    {
        t += mod;
    }
    int c = combMulMod(a, t, mod);
    int x = combMulMod(a, a, mod);
    int y = combMulMod(b, b, mod);
    int d = x >= mod - y ? x - (mod - y) : x + y;
    if (n & 1)
    {
        int next = c >= mod - d ? c - (mod - d) : c + d;
        return {d, next};
    }
    return {c, d};
}

int lucasNumber(unsigned long long n, int mod)
{
    // n 是下标，mod 是正模数；返回 Lucas 数 L_n mod mod。
    assert(mod > 0); // 调试检查，可删。
    if (n == 0)
    {
        return 2 % mod;
    }
    auto [fn, fn1] = fibPair(n, mod);
    int ans = fn1 >= mod - fn1 ? fn1 - (mod - fn1) : fn1 + fn1;
    ans -= fn;
    return ans < 0 ? ans + mod : ans;
}
```

`fibPair(n,mod)` 返回 $\{F_n,F_{n+1}\}\bmod mod$，`lucasNumber(n,mod)` 返回 $L_n\bmod mod$；模 $10^9+7$ 下 `fibPair(10)={55,89}`、`lucasNumber(10)=123`。$O(\log n)$ 时间、$O(\log n)$ 递归栈。需要精确大整数时沿用倍增式，改以 [14.1.001 高精度整数（BigInt）](#141001) 保存两项并去掉取模。这是唯一按下标 $O(\log n)$ 求值的数列；下面 Catalan、Motzkin、Bell 等都是 $O(n^2)$ 或 $O(n\sqrt n)$ 的整段递推，求单个高位项时反而应回到对应的闭式或矩阵幂。

### 8.2.002 Catalan 数

合法括号序列、凸多边形三角剖分、二叉树形态等都由 Catalan 数计数。按第一个匹配位置（或根）把对象拆成互相独立的左右两块，得卷积递推 $C_i=\sum_{j=0}^{i-1}C_j C_{i-1-j}$，$C_0=1$；闭式 $C_n=\frac{1}{n+1}\binom{2n}{n}$ 只在分母可逆或整数约分时才直接用。下面返回 $C_0,\ldots,C_n$，`T` 只需支持加法、乘法和从 $1$ 构造。

```cpp
template <class T> vector<T> catalan(int n)
{
    // n 是要计算的最大下标；返回 C_0 到 C_n 的 Catalan 数。
    assert(n >= 0); // 调试检查，可删。
    vector<T> a(n + 1);
    a[0] = 1;
    for (int i = 1; i <= n; i++)
    {
        for (int j = 0; j < i; j++)
        {
            a[i] += a[j] * a[i - 1 - j];
        }
    }
    return a;
}
```

`catalan<T>(n)` 返回 $C_0,\ldots,C_n$，卷积递推为 $O(n^2)$ 时间、$O(n)$ 输出空间。模素数且 $2n$ 小于模数时，用 [8.1.001 阶乘、逆阶乘与组合数](#81001) 按闭式预处理整段只需 $O(n)$；用形式幂级数时则由 $C(x)=1+xC(x)^2$ 做 Newton 迭代。图的生成树计数不属于这类一维递推，见 [8.2.011 Matrix-Tree / Kirchhoff 生成树计数](#82011)。

前若干项 $C_0,C_1,\dots$：$1,1,2,5,14,42,132,429,1430,4862,16796,58786,208012,742900$（$64$ 位有符号内到 $C_{35}=3116285494907301262$ 不溢出，$C_{36}$ 溢出）。

### 8.2.003 Motzkin 数

Motzkin 数统计不越过横轴、每步为上/下/平的格路（也对应圆上非交弦等结构）。按首步分类：若首步是水平步，剩下是长度 $i-1$ 的 Motzkin 路；若首步上升，则到首次回零之间夹一段 Motzkin 路、之后再接一段，于是 $M_i=M_{i-1}+\sum_{j=0}^{i-2}M_j M_{i-2-j}$，$M_0=1$，即一个线性项加 Catalan 型卷积。

```cpp
template <class T> vector<T> motzkin(int n)
{
    // n 是要计算的最大下标；返回 M_0 到 M_n 的 Motzkin 数。
    assert(n >= 0); // 调试检查，可删。
    vector<T> a(n + 1);
    a[0] = 1;
    for (int i = 1; i <= n; i++)
    {
        a[i] = a[i - 1];
        for (int j = 0; j + 2 <= i; j++)
        {
            a[i] += a[j] * a[i - 2 - j];
        }
    }
    return a;
}
```

`motzkin<T>(n)` 返回 $M_0,\ldots,M_n$，卷积递推 $O(n^2)$ 时间、$O(n)$ 空间。与 [8.2.002 Catalan 数](#82002) 同族但多了水平步：只有上、下步且必须回到起点时是 Catalan，允许平步则是 Motzkin。

前若干项 $M_0,M_1,\dots$：$1,1,2,4,9,21,51,127,323,835,2188,5798,15511$。

### 8.2.004 Bell 数

Bell 数 $B_n$ 数把 $n$ 个带标号元素划分成若干非空无序集合的方案。递推 $B_{n+1}=\sum_{k=0}^{n}\binom{n}{k}B_k$ 来自按「含最后一个元素的块」大小分类。代码用等价的 Bell 三角实现：每行首项抄上一行末项，同行向右做前缀累加 $a[i][j]=a[i][j-1]+a[i-1][j-1]$，每行首项 `a[i][0]` 即 $B_i$，避免显式算组合数。

```cpp
template <class T> vector<T> bell(int n)
{
    // n 是要计算的最大下标；返回 B_0 到 B_n 的 Bell 数。
    assert(n >= 0); // 调试检查，可删。
    vector<vector<T>> a(n + 1, vector<T>(n + 1));
    vector<T> b(n + 1);
    a[0][0] = 1;
    b[0] = 1;
    for (int i = 1; i <= n; i++)
    {
        a[i][0] = a[i - 1][i - 1];
        b[i] = a[i][0];
        for (int j = 1; j <= i; j++)
        {
            a[i][j] = a[i][j - 1] + a[i - 1][j - 1];
        }
    }
    return b;
}
```

`bell<T>(n)` 返回 $B_0,\ldots,B_n$，Bell 三角为 $O(n^2)$ 时间与空间。Bell 数是第二类 Stirling 数按块数求和 $B_n=\sum_k S(n,k)$，需要按块数分开计数时用 [8.2.005 Stirling 数](#82005) 的第二类；只关心总划分数则本条足够。模素数下需要整段大规模结果时，可由 EGF $e^{e^x-1}$ 调用第 10 章形式幂级数指数。

前若干项 $B_0,B_1,\dots$：$1,1,2,5,15,52,203,877,4140,21147,115975,678570$。

### 8.2.005 Stirling 数

第一类 Stirling 数按下标计排列的环数，第二类计集合划分成指定块数的方案。加入第 $i$ 个新元素时，无符号第一类满足 $\left[{i\atop j}\right]=\left[{i-1\atop j-1}\right]+(i-1)\left[{i-1\atop j}\right]$；代码采用对应下降幂 $x^{\underline n}$ 系数的有符号版本 $s(i,j)=s(i-1,j-1)-(i-1)s(i-1,j)$，需要环数时取绝对值。第二类中，新元素要么独占新块，要么并入已有 $j$ 块之一，故 $\left\{{i\atop j}\right\}=\left\{{i-1\atop j-1}\right\}+j\left\{{i-1\atop j}\right\}$。

```cpp
template <class T> vector<vector<T>> stirling1(int n)
{
    // n 是最大行号；返回有符号第一类 Stirling 数 s(i,j) 的第 0..n 行。
    assert(n >= 0); // 调试检查，可删。
    vector<vector<T>> a(n + 1, vector<T>(n + 1));
    a[0][0] = 1;
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= i; j++)
        {
            a[i][j] = a[i - 1][j - 1] - T(i - 1) * a[i - 1][j];
        }
    }
    return a;
}

template <class T> vector<vector<T>> stirling2(int n)
{
    // n 是最大行号；返回第二类 Stirling 数 S(i,j) 的第 0..n 行。
    assert(n >= 0); // 调试检查，可删。
    vector<vector<T>> a(n + 1, vector<T>(n + 1));
    a[0][0] = 1;
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= i; j++)
        {
            a[i][j] = a[i - 1][j - 1] + T(j) * a[i - 1][j];
        }
    }
    return a;
}
```

`stirling1<T>(n)`、`stirling2<T>(n)` 各返回一张 $(n+1)\times(n+1)$ 的三角表，$a[i][j]$ 为对应 Stirling 数；均 $O(n^2)$ 时间与空间。只查一个第二类 $S(n,k)$ 且模素数时，可用 $S(n,k)=\frac{1}{k!}\sum_j(-1)^j\binom{k}{j}(k-j)^n$ 配 [8.1.001 阶乘、逆阶乘与组合数](#81001)，以 $O(k\log n)$ 时间求值；需要小规模整表时用本条。集合划分不限块数就求和成 [8.2.004 Bell 数](#82004)。

在 $m,n\ge0$ 时，交错幂和与第二类 Stirling 数满足下式；当 $m=n=0$ 时约定 $0^0=1$：

$$
\sum_{k=0}^{n}(-1)^k\binom nk k^m
=(-1)^n n!\left\{\!\!\begin{matrix}m\\n\end{matrix}\!\!\right\}.
$$

因此 $m<n$ 时为 $0$，$m=n$ 时为 $(-1)^n n!$。当模素数 $p$ 且 $n<p$ 时，因 $n!$ 可逆，可以反解为

$$
\left\{\!\!\begin{matrix}m\\n\end{matrix}\!\!\right\}
=\frac1{n!}\sum_{k=0}^{n}(-1)^{n-k}\binom nk k^m\pmod p.
$$

若模数为合数，或 $p\mid n!$，不能直接使用这个分式形式。

### 8.2.006 Eulerian 数

Eulerian 数 $\left\langle{n\atop k}\right\rangle$ 计 $n$ 元排列中恰有 $k$ 个下降位置（$a_i>a_{i+1}$）的排列数，下标 $k$ 从 $0$ 起。把最大元素插入长度 $i-1$ 排列的 $i$ 个空隙：插进已有的某个下降处或末尾（保持下降数，共 $j+1$ 种），或插进上升处新增一个下降（$i-j$ 种），得递推 $\left\langle{i\atop j}\right\rangle=(j+1)\left\langle{i-1\atop j}\right\rangle+(i-j)\left\langle{i-1\atop j-1}\right\rangle$。有些资料把递增段数 $k+1$ 作为列下标，本实现始终用下降位置数 $k$。

```cpp
template <class T> vector<vector<T>> eulerian(int n)
{
    // n 是最大排列长度；返回第 0..n 行 Eulerian 数，列下标是下降位置数。
    assert(n >= 0); // 调试检查，可删。
    vector<vector<T>> a(n + 1, vector<T>(n + 1));
    a[0][0] = 1;
    for (int i = 1; i <= n; i++)
    {
        for (int j = 0; j < i; j++)
        {
            if (j)
            {
                a[i][j] += T(i - j) * a[i - 1][j - 1];
            }
            a[i][j] += T(j + 1) * a[i - 1][j];
        }
    }
    return a;
}
```

`eulerian<T>(n)` 返回三角表，下标为元素数与下降数，$O(n^2)$ 时间与空间。Eulerian 数与 Worpitzky 恒等式、$\sum k^n$ 幂和相关联；若只需幂和本身，[8.2.007 Bernoulli 数与幂和](#82007) 的 Faulhaber 公式更直接。它与第二类 Stirling 数可互相线性表出，按题目要「下降数分布」还是「块数分布」二选一。

### 8.2.007 Bernoulli 数与幂和

本条采用 $B_1=+\frac12$ 的 Bernoulli 约定，由差分表以 $a[j]=\frac1{j+1}$ 起、反复做 $a[j-1]\leftarrow j(a[j-1]-a[j])$ 得到。对应的 Faulhaber 公式是 $\sum_{i=1}^{n}i^m=\frac1{m+1}\sum_{k=0}^{m}\binom{m+1}{k}B_k n^{m+1-k}$；例如 $m=1$ 给出 $\frac{n^2+n}{2}$。因递推含除法，系数类型 `T` 必须支持除法，模素数时各分母须可逆。

```cpp
template <class T> vector<T> bernoulli(int n)
{
    // n 是要计算的最大下标；返回 B_0 到 B_n，采用 B_1=+1/2 的约定。
    assert(n >= 0); // 调试检查，可删。
    vector<T> a(n + 1), b(n + 1);
    for (int m = 0; m <= n; m++)
    {
        a[m] = T(1) / T(m + 1);
        for (int j = m; j; j--)
        {
            a[j - 1] = T(j) * (a[j - 1] - a[j]);
        }
        b[m] = a[0];
    }
    return b;
}
```

`bernoulli<T>(n)` 返回 $B_0,\ldots,B_n$，$O(n^2)$ 时间、$O(n)$ 空间。本条符号约定下前四项为 $1,\frac12,\frac16,0$，可优先用 $B_1=+\frac12$ 排查约定混用。要算 $\sum_{i=1}^{n}i^m$ 时，先取 $B_0,\ldots,B_m$ 再代入 Faulhaber 公式；系数预处理后可把它作为关于 $n$ 的 $m+1$ 次多项式求值。需要更快地求整段 Bernoulli 数时，用 EGF $\frac{x}{e^x-1}$ 配 [10.2.003 乘法逆元](#102003)。

### 8.2.008 错排数

错排数 $D_n$ 是没有任何元素留在原位的排列数（空排列计为 $1$）。考虑元素 $1$ 被放到位置 $j$：若 $j$ 上的元素回放到位置 $1$（对换），剩下是 $n-2$ 个元素的错排；否则把「位置 $1$ 禁放 $1$」的约束转嫁给 $j$，化为 $n-1$ 个元素的错排。$j$ 有 $n-1$ 种取法，故 $D_n=(n-1)(D_{n-1}+D_{n-2})$，$D_0=1,D_1=0$。

```cpp
template <class T> vector<T> derangements(int n)
{
    // n 是最大规模；返回 d[0..n]，d[i] 是 i 个元素无固定点排列数。
    assert(n >= 0); // 调试检查，可删。
    vector<T> d(n + 1);
    d[0] = 1;
    if (n >= 1)
    {
        d[1] = 0;
    }
    for (int i = 2; i <= n; i++)
    {
        d[i] = T(i - 1) * (d[i - 1] + d[i - 2]);
    }
    return d;
}
```

`derangements<T>(n)` 返回 `d[0..n]`，`T` 支持加法、乘法与整数构造即可（大整数或模整数）；$O(n)$ 时间与输出空间。前七项为 $1,0,1,2,9,44,265$。等价的容斥闭式是 $D_n=n!\sum_{k=0}^{n}\frac{(-1)^k}{k!}$，模素数下想 $O(1)$ 单点查询就配 [8.1.001 阶乘、逆阶乘与组合数](#81001) 预处理；只需整段小 $n$ 时本条线性递推更短。

### 8.2.009 分拆数与五边形数

分拆数 $p(n)$ 数把 $n$ 写成无序正整数之和的方案数，生成函数为 $\prod_{k\ge1}(1-x^k)^{-1}$。对其倒数 $\prod(1-x^k)$ 用 Euler 五边形定理 $\prod_{k\ge1}(1-x^k)=\sum_{j}(-1)^j x^{j(3j-1)/2}$（$j$ 取全体整数），只在广义五边形数 $g=\frac{k(3k\pm1)}2$ 处非零，于是得正负交替递推 $p(n)=\sum_{k\ge1}(-1)^{k-1}\big(p(n-g_k^-)+p(n-g_k^+)\big)$。每个 $n$ 只枚举到 $g\le n$ 的 $O(\sqrt n)$ 个五边形数，把朴素 $O(n^2)$ 降到 $O(n\sqrt n)$。

```cpp
template <class T> vector<T> partitions(int n)
{
    // n 是要计算的最大整数；返回 p(0) 到 p(n)。
    assert(n >= 0); // 调试检查，可删。
    vector<T> a(n + 1);
    a[0] = 1;
    for (int i = 1; i <= n; i++)
    {
        for (int k = 1;; k++)
        {
            int x = k / 2;
            int y = 3 * k - 1;
            if (k & 1)
            {
                x = k;
                y /= 2;
            }
            if (x > i / y)
            {
                break;
            }
            int g = x * y;
            if (k & 1)
            {
                a[i] += a[i - g];
            }
            else
            {
                a[i] -= a[i - g];
            }
            if (g <= i - k)
            {
                g += k;
                if (k & 1)
                {
                    a[i] += a[i - g];
                }
                else
                {
                    a[i] -= a[i - g];
                }
            }
        }
    }
    return a;
}
```

`partitions<T>(n)` 返回 $p(0),\ldots,p(n)$，固定查验值为 $p(0)=1$、$p(5)=7$、$p(10)=42$；$O(n\sqrt n)$ 时间、$O(n)$ 空间。这是无限制的整数分拆；若限制份数或每份大小（如「至多 $k$ 份」「各份互异」），改用完全背包式的二维 DP 或对应生成函数。模素数下需要更长前缀时，可按五边形定理写出分母的稀疏系数，再调用 [10.2.003 乘法逆元](#102003) 在 $O(n\log n)$ 时间求逆。

### 8.2.010 有根树、无根树计数

树计数要先分清是否有根、节点是否带标号、儿子是否有序。本条实现最干净的一类——带标号无根树，即 Cayley 公式：$n$ 个带标号点的无根树共 $n^{n-2}$ 棵；若根标号预先固定，每棵无根树只有一种置根方式，数量仍为 $n^{n-2}$；若把“根是哪一个标号点”也视为结构的一部分，数量才是 $n\cdot n^{n-2}=n^{n-1}$。`cayley` 返回无根树数和后一种“根被区分”的有根树数，$n=1$ 单独处理。无标号树不在本条闭式范围内。

```cpp
namespace tree_count
{
int qpow(int a, int n, int mod)
{
    // a、n、mod 是底数、非负指数和正模数；返回模幂。
    assert(n >= 0 && mod > 0); // 调试检查，可删。
    a %= mod;
    if (a < 0)
    {
        a += mod;
    }
    int r = 1 % mod;
    while (n)
    {
        if (n & 1)
        {
            r = combMulMod(r, a, mod);
        }
        a = combMulMod(a, a, mod);
        n >>= 1;
    }
    return r;
}

array<int, 2> cayley(int n, int mod)
{
    // n 是有标号点数，mod 是正模数；返回 {无根树数, 根标号被区分的有根树数}。
    assert(n >= 1 && mod > 0); // 调试检查，可删。
    int unrooted = n == 1 ? 1 % mod : qpow(n % mod, n - 2, mod);
    int rooted = qpow(n % mod, n - 1, mod);
    return {unrooted, rooted};
}
}
```

`cayley(n, mod)` 返回的两维依次是无根树数、根标号被区分的有根树数，固定查验值为 `cayley(4,1000000007)` 返回 `{16,64}`。两次模幂共 $O(\log n)$ 时间、$O(1)$ 空间；若根固定为某个已知标号，答案取第一项。Cayley 公式只适用于完全图的标号树；给定图上数生成树要用 [8.2.011 Matrix-Tree / Kirchhoff 生成树计数](#82011) 的 Laplacian 行列式，带度数限制的标号树则用 Prüfer 序列计数。

### 8.2.011 Matrix-Tree / Kirchhoff 生成树计数

Matrix-Tree（Kirchhoff）定理把生成树计数化成行列式。无向图的 Laplacian $L=D-A$（$D$ 度数对角阵、$A$ 邻接矩阵）任意删掉同编号的一行一列，剩下 $(n-1)$ 阶子式的行列式就是生成树数量；证明来自 Cauchy–Binet 展开到关联矩阵。有向图则按朝向选入度或出度 Laplacian，`arboCount` 用入度版统计从 `root` 向外、每个非根点恰有一条入边的树形图数。平行边按重数累加；自环在 Laplacian 的度数项与邻接项中抵消，不属于生成树。行列式在有限域 `Z` 上用高斯消元 `detField` 求：逐列找非零主元（找不到说明奇异，返回 $0$），换行时翻转符号，消元后累乘对角元。

```cpp
Z detField(vector<vector<Z>> a)
{
    // a 是有限域上的方阵；返回其行列式。
    int n = a.size();
    for (const auto &row : a)
    {
        assert((int)row.size() == n); // 调试检查，可删。
    }
    Z ans = 1;
    for (int c = 0; c < n; c++)
    {
        int p = c;
        while (p < n && a[p][c] == Z(0))
        {
            p++;
        }
        if (p == n)
        {
            return 0;
        }
        if (p != c)
        {
            swap(a[p], a[c]);
            ans = -ans;
        }
        ans *= a[c][c];
        Z iv = a[c][c].inv();
        for (int i = c + 1; i < n; i++)
        {
            Z q = a[i][c] * iv;
            for (int j = c; j < n; j++)
            {
                a[i][j] -= q * a[c][j];
            }
        }
    }
    return ans;
}

Z spanTreeCnt(int n, const vector<pair<int, int>> &e)
{
    // n 是点数，e 是无向边端点列表且允许重边；返回生成树数量。
    assert(n >= 0); // 调试检查，可删。
    for (auto [u, v] : e)
    {
        assert(0 <= u && u < n && 0 <= v && v < n); // 调试检查，可删。
    }
    if (n <= 1)
    {
        return 1;
    }
    vector a(n - 1, vector<Z>(n - 1));
    for (auto [u, v] : e)
    {
        if (u < n - 1)
        {
            a[u][u] += 1;
        }
        if (v < n - 1)
        {
            a[v][v] += 1;
        }
        if (u < n - 1 && v < n - 1)
        {
            a[u][v] -= 1;
            a[v][u] -= 1;
        }
    }
    return detField(move(a));
}

Z arboCount(int n, int root, const vector<pair<int, int>> &e)
{
    // n 是点数，root 是根，e 是有向边端点列表且允许重边；返回从 root 向外的生成树数量。
    assert(n >= 1 && 0 <= root && root < n); // 调试检查，可删。
    for (auto [u, v] : e)
    {
        assert(0 <= u && u < n && 0 <= v && v < n); // 调试检查，可删。
    }
    if (n == 1)
    {
        return 1;
    }
    vector<int> id(n, -1);
    for (int u = 0, k = 0; u < n; u++)
    {
        if (u != root)
        {
            id[u] = k++;
        }
    }
    vector a(n - 1, vector<Z>(n - 1));
    for (auto [u, v] : e)
    {
        if (v == root)
        {
            continue;
        }
        a[id[v]][id[v]] += 1;
        if (u != root)
        {
            a[id[v]][id[u]] -= 1;
        }
    }
    return detField(move(a));
}
```

先准备 [1.1.008 静态取模类](01-基础与工程.md#11008) 的素数域 `Z`。`spanTreeCnt(n, e)` 接受点编号为 `0..n-1` 的无向边，`arboCount(n, root, e)` 接受有向边 $u\to v$ 并统计从 `root` 向外的树形图；空无向图 `n=0` 约定返回 $1$，有向接口要求 `n>=1`。自环可保留在边表中并自动抵消，平行边分别计数，非单点图存在孤立点时行列式为 $0$。整体 $O(n^3+m)$ 时间、$O(n^2)$ 额外空间。完全图上有闭式时优先 [8.2.010 有根树、无根树计数](#82010) 的 Cayley 公式；只有一般图才动用行列式。[5.9.003 BEST 定理与有向欧拉回路计数](#59003) 需要朝根的树形图，因此应反转原图全部边后调用 `arboCount`。

下面用三角形、孤立点和有向图各核对一个已知答案：

```cpp
signed main()
{
    assert(spanTreeCnt(3, {{0, 1}, {1, 2}, {2, 0}}).val() == 3);
    assert(spanTreeCnt(3, {{0, 1}}).val() == 0);
    assert(arboCount(3, 0, {{0, 1}, {0, 2}, {1, 2}}).val() == 2);
}
```

### 8.2.012 Pólya / Burnside 置换计数

Burnside 引理说群 $G$ 作用下的轨道数（本质不同方案数）等于各元素不动点数的平均：$|X/G|=\frac1{|G|}\sum_{g\in G}|\mathrm{Fix}(g)|$。用 $c$ 种颜色给位置染色时，一个置换 $g$ 的染色不动点要求同一循环上位置同色，故 $|\mathrm{Fix}(g)|=c^{\text{cyc}(g)}$，这就是 Pólya 计数把循环长度代入库存多项式的常数版本。`burnside` 对群里每个位置置换数出循环数 `cyc`，累加 $c^{\text{cyc}}$，最后除以 $|G|$。传入的 `group` 必须是真正的群（含单位元、对合成封闭），否则平均无意义。

```cpp
namespace polya
{
template <class T = long long>
T burnside(const vector<vector<int>> &group, int colors)
{
    // group 中每项是位置置换，colors 是颜色数；返回本质不同染色数。
    static_assert(numeric_limits<T>::is_integer && numeric_limits<T>::is_signed &&
                  sizeof(T) >= sizeof(long long));
    assert(!group.empty() && colors >= 0); // 调试检查，可删。
    int n = group[0].size();
    T sum = 0;
    for (const auto &p : group)
    {
        assert((int)p.size() == n); // 调试检查，可删。
        vector<int> cnt(n);
        for (int x : p)
        {
            assert(0 <= x && x < n); // 调试检查，可删。
            cnt[x]++;
        }
        for (int x : cnt)
        {
            assert(x == 1); // 调试检查，可删。
        }
        vector<int> vis(n);
        int cyc = 0;
        for (int i = 0; i < n; i++)
        {
            if (vis[i])
            {
                continue;
            }
            cyc++;
            int x = i;
            while (!vis[x])
            {
                vis[x] = 1;
                x = p[x];
            }
        }
        T ways = 1;
        for (int i = 0; i < cyc; i++)
        {
            ways *= colors;
        }
        sum += ways;
    }
    int sz = group.size();
    assert(sum % sz == 0); // 调试检查，可删。
    return sum / sz;
}
}
```

`burnside(group, colors)` 要求 `group` 是完整有限群，并返回本质不同染色数；调用者须保证幂和总和能由 `T` 表示，需要 128 位精确计数时调用 `burnside<__int128>`。算法访问 $O(|G|n)$ 个置换位置。结果超过 128 位时换 [14.1.001 高精度整数（BigInt）](#141001)；群规模本身指数级时应改用循环指标解析求和。

四个位置只考虑旋转、使用两种颜色时共有 $6$ 个轨道，可用这组固定群检查循环分解与平均数：

```cpp

{
    // 无参数；核对四点旋转群的二染色轨道数，返回 0。
    vector<vector<int>> rot{
        {0, 1, 2, 3}, {1, 2, 3, 0},
        {2, 3, 0, 1}, {3, 0, 1, 2}};
    assert(polya::burnside(rot, 2) == 6);
    return 0;
}
```

### 8.2.013 杨氏矩阵与钩子公式

形状为分拆 $\lambda$（行长非增）的 Young 图，其标准 Young 表（每行每列严格递增地填 $1..n$）数量由钩长公式给出：$f^\lambda=\dfrac{n!}{\prod_{\text{格 }c}h(c)}$，其中格 $c$ 的钩长 $h(c)=1+\text{右侧同行格数}+\text{下方同列格数}$。直接形成 $n!$ 可能在除掉钩长前溢出。代码用线性筛得到最小质因子，先按 Legendre 公式统计 $n!$ 的质因数指数，再逐格分解钩长并扣除指数，最后只乘答案真正保留的质因子；列高 `col[j]` 让钩长可在 $O(1)$ 时间写成 `row[i]-j+col[j]-i-1`。

```cpp
namespace young_tab
{
template <class T = long long>
T youngTab(const vector<int> &row)
{
    // row 是非增的行长分拆；返回标准 Young 表数量。
    static_assert(numeric_limits<T>::is_integer && numeric_limits<T>::is_signed &&
                  sizeof(T) >= sizeof(long long));
    size_t cells = 0;
    for (int i = 0; i < (int)row.size(); i++)
    {
        assert(row[i] >= 0); // 调试检查，可删。
        if (i)
        {
            assert(row[i - 1] >= row[i]); // 调试检查，可删。
        }
        cells += row[i];
    }
    int n = (int)cells;
    vector<int> spf(n + 1), ps;
    for (int i = 2; i <= n; i++)
    {
        if (!spf[i])
        {
            spf[i] = i;
            ps.push_back(i);
        }
        for (int p : ps)
        {
            if (p > spf[i] || i > n / p)
            {
                break;
            }
            spf[i * p] = p;
        }
    }
    vector<int> e(n + 1);
    for (int p : ps)
    {
        for (int x = n / p; x; x /= p)
        {
            e[p] += x;
        }
    }
    vector<int> col(row.empty() ? 0 : row[0]);
    for (int x : row)
    {
        for (int j = 0; j < x; j++)
        {
            col[j]++;
        }
    }
    for (int i = 0; i < (int)row.size(); i++)
    {
        for (int j = 0; j < row[i]; j++)
        {
            int x = row[i] - j + col[j] - i - 1;
            while (x > 1)
            {
                int p = spf[x];
                e[p]--;
                x /= p;
            }
        }
    }
    T ans = 1;
    for (int p : ps)
    {
        assert(e[p] >= 0); // 调试检查，可删。
        for (int i = 0; i < e[p]; i++)
        {
            ans *= p;
        }
    }
    return ans;
}

}
```

`youngTab(row)` 要求行长非负且非增，空形状返回 $1$；调用者须保证结果能由 `T` 表示，需要 128 位时调用 `youngTab<__int128>`。线性筛、列高与指数表使用 $O(n+\lambda_1)$ 空间，全部分解最坏为 $O(n\log n)$ 时间。超过 128 位的精确答案换 [14.1.001 高精度整数（BigInt）](#141001)。

下面同时检查非矩形小形状与 $2\times10$ 矩形；后者的标准 Young 表数恰为 Catalan 数 $C_{10}=16796$：

```cpp
signed main()
{
    // 无参数；核对两个 Young 图形状的标准表数量，返回 0。
    assert(young_tab::youngTab({2, 1}) == 2);
    assert(young_tab::youngTab({10, 10}) == 16796);
    return 0;
}
```

### 8.2.014 固定 k 的第一类 Stirling 数

有符号第一类 Stirling 数的指数生成函数为 $\frac{\log(1+x)^k}{k!}=\sum_{n\ge k}s(n,k)\frac{x^n}{n!}$。先写出 $\log(1+x)=\sum_{i\ge1}(-1)^{i-1}\frac{x^i}{i}$，调用多项式快速幂，最后把第 $n$ 项乘回 $n!/k!$，即可一次得到固定 `k` 的整列；符号与 [8.2.005 Stirling 数](#82005) 的 `stirling1` 完全一致。

```cpp
vector<Z> stirling1K(int n, int k)
{
    // n 是最大行号，k 是固定列号；返回 s(k,k) 到 s(n,k)，模数由 Z 决定。
    assert(0 <= k && k <= n && n < mod); // 调试检查，可删。
    Poly f(n + 1);
    for (int i = 1; i <= n; i++)
    {
        f[i] = (i & 1 ? Z(1) : Z(-1)) / Z(i);
    }
    Poly g = fpsPow(f, k, n + 1);
    Z fk = 1;
    for (int i = 1; i <= k; i++)
    {
        fk *= i;
    }
    Z ifk = fk.inv();
    Z fac = 1;
    vector<Z> ans(n - k + 1);
    for (int i = 0; i <= n; i++)
    {
        if (i >= k)
        {
            ans[i - k] = g[i] * fac * ifk;
        }
        fac *= i + 1;
    }
    return ans;
}
```

本函数依赖 [10.2.001 多项式封装](#102001) 的 `Poly/Z` 与 [10.2.005 `ln`、`exp`、开根、快速幂](#102005) 的 `fpsPow`。`stirling1K(n,k)[i-k]` 等于 $s(i,k)$，要求 $n<\operatorname{mod}$ 以保证积分分母和阶乘可逆；时间 $O(M(n))$、输出与工作空间 $O(n)$。若只查一个小 `n,k`，直接用 [8.2.005 Stirling 数](#82005) 的二维递推更简单。

### 8.2.015 固定 k 的第二类 Stirling 数

固定块数的第二类 Stirling 数满足普通生成函数 $\sum_{n\ge k}S(n,k)x^{n-k}=\prod_{i=1}^{k}(1-ix)^{-1}$。代码用平衡乘积树构造分母，避免从左到右相乘退化成二次复杂度，再调用 FPS 逆元截断到 `n-k+1` 项；第 $t$ 项就是 $S(k+t,k)$。

```cpp
vector<Z> stirling2K(int n, int k)
{
    // n 是最大行号，k 是固定列号；返回 S(k,k) 到 S(n,k)，模数由 Z 决定。
    assert(0 <= k && k <= n); // 调试检查，可删。
    auto build = [&](auto &&self, int l, int r) -> Poly
    {
        // self 是递归自身，[l,r) 是因子下标；返回这一段 (1-ix) 的乘积。
        if (r - l == 1)
        {
            return Poly{1, -Z(l)};
        }
        int m = (l + r) / 2;
        return self(self, l, m) * self(self, m, r);
    };
    Poly den = k == 0 ? Poly{1} : build(build, 1, k + 1);
    Poly ans = fpsInv(den, n - k + 1);
    return vector<Z>(ans.begin(), ans.end());
}
```

本函数依赖 [10.2.001 多项式封装](#102001) 的乘法和 [10.2.003 乘法逆元](#102003) 的 `fpsInv`。构造乘积树为 $O(M(k)\log k)$，求逆为 $O(M(n-k))$，峰值空间 $O(n+k)$；`k=0` 返回 $S(0,0)=1$ 及其后的全零项。若只需固定 `n` 的整行，仍用 [8.2.005 Stirling 数](#82005) 或第二类的有限和公式。

下面直接检查四条可手算的固定列，不在正文再放二维递推作参考实现：

```cpp
signed main()
{
    assert(stirling1K(5, 1) == vector<Z>({1, -1, 2, -6, 24}));
    assert(stirling1K(6, 2) == vector<Z>({1, -3, 11, -50, 274}));
    assert(stirling2K(5, 1) == vector<Z>({1, 1, 1, 1, 1}));
    assert(stirling2K(6, 2) == vector<Z>({1, 3, 7, 15, 31}));
    return 0;
}
```

### 8.2.016 小素数大下标第一类 Stirling 数

当素数 $p$ 很小而 $n,k$ 可到 64 位时，二维递推不再可行。先用 Pascal 表保存 $0\le a,b<p$ 的组合数，`comb` 按 Lucas 定理逐位查询。对 $n=ip+j$，再把 $k-i=a(p-1)+b$ 调整为公式要求的余数范围，有
$s(n,k)=(-1)^{i+a}\binom{i}{a}s(j,b)\pmod p$；越界条件直接贡献 $0$。低位 $s(j,b)$ 只需预处理到 $p-1$。

```cpp
class SmallPComb
{
  protected:
    // p 是小素数，c[i][j] 是 C(i,j) mod p，其中 0<=j<=i<p。
    int p;
    vector<vector<int>> c;

    int comb(int n, int k) const
    {
        // n、k 是非负 64 位组合数参数；返回 C(n,k) mod p，k 越界时返回 0。
        if (k < 0 || k > n)
        {
            return 0;
        }
        int ans = 1;
        while (n)
        {
            int a = n % p;
            int b = k % p;
            if (b > a)
            {
                return 0;
            }
            ans = ans * c[a][b] % p;
            n /= p;
            k /= p;
        }
        return ans;
    }

  public:
    explicit SmallPComb(int p) : p(p)
    {
        // p 是可承担 O(p^2) 表的素数；建立一位 Lucas 所需的 Pascal 三角。
        assert(2 <= p && p <= numeric_limits<int>::max()); // 调试检查，可删。
        c.resize(p);
        c[0] = {1};
        for (int i = 1; i < p; i++)
        {
            c[i].assign(i + 1, 0);
            for (int j = 0; j <= i; j++)
            {
                int x = 0;
                if (j < i)
                {
                    x += c[i - 1][j];
                }
                if (j)
                {
                    x += c[i - 1][j - 1];
                }
                c[i][j] = x % p;
            }
        }
    }
};

class Stirling1P : public SmallPComb
{
    // s[i][j] 是有符号第一类 Stirling 数 s(i,j) mod p，其中 0<=j<=i<p。
    vector<vector<int>> s;

  public:
    explicit Stirling1P(int p) : SmallPComb(p), s(p)
    {
        // p 是小素数；建立第一类低位表与 Lucas 表。
        s[0] = {1};
        for (int i = 1; i < p; i++)
        {
            s[i].assign(i + 1, 0);
            for (int j = 0; j <= i; j++)
            {
                int x = 0;
                if (j)
                {
                    x += s[i - 1][j - 1];
                }
                if (j < i)
                {
                    x += s[i - 1][j] * (p - i + 1) % p;
                }
                s[i][j] = x % p;
            }
        }
    }

    int get(int n, int k) const
    {
        // n、k 是非负 64 位下标；返回有符号 s(n,k) mod p，k>n 时返回 0。
        assert(n >= 0); // 调试检查，可删。
        if (k < 0 || k > n)
        {
            return 0;
        }
        int i = n / p;
        int j = n % p;
        if (i > k)
        {
            return 0;
        }
        int a = (k - i) / (p - 1);
        int b = (k - i) % (p - 1);
        if (b == 0 && j > 0)
        {
            b += p - 1;
            a--;
        }
        if (a < 0 || i < a || b > j)
        {
            return 0;
        }
        int ans = comb(i, a) * s[j][b] % p;
        if (((i + a) & 1) && ans)
        {
            ans = p - ans;
        }
        return ans;
    }
};
```

构造 `Stirling1P q(p)` 后 `q.get(n,k)` 在 $O(\log_p n)$ 时间返回有符号 $s(n,k)\bmod p$；预处理为 $O(p^2)$ 时间与空间，表项显式使用 `int`，避免 `#define int long long` 把 $p=5000$ 时的内存翻倍。`p` 必须是不超过 `int` 上界的小素数，且内存要能容纳两张三角表；中等规模整表仍优先 [8.2.005 Stirling 数](#82005)。

### 8.2.017 小素数大下标第二类 Stirling 数

第二类需要不同的周期式，不能复用第一类的符号公式。对 $n<p$ 直接查低位表；否则写 $k=ip+j$，再把 $n-i=a(p-1)+b$ 调整到 $1\le b\le p-1$。当 $b<p-1$ 时答案为 $\binom{a}{i}S(b,j)$；当 $b=p-1$ 时，`j=0` 的答案是 $\binom{a}{i-1}$，否则为 $\binom{a}{i}S(p-1,j)$。

```cpp
class Stirling2P : public SmallPComb
{
    // s[i][j] 是第二类 Stirling 数 S(i,j) mod p，其中 0<=j<=i<p。
    vector<vector<int>> s;

  public:
    explicit Stirling2P(int p) : SmallPComb(p), s(p)
    {
        // p 是小素数；建立第二类低位表与 Lucas 表。
        s[0] = {1};
        for (int i = 1; i < p; i++)
        {
            s[i].assign(i + 1, 0);
            for (int j = 0; j <= i; j++)
            {
                int x = 0;
                if (j)
                {
                    x += s[i - 1][j - 1];
                }
                if (j < i)
                {
                    x += s[i - 1][j] * j % p;
                }
                s[i][j] = x % p;
            }
        }
    }

    int get(int n, int k) const
    {
        // n、k 是非负 64 位下标；返回 S(n,k) mod p，k>n 时返回 0。
        assert(n >= 0); // 调试检查，可删。
        if (k < 0 || k > n)
        {
            return 0;
        }
        if (n < p)
        {
            return s[n][k];
        }
        int i = k / p;
        int j = k % p;
        int a = (n - i) / (p - 1);
        int b = (n - i) % (p - 1);
        if (b == 0)
        {
            b = p - 1;
            a--;
        }
        if (a < 0 || j > b)
        {
            return 0;
        }
        if (b < p - 1)
        {
            return comb(a, i) * s[b][j] % p;
        }
        if (j == 0)
        {
            return comb(a, i - 1);
        }
        return comb(a, i) * s[p - 1][j] % p;
    }
};
```

`Stirling2P q(p)` 同样以 $O(p^2)$ 时间与空间预处理，`q.get(n,k)` 为 $O(\log_p n)$；它与 [8.2.016 小素数大下标第一类 Stirling 数](#82016) 只共享 `SmallPComb` 的 Lucas 表，低位递推和高位公式彼此独立。固定 $k$ 的整列输出用 [8.2.015 固定 k 的第二类 Stirling 数](#82015)，不要为单次大下标查询构造长多项式。

模 $5$ 时先用小值核对定义递推，再用 $s(n,n-1)=-\binom n2$ 与 $S(n,2)=2^{n-1}-1$ 检查大下标分支：

```cpp
signed main()
{
    int n = 1000000000000000003LL;
    Stirling1P a(5);
    Stirling2P b(5);
    assert(a.get(6, 2) == 4);
    assert(a.get(n, n - 1) == 2);
    assert(b.get(6, 3) == 0);
    assert(b.get(n, 2) == 3);
    cout << "OK\n";
    return 0;
}
```

## 8.3 生成与构造

### 8.3.001 排列组合生成

这四种枚举都短到不该封装，直接展开在使用处即可。排序后反复 `next_permutation` 无重地枚举多重集合的不同排列；`0..2^n-1` 的二进制掩码天然枚举全部子集；Gosper Hack（`x&-x` 取最低位、`s+lowbit` 进位后补齐低位）在相同 popcount 的掩码间按数值递增跳转，等价于组合的 colex 顺序，并非元素下标序列的字典序；Gray 编码 $g=x\oplus(x\gg1)$ 使相邻编号只差一位，解码则对不断右移的前缀连续异或还原。下面的 `main` 把四段写在一起，末尾 `assert(y == x)` 验证 Gray 编解码互逆。

```cpp
signed main()
{
    // 无参数；直接展示排列、子集、固定大小子集与 Gray Code 的展开写法，返回 0。
    vector<int> a{1, 1, 2};
    sort(a.begin(), a.end());
    do
    {
        // 此处直接处理当前字典序排列 a。
    } while (next_permutation(a.begin(), a.end()));

    int n = 5;
    for (int s = 0; s < (1LL << n); s++)
    {
        // s 的第 i 位表示是否选择编号 i 的元素。
    }

    int k = 3;
    if (k == 0)
    {
        int s = 0;
        // 空集是唯一的 0 元子集。
    }
    else
    {
        int s = (1LL << k) - 1;
        while (s < (1LL << n))
        {
            // 此处直接处理当前恰含 k 个元素的子集 s。
            int x = s & -s;
            int y = s + x;
            s = y | ((s ^ y) >> 2) / x;
        }
    }

    int x = 23;
    int g = x ^ (x >> 1);
    int y = g;
    for (int t = y >> 1; t; t >>= 1)
    {
        y ^= t;
    }
    assert(y == x);
}
```

直接把对应循环抄到 `solve`/`main` 里，不要包装成回调。示例要求 $0\le n<63$、$0\le k\le n$，Gray 编解码按非负整数处理。设不同排列数为 $P$，排列枚举 $O(Pn)$；全部子集 $O(2^n)$；固定大小子集 $O(\binom{n}{k})$；Gray 编码 $O(1)$、解码 $O(\log x)$，都只用 $O(1)$ 额外空间。只需要「第几个排列」而非全部枚举时用 [8.3.002 康托展开与逆展开](#83002) 直接跳到目标；子集枚举里要按权和分组则配合状压 DP。

### 8.3.002 康托展开与逆展开

康托展开把排列映到它的字典序排名（从 $0$ 起）：第 $i$ 位的权是 $(n-1-i)!$，系数是「$p_i$ 右边比它小的元素个数」，累加即排名。逆展开反过来，从高位到低位用 $k/(i-1)!$ 选出当前剩余元素里第几小的、再取模递降。`cantor` 用树状数组维护右侧更小计数把逐位统计降到 $O(n\log n)$；`invCantor` 每步从候选 `vector` 删除选中元素，故为 $O(n^2)$。两向都要求元素互异，且排名从 $0$ 还是 $1$ 起必须全程统一。

```cpp
unsigned long long cantor(const vector<int> &p)
{
    // p 是由 0 到 n-1 组成的排列；返回其从零开始的字典序排名。
    int n = p.size();
    assert(n <= 20); // 调试检查，可删。
    vector<int> seen(n);
    for (int x : p)
    {
        assert(0 <= x && x < n && !seen[x]); // 调试检查，可删。
        seen[x] = 1;
    }
    vector<unsigned long long> f(n + 1, 1);
    for (int i = 1; i <= n; i++)
    {
        f[i] = f[i - 1] * i;
    }
    vector<int> bit(n + 1);
    auto add = [&](int x)
    {
        // x 是要加入的排列值；把它在树状数组中的计数加一。
        for (++x; x <= n; x += x & -x)
        {
            bit[x]++;
        }
    };
    auto sum = [&](int x)
    {
        // x 是值域右端；返回已加入值中严格小于 x 的个数。
        int ans = 0;
        for (; x; x -= x & -x)
        {
            ans += bit[x];
        }
        return ans;
    };
    unsigned long long ans = 0;
    for (int i = n - 1; i >= 0; i--)
    {
        ans += f[n - 1 - i] * sum(p[i]);
        add(p[i]);
    }
    return ans;
}

vector<int> invCantor(int n, unsigned long long k)
{
    // n 是排列长度，k 是从零开始的排名；返回由 0 到 n-1 组成的对应排列。
    assert(0 <= n && n <= 20); // 调试检查，可删。
    vector<unsigned long long> f(n + 1, 1);
    for (int i = 1; i <= n; i++)
    {
        f[i] = f[i - 1] * i;
    }
    assert(k < f[n]); // 调试检查，可删。
    vector<int> a(n), ans;
    iota(a.begin(), a.end(), 0);
    for (int i = n; i; i--)
    {
        unsigned long long q = k / f[i - 1];
        k %= f[i - 1];
        assert(q < a.size()); // 调试检查，可删。
        ans.push_back(a[q]);
        a.erase(a.begin() + q);
    }
    return ans;
}
```

`cantor(p)` 接受由 $0,1,\ldots,n-1$ 组成的排列并返回其康托排名，`invCantor(n, k)` 由排名还原排列。`cantor` 为 $O(n\log n)$、`invCantor` 为 $O(n^2)$，都 $O(n)$ 空间；排名用 `unsigned long long` 存，故 $n\le20$（$21!$ 已溢出 $64$ 位）。更大的精确排名改用 [14.1.001 高精度整数（BigInt）](#141001) 保存阶乘与答案；若只求排名模某个数，则把 `f[i]` 和累加过程取模，但模排名不能再唯一逆展开。只需按字典序枚举而非跳转到指定名次时用 [8.3.001 排列组合生成](#83001) 的 `next_permutation`。

下面用三组已知排名检查重复换位、首尾边界与逆展开：

```cpp
signed main()
{
    assert(cantor({0, 1, 2, 3}) == 0);
    assert(cantor({1, 0, 3, 2}) == 7);
    assert(cantor({3, 2, 1, 0}) == 23);
    assert((invCantor(4, 7) == vector<int>{1, 0, 3, 2}));
    cout << "OK\n";
    return 0;
}
```

### 8.3.003 勾股数枚举

Euclid 参数化给出全部本原勾股数组：取 $m>n\ge1$、$\gcd(m,n)=1$ 且 $m,n$ 一奇一偶，则 $(m^2-n^2,\,2mn,\,m^2+n^2)$ 是本原三元组，且每个本原解恰对应一组这样的 $(m,n)$；再乘正整数就得到全部非本原解。`pythagorean` 按此枚举斜边 $c=m^2+n^2\le c_{\max}$ 的本原组，`(m-n)%2==0`（同奇偶）或 $\gcd\ne1$ 时跳过，并把两条直角边排成 $a\le b$。

```cpp
// PTriple 的三个分量依次是两条直角边 a、b 与斜边 c。
using PTriple = array<int, 3>;

vector<PTriple> pythagorean(long long cmax)
{
    // cmax 是非负斜边上界；返回全部满足 a<=b、c<=cmax 的本原三元组 {a,b,c}。
    assert(cmax >= 0); // 调试检查，可删。
    vector<PTriple> ans;
    for (long long m = 2; m <= (cmax - 1) / m; m++)
    {
        long long mm = m * m;
        for (long long n = 1; n < m; n++)
        {
            if ((m - n) % 2 == 0 || gcd(m, n) != 1)
            {
                continue;
            }
            long long nn = n * n;
            if (mm > cmax - nn)
            {
                break;
            }
            long long a = mm - nn;
            long long b = 2 * m * n;
            long long c = mm + nn;
            if (a > b)
            {
                swap(a, b);
            }
            ans.push_back({a, b, c});
        }
    }
    return ans;
}
```

`pythagorean(cmax)` 返回斜边不超过 `cmax` 的全部本原三元组，`PTriple[0..2]` 为两直角边与斜边。$m$ 到约 $\sqrt{c_{\max}}$、内层 $n<m$，实际有效 $(m,n)$ 对约 $O(c_{\max})$，空间为答案数。需要全部（含非本原）解时把每个本原组乘 $1,2,\dots$ 直到超限；只统计数量而不列举时用数论方法按 $c$ 的因子结构计数，不必逐个枚举。

### 8.3.004 阿达马矩阵

Sylvester 阿达马矩阵元素全为 $\pm1$、任意两行正交，用于正交编码和构造题。从 $H_1=[1]$ 出发按分块递推 $H_{2m}=\begin{bmatrix}H_m&H_m\\H_m&-H_m\end{bmatrix}$ 倍增：若 $H_mH_m^{T}=mI$，则展开可验证 $H_{2m}H_{2m}^{T}=2mI$，故正交性对全部 $2$ 的幂阶成立。代码每轮把 $m\times m$ 的块复制到左上、右上、左下，右下取相反数，直到边长达到 $n$。

```cpp
vector<vector<int>> hadamard(int n)
{
    // n 是正的 2 的幂；返回元素只取 1 和 -1、任意两条不同行正交的 Sylvester 阿达马矩阵。
    assert(n > 0 && has_single_bit((unsigned long long)n)); // 调试检查，可删。
    assert((unsigned long long)n <= numeric_limits<size_t>::max()); // 调试检查，可删。
    size_t side = (size_t)n;
    assert(side <= numeric_limits<size_t>::max() / sizeof(int) / side); // 调试检查，可删。
    vector<vector<int>> a{{1}};
    while ((int)a.size() < n)
    {
        int m = a.size();
        vector b(2 * m, vector<int>(2 * m));
        for (int i = 0; i < m; i++)
        {
            for (int j = 0; j < m; j++)
            {
                b[i][j] = a[i][j];
                b[i][j + m] = a[i][j];
                b[i + m][j] = a[i][j];
                b[i + m][j + m] = -a[i][j];
            }
        }
        a.swap(b);
    }
    return a;
}
```

`hadamard(n)` 要求 `n` 是正的 $2$ 的幂（`has_single_bit` 断言），返回 $n\times n$ 的 $\pm1$ 矩阵；$O(n^2)$ 时间与输出空间。$n=4$ 时每个元素为 $\pm1$，每行与自身内积为 $4$、与其他行内积为 $0$。题目要 $0/1$ 矩阵时把 $-1$ 映射成 $0$，但映射后普通整数内积不再为 $0$（需改用异或/汉明距离度量正交）。非 $2$ 的幂阶的阿达马矩阵（如 $12$、$20$ 阶）需要 Paley 等其他构造，本条不覆盖；填数构造题里的幻方见 [8.3.005 幻方](#83005)。

### 8.3.005 幻方

幻方要把 $1,2,\ldots,n^2$ 各恰用一次填进 $n\times n$ 方阵，使每行、每列、两条主对角线的和都等于幻和 $\frac{n(n^2+1)}{2}$。除 $n=2$ 无解外，按阶数奇偶分三套经典构造：奇数阶用 Siamese（罗伯法），从中上格填 $1$，每步向右上移动，越界则绕环、遇占用则改为下移一格；双偶阶（$4\mid n$）先按行填 $1,2,\ldots,n^2$，再对满足 $i\bmod4=j\bmod4$ 或 $(i\bmod4)+(j\bmod4)=3$ 的格取补数 $n^2+1-v$；单偶阶（$n\equiv2\pmod4$）用 Strachey/LUX 法，把方阵分成四个 $m=\frac n2$ 阶子块，各子块填同一个奇数阶幻方 $b$ 加偏移 $0,2m^2,3m^2,m^2$（左上、右上、左下、右下），再交换左上/左下块的前 $k=\frac{n-2}4$ 列（跳过中心行首列并单独换中心行的第 $k$ 列）与后 $k-1$ 列，使四块的行列和重新配平。

```cpp
vector<vector<int>> magicSquare(int n)
{
    // n 是阶数；返回恰好填入 1 到 n^2 的幻方，n=2 时无解并返回空矩阵。
    assert(n > 0); // 调试检查，可删。
    assert(n <= numeric_limits<int>::max() / n); // 调试检查，可删。
    assert((unsigned long long)n <= numeric_limits<size_t>::max()); // 调试检查，可删。
    size_t side = (size_t)n;
    assert(side <= numeric_limits<size_t>::max() / sizeof(int) / side); // 调试检查，可删。
    int total = n * n;
    if (n == 2)
    {
        return {};
    }
    if (n & 1)
    {
        vector a(n, vector<int>(n));
        int x = 0;
        int y = n / 2;
        for (int v = 1; v <= total; v++)
        {
            a[x][y] = v;
            int nx = (x + n - 1) % n;
            int ny = (y + 1) % n;
            if (a[nx][ny])
            {
                x = (x + 1) % n;
            }
            else
            {
                x = nx;
                y = ny;
            }
        }
        return a;
    }
    if (n % 4 == 0)
    {
        vector a(n, vector<int>(n));
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < n; j++)
            {
                int v = i * n + j + 1;
                bool rev = i % 4 == j % 4 || i % 4 + j % 4 == 3;
                a[i][j] = rev ? total + 1 - v : v;
            }
        }
        return a;
    }
    int m = n / 2;
    int k = (n - 2) / 4;
    int z = m * m;
    auto b = magicSquare(m);
    vector a(n, vector<int>(n));
    for (int i = 0; i < m; i++)
    {
        for (int j = 0; j < m; j++)
        {
            a[i][j] = b[i][j];
            a[i][j + m] = b[i][j] + 2 * z;
            a[i + m][j] = b[i][j] + 3 * z;
            a[i + m][j + m] = b[i][j] + z;
        }
    }
    for (int i = 0; i < m; i++)
    {
        for (int j = 0; j < k; j++)
        {
            if (i != m / 2 || j)
            {
                swap(a[i][j], a[i + m][j]);
            }
        }
        for (int j = n - k + 1; j < n; j++)
        {
            swap(a[i][j], a[i + m][j]);
        }
    }
    swap(a[m / 2][k], a[m / 2 + m][k]);
    return a;
}
```

`magicSquare(n)` 要求 `n>0` 且 $n^2$、所有块偏移和填充值均能由当前 64 位 `int` 表示；代码用除法断言检查乘法前边界，并另行检查矩阵元素总数可由 `size_t` 分配。$n=1$ 得 `{{1}}`，$n=2$ 返回空矩阵表示无解，其余阶数均有构造。时间与输出空间均为 $O(n^2)$；同为矩阵构造的 $\pm1$ 正交阵见 [8.3.004 阿达马矩阵](#83004)。

```cpp
signed main()
{
    assert(magicSquare(2).empty());
    auto a = magicSquare(3);
    assert(a.size() == 3 && a[0].size() == 3 && a[1][1] == 5);
}
```
