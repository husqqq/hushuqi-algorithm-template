```cpp
#include <bits/stdc++.h>

template <class T> bool chmin(T &a, const T &b)
{
    // a 是当前值，b 是候选值；若 b 更小则更新 a 并返回 true。
    if (b < a)
    {
        a = b;

    }

}

{
    // a 是当前值，b 是候选值；若 b 更大则更新 a 并返回 true。

    {

    }

inline long long linearMulMod(long long a, long long b, long long mod)
{
```

## 9.1 矩阵

### 9.1.001 普通矩阵与取模矩阵

矩阵 $A$ 记作行数 $n$、列数 $m$ 的二维数组，`Mat<T>` 把元素类型 $T$ 参数化，只要 $T$ 支持所在环的加法与乘法就能复用同一套加乘与快速幂。乘积 $C=AB$ 定义为 $C_{ij}=\sum_k A_{ik}B_{kj}$，要求左矩阵列数等于右矩阵行数，一般不交换；实现里三重循环特意按 $i,k,j$ 顺序累加，使最内层对 `x[i]`、`y[k]` 都连续访问，比教科书的 $i,j,k$ 更缓存友好。`eye(n)` 造的 $n$ 阶单位矩阵既是快速幂的初值，也是空乘积的单位元。元素类型若用原生整数，连乘会溢出；需要取模时把 `T` 换成模整数类型（每次乘加后自动约减），本条不内置取模、由 `T` 承担。

```cpp
template <class T> struct Mat
{
    // n、m 是行数与列数；a 始终应有 n 行，每行恰有 m 个元素。
    int n, m;
    vector<vector<T>> a;

    Mat(int n = 0, int m = 0) : n(n), m(m), a(n, vector<T>(m))
    {
        // 建立 n 行 m 列的零矩阵；不要直接改变 a 的行数或行宽。
    }

    static Mat eye(int n)
    {
        // n 是矩阵阶数；返回 n 阶单位矩阵。
        Mat a(n, n);
        for (int i = 0; i < n; i++)
        {
            a[i][i] = 1;
        }
        return a;
    }

    vector<T> &operator[](int x)
    {
        // x 是行下标；返回该行的可修改引用。
        return a[x];
    }
    const vector<T> &operator[](int x) const
    {
        // x 是行下标；返回该行的只读引用。
        return a[x];
    }

    friend Mat operator*(const Mat &x, const Mat &y)
    {
        // x、y 是乘法左右两侧的矩阵；返回维数相容时的普通乘积。
        // 调试检查，可删。
        assert((int)x.a.size() == x.n && (int)y.a.size() == y.n && x.m == y.n);
        // 调试检查，可删。
        for (const auto &r : x.a)
        {
            assert((int)r.size() == x.m);
        }
        // 调试检查，可删。
        for (const auto &r : y.a)
        {
            assert((int)r.size() == y.m);
        }
        Mat z(x.n, y.m);
        for (int i = 0; i < x.n; i++)
        {
            for (int k = 0; k < x.m; k++)
            {
                for (int j = 0; j < y.m; j++)
                {
                    z[i][j] += x[i][k] * y[k][j];
                }
            }
        }
        return z;
    }

    Mat pow(unsigned long long b) const
    {
        // b 是非负指数；返回当前方阵的 b 次幂。
        // 调试检查，可删。
        assert(n == m);
        Mat x = *this;
        Mat ans = eye(n);
        while (b)
        {
            if (b & 1)
            {
                ans = ans * x;
            }
            b >>= 1;
            if (b)
            {
                x = x * x;
            }
        }
        return ans;
    }
};
```

以 `Mat<T>(n,m)` 建零矩阵、`Mat<T>::eye(n)` 建单位阵，`operator*` 要求维数相容，`pow(b)` 要求方阵，$b=0$ 时返回单位阵。由于 `a` 是公开成员，若直接改坏其行数或行宽，后续运算不再合法；代码中的断言只用于调试。$n\times m$ 乘 $m\times k$ 是 $O(nmk)$ 时间、$O(nk)$ 结果空间；$n$ 阶方阵的 $b$ 次幂是 $O(n^3\log(b+1))$ 时间、$O(n^2)$ 空间。转移是普通加乘时本条即够；只需裸函数时用 [9.1.002 矩阵快速幂](#91002)；转移是 $\min$-加、$\max$-加等半环运算时改用 [9.1.004 `+min` / `+max` 半环矩阵](#91004)；非零元远少于 $n^2$ 时用 [9.1.008 稀疏矩阵](#91008)；要行列式或逆矩阵则转 [9.1.006 行列式（质数模数 / 任意模数）](#91006) 与 [9.1.005 矩阵求逆（质数模数 / 任意模数）](#91005)。

### 9.1.002 矩阵快速幂

常系数线性递推 $f_n=\sum_i c_i f_{n-i}$ 与固定状态转移都能写成向量左乘同一个转移方阵 $M$，第 $b$ 步的状态即 $M^b$ 乘初始向量，于是把递推的 $O(b)$ 步压成对 $M$ 求幂。快速幂把指数按二进制拆开：$M^b=\prod_{b\text{ 的置位 }i}M^{2^i}$，边平方边在对应位乘入答案。只有方阵能做普通幂，答案初值必须是同阶单位矩阵。本条给出不依赖 [9.1.001 普通矩阵与取模矩阵](#91001) 类封装的裸函数版，`T` 只需支持加法、乘法与从 $0$、$1$ 构造。

```cpp
template <class T> vector<vector<T>> matPow(vector<vector<T>> a, unsigned long long b)
{
    // a 是方阵，b 是非负指数；返回 a 的 b 次幂，T 必须支持加法、乘法和从 0、1 构造。
    int n = a.size();
    // 调试检查，可删。
    for (const auto &r : a)
    {
        assert((int)r.size() == n);
    }
    vector<vector<T>> ans(n, vector<T>(n));
    for (int i = 0; i < n; i++)
    {
        ans[i][i] = 1;
    }
    auto mul = [&](const vector<vector<T>> &x, const vector<vector<T>> &y)
    {
        // x、y 是同阶方阵；返回普通矩阵乘积。
        vector<vector<T>> z(n, vector<T>(n));
        for (int i = 0; i < n; i++)
        {
            for (int k = 0; k < n; k++)
            {
                for (int j = 0; j < n; j++)
                {
                    z[i][j] += x[i][k] * y[k][j];
                }
            }
        }
        return z;
    };
    while (b)
    {
        if (b & 1)
        {
            ans = mul(ans, a);
        }
        b >>= 1;
        if (b)
        {
            a = mul(a, a);
        }
    }
    return ans;
}
```

调用 `matPow(a,b)`，`a` 必须是方阵，$b=0$ 时返回单位矩阵；斐波那契转移 $\begin{pmatrix}1&1\\1&0\end{pmatrix}$ 的 $10$ 次幂应为 $\begin{pmatrix}89&55\\55&34\end{pmatrix}$。$O(n^3\log(b+1))$ 时间、$O(n^2)$ 额外空间。转移是 $\min$-加或 $\max$-加运算时改用 [9.1.004 `+min` / `+max` 半环矩阵](#91004)；需要携带行列信息或下标运算符时用 [9.1.001 普通矩阵与取模矩阵](#91001) 的 `Mat<T>::pow`。

下面只调用本条的裸函数，检查手算三次幂、零次幂、单位阵与 Fibonacci 转移：

```cpp
signed main()
{
    vector<vector<long long>> a{{1, 2}, {3, 4}};
    vector<vector<long long>> cube{{37, 54}, {81, 118}};
    assert(matPow(a, 3) == cube);
    assert(matPow(a, 0) == vector<vector<long long>>({{1, 0}, {0, 1}}));
    assert(matPow(vector<vector<long long>>{{1, 0}, {0, 1}}, 7) == vector<vector<long long>>({{1, 0}, {0, 1}}));
    auto f = matPow(vector<vector<long long>>{{1, 1}, {1, 0}}, 10);
    cout << f[0][0] << ' ' << f[0][1] << ' ' << f[1][0] << ' ' << f[1][1] << '\n'; // 89 55 55 34
    return 0;
}
```

### 9.1.003 Frobenius 标准形矩阵快速幂（高级版）

基础二进制矩阵幂每个指数位都做一次 $n^3$ 乘法。若指数很大且矩阵阶数也高，可先随机选 Krylov 向量，把空间拆成循环子空间，使 $P^{-1}AP$ 成为若干伴随块；每个 $d$ 阶块上的幂只需算 $x^k$ 对块多项式取模，最后再乘 $P$ 与 $P^{-1}$ 还原。构造基底时会验证新块多项式对旧块的整除关系；验证失败就丢弃本轮随机向量并重建，因此这是 Las Vegas 算法：运行时间随机，返回结果不带错误概率。

```cpp
template <class T> struct NaivePolyMul
{
    vector<T> operator()(const vector<T> &a, const vector<T> &b) const
    {
        // a、b 是系数从低次到高次排列的多项式；返回普通卷积。
        if (a.empty() || b.empty())
        {
            return {};
        }
        vector<T> c(a.size() + b.size() - 1);
        for (int i = 0; i < (int)a.size(); i++)
        {
            for (int j = 0; j < (int)b.size(); j++)
            {
                c[i + j] += a[i] * b[j];
            }
        }
        return c;
    }
};

template <class T> struct FieldBasis
{
    // n 是向量维数，rk 是当前秩。
    int n, rk = 0;
    // basis 保存插入的原向量；row 是按首个非零位置存放的阶梯基，way 给出 row 关于 basis 的系数。
    vector<vector<T>> basis, row, way;

    FieldBasis(int n) : n(n), row(n, vector<T>(n)), way(n, vector<T>(n))
    {
        // 建立 n 维空线性基。
    }

    pair<bool, vector<T>> solveOrAdd(vector<T> x)
    {
        // x 是待查询向量；可表示时返回 true 及其关于 basis 的系数，否则插入 x 并返回 false。
        vector<T> original = x;
        vector<T> coef(n);
        for (int i = 0; i < n; i++)
        {
            if (row[i][i] == T(1))
            {
                coef[i] = x[i];
                for (int j = i; j < n; j++)
                {
                    x[j] -= coef[i] * row[i][j];
                }
            }
        }
        int p = 0;
        while (p < n && x[p] == T(0))
        {
            p++;
        }
        if (p == n)
        {
            vector<T> ans(rk);
            for (int i = 0; i < n; i++)
            {
                for (int j = 0; j < rk; j++)
                {
                    ans[j] += coef[i] * way[i][j];
                }
            }
            return {true, ans};
        }
        T z = T(1) / x[p];
        for (int j = p; j < n; j++)
        {
            x[j] *= z;
        }
        for (auto &v : coef)
        {
            v *= z;
        }
        basis.push_back(move(original));
        row[p] = move(x);
        way[p][rk] = z;
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < rk; j++)
            {
                way[p][j] -= coef[i] * way[i][j];
            }
        }
        rk++;
        return {false, {}};
    }

    void rollback(int r)
    {
        // r 是此前的秩；撤销最近插入的向量，直到秩恢复为 r。
        while (rk > r)
        {
            rk--;
            basis.pop_back();
            for (int i = 0; i < n; i++)
            {
                if (way[i][rk] != T(0))
                {
                    fill(row[i].begin(), row[i].end(), T(0));
                    fill(way[i].begin(), way[i].end(), T(0));
                }
            }
        }
    }
};

template <class T, class Conv = NaivePolyMul<T>> struct FrobPower
{
    using Poly = vector<T>;
    using Matrix = vector<vector<T>>;

    // n 是矩阵阶数；a 是原矩阵，p 的列是 Krylov 基，ip 是 p 的逆矩阵。
    int n;
    Matrix a, p, ip;
    // start 保存各循环子空间的起始向量；relation 保存对应伴随块的递推系数。
    vector<Poly> start, relation;
    // conv 计算多项式卷积，rng 产生试探向量，trials 记录完整重建次数。
    Conv conv;
    mt19937_64 rng;
    int trials = 0;

    FrobPower(Matrix a, Conv conv = Conv(), unsigned long long seed = 0)
        : n(a.size()), a(move(a)), conv(move(conv)), rng(seed)
    {
        // a 是域上的方阵，conv 是卷积器，seed 是固定随机种子；构造时完成 Frobenius 分解。
        // 调试检查，可删。
        for (const auto &r : this->a)
        {
            assert((int)r.size() == n);
        }
        do
        {
            trials++;
        } while (!trial());
    }

    Matrix pow(unsigned long long b)
    {
        // b 是非负指数；返回原矩阵的 b 次幂。
        Matrix x(n, vector<T>(n));
        int s = 0;
        for (const auto &g : relation)
        {
            int d = g.size();
            Poly f = polyPowMod(g, b);
            for (int j = 0; j < d; j++)
            {
                for (int i = 0; i < (int)f.size(); i++)
                {
                    x[s + i][s + j] = f[i];
                }
                if (j + 1 < d)
                {
                    f.insert(f.begin(), T(0));
                    divmod(f, g);
                }
            }
            s += d;
        }
        return matrixMul(matrixMul(p, x), ip);
    }

private:
    static void trim(Poly &f)
    {
        // f 的系数按低次到高次排列；原地删除高次零项。
        while (!f.empty() && f.back() == T(0))
        {
            f.pop_back();
        }
    }

    Poly divmod(Poly &f, const Poly &g)
    {
        // f 是被除式，g 表示 x^d-sum(g[i]x^i)；原地留下余数并返回商。
        trim(f);
        int d = g.size();
        Poly q;
        for (int i = (int)f.size() - 1; i >= d; i--)
        {
            q.push_back(f[i]);
            for (int j = 0; j < d; j++)
            {
                f[i - d + j] += f[i] * g[j];
            }
            f[i] = T(0);
        }
        trim(f);
        reverse(q.begin(), q.end());
        return q;
    }

    Poly polyMulMod(const Poly &x, const Poly &y, const Poly &g)
    {
        // x、y 是被乘多项式，g 表示模多项式；返回乘积的余式。
        Poly z = conv(x, y);
        divmod(z, g);
        return z;
    }

    Poly polyPowMod(const Poly &g, unsigned long long b)
    {
        // g 表示模多项式，b 是非负指数；返回 x^b 的余式。
        Poly ans{T(1)};
        Poly x{T(0), T(1)};
        divmod(x, g);
        while (b)
        {
            if (b & 1)
            {
                ans = polyMulMod(ans, x, g);
            }
            b >>= 1;
            if (b)
            {
                x = polyMulMod(x, x, g);
            }
        }
        return ans;
    }

    Matrix matrixMul(const Matrix &x, const Matrix &y)
    {
        // x、y 是 n 阶方阵；返回普通矩阵乘积。
        Matrix z(n, vector<T>(n));
        for (int i = 0; i < n; i++)
        {
            for (int k = 0; k < n; k++)
            {
                for (int j = 0; j < n; j++)
                {
                    z[i][j] += x[i][k] * y[k][j];
                }
            }
        }
        return z;
    }

    optional<Matrix> matrixInv(Matrix x)
    {
        // x 是 n 阶方阵；可逆时返回逆矩阵，否则返回空。
        Matrix y(n, vector<T>(n));
        for (int i = 0; i < n; i++)
        {
            y[i][i] = T(1);
        }
        for (int col = 0; col < n; col++)
        {
            int row = col;
            while (row < n && x[row][col] == T(0))
            {
                row++;
            }
            if (row == n)
            {
                return nullopt;
            }
            swap(x[row], x[col]);
            swap(y[row], y[col]);
            T z = T(1) / x[col][col];
            for (int j = 0; j < n; j++)
            {
                x[col][j] *= z;
                y[col][j] *= z;
            }
            for (int i = 0; i < n; i++)
            {
                if (i == col)
                {
                    continue;
                }
                T q = x[i][col];
                for (int j = 0; j < n; j++)
                {
                    x[i][j] -= q * x[col][j];
                    y[i][j] -= q * y[col][j];
                }
            }
        }
        return y;
    }

    vector<T> apply(const vector<T> &v)
    {
        // v 是 n 维列向量；返回 a*v。
        vector<T> w(n);
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < n; j++)
            {
                w[i] += a[i][j] * v[j];
            }
        }
        return w;
    }

    vector<T> randomVector()
    {
        // 返回由 rng 生成的 n 维试探向量。
        vector<T> v(n);
        for (auto &x : v)
        {
            x = T((long long)(rng() & 0x7FFFFFFFFFFFFFFFULL));
        }
        return v;
    }

    bool trial()
    {
        // 从空基底尝试构造一次 Frobenius 分解；成功时写入 p、ip 与 relation。
        start.clear();
        relation.clear();
        FieldBasis<T> s(n);
        while (s.rk < n)
        {
            int oldRank = s.rk;
            vector<T> v = randomVector();
            start.push_back(v);
            while (true)
            {
                auto [solved, coef] = s.solveOrAdd(v);
                if (!solved)
                {
                    v = apply(v);
                    continue;
                }
                Poly f(coef.begin() + oldRank, coef.end());
                if (f.empty())
                {
                    return false;
                }
                relation.push_back(f);
                if (start.size() == 1)
                {
                    break;
                }
                v = start.back();
                int pos = 0;
                for (int k = 0; k + 1 < (int)start.size(); k++)
                {
                    int d = relation[k].size();
                    Poly rem(coef.begin() + pos, coef.begin() + pos + d);
                    Poly q = divmod(rem, f);
                    if (!rem.empty())
                    {
                        return false;
                    }
                    for (int i = 0; i < (int)q.size(); i++)
                    {
                        for (int j = 0; j < n; j++)
                        {
                            v[j] -= q[i] * s.basis[pos + i][j];
                        }
                    }
                    pos += d;
                }
                s.rollback(oldRank);
                start.back() = v;
                for (int i = 0; i < (int)f.size(); i++)
                {
                    if (s.solveOrAdd(v).first)
                    {
                        return false;
                    }
                    if (i + 1 < (int)f.size())
                    {
                        v = apply(v);
                    }
                }
                break;
            }
        }
        p.assign(n, vector<T>(n));
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < n; j++)
            {
                p[i][j] = s.basis[j][i];
            }
        }
        auto inv = matrixInv(p);
        if (!inv)
        {
            return false;
        }
        ip = move(*inv);
        return true;
    }
};
```

`FrobPower<T> fp(a,{},seed)` 构造后用 `fp.pow(k)`。`T` 必须是域；每轮分解与求逆为 $O(n^3)$，失败后会从空基底重试，`trials` 可查看实际轮数。一次 `pow` 的块内求幂为 $O(\sum_i d_i^2\log(k+1))$，乘 $P$、$P^{-1}$ 还原为 $O(n^3)$，峰值空间为 $O(n^2)$。第二模板参数只替换卷积器；当前 `divmod` 仍是二次取模，因此仅接入快速卷积只能减小卷积部分的常数。若 $M(d)$ 表示长度 $d$ 的卷积复杂度，要达到 $O(\sum_i M(d_i)\log(k+1))$ 还必须同时换成快速多项式取模。有限域大小为 $p$ 时单轮失败概率为 $O(n/p)$。它依赖域除法、卷积和随机 Krylov 分解，只适合大矩阵或超大指数；一般情况优先使用 [9.1.002 矩阵快速幂](#91002)。

固定查验时可在模 $998244353$ 的域上取 `seed=0` 与 $A=\begin{pmatrix}1&1&0\\0&1&1\\0&0&1\end{pmatrix}$；`fp.pow(5)` 应为 $\begin{pmatrix}1&5&10\\0&1&5\\0&0&1\end{pmatrix}$。再分别对零矩阵、单位阵和两个 Krylov 块组成的分块矩阵核对指数 $0,1,2,17$ 的预先算定结果，可覆盖空乘积、不可逆矩阵和多块分解；完整随机检验放在本地测试文件。

### 9.1.004 `+min` / `+max` 半环矩阵

普通矩阵乘法 $C_{ij}=\sum_k A_{ik}B_{kj}$ 只要求标量加法与乘法构成半环——满足结合律、乘法对加法分配、有各自单位元——就能整套照搬。把 $(+,\times)$ 换成 $(\min,+)$ 得 min-plus（热带）半环，此时 $M^b$ 的 $(i,j)$ 项恰是 $i\to j$ 恰好 $b$ 条边的最短路，正是 Floyd 分治与定长最短路的代数形式；换成 $(\max,+)$ 则给定长最长路。两者的加法单位元分别是 $+\infty$、$-\infty$ 哨兵，乘法单位元都是 $0$。`semiringMul` 与 `semiringPow` 把 `add`、`mul` 与单位元 `zero`、`one` 抽成回调，因此同一份幂运算能服务任意半环，不必为每种运算重写。

```cpp
template <class T, class Add, class Mul>
vector<vector<T>> semiringMul(const vector<vector<T>> &a, const vector<vector<T>> &b, T zero, Add add, Mul mul)
{
    // a、b 是待乘矩阵，zero 是加法单位元；add、mul 定义半环加法与乘法。
    int n = a.size();
    int m = b.size();
    int k = m ? b[0].size() : 0;
    // 调试检查，可删。
    for (const auto &r : a)
    {
        assert((int)r.size() == m);
    }
    // 调试检查，可删。
    for (const auto &r : b)
    {
        assert((int)r.size() == k);
    }
    vector c(n, vector<T>(k, zero));
    for (int i = 0; i < n; i++)
    {
        for (int z = 0; z < m; z++)
        {
            for (int j = 0; j < k; j++)
            {
                c[i][j] = add(c[i][j], mul(a[i][z], b[z][j]));
            }
        }
    }
    return c;
}

template <class T, class Add, class Mul>
vector<vector<T>> semiringPow(vector<vector<T>> a, unsigned long long b, T zero, T one, Add add, Mul mul)
{
    // a 是方阵，b 是非负指数；zero、one 是加法与乘法单位元。
    int n = a.size();
    // 调试检查，可删。
    for (const auto &r : a)
    {
        assert((int)r.size() == n);
    }
    vector ans(n, vector<T>(n, zero));
    for (int i = 0; i < n; i++)
    {
        ans[i][i] = one;
    }
    while (b)
    {
        if (b & 1)
        {
            ans = semiringMul(ans, a, zero, add, mul);
        }
        b >>= 1;
        if (b)
        {
            a = semiringMul(a, a, zero, add, mul);
        }
    }
    return ans;
}
```

`semiringMul(a,b,zero,add,mul)` 做一次半环乘法，`semiringPow(a,b,zero,one,add,mul)` 对方阵求 $b$ 次幂。min-plus 不能把普通 `+` 直接作为 `mul`：不可达哨兵与负边相加后会变成有限值。安全回调可写成：

```cpp
auto minAdd = [](long long x, long long y)
{
    return min(x, y);
};
auto minMul = [](long long x, long long y)
{
    return x == INF || y == INF ? INF : x + y;
};
auto d = semiringPow(g, k, INF, 0LL, minAdd, minMul);
```

这里还要求所有有限距离及其加法都在 `long long` 范围内且不等于 `INF`。max-plus 同理使用 `-INF`，只要任一因子为 `-INF` 就返回 `-INF`。$n\times m$ 与 $m\times k$ 的半环乘法是 $O(nmk)$，$n$ 阶方阵快速幂是 $O(n^3\log(b+1))$，空间 $O(n^2)$。转移是普通加乘时回到 [9.1.001 普通矩阵与取模矩阵](#91001)、[9.1.002 矩阵快速幂](#91002)；若求任意步数最短路而非恰好 $k$ 条边，用图论章的 Floyd 或 Bellman-Ford 更直接。

### 9.1.005 矩阵求逆（质数模数 / 任意模数）

求方阵 $A$ 的模逆矩阵，即找 $B$ 使 $AB\equiv I$。素数模 $p$ 下每个非零元都可逆，直接 Gauss-Jordan 消元即可；难点在合数模：此时非零元未必可逆，不能随便拿主元求逆。关键事实是 $A$ 模 $p^k$ 可逆当且仅当 $\det(A)$ 不被 $p$ 整除，等价于消元时每一列都能找到模 $p$ 非零的主元（模 $p^k$ 的单位）。于是把模数分解成互质的质数幂 $\prod p_i^{k_i}$，`matInvPP` 在每个 $p_i^{k_i}$ 上单独消元（主元用扩欧求 `invInt` 得到模 $p^k$ 逆元，普通模数乘法调用公共的 `linearMulMod`，接近 `long long` 上界时才局部使用 `__int128`），再对每个位置用 CRT 把各分量唯一合并回模 $mod$。任一分量不可逆则整体不可逆，返回空。

```cpp
int invInt(int a, int mod)
{
    // a 与 mod 互质且 mod>1；返回 a 模 mod 的最小非负逆元。
    // 调试检查，可删。
    assert(mod > 1);
    a %= mod;
    if (a < 0)
    {
        a += mod;
    }
    int b = mod;
    __int128 x = 1;
    __int128 y = 0;
    while (b)
    {
        int q = a / b;
        a -= q * b;
        swap(a, b);
        x -= q * y;
        swap(x, y);
    }
    // 调试检查，可删。
    assert(a == 1);
    x %= mod;
    if (x < 0)
    {
        x += mod;
    }
    return (int)x;
}

optional<vector<vector<int>>> matInvPP(vector<vector<int>> a, int p, int pk)
{
    // a 是方阵，p 是质数，pk 是 p 的正次幂；返回模 pk 的逆矩阵，不可逆时返回空。
    int n = a.size();
    // 调试检查，可删。
    assert(p >= 2 && pk >= p);
    for (auto &row : a)
    {
        // 调试检查，可删。
        assert((int)row.size() == n);
        for (auto &x : row)
        {
            x %= pk;
            if (x < 0)
            {
                x += pk;
            }
        }
    }
    vector b(n, vector<int>(n));
    for (int i = 0; i < n; i++)
    {
        b[i][i] = 1;
    }
    for (int c = 0; c < n; c++)
    {
        int r = c;
        while (r < n && a[r][c] % p == 0)
        {
            r++;
        }
        if (r == n)
        {
            return nullopt;
        }
        swap(a[r], a[c]);
        swap(b[r], b[c]);
        int z = invInt(a[c][c], pk);
        for (int j = 0; j < n; j++)
        {
            a[c][j] = linearMulMod(a[c][j], z, pk);
            b[c][j] = linearMulMod(b[c][j], z, pk);
        }
        for (int i = 0; i < n; i++)
        {
            if (i == c)
            {
                continue;
            }
            int q = a[i][c];
            for (int j = 0; j < n; j++)
            {
                a[i][j] = (a[i][j] - linearMulMod(q, a[c][j], pk)) % pk;
                b[i][j] = (b[i][j] - linearMulMod(q, b[c][j], pk)) % pk;
                if (a[i][j] < 0)
                {
                    a[i][j] += pk;
                }
                if (b[i][j] < 0)
                {
                    b[i][j] += pk;
                }
            }
        }
    }
    return b;
}

optional<vector<vector<int>>> matInvAny(const vector<vector<int>> &a, int mod)
{
    // a 是方阵，mod>1 可为素数或合数；返回模 mod 的逆矩阵，不存在时返回空。
    // 调试检查，可删。
    assert(mod > 1);
    vector<pair<int, int>> fac;
    int x = mod;
    for (int p = 2; p <= x / p; p++)
    {
        if (x % p)
        {
            continue;
        }
        int pk = 1;
        while (x % p == 0)
        {
            x /= p;
            pk *= p;
        }
        fac.push_back({p, pk});
    }
    if (x > 1)
    {
        fac.push_back({x, x});
    }
    int n = a.size();
    int cur = 1;
    vector ans(n, vector<int>(n));
    for (auto [p, pk] : fac)
    {
        auto b = matInvPP(a, p, pk);
        if (!b)
        {
            return nullopt;
        }
        int z = invInt(cur % pk, pk);
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < n; j++)
            {
                int t = ((*b)[i][j] - ans[i][j]) % pk;
                if (t < 0)
                {
                    t += pk;
                }
                t = linearMulMod(t, z, pk);
                // 已处理模数 cur 与当前 pk 的乘积不超过总模数，故 cur*t 不溢出。
                ans[i][j] += cur * t;
            }
        }
        cur *= pk;
    }
    return ans;
}
```

调用 `matInvAny(a,mod)`，`a` 是方阵、`mod>1`，返回空表示不可逆；素数模也走同一接口，此时只有一个质数幂分量。矩阵 $\begin{pmatrix}2&3\\3&2\end{pmatrix}$ 模 $6$ 可逆且逆恰为自身，$\begin{pmatrix}2&0\\0&1\end{pmatrix}$ 模 $6$ 不可逆。若记模数为 $M$，试除分解最坏为 $O(\sqrt M)$，随后是 $O(\omega(M)n^3)$ 时间、$O(n^2)$ 空间，其中 $\omega(M)$ 是 $M$ 的不同质因子数。只要行列式而不需要整个逆矩阵时用 [9.1.006 行列式（质数模数 / 任意模数）](#91006)；解线性方程组时用 [9.2.001 高斯消元：唯一解、无解、多解](#92001)，避免多算 $n$ 列。

下面直接检查两组已知逆矩阵和一组不可逆矩阵；不可逆样例的行列式为 $2$，与模数 $6$ 不互质：

```cpp
signed main()
{
    auto a = matInvAny({{1, 2}, {3, 4}}, 5);
    assert(a && *a == vector<vector<int>>({{3, 1}, {4, 2}}));
    auto b = matInvAny({{2, 3}, {3, 2}}, 6);
    assert(b && *b == vector<vector<int>>({{2, 3}, {3, 2}}));
    assert(!matInvAny({{2, 0}, {0, 1}}, 6));
    cout << (*b)[0][0] << ' ' << (*b)[0][1] << ' ' << (*b)[1][0] << ' ' << (*b)[1][1] << '\n'; // 2 3 3 2
    return 0;
}
```

### 9.1.006 行列式（质数模数 / 任意模数）

行列式在行初等变换下的行为完全确定：交换两行变号，某行乘常数 $c$ 则整体乘 $c$，某行加另一行的倍数不变。把矩阵消成上三角后行列式就是对角线乘积（乘上累计的交换符号）。素数模下 `detPrime` 直接取主元逆元（费马小定理求 $a^{mod-2}$）消去下方各行。合数模不能求逆，`detAny` 改用辗转相除式消元：对同列的两行反复取商作整倍数相减再交换，就像对这两个元素跑欧几里得算法，把下方元素辗转消成 $0$，全程只用整数乘减、绝不请求模逆——代价是每列多花 $O(\log mod)$ 轮相减。任一列主元最终为 $0$ 则行列式为 $0$。

```cpp
int detPrime(vector<vector<int>> a, int mod)
{
    // a 是方阵，mod 是素数；返回行列式对 mod 的最小非负余数。
    // 调试检查，可删。
    assert(mod > 1);
    int n = a.size();
    int ans = 1;
    for (auto &row : a)
    {
        // 调试检查，可删。
        assert((int)row.size() == n);
        for (auto &x : row)
        {
            x %= mod;
            if (x < 0)
            {
                x += mod;
            }
        }
    }
    for (int c = 0; c < n; c++)
    {
        int r = c;
        while (r < n && a[r][c] == 0)
        {
            r++;
        }
        if (r == n)
        {
            return 0;
        }
        if (r != c)
        {
            swap(a[r], a[c]);
            ans = (mod - ans) % mod;
        }
        ans = linearMulMod(ans, a[c][c], mod);
        int z = 1;
        int b = mod - 2;
        int x = a[c][c];
        while (b)
        {
            if (b & 1)
            {
                z = linearMulMod(z, x, mod);
            }
            b >>= 1;
            if (b)
            {
                x = linearMulMod(x, x, mod);
            }
        }
        for (int i = c + 1; i < n; i++)
        {
            int q = linearMulMod(a[i][c], z, mod);
            for (int j = c; j < n; j++)
            {
                a[i][j] = (a[i][j] - linearMulMod(q, a[c][j], mod)) % mod;
                if (a[i][j] < 0)
                {
                    a[i][j] += mod;
                }
            }
        }
    }
    return ans;
}

int detAny(vector<vector<int>> a, int mod)
{
    // a 是方阵，mod>1 可为合数；返回行列式对 mod 的最小非负余数，全程不做模逆。
    // 调试检查，可删。
    assert(mod > 1);
    int n = a.size();
    int ans = 1;
    for (auto &row : a)
    {
        // 调试检查，可删。
        assert((int)row.size() == n);
        for (auto &x : row)
        {
            x %= mod;
            if (x < 0)
            {
                x += mod;
            }
        }
    }
    for (int c = 0; c < n; c++)
    {
        for (int i = c + 1; i < n; i++)
        {
            while (a[i][c])
            {
                int q = a[c][c] / a[i][c];
                for (int j = c; j < n; j++)
                {
                    a[c][j] = (a[c][j] - linearMulMod(q, a[i][j], mod)) % mod;
                    if (a[c][j] < 0)
                    {
                        a[c][j] += mod;
                    }
                }
                swap(a[c], a[i]);
                ans = (mod - ans) % mod;
            }
        }
        if (a[c][c] == 0)
        {
            return 0;
        }
        ans = linearMulMod(ans, a[c][c], mod);
    }
    return ans;
}
```

模数是编译期已知且小于 $2^{30}$ 的素数时，热点可改用 `detPrimeFast<P>`：矩阵改成连续 64 位数组，每行累计 8 次消元后才统一取模，避免内层每个元素一次 `%`。核心消元如下，完整初始化、换行与收尾见对应头文件：

```cpp
template <int P> int detPrimeFast(const vector<vector<long long>> &src)
{
    static_assert(1 < P && P < (1U << 30));
    int n = src.size();
    vector<unsigned long long> a((size_t)n * n);
    vector<int> cnt(n);
    for (int i = 0; i < n; i++)
    {
        assert(src[i].size() == n); // 调试检查，可删
        for (int j = 0; j < n; j++)
        {
            long long x = src[i][j] % (long long)P;
            if (x < 0)
            {
                x += P;
            }
            a[(size_t)i * n + j] = x;
        }
    }
    auto power = [](unsigned long long x, int b)
    {
        unsigned long long ans = 1;
        for (; b; b >>= 1, x = x * x % P)
        {
            if (b & 1)
            {
                ans = ans * x % P;
            }
        }
        return (int)ans;
    };
    bool neg = false;
    for (int i = 0; i < n; i++)
    {
        for (int j = i; j < n; j++)
        {
            a[(size_t)j * n + i] %= P;
        }
        int p = i;
        while (p < n && !a[(size_t)p * n + i])
        {
            p++;
        }
        if (p == n)
        {
            return 0;
        }
        if (p != i)
        {
            for (int j = i; j < n; j++)
            {
                swap(a[(size_t)i * n + j], a[(size_t)p * n + j]);
            }
            swap(cnt[i], cnt[p]);
            neg = !neg;
        }
        auto row = a.data() + (size_t)i * n;
        for (int j = i; j < n; j++)
        {
            row[j] %= P;
        }
        int iv = power(row[i], P - 2);
        for (int k = i + 1; k < n; k++)
        {
            auto cur = a.data() + (size_t)k * n;
            int q = cur[i] ? P - (unsigned long long)iv * cur[i] % P : 0;
            int j = i;
            for (; j + 4 <= n; j += 4)
            {
                cur[j] += (unsigned long long)q * row[j];
                cur[j + 1] += (unsigned long long)q * row[j + 1];
                cur[j + 2] += (unsigned long long)q * row[j + 2];
                cur[j + 3] += (unsigned long long)q * row[j + 3];
            }
            for (; j < n; j++)
            {
                cur[j] += (unsigned long long)q * row[j];
            }
            if (++cnt[k] == 8)
            {
                cnt[k] = 0;
                for (int j = i; j < n; j++)
                {
                    cur[j] %= P;
                }
            }
        }
    }
    unsigned long long ans = neg ? P - 1 : 1;
    for (int i = 0; i < n; i++)
    {
        ans = ans * (a[(size_t)i * n + i] % P) % P;
    }
    return ans;
}
```

已知 `mod` 为素数时调用 `detPrime(a,mod)`，可能为合数时调用 `detAny(a,mod)`；两者都接受负元素并返回最小非负余数。固定小素数且矩阵较大时调用 `detPrimeFast<P>(a)`；它仍是便携 C++，未绑定 AVX2，但要求 $P<2^{30}$ 以证明 8 轮延迟取模不溢出 64 位。矩阵 $\begin{pmatrix}2&3\\3&2\end{pmatrix}$ 的整数行列式为 $-5$，故 `detPrime(a,7)` 返回 $2$、`detAny(a,6)` 返回 $1$。三版渐进复杂度不变；本机 $400\times400$ 随机矩阵的 5 轮中位数约为 `8.2ms`，逐元素取模版约为 `25.8ms`。运行时模数或大模数不能用固定版。若记模数为 $M$，素数模版本为 $O(n^3+n\log M)$，任意模版本为 $O(n^3\log M)$ 次基本算术，空间均为 $O(n^2)$。需要在模意义下解方程或求逆而非只要行列式时用 [9.2.001 高斯消元：唯一解、无解、多解](#92001)、[9.1.005 矩阵求逆（质数模数 / 任意模数）](#91005)；要 $\det(xI-A)$ 的整条特征多项式时用 [9.1.007 矩阵特征多项式](#91007)。

下面用两个手算行列式的固定矩阵检查素数模、合数模、负元素和奇异矩阵；只保留一个 $3\times3$ 例，不做大规模随机对拍：

```cpp
signed main()
{
    vector<vector<int>> a{{2, 3}, {3, 2}};
    assert(detPrime(a, 7) == 2);
    assert(detAny(a, 6) == 1);
    vector<vector<int>> b{{1, 2, 3}, {0, 1, 4}, {5, 6, 0}};
    assert(detPrime(b, 7) == 1 && detAny(b, 6) == 1);
    assert(detAny({{1, 2}, {2, 4}}, 6) == 0);
    cout << detAny(a, 6) << ' ' << detPrime(b, 7) << '\n'; // 1 1
    return 0;
}
```

### 9.1.007 矩阵特征多项式

特征多项式 $p(\lambda)=\det(\lambda I-A)=\lambda^n+c_1\lambda^{n-1}+\dots+c_n$，其系数装着行列式、迹等谱信息，也是 Cayley-Hamilton（$p(A)=0$）与常系数矩阵递推降幂的基础。本条用 Faddeev–LeVerrier 递推：置 $M_0=I$、$c_0=1$，每步令 $M_k=A M_{k-1}+c_k I$、$c_k=-\dfrac{1}{k}\operatorname{tr}(A M_{k-1})$，$n$ 步后 $c_0,\dots,c_n$ 就是各次系数。递推里出现除以 $k$，故 `T` 必须是域（模素数或有理/浮点），且需域特征大于 $n$，否则 $T(k)$ 不可逆。返回顺序是从高次到低次：`c[0]` 是 $\lambda^n$ 的系数（恒为 $1$），`c[n]` 是常数项 $(-1)^n\det A$；霍纳求值应从 `c[0]` 起累乘。

```cpp
template <class T> vector<T> charPoly(const vector<vector<T>> &a)
{
    // a 是方阵；返回特征多项式系数，c[0]=1 为最高次 x^n 项、c[n] 为常数项。
    int n = a.size();
    // 调试检查，可删。
    for (const auto &r : a)
    {
        assert((int)r.size() == n);
    }
    vector b(n, vector<T>(n));
    for (int i = 0; i < n; i++)
    {
        b[i][i] = 1;
    }
    vector<T> c(n + 1);
    c[0] = 1;
    for (int k = 1; k <= n; k++)
    {
        vector q(n, vector<T>(n));
        for (int i = 0; i < n; i++)
        {
            for (int z = 0; z < n; z++)
            {
                for (int j = 0; j < n; j++)
                {
                    q[i][j] += a[i][z] * b[z][j];
                }
            }
        }
        T tr = 0;
        for (int i = 0; i < n; i++)
        {
            tr += q[i][i];
        }
        c[k] = -tr / T(k);
        for (int i = 0; i < n; i++)
        {
            q[i][i] += c[k];
        }
        b.swap(q);
    }
    return c;
}
```

调用 `charPoly(a)` 传入方阵，返回长度 $n+1$ 的系数向量。每步一次矩阵乘法，Faddeev–LeVerrier 共 $O(n^4)$ 时间、$O(n^2)$ 空间，且要求 `T(k)` 对所有 $1\le k\le n$ 可逆；素数模下通常要求 `n<mod`。常数项 `c[n]` 等于 $(-1)^n\det(A)$，不能不看奇偶直接当作行列式。只要行列式时 [9.1.006 行列式（质数模数 / 任意模数）](#91006) 的 $O(n^3)$ 消元更快；线性递推可借本条系数配合 [9.1.002 矩阵快速幂](#91002) 或第 10 章的多项式取模加速。

下面直接核对空矩阵、对角矩阵与一个非对角矩阵的已知特征多项式：

```cpp
signed main()
{
    auto same = [](const vector<long double> &a, const vector<long double> &b)
    {
        if (a.size() != b.size())
        {
```

### 9.1.008 稀疏矩阵

当矩阵非零元个数 $\operatorname{nnz}$ 远少于 $n\times m$ 时，稠密存储既费内存又让每次乘法白算大量零。`SparseMat` 按行存（列下标，值），只保留非零项，矩阵乘向量 $y=Ax$ 就退化为遍历每行的非零表累加 $y_i\mathrel{+}=v\cdot x_j$。它专为幂法、共轭梯度等只需反复算 $Ax$ 的线性迭代设计；若做稀疏乘稀疏且结果趋于稠密，或反复自乘导致填充增长，稀疏表的优势会消失，此时应转回稠密矩阵。`add(i,j,x)` 中 `i`、`j` 分别是行、列下标，同一位置可加入多项并在乘法时累加。

```cpp
template <class T> struct SparseMat
{
    // n、m 是行数与列数，a[i] 存第 i 行的 (列下标, 值)。
    int n, m;
    vector<vector<pair<int, T>>> a;

    SparseMat(int n, int m) : n(n), m(m), a(n)
    {
        // 建立 n 行 m 列的零矩阵。
    }

    void add(int i, int j, T x)
    {
        // 在 (i,j) 加一项 x；重复位置会在乘法时累加，零项应由调用方过滤。
        // 调试检查，可删。
        assert(0 <= i && i < n && 0 <= j && j < m);
        a[i].push_back({j, x});
    }

    vector<T> operator*(const vector<T> &x) const
    {
        // x 是 m 维列向量；返回 A*x。
        // 调试检查，可删。
        assert((int)x.size() == m);
        vector<T> y(n);
        for (int i = 0; i < n; i++)
        {
            for (auto [j, v] : a[i])
            {
                y[i] += v * x[j];
            }
        }
        return y;
    }
};
```

以 `SparseMat<T> s(n,m)` 建 $n\times m$ 零矩阵，`s.add(i,j,x)` 只加入非零项，`s*x` 返回长度 $n$ 的乘积向量。同一位置可加入多次，乘法时各项自然相加；若把零值也放入表中，算法仍正确但会浪费时间和空间。单次矩阵乘向量为 $O(\operatorname{nnz})$，存储为 $O(n+\operatorname{nnz})$。矩阵稠密时回到 [9.1.001 普通矩阵与取模矩阵](#91001)；固定步数的半环最短路用 [9.1.004 `+min` / `+max` 半环矩阵](#91004)，那是稠密表示。

### 9.1.009 Strassen 矩阵乘法

把两个 $n\times n$ 矩阵各切成四块 $m\times m$（$m=n/2$），朴素分块乘法要算八次子块乘积。Strassen 用七次线性组合乘积 $p_1,\dots,p_7$ 拼出全部四块结果，把递归式从 $T(n)=8T(n/2)$ 降到 $T(n)=7T(n/2)+O(n^2)$，复杂度为 $O(n^{\log_2 7})$。本实现在规模 $\le32$ 时退回朴素乘法；入口先把行数、公共维数和列数补齐到同一个二次幂，算完再裁回原形。实现放在 `strassen_mul` 命名空间内，避免其中的矩阵别名与 [9.1.001 普通矩阵与取模矩阵](#91001) 的 `Mat` 结构体重名。

```cpp
namespace strassen_mul
{
template <class T> using Mat = vector<vector<T>>;

template <class T> Mat<T> matAdd(const Mat<T> &a, const Mat<T> &b, int sgn = 1)
{
    // a、b 是同阶方阵；sgn=1 时相加，sgn=-1 时相减。
    int n = a.size();
    Mat<T> c(n, vector<T>(n));
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < n; j++)
        {
            c[i][j] = a[i][j] + (sgn == 1 ? b[i][j] : -b[i][j]);
        }
    }
    return c;
}

template <class T> Mat<T> strassenRec(const Mat<T> &a, const Mat<T> &b)
{
    // a、b 是同阶二次幂方阵；返回其乘积。
    int n = a.size();
    if (n <= 32)
    {
        Mat<T> c(n, vector<T>(n));
        for (int i = 0; i < n; i++)
        {
            for (int k = 0; k < n; k++)
            {
                for (int j = 0; j < n; j++)
                {
                    c[i][j] += a[i][k] * b[k][j];
                }
            }
        }
        return c;
    }
    int m = n / 2;
    Mat<T> x[2][2], y[2][2];
    for (int p = 0; p < 2; p++)
    {
        for (int q = 0; q < 2; q++)
        {
            x[p][q].assign(m, vector<T>(m));
            y[p][q].assign(m, vector<T>(m));
            for (int i = 0; i < m; i++)
            {
                copy_n(a[i + p * m].begin() + q * m, m, x[p][q][i].begin());
                copy_n(b[i + p * m].begin() + q * m, m, y[p][q][i].begin());
            }
        }
    }
    auto p1 = strassenRec(matAdd(x[0][0], x[1][1]), matAdd(y[0][0], y[1][1]));
    auto p2 = strassenRec(matAdd(x[1][0], x[1][1]), y[0][0]);
    auto p3 = strassenRec(x[0][0], matAdd(y[0][1], y[1][1], -1));
    auto p4 = strassenRec(x[1][1], matAdd(y[1][0], y[0][0], -1));
    auto p5 = strassenRec(matAdd(x[0][0], x[0][1]), y[1][1]);
    auto p6 = strassenRec(matAdd(x[1][0], x[0][0], -1), matAdd(y[0][0], y[0][1]));
    auto p7 = strassenRec(matAdd(x[0][1], x[1][1], -1), matAdd(y[1][0], y[1][1]));
    Mat<T> c(n, vector<T>(n));
    for (int i = 0; i < m; i++)
    {
        for (int j = 0; j < m; j++)
        {
            c[i][j] = p1[i][j] + p4[i][j] - p5[i][j] + p7[i][j];
            c[i][j + m] = p3[i][j] + p5[i][j];
            c[i + m][j] = p2[i][j] + p4[i][j];
            c[i + m][j + m] = p1[i][j] - p2[i][j] + p3[i][j] + p6[i][j];
        }
    }
    return c;
}

template <class T> Mat<T> strassen(const Mat<T> &a, const Mat<T> &b)
{
    // a 是 n*m 矩阵，b 是 m*k 矩阵；返回 n*k 乘积。
    int n = a.size();
    int m = b.size();
    int k = m ? b[0].size() : 0;
    // 调试检查，可删。
    for (const auto &r : a)
    {
        assert((int)r.size() == m);
    }
    // 调试检查，可删。
    for (const auto &r : b)
    {
        assert((int)r.size() == k);
    }
    if (!n || !m || !k)
    {
        return Mat<T>(n, vector<T>(k));
    }
    int z = (int)bit_ceil((unsigned long long)max({n, m, k}));
    Mat<T> x(z, vector<T>(z));
    Mat<T> y(z, vector<T>(z));
    for (int i = 0; i < n; i++)
    {
        copy(a[i].begin(), a[i].end(), x[i].begin());
    }
    for (int i = 0; i < m; i++)
    {
        copy(b[i].begin(), b[i].end(), y[i].begin());
    }
    x = strassenRec(x, y);
    x.resize(n);
    for (auto &v : x)
    {
        v.resize(k);
    }
    return x;
}
}
```

调用 `strassen_mul::strassen(a,b)`，`a` 是 $n\times m$、`b` 是 $m\times k$，返回 $n\times k$ 乘积；零行或零列矩阵返回对应零形。补零尺寸用 64 位无符号整数求 $N=\text{bit\_ceil}(\max(n,m,k))$，要求该尺寸和对应矩阵均可分配。元素类型 `T` 必须同时容纳最终乘积与递归中的块加减，最终答案可表示并不能单独保证 Strassen 的临时值可表示。时间 $O(N^{\log_2 7})$、峰值空间 $O(N^2)$；常规规模或模矩阵仍优先用 [9.1.001 普通矩阵与取模矩阵](#91001)。

### 9.1.010 Woodbury 矩阵恒等式

已知 $A^{-1}$，若 $A$ 又叠加一个低秩修正变成 $A+UCV$（$U$ 是 $n\times k$、$C$ 是 $k\times k$、$V$ 是 $k\times n$，$k\ll n$），从头重算逆是 $O(n^3)$，太浪费。Woodbury 恒等式 $(A+UCV)^{-1}=A^{-1}-A^{-1}U(C^{-1}+VA^{-1}U)^{-1}VA^{-1}$ 把它转成只对一个 $k\times k$ 矩阵 $C^{-1}+VA^{-1}U$ 求逆，其余都是矩阵乘法。要求 $A$、$C$ 以及那个 $k\times k$ 中间矩阵都可逆。$k=1$ 的特例即 Sherman–Morrison。下面 `inv` 用带部分选主元的高斯消元求逆（`long double`），`woodbury(ai,u,ci,v)` 直接吃 $A^{-1}$、$C^{-1}$ 与低秩因子。

```cpp
namespace woodbury
{
using Mat = vector<vector<long double>>;

Mat mul(const Mat &a, const Mat &b)
{
    // a、b 是维数相容矩阵；返回矩阵乘积。
    if (a.empty())
    {
        return {};
    }
    if (b.empty())
    {
        for (const auto &r : a)
        {
            assert(r.empty()); // 调试检查，可删。
        }
        return Mat(a.size());
    }
    int n = a.size();
    int k = b.size();
    int m = b[0].size();
    // 调试检查，可删。
    for (const auto &r : b)
    {
        assert((int)r.size() == m);
    }
    Mat c(n, vector<long double>(m));
    for (int i = 0; i < n; i++)
    {
        // 调试检查，可删。
        assert((int)a[i].size() == k);
        for (int p = 0; p < k; p++)
        {
            for (int j = 0; j < m; j++)
            {
                c[i][j] += a[i][p] * b[p][j];
            }
        }
    }
    return c;
}

Mat inv(Mat a)
{
    // a 是可逆方阵；返回高斯消元求得的逆矩阵。
    int n = a.size();
    Mat b(n, vector<long double>(2 * n));
    for (int i = 0; i < n; i++)
    {
        // 调试检查，可删。
        assert((int)a[i].size() == n);
        copy(a[i].begin(), a[i].end(), b[i].begin());
        b[i][n + i] = 1;
    }
    for (int j = 0; j < n; j++)
    {
        int p = j;
        for (int i = j + 1; i < n; i++)
        {
            if (abs(b[i][j]) > abs(b[p][j]))
            {
                p = i;
            }
        }
        // 调试检查，可删。
        assert(abs(b[p][j]) > 1E-18L);
        swap(b[p], b[j]);
        long double x = b[j][j];
        for (auto &v : b[j])
        {
            v /= x;
        }
        for (int i = 0; i < n; i++)
        {
            if (i == j)
            {
                continue;
            }
            x = b[i][j];
            for (int k = 0; k < 2 * n; k++)
            {
                b[i][k] -= x * b[j][k];
            }
        }
    }
    Mat ans(n, vector<long double>(n));
    for (int i = 0; i < n; i++)
    {
        copy(b[i].begin() + n, b[i].end(), ans[i].begin());
    }
    return ans;
}

Mat woodbury(const Mat &ai, const Mat &u, const Mat &ci, const Mat &v)
{
    // ai、ci 分别是 A 与 C 的逆，u、v 是低秩因子；返回 (A+UCV) 的逆。
    int n = ai.size();
    int k = ci.size();
    // 调试检查，可删。
    assert((int)u.size() == n && (int)v.size() == k);
    // 调试检查，可删。
    for (const auto &r : ai)
    {
        assert((int)r.size() == n);
    }
    // 调试检查，可删。
    for (const auto &r : u)
    {
        assert((int)r.size() == k);
    }
    // 调试检查，可删。
    for (const auto &r : ci)
    {
        assert((int)r.size() == k);
    }
    // 调试检查，可删。
    for (const auto &r : v)
    {
        assert((int)r.size() == n);
    }
    if (n == 0)
    {
        return {};
    }
    if (k == 0)
    {
        return ai;
    }
    Mat left = mul(ai, u);
    Mat right = mul(v, ai);
    Mat mid = ci;
    Mat vaiu = mul(right, u);
    for (int i = 0; i < (int)mid.size(); i++)
    {
        for (int j = 0; j < (int)mid.size(); j++)
        {
            mid[i][j] += vaiu[i][j];
        }
    }
    Mat sub = mul(mul(left, inv(mid)), right);
    Mat ans = ai;
    for (int i = 0; i < (int)ans.size(); i++)
    {
        for (int j = 0; j < (int)ans.size(); j++)
        {
            ans[i][j] -= sub[i][j];
        }
    }
    return ans;
}
}
```

调用 `woodbury::woodbury(ai,u,ci,v)`，其中 `ai=A^{-1}`、`ci=C^{-1}` 需调用方预先备好（或用同命名空间的 `inv` 现算）。`k=0` 是合法的零秩修正，此时 `ci`、`v` 为空、`u` 每行为空，函数直接返回 `ai`。低秩为 $k$、原维数为 $n$ 时 $O(n^2k+nk^2+k^3)$ 时间、$O(n^2+k^2)$ 空间，当 $k\ll n$ 时远优于重新求逆。若修正不是低秩、或没有现成 $A^{-1}$，直接对 $A+UCV$ 用 [9.1.005 矩阵求逆（质数模数 / 任意模数）](#91005)（模意义）或本条 `inv`（浮点）从头求逆。

### 9.1.011 伴随矩阵

伴随矩阵满足 $A\operatorname{adj}(A)=\det(A)I$。满秩时直接用 $\operatorname{adj}(A)=\det(A)A^{-1}$；秩为 $n-1$ 时，右零空间与左零空间都为一维，伴随矩阵是两个零向量外积乘一个标量，标量由一个 $(n-1)\times(n-1)$ 余子式确定；秩至多 $n-2$ 时所有余子式都为零，伴随矩阵就是零矩阵。下面实现只要求素数模，按行最简形确定秩和零向量，整个过程是确定性的，不依赖随机增广。

```cpp
vector<vector<int>> adjugateP(vector<vector<int>> a, int mod)
{
    // a 是方阵，mod 是素数；返回 A 的伴随矩阵。
    // 调试检查，可删。
    assert(mod > 1);
    int n = a.size();
    for (auto &r : a)
    {
        // 调试检查，可删。
        assert((int)r.size() == n);
        for (auto &x : r)
        {
            x %= mod;
            if (x < 0)
            {
                x += mod;
            }
        }
    }
    if (n == 0)
    {
        return {};
    }
    auto power = [&](int x, int b)
    {
        // x 是非零剩余，b 是非负指数；返回 x 的 b 次幂对 mod 的余数。
        int ans = 1;
        while (b)
        {
            if (b & 1)
            {
                ans = linearMulMod(ans, x, mod);
            }
            x = linearMulMod(x, x, mod);
            b >>= 1;
        }
        return ans;
    };
    auto reduce = [&](vector<vector<int>> m)
    {
        // m 是 n 阶方阵；返回秩、行最简形和每列对应的主元行，非主元列记为 -1。
        vector<int> pivot(n, -1);
        int rk = 0;
        for (int col = 0; col < n && rk < n; col++)
        {
            int row = rk;
            while (row < n && m[row][col] == 0)
            {
                row++;
            }
            if (row == n)
            {
                continue;
            }
            swap(m[row], m[rk]);
            int z = power(m[rk][col], mod - 2);
            for (int j = 0; j < n; j++)
            {
                m[rk][j] = linearMulMod(m[rk][j], z, mod);
            }
            for (int i = 0; i < n; i++)
            {
                if (i == rk)
                {
                    continue;
                }
                int q = m[i][col];
                for (int j = 0; j < n; j++)
                {
                    m[i][j] = (m[i][j] - linearMulMod(q, m[rk][j], mod)) % mod;
                    if (m[i][j] < 0)
                    {
                        m[i][j] += mod;
                    }
                }
            }
            pivot[col] = rk++;
        }
        return tuple<int, vector<vector<int>>, vector<int>>(rk, move(m), move(pivot));
    };
    auto [rk, r, pivot] = reduce(a);
    vector ans(n, vector<int>(n));
    if (rk == n)
    {
        auto inv = matInvPP(a, mod, mod);
        int det = detPrime(a, mod);
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < n; j++)
            {
                ans[i][j] = linearMulMod(det, (*inv)[i][j], mod);
            }
        }
        return ans;
    }
    if (rk <= n - 2)
    {
        return ans;
    }
    int fx = find(pivot.begin(), pivot.end(), -1) - pivot.begin();
    vector<int> x(n);
    x[fx] = 1;
    for (int col = 0; col < n; col++)
    {
        if (pivot[col] != -1)
        {
            x[col] = (mod - r[pivot[col]][fx]) % mod;
        }
    }
    vector<vector<int>> at(n, vector<int>(n));
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < n; j++)
        {
            at[i][j] = a[j][i];
        }
    }
    auto [rk2, rt, pivot2] = reduce(at);
    (void)rk2;
    int fy = find(pivot2.begin(), pivot2.end(), -1) - pivot2.begin();
    vector<int> y(n);
    y[fy] = 1;
    for (int col = 0; col < n; col++)
    {
        if (pivot2[col] != -1)
        {
            y[col] = (mod - rt[pivot2[col]][fy]) % mod;
        }
    }
    int i0 = 0, j0 = 0;
    while (x[i0] == 0)
    {
        i0++;
    }
    while (y[j0] == 0)
    {
        j0++;
    }
    vector<vector<int>> minor;
    for (int i = 0; i < n; i++)
    {
        if (i == j0)
        {
            continue;
        }
        vector<int> row;
        for (int j = 0; j < n; j++)
        {
            if (j != i0)
            {
                row.push_back(a[i][j]);
            }
        }
        minor.push_back(move(row));
    }
    int cof = detPrime(minor, mod);
    if ((i0 + j0) & 1 && cof)
    {
        cof = mod - cof;
    }
    int scale = linearMulMod(cof, power(x[i0], mod - 2), mod);
    scale = linearMulMod(scale, power(y[j0], mod - 2), mod);
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < n; j++)
        {
            ans[i][j] = linearMulMod(linearMulMod(scale, x[i], mod), y[j], mod);
        }
    }
    return ans;
}
```

`adjugateP(a,mod)` 覆盖满秩、秩 $n-1$ 和秩至多 $n-2$ 三种情况，返回值总在 $[0,mod)$。时间为 $O(n^3)$，空间为 $O(n^2)$；满秩分支复用本章的 [9.1.005 矩阵求逆（质数模数 / 任意模数）](#91005) 和 [9.1.006 行列式（质数模数 / 任意模数）](#91006)。只要逆矩阵而不要伴随矩阵时不要额外乘行列式。

### 9.1.012 Hafnian 与一般图完美匹配计数

对 $2n\times2n$ 对称矩阵 $A$，Hafnian 是所有完美匹配边权乘积之和：$\operatorname{haf}(A)=\sum_M\prod_{(i,j)\in M}A_{ij}$。邻接矩阵取 $0/1$ 时，它就是一般图完美匹配数。直接枚举匹配约有 $(2n-1)!!$ 项；Bjorklund 型算法把顶点预先两两分组，把交错环贡献写成集合函数 `cyc`，再调用 [10.1.013 集合形式幂级数](10-多项式、变换与线性递推.md#101013) 的 `setExp`。子集卷积与集合指数的表示、工作区和边界完全适用，因此本条只保留 Hafnian 自己的状态转移。

```cpp
template <class T> T hafnian(const vector<vector<T>> &a)
{
    // a 是 2n 阶对称矩阵；返回其 Hafnian。
    int N = a.size();
    // 调试检查，可删。
    assert(N % 2 == 0);
    int n = N / 2;
    // 调试检查，可删。
    assert(n <= 20);
    // 调试检查，可删。
    for (int i = 0; i < N; i++)
    {
        assert((int)a[i].size() == N);
        for (int j = 0; j < N; j++)
        {
            assert(a[i][j] == a[j][i]);
        }
    }
    vector<T> cyc(1LL << n);
    for (int i = 0; i < n; i++)
    {
        int u = 2 * i;
        int v = u + 1;
        int w = 2 * i;
        cyc[1LL << i] += a[u][v];
        vector<T> dp(w * (1LL << i));
        for (int j = 0; j < i; j++)
        {
            int x = 2 * j;
            int y = x + 1;
            dp[w * (1LL << j) + x] += a[u][y];
            dp[w * (1LL << j) + y] += a[u][x];
        }
        for (int mask = 0; mask < (1LL << i); mask++)
        {
            for (int j = 0; j < i; j++)
            {
                int x = 2 * j;
                int y = x + 1;
                cyc[mask | (1LL << i)] += dp[w * mask + x] * a[v][x];
                cyc[mask | (1LL << i)] += dp[w * mask + y] * a[v][y];
                int rest = ((1LL << i) - 1) ^ mask;
                while (rest)
                {
                    int k = countr_zero((unsigned)rest);
                    rest &= rest - 1;
                    int p = 2 * k;
                    int q = p + 1;
                    int to = mask | (1LL << k);
                    dp[w * to + p] += dp[w * mask + x] * a[x][q];
                    dp[w * to + p] += dp[w * mask + y] * a[y][q];
                    dp[w * to + q] += dp[w * mask + x] * a[x][p];
                    dp[w * to + q] += dp[w * mask + y] * a[y][p];
                }
            }
        }
    }
    return setExp(cyc).back();
}
```

`hafnian(a)` 要求矩阵阶数为 $2n$ 且对称，主对角线不会参与答案；`T` 必须构成交换环，空矩阵的 Hafnian 按空乘积定义为 $1$。本条显式依赖第 10 章的 `setExp`，拼接或单独使用头文件时都要带上该公共头。按子集大小分层计数，时间为 $O(n^2 2^n)$，峰值空间为 $O(n2^n)$，因此代码把 `n` 限为 $20$。它计算的是一般图完美匹配的带权和；二分图完美匹配计数对应行列式或 permanent 的其他专用算法，不能混用。

下面用 $4\times4$ 矩阵直接核对三种配对之和 $1\cdot6+2\cdot5+3\cdot4=28$：

```cpp
signed main()
{
    vector<vector<long long>> a{{0, 1, 2, 3}, {1, 0, 4, 5}, {2, 4, 0, 6}, {3, 5, 6, 0}};
    assert(hafnian(vector<vector<long long>>{}) == 1);
    assert(hafnian(vector<vector<long long>>{{0, 7}, {7, 0}}) == 7);
    cout << hafnian(a) << '\n'; // 28
    return 0;
}
```

## 9.2 线性方程与规划

### 9.2.001 高斯消元：唯一解、无解、多解

在素数模域 $\mathbb{F}_p$ 上解 $Ax\equiv b$，并完整区分无解、唯一解与多解。Gauss-Jordan 消元逐列找非零主元、用主元逆元把该列化成单位列（主元行系数变 $1$、同列其他行清零），处理完得到行最简形。判定规则：若某行左侧系数全零而右端非零，方程 $0\equiv c\neq0$ 矛盾即无解；否则每个主元列的变量由该行右端唯一定出（特解），每个非主元列是自由变量，对应一条齐次解基——把该自由变量置 $1$、其余自由变量置 $0$，主元变量取相应负系数，就得到 $Ax\equiv0$ 的一个基向量。通解即“特解 $+$ 齐次基的任意线性组合”。`basis` 为空即唯一解。

```cpp
struct ModLinearSol
{
    // part 是一组特解，basis 是齐次解空间基；全部值在 [0,mod) 内。
    vector<int> part;
    vector<vector<int>> basis;
};

optional<ModLinearSol> gaussMod(vector<vector<int>> a, int n, int mod)
{
    // a 有 n+1 列且最后一列是右端，mod 是素数；返回特解与齐次基，无解时返回空。
    // 调试检查，可删。
    assert(n >= 0 && mod > 1);
    for (auto &r : a)
    {
        // 调试检查，可删。
        assert((int)r.size() == n + 1);
        for (auto &x : r)
        {
            x %= mod;
            if (x < 0)
            {
                x += mod;
            }
        }
    }
    int m = a.size();
    int rk = 0;
    vector<int> pivot(n, -1);
    auto power = [&](int x, int b)
    {
        // x 是非零剩余，b 是非负指数；返回 x^b mod mod。
        int ans = 1;
        while (b)
        {
            if (b & 1)
            {
                ans = linearMulMod(ans, x, mod);
            }
            x = linearMulMod(x, x, mod);
            b >>= 1;
        }
        return ans;
    };
    for (int col = 0; col < n && rk < m; col++)
    {
        int row = rk;
        while (row < m && a[row][col] == 0)
        {
            row++;
        }
        if (row == m)
        {
            continue;
        }
        swap(a[row], a[rk]);
        int inv = power(a[rk][col], mod - 2);
        for (int j = col; j <= n; j++)
        {
            a[rk][j] = linearMulMod(a[rk][j], inv, mod);
        }
        for (int i = 0; i < m; i++)
        {
            if (i == rk)
            {
                continue;
            }
            int q = a[i][col];
            for (int j = col; j <= n; j++)
            {
                a[i][j] -= linearMulMod(q, a[rk][j], mod);
                if (a[i][j] < 0)
                {
                    a[i][j] += mod;
                }
            }
        }
        pivot[col] = rk++;
    }
    for (int i = rk; i < m; i++)
    {
        if (a[i][n])
        {
            return nullopt;
        }
    }
    ModLinearSol sol{vector<int>(n), {}};
    for (int col = 0; col < n; col++)
    {
        if (pivot[col] != -1)
        {
            sol.part[col] = a[pivot[col]][n];
        }
    }
    for (int free = 0; free < n; free++)
    {
        if (pivot[free] != -1)
        {
            continue;
        }
        vector<int> v(n);
        v[free] = 1;
        for (int col = 0; col < n; col++)
        {
            if (pivot[col] != -1)
            {
                v[col] = (mod - a[pivot[col]][free]) % mod;
            }
        }
        sol.basis.push_back(move(v));
    }
    return sol;
}
```

调用 `gaussMod(a,n,mod)`，`a` 是 $m$ 行 $n+1$ 列增广矩阵（最后一列是右端），`mod` 必须是素数；显式传 `n` 是为了让 `m=0` 时仍能表示变量数。返回空即无解，`sol.basis.empty()` 即唯一解，齐次基数量恒为 `n-rk`。零条方程会返回全零特解和 $n$ 个单位基向量；零个变量时，每个合法方程只剩右端，全部为零才相容。连同输入归一化和齐次基输出，时间为 $O(m(n+1)+mn\min(m,n)+n^2)$，空间为 $O(m(n+1)+n^2)$。实数方程用 [9.2.002 高斯消元：列主元](#92002) 或 [9.2.003 高斯消元最小范数解](#92003)；异或方程参见 [9.2.004 线性基（参见数据结构）](#92004)；三对角结构用 [9.2.007 追赶法解三对角/周期方程](#92007)。

下面用五个固定系统覆盖唯一解、多解、无解、零方程和零变量，并直接核对齐次基维数：

```cpp
signed main()
{
    auto one = gaussMod({{1, 1, 3}, {1, -1, 1}}, 2, 5);
    assert(one && one->part == vector<int>({2, 1}) && one->basis.empty());
    auto many = gaussMod({{1, 1, 1}}, 2, 5);
    assert(many && many->basis.size() == 1);
    assert(many->basis[0] == vector<int>({4, 1}));
    assert(!gaussMod({{1, 1}, {1, 2}}, 1, 5));
    auto none = gaussMod({}, 3, 5);
    assert(none && none->part == vector<int>(3, 0) && none->basis.size() == 3);
    assert(gaussMod({{0}}, 0, 5) && !gaussMod({{1}}, 0, 5));
    cout << one->part[0] << ' ' << one->part[1] << ' ' << none->basis.size() << '\n'; // 2 1 3
    return 0;
}
```

### 9.2.002 高斯消元：列主元

浮点消元里若拿一个绝对值极小的数当主元（除数），舍入误差会被放大很多倍，最终解可能面目全非。列主元（部分选主元）法在当前列的候选行中挑绝对值最大者交换到主元位，让除数尽量大、误差最小。这里用它求浮点线性方程的唯一解：消元后统计秩，秩等于变量数才有唯一解并回代；否则（出现矛盾行或秩不足）返回空，把多解、最小范数解交给专门接口处理。判零阈值取公共前导里的 `eps`。

```cpp
optional<vector<long double>> gaussPivot(vector<vector<long double>> a, int n)
{
    // a 有 n+1 列且最后一列是右端；唯一解时返回解，否则返回空。
    // 调试检查，可删。
    assert(n >= 0);
    // 调试检查，可删。
    for (const auto &r : a)
    {
        assert((int)r.size() == n + 1);
    }
    int m = a.size();
    int rk = 0;
    vector<int> pivot(n, -1);
    for (int col = 0; col < n && rk < m; col++)
    {
        int row = rk;
        for (int i = rk + 1; i < m; i++)
        {
            if (fabsl(a[i][col]) > fabsl(a[row][col]))
            {
                row = i;
            }
        }
        if (fabsl(a[row][col]) <= eps)
        {
            continue;
        }
        swap(a[row], a[rk]);
        for (int j = n; j >= col; j--)
        {
            a[rk][j] /= a[rk][col];
        }
        for (int i = 0; i < m; i++)
        {
            if (i == rk)
            {
                continue;
            }
            long double q = a[i][col];
            for (int j = col; j <= n; j++)
            {
                a[i][j] -= q * a[rk][j];
            }
        }
        pivot[col] = rk++;
    }
    for (int i = rk; i < m; i++)
    {
        if (fabsl(a[i][n]) > eps)
        {
            return nullopt;
        }
    }
    if (rk < n)
    {
        return nullopt;
    }
    vector<long double> x(n);
    for (int col = 0; col < n; col++)
    {
        x[col] = a[pivot[col]][n];
    }
    return x;
}
```

调用 `gaussPivot(a,n)`，`a` 是 $m$ 行 $n+1$ 列的增广矩阵；显式变量数使空方程组也有确定含义：$m=n=0$ 有唯一空解，$m=0<n$ 时是多解。函数只用于求浮点唯一解，返回空可能是无解也可能是多解。连同输入检查，时间为 $O(m(n+1)+mn\min(m,n))$，空间为 $O(m(n+1))$。素数模方程用 [9.2.001 高斯消元：唯一解、无解、多解](#92001)；相容多解的最小范数解用 [9.2.003 高斯消元最小范数解](#92003)；三对角结构用 [9.2.007 追赶法解三对角/周期方程](#92007)。

下面固定检查唯一解、需要换行的主元、多解、不相容与空系统：

```cpp
signed main()
{
    auto x = gaussPivot({{0, 1, 2}, {1, 1, 3}}, 2);
    assert(x && fabsl((*x)[0] - 1) < 1E-10L && fabsl((*x)[1] - 2) < 1E-10L);
    assert(!gaussPivot({{1, 1, 2}}, 2));
    assert(!gaussPivot({{1, 1}, {1, 2}}, 1));
    assert(gaussPivot({}, 0) && !gaussPivot({}, 2));
    cout << llround((*x)[0]) << ' ' << llround((*x)[1]) << '\n'; // 1 2
    return 0;
}
```

### 9.2.003 高斯消元最小范数解

当实数约束 $Ax=b$ 相容但欠定（解不唯一）时，把自由变量随手置零得到的解依赖消元顺序、没有几何意义。这里求的是欧氏范数 $\lVert x\rVert_2$ 最小的那个唯一解。它是带约束极小化 $\min\lVert x\rVert^2\text{ s.t. }Ax=b$，拉格朗日 KKT 条件给出 $x+A^T\lambda=0$、$Ax=b$，把 $x$ 与乘子 $\lambda$ 拼成未知量就是一个对称增广线性系统 $\begin{pmatrix}I&A^T\\A&0\end{pmatrix}\begin{pmatrix}x\\\lambda\end{pmatrix}=\begin{pmatrix}0\\b\end{pmatrix}$。用列主元消元解这个 $(n+m)$ 维系统，约束相关只让 $\lambda$ 不唯一，$x$ 部分仍唯一确定，取出前 $n$ 维即最小范数解。

```cpp
optional<vector<long double>> minNormSolve(const vector<vector<long double>> &a, const vector<long double> &b, int n)
{
    // a 是 m*n 系数矩阵，b 是 m 维右端；返回 Ax=b 的最小二范数解，不相容时返回空。
    int m = a.size();
    int s = n + m;
    // 调试检查，可删。
    assert(n >= 0);
    // 调试检查，可删。
    assert((int)b.size() == m);
    // 调试检查，可删。
    for (const auto &r : a)
    {
        assert((int)r.size() == n);
    }
    vector mat(s, vector<long double>(s + 1));
    for (int j = 0; j < n; j++)
    {
        mat[j][j] = 1;
        for (int i = 0; i < m; i++)
        {
            mat[j][n + i] = a[i][j];
        }
    }
    for (int i = 0; i < m; i++)
    {
        for (int j = 0; j < n; j++)
        {
            mat[n + i][j] = a[i][j];
        }
        mat[n + i][s] = b[i];
    }
    int rk = 0;
    vector<int> pivot(s, -1);
    for (int col = 0; col < s && rk < s; col++)
    {
        int row = rk;
        for (int i = rk + 1; i < s; i++)
        {
            if (fabsl(mat[i][col]) > fabsl(mat[row][col]))
            {
                row = i;
            }
        }
        if (fabsl(mat[row][col]) <= eps)
        {
            continue;
        }
        swap(mat[row], mat[rk]);
        for (int j = s; j >= col; j--)
        {
            mat[rk][j] /= mat[rk][col];
        }
        for (int i = 0; i < s; i++)
        {
            if (i == rk)
            {
                continue;
            }
            long double q = mat[i][col];
            for (int j = col; j <= s; j++)
            {
                mat[i][j] -= q * mat[rk][j];
            }
        }
        pivot[col] = rk++;
    }
    for (int i = rk; i < s; i++)
    {
        if (fabsl(mat[i][s]) > 1E-10L)
        {
            return nullopt;
        }
    }
    vector<long double> z(s);
    for (int col = 0; col < s; col++)
    {
        if (pivot[col] != -1)
        {
            z[col] = mat[pivot[col]][s];
        }
    }
    z.resize(n);
    return z;
}
```

调用 `minNormSolve(a,b,n)`，`a` 各行长度为 `n`、`b` 长度等于行数，返回空表示约束不相容。显式传 `n` 后，零条约束会正确返回 $n$ 维零向量。单个方程 $x+y=1$ 的最小范数解是 $(0.5,0.5)$，重复约束不改变结果。复杂度为 $O((m+n)^3)$，空间为 $O((m+n)^2)$。这里解的是 KKT 增广系统，不是正规方程；它不会把条件数直接平方，但矩阵不定且在冗余约束下奇异，病态问题仍应换 QR 或 SVD。唯一解用 [9.2.002 高斯消元：列主元](#92002) 更省；素数模多解结构用 [9.2.001 高斯消元：唯一解、无解、多解](#92001) 的齐次基表示。

下面检查单约束、重复约束、不相容约束和零约束：

```cpp
signed main()
{
    auto x = minNormSolve({{1, 1}}, {1}, 2);
    assert(x && fabsl((*x)[0] - 0.5L) < 1E-10L && fabsl((*x)[1] - 0.5L) < 1E-10L);
    auto y = minNormSolve({{1, 1}, {1, 1}}, {1, 1}, 2);
    assert(y && fabsl((*y)[0] - 0.5L) < 1E-10L && fabsl((*y)[1] - 0.5L) < 1E-10L);
    assert(!minNormSolve({{1, 1}, {1, 1}}, {1, 2}, 2));
    auto z = minNormSolve({}, {}, 2);
    assert(z && *z == vector<long double>({0, 0}));
    cout << fixed << setprecision(1) << (*x)[0] << ' ' << (*x)[1] << '\n'; // 0.5 0.5
    return 0;
}
```

### 9.2.004 线性基（参见数据结构）

整数按二进制看成 $\mathbb{F}_2$（二元域）上的向量，异或就是向量加法，一组数的所有异或子集和正是这些向量张成的线性子空间。线性基按最高位存一组极大无关向量：插入 $x$ 时从高位向低位消元，若最终在空档留下非零主元则秩加一，否则它可由现有基表示。`has` 判可表示性，`maxXor` 贪心得最大异或值，`rk` 是秩，`kth` 在化简后的基上取第 $k$ 小可表示值。本条只给出线性代数视角的索引，代表实现与增强接口统一放在数据结构章，避免同一代码维护两份。

本条不重复打印代码；请直接使用 [3.8.001 线性基](#38001) 的代表实现。模板参数 `W` 是位宽，`add(x)` 返回是否使秩增加，`has(x)`、`maxXor(x)`、`kth(k)`、`merge(o)` 的语义和边界检查均以该条为准。`add`、`has`、`maxXor` 均 $O(W)$，`kth` 需一次 $O(W^2)$ 化简，空间 $O(W)$。解异或方程组也使用同一按位消元思路；系数不在 $\mathbb{F}_2$ 而在素数模域时改用 [9.2.001 高斯消元：唯一解、无解、多解](#92001)。

### 9.2.005 Simplex 单纯形

线性规划求 $\max c^Tx$ 使 $Ax\le b$、$x\ge0$。可行域是若干半空间的交；若有限最优值存在，就能在某个顶点取到。单纯形维护一个基可行解，每轮选一个可改进目标的非基变量入基，再按最小比值法则选基变量出基。退化换基时目标值可以不变，直到没有可改进方向才得到最优解。初始基未必可行，故先引入一个人工变量，第一阶段把它压到 $0$ 以找到可行基，再在第二阶段优化原目标。三种结局必须分清——不可行（第一阶段无法把人工变量压到 $0$）、无界（存在改进方向却没有正系数约束它）、最优。本实现按 Bland 规则选编号最小的可改进入基变量，最小比值相等时选编号最小的离基变量；精确算术下这能避免退化循环，浮点版仍由成员容差 `e` 决定何时视为相等。

```cpp
struct Simplex
{
    // m 是约束数，n 是变量数。
    int m, n;
    // b、z 分别记录各行的基变量和各列的非基变量编号。
    vector<int> b, z;
    // a 是单纯形表。
    vector<vector<long double>> a;
    // e 是浮点比较容差，linf 标记无界或不可行。
    static constexpr long double e = 1E-12L;
    static constexpr long double linf = numeric_limits<long double>::infinity();

    Simplex(const vector<vector<long double>> &A, const vector<long double> &B, const vector<long double> &C)
        : m(B.size()), n(C.size()), b(m), z(n + 1), a(m + 2, vector<long double>(n + 2))
    {
        // A、B 给出 Ax<=B，C 是最大化目标系数；隐含 x>=0。
        // 调试检查，可删。
        assert((int)A.size() == m);
        for (int i = 0; i < m; i++)
        {
            // 调试检查，可删。
            assert((int)A[i].size() == n);
            for (int j = 0; j < n; j++)
            {
                a[i][j] = A[i][j];
            }
        }
        for (int i = 0; i < m; i++)
        {
            b[i] = n + i;
            a[i][n] = -1;
            a[i][n + 1] = B[i];
        }
        for (int j = 0; j < n; j++)
        {
            z[j] = j;
            a[m][j] = -C[j];
        }
        z[n] = -1;
        a[m + 1][n] = 1;
    }

private:
    void pivot(int r, int s)
    {
        // r 是离基变量所在行，s 是入基变量所在列；用 a[r][s] 换基并更新整张表。
        long double v = 1 / a[r][s];
        for (int i = 0; i < m + 2; i++)
        {
            if (i != r)
            {
                for (int j = 0; j < n + 2; j++)
                {
                    if (j != s)
                    {
                        a[i][j] -= a[r][j] * a[i][s] * v;
                    }
                }
            }
        }
        for (int j = 0; j < n + 2; j++)
        {
            if (j != s)
            {
                a[r][j] *= v;
            }
        }
        for (int i = 0; i < m + 2; i++)
        {
            if (i != r)
            {
                a[i][s] *= -v;
            }
        }
        a[r][s] = v;
        swap(b[r], z[s]);
    }

    bool phase(int p)
    {
        // p=1 时找可行基，p=2 时优化原目标；返回 false 表示存在无界改进方向。
        int x = p == 1 ? m + 1 : m;
        while (true)
        {
            int s = -1;
            for (int j = 0; j <= n; j++)
            {
                if (z[j] == -1 || a[x][j] >= -e)
                {
                    continue;
                }
                if (s < 0 || z[j] < z[s])
                {
                    s = j;
                }
            }
            if (s < 0)
            {
                return true;
            }
            int r = -1;
            for (int i = 0; i < m; i++)
            {
                if (a[i][s] <= e)
                {
                    continue;
                }
                if (r < 0 || a[i][n + 1] / a[i][s] < a[r][n + 1] / a[r][s] - e ||
                    (fabsl(a[i][n + 1] / a[i][s] - a[r][n + 1] / a[r][s]) <= e && b[i] < b[r]))
                {
                    r = i;
                }
            }
            if (r < 0)
            {
                return false;
            }
            pivot(r, s);
        }
    }

public:
    pair<long double, vector<long double>> solve()
    {
        // 返回最优值与变量；不可行返回 -linf，无界返回 linf，二者的解向量均为空。
        if (m == 0)
        {
            for (int j = 0; j < n; j++)
            {
                if (a[m][j] < -e)
                {
                    return {linf, {}};
                }
            }
            return {0, vector<long double>(n)};
        }
        int r = min_element(a.begin(),
                            a.begin() + m,
                            [&](auto &x, auto &y)
                            {
                                return x[n + 1] < y[n + 1];
                            }) -
                a.begin();
        if (a[r][n + 1] < -e)
        {
            pivot(r, n);
            if (!phase(1) || a[m + 1][n + 1] < -e)
            {
                return {-linf, {}};
            }
            if (fabsl(a[m + 1][n + 1]) > e)
            {
                return {-linf, {}};
            }
            auto it = find(b.begin(), b.end(), -1);
            if (it != b.end())
            {
                r = it - b.begin();
                int s = 0;
                for (int j = 1; j <= n; j++)
                {
                    if (fabsl(a[r][j]) > fabsl(a[r][s]))
                    {
                        s = j;
                    }
                }
                if (fabsl(a[r][s]) > e)
                {
                    pivot(r, s);
                }
            }
        }
        if (!phase(2))
        {
            return {linf, {}};
        }
        vector<long double> x(n);
        for (int i = 0; i < m; i++)
        {
            if (b[i] < n)
            {
                x[b[i]] = a[i][n + 1];
            }
        }
        return {a[m][n + 1], x};
    }
};
```

以 `Simplex(A,B,C)` 构造（$A$ 是 $m\times n$ 约束系数、$B$ 是 $m$ 维右端、$C$ 是 $n$ 维目标系数，隐含 $x\ge0$），`solve()` 返回最优值与解向量：不可行时值为 $-\infty$、无界时为 $+\infty$，均配空向量。每次换基为 $O(mn)$，最坏换基次数是指数级，空间为 $O(mn)$。等式约束要拆成两个不等式；纯粹解方程用 [9.2.001 高斯消元：唯一解、无解、多解](#92001) 或 [9.2.002 高斯消元：列主元](#92002)；分式目标的恰选问题用 [9.2.006 0-1 分数规划](#92006)。

下面固定检查普通最优、第一阶段、不可行、无界，再用经典退化循环例检查 Bland 选元：

```cpp
signed main()
{
    auto [val, x] = Simplex({{1, 1}, {1, 0}, {0, 1}}, {4, 2, 3}, {3, 2}).solve();
    assert(fabsl(val - 10) < 1E-10L && x.size() == 2);
    auto [phaseOne, y] = Simplex({{-1}}, {-1}, {-1}).solve();
    assert(fabsl(phaseOne + 1) < 1E-10L && y.size() == 1);
    auto [infeasible, p] = Simplex({{1}, {-1}}, {0, -1}, {1}).solve();
    assert(isinf(infeasible) && infeasible < 0 && p.empty());
    auto [unbounded, q] = Simplex({}, {}, {1}).solve();
    assert(isinf(unbounded) && unbounded > 0 && q.empty());
    auto [degenerate, z] = Simplex({{0.5L, -5.5L, -2.5L, 9},
                                    {0.5L, -1.5L, -0.5L, 1},
                                    {1, 0, 0, 0}},
                                   {0, 0, 1}, {10, -57, -9, -24})
                               .solve();
    assert(fabsl(degenerate - 1) < 1E-10L && z.size() == 4);
    cout << llround(val) << ' ' << llround(phaseOne) << '\n'; // 10 -1
    return 0;
}
```

### 9.2.006 0-1 分数规划

要在若干物品里恰选 $k$ 个，最大化比值 $\dfrac{\sum a_i}{\sum b_i}$（$b_i>0$）。分式不能直接贪心，但可以参数化：比值 $\ge r$ 当且仅当 $\sum a_i-r\sum b_i=\sum(a_i-rb_i)\ge0$。固定 $r$ 后 $a_i-rb_i$ 就是每个物品的独立权重，取最大的 $k$ 个之和判它是否非负即可——这个判定关于 $r$ 单调，于是对 $r$ 二分。所有分母 $b_i$ 必须为正，否则单调性不成立。（另一路线是 Dinkelbach 迭代，收敛更快但实现更绕，这里用固定轮数二分足够。）

```cpp
namespace frac_prog
{
long double maxRatio(const vector<long double> &a, const vector<long double> &b, int k)
{
    // a、b 是分子分母贡献且 b[i] 为正，k 是恰选数量；返回最大比值。
    int n = a.size();
    // 调试检查，可删。
    assert((int)b.size() == n && 1 <= k && k <= n);
    long double l = numeric_limits<long double>::infinity();
    long double r = -l;
    for (int i = 0; i < n; i++)
    {
        // 调试检查，可删。
        assert(b[i] > 0);
        l = min(l, a[i] / b[i]);
        r = max(r, a[i] / b[i]);
    }
    vector<long double> c(n);
    for (int it = 0; it < 100; it++)
    {
        long double mid = (l + r) / 2;
        for (int i = 0; i < n; i++)
        {
            c[i] = a[i] - mid * b[i];
        }
        nth_element(c.begin(), c.end() - k, c.end());
        long double s = accumulate(c.end() - k, c.end(), 0.0L);
        if (s >= 0)
        {
            l = mid;
        }
        else
        {
            r = mid;
        }
    }
    return (l + r) / 2;
}
}
```

调用 `frac_prog::maxRatio(a,b,k)`，`a`、`b` 是各物品分子分母贡献（`b[i]>0`）、`k` 是恰选数量，返回最大比值。上下界取所有单项比值的最小值与最大值，因此最优值为负时也正确。固定 $100$ 轮二分，每轮用 `nth_element` 在线性期望时间取前 $k$ 大，总时间为 $O(n)$ 期望、空间为 $O(n)$。复杂可行性约束可把内层判定换成对应最优化；连续线性规划用 [9.2.005 Simplex 单纯形](#92005)。

下面用负最优值和普通正值各检查一次：

```cpp

{

    long double y = frac_prog::maxRatio({1, 8}, {1, 2}, 1);
    assert(fabsl(x + 3.5L) < 1E-10L && fabsl(y - 4) < 1E-10L);
    cout << fixed << setprecision(1) << x << ' ' << y << '\n'; // -3.5 4.0
    return 0;
}
```

### 9.2.007 追赶法解三对角/周期方程

三对角矩阵每行只有主对角 $b_i$ 与相邻的下对角 $a_i$、上对角 $c_i$ 三个非零元。对它做一般高斯消元是 $O(n^3)$ 的浪费，因为消去下对角后不会产生新的填充。追赶法（Thomas 算法）就是这个专门化：正向“追”一遍，用 $a_i/b_{i-1}$ 把每行下对角消掉、同步更新主对角与右端；再反向“赶”一遍回代求解，总共 $O(n)$。周期三对角在左下角 $A_{n-1,0}$ 与右上角 $A_{0,n-1}$ 多两个耦合元素，破坏了带状结构；这里用 Sherman-Morrison 把周期矩阵写成“三对角 $+$ 秩一修正”，解两次普通三对角方程（一次对原右端、一次对修正向量）再线性组合得到答案。所有主元 $b_{i-1}$ 消元时必须非零可除（对角占优时天然成立）。

```cpp
template <class T> optional<vector<T>> tridiagonal(vector<T> a, vector<T> b, vector<T> c, vector<T> d)
{
    // a、b、c 是下、主、上对角线，d 是右端；零主元时返回空。
    int n = b.size();
    // 调试检查，可删。
    assert((int)a.size() == n && (int)c.size() == n && (int)d.size() == n);
    if (n == 0)
    {
        return vector<T>{};
    }
    for (int i = 1; i < n; i++)
    {
        if (b[i - 1] == T(0))
        {
            return nullopt;
        }
        T q = a[i] / b[i - 1];
        b[i] -= q * c[i - 1];
        d[i] -= q * d[i - 1];
    }
    if (b[n - 1] == T(0))
    {
        return nullopt;
    }
    vector<T> x(n);
    x[n - 1] = d[n - 1] / b[n - 1];
    for (int i = n - 2; i >= 0; i--)
    {
        if (b[i] == T(0))
        {
            return nullopt;
        }
        x[i] = (d[i] - c[i] * x[i + 1]) / b[i];
    }
    return x;
}

template <class T>
optional<vector<T>> cyclicTri(vector<T> a, vector<T> b, vector<T> c, T tr, T bl, const vector<T> &d)
{
    // tr、bl 是右上角与左下角系数；追赶法或秩一修正失败时返回空。
    int n = b.size();
    // 调试检查，可删。
    assert(n >= 3);
    // 调试检查，可删。
    assert((int)a.size() == n && (int)c.size() == n && (int)d.size() == n);
    T g = b[0] == T(0) ? T(1) : -b[0];
    b[0] -= g;
    b[n - 1] -= tr * bl / g;
    vector<T> u(n);
    u[0] = g;
    u[n - 1] = bl;
    auto x = tridiagonal(a, b, c, d);
    auto z = tridiagonal(a, b, c, u);
    if (!x || !z)
    {
        return nullopt;
    }
    T den = T(1) + (*z)[0] + tr * (*z)[n - 1] / g;
    if (den == T(0))
    {
        return nullopt;
    }
    T q = ((*x)[0] + tr * (*x)[n - 1] / g) / den;
    for (int i = 0; i < n; i++)
    {
        (*x)[i] -= q * (*z)[i];
    }
    return x;
}
```

普通三对角方程调用 `tridiagonal(a,b,c,d)`，约定 `a[0]`、`c[n-1]` 不使用；空系统返回一个存在但长度为零的解。周期方程调用 `cyclicTri(a,b,c,tr,bl,d)`，要求 $n\ge3$。两者返回 `nullopt` 表示无主元交换的追赶过程失败，并不等价于原矩阵一定奇异。`T` 必须提供域上的除法；若 `T` 是整数，则每一步除法都必须整除，否则应改用有理数或模整数。两者均为 $O(n)$ 时间、$O(n)$ 空间。一般矩阵用 [9.2.002 高斯消元：列主元](#92002)；模意义下用 [9.2.001 高斯消元：唯一解、无解、多解](#92001)。

下面检查空系统、普通三对角、周期三对角与非零行列式但首主元为零的失败分支：

```cpp

{
    auto empty = tridiagonal<long double>({}, {}, {}, {});
    assert(empty && empty->empty());
    auto x = tridiagonal<long double>({0, -1, -1}, {2, 2, 2}, {-1, -1, 0}, {1, 0, 1});
    assert(x && *x == vector<long double>({1, 1, 1}));
    auto y = cyclicTri<long double>({0, -1, -1, -1}, {4, 4, 4, 4},
                                             {-1, -1, -1, 0}, -1, -1, {2, 2, 2, 2});
    assert(y);
    for (auto v : *y)
    {
        assert(fabsl(v - 1) < 1E-10L);
    }
    assert(!tridiagonal<long double>({0, 1}, {0, 1}, {1, 0}, {1, 2}));
    cout << llround((*x)[0]) << ' ' << llround((*y)[0]) << '\n'; // 1 1
    return 0;
}
```

### 9.2.008 Pfaffian

偶数阶斜对称矩阵 $A^T=-A$ 的 Pfaffian 满足 $\operatorname{pf}(A)^2=\det(A)$。对前两个下标选非零配对主元后，其余部分做一次斜对称 Schur 补，Pfaffian 就乘上该主元；若交换一个下标到主元位置，必须同步交换对应行列并让答案变号。重复成对消元即可在 $O(n^3)$ 内求值，不需要先算行列式再开平方。

```cpp
template <class T> T pfaffian(vector<vector<T>> a)
{
    // a 是偶数阶斜对称矩阵，T 是域；返回 Pfaffian。
    int n = a.size();
    // 调试检查，可删。
    assert(n % 2 == 0);
    // 调试检查，可删。
    for (int i = 0; i < n; i++)
    {
        assert((int)a[i].size() == n && a[i][i] == T(0));
        for (int j = 0; j < n; j++)
        {
            assert(a[i][j] == -a[j][i]);
        }
    }
    T ans = 1;
    for (int k = 0; k < n; k += 2)
    {
        int p = k + 1;
        while (p < n && a[k][p] == T(0))
        {
            p++;
        }
        if (p == n)
        {
            return T(0);
        }
        if (p != k + 1)
        {
            swap(a[p], a[k + 1]);
            for (int i = 0; i < n; i++)
            {
                swap(a[i][p], a[i][k + 1]);
            }
            ans = -ans;
        }
        T v = a[k][k + 1];
        ans *= v;
        for (int i = k + 2; i < n; i++)
        {
            for (int j = i + 1; j < n; j++)
            {
                a[i][j] -= (a[k][i] * a[k + 1][j] - a[k][j] * a[k + 1][i]) / v;
                a[j][i] = -a[i][j];
            }
        }
    }
    return ans;
}
```

`pfaffian(a)` 要求偶数阶斜对称矩阵，`T` 必须是可除的域；模意义下通常使用奇素数模的模整数。空矩阵返回空乘积 $1$。时间为 $O(n^3)$，按值传入的矩阵副本占 $O(n^2)$ 空间，除此之外只需 $O(1)$ 工作空间。Pfaffian 是带符号的配对和；不带符号的一般图完美匹配计数应使用 [9.1.012 Hafnian 与一般图完美匹配计数](#91012)，二者不能互换。

下面用 $4\times4$ 闭式 $a_{01}a_{23}-a_{02}a_{13}+a_{03}a_{12}$ 检查普通主元与交换主元：

```cpp
signed main()
{
    vector<vector<long double>> a{{0, 1, 2, 3}, {-1, 0, 4, 5}, {-2, -4, 0, 6}, {-3, -5, -6, 0}};
    vector<vector<long double>> b{{0, 0, 2, 3}, {0, 0, 4, 5}, {-2, -4, 0, 6}, {-3, -5, -6, 0}};
    assert(fabsl(pfaffian(vector<vector<long double>>{}) - 1) < 1E-10L);
    assert(fabsl(pfaffian(a) - 8) < 1E-10L);
    cout << llround(pfaffian(b)) << '\n'; // 2
    return 0;
}
```

### 9.2.009 对称矩阵 $LDL^T$ 分解

对称矩阵在不交换主元且各步枢轴非零时可分解为 $A=LDL^T$，其中 $L$ 是单位下三角矩阵、$D=\operatorname{diag}(d_0,\ldots,d_{n-1})$。比较等式两侧第 $(j,j)$ 与 $(i,j)$ 项可依次得到

$$
\begin{aligned}
d_j&=a_{jj}-\sum_{k<j}l_{jk}^2d_k,\\
l_{ij}&=\frac{a_{ij}-\sum_{k<j}l_{ik}d_kl_{jk}}{d_j}\qquad(i>j),
\end{aligned}
$$

因此按列递推即可原地确定全部因子。解 $Ax=b$ 时依次做单位下三角前代 $Ly=b$、对角求解 $Dz=y$ 和单位上三角回代 $L^Tx=z$。与 Cholesky 不同，这个分解允许 $D$ 中出现负数，因而能处理一部分不定矩阵；但当前版本不做对称主元交换，某个 $d_j$ 为零时即使 $A$ 可逆也会失败。

```cpp
struct LDLT
{
    // l 是单位下三角因子，满足 A=l*diag(d)*l^T；只使用其下三角部分。
    vector<vector<long double>> l;
    // d 是对角因子；每个元素都必须远离零枢轴。
    vector<long double> d;
    // eps 是判断对称误差和零枢轴的绝对容差。
    long double eps;
    // valid 表示构造时是否成功完成无主元交换的分解。
    bool valid;

    explicit LDLT(const vector<vector<long double>> &a, long double tol = 1E-12L)
        : l(a.size(), vector<long double>(a.size())), d(a.size()), eps(tol), valid(true)
    {
        // a 是方阵且应对称，tol 是非负的对称性与枢轴判定容差；保存 A 的 LDL^T 分解。
        assert(tol >= 0); // 调试检查，可删。
        int n = (int)a.size();
        for (int i = 0; i < n; i++)
        {
            assert((int)a[i].size() == n); // 调试检查，可删。
        }
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < n; j++)
            {
                assert(fabsl(a[i][j] - a[j][i]) <= eps); // 调试检查，可删。
            }
            l[i][i] = 1;
        }
        for (int j = 0; j < n && valid; j++)
        {
            long double diag = a[j][j];
            for (int k = 0; k < j; k++)
            {
                diag -= l[j][k] * l[j][k] * d[k];
            }
            if (fabsl(diag) <= eps)
            {
                valid = false;
                break;
            }
            d[j] = diag;
            for (int i = j + 1; i < n; i++)
            {
                long double value = a[i][j];
                for (int k = 0; k < j; k++)
                {
                    value -= l[i][k] * d[k] * l[j][k];
                }
                l[i][j] = value / d[j];
            }
        }
    }

    optional<vector<long double>> solve(const vector<long double> &b) const
    {
        // b 是长度等于矩阵阶数的右端向量；返回 Ax=b 的解，无效分解时返回空。
        assert((int)b.size() == (int)d.size()); // 调试检查，可删。
        if (!valid)
        {
            return nullopt;
        }
        int n = (int)d.size();
        vector<long double> y = b;
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < i; j++)
            {
                y[i] -= l[i][j] * y[j];
            }
        }
        vector<long double> z(n);
        for (int i = 0; i < n; i++)
        {
            z[i] = y[i] / d[i];
        }
        vector<long double> x = z;
        for (int i = n - 1; i >= 0; i--)
        {
            for (int j = i + 1; j < n; j++)
            {
                x[i] -= l[j][i] * x[j];
            }
        }
        return x;
    }
};
```

构造 `LDLT fact(a,e)` 后先检查 `fact.valid`，再可对多个右端重复调用 `fact.solve(b)`；输入必须是 $n\times n$ 对称矩阵，`e` 是同时用于对称性断言与零枢轴判断的绝对容差，调用者应按数据尺度选取。空矩阵有唯一空分解和空解；失败只说明当前顺序下无主元分解不可用，不等价于矩阵奇异，此时改用 [9.2.002 高斯消元：列主元](#92002)，病态对称系统则应使用带对称主元交换的 $LDL^T$ 或 QR。分解耗时 $O(n^3)$、空间 $O(n^2)$，每个右端的前代与回代耗时 $O(n^2)$、额外空间 $O(n)$。

下面用一个 $D$ 含负数的不定矩阵核对解 $(1,-2,3)$，并检查空矩阵和“矩阵可逆但首枢轴为零”的失败分支：

```cpp
signed main()
{
    LDLT fact({{4, 8, -4}, {8, 21, 7}, {-4, 7, 47}});
    auto x = fact.solve({-24, -13, 123});
    assert(fact.valid && x);
    assert(fabsl((*x)[0] - 1) < 1E-10L);
    assert(fabsl((*x)[1] + 2) < 1E-10L);
    assert(fabsl((*x)[2] - 3) < 1E-10L);
    auto empty = LDLT({}).solve({});
    assert(empty && empty->empty());
    LDLT zeroPivot({{0, 1}, {1, 0}});
    assert(!zeroPivot.valid && !zeroPivot.solve({1, 1}));
    cout << llround((*x)[0]) << ' ' << llround((*x)[1]) << ' ' << llround((*x)[2]) << '\n'; // 1 -2 3
    return 0;
}
```

## 9.3 数值计算

### 9.3.001 自适应 Simpson

Simpson 公式用一条抛物线拟合三点近似定积分：$\int_l^r f\approx\frac{r-l}{6}\big(f(l)+4f(\frac{l+r}{2})+f(r)\big)$，对三次以内多项式精确。自适应细分是它的实用化：把区间二分后左右两段的 Simpson 和 $S_{lr}$ 与整段 $S$ 相比；对足够光滑的函数，两半的总误差约为整段误差的 $1/16$，故差值 $S_{lr}-S$ 能估计误差。当 $|S_{lr}-S|\le15\varepsilon$ 时，用 $S_{lr}+\frac{S_{lr}-S}{15}$ 做 Richardson 外推；否则把误差预算 $\varepsilon$ 均分给两半递归。变化平缓处自然少分、剧烈处自动加密，代码中的 `err` 就是 $\varepsilon$。

```cpp
template <class F>
long double simpson(F f, long double l, long double r, long double err = 1E-12L, int dep = 24)
{
    // f 是被积函数，[l,r] 是区间，err 是绝对误差目标，dep 是递归上限；返回定积分近似值。
    // 调试检查，可删。
    assert(err > 0 && dep >= 0);
    auto simpson = [](long double ql, long double qr, long double vl, long double vm, long double vr)
    {
        // ql、qr 是端点，vl、vm、vr 是左端、中点、右端函数值；返回一次 Simpson 近似。
        return (vl + 4 * vm + vr) * (qr - ql) / 6;
    };
    long double m = (l + r) / 2;
    long double fl = f(l);
    long double fm = f(m);
    long double fr = f(r);
    auto dfs = [&](auto &&self,
                   long double ql,
                   long double qr,
                   long double vl,
                   long double vm,
                   long double vr,
                   long double whole,
                   long double tol,
                   int depth) -> long double
    {
        // self 是递归入口，whole 是当前段近似，tol 是误差预算，depth 是剩余层数；返回当前段积分值。
        long double mid = (ql + qr) / 2;
        long double x = (ql + mid) / 2;
        long double y = (mid + qr) / 2;
        long double fx = f(x);
        long double fy = f(y);
        long double a = simpson(ql, mid, vl, fx, vm);
        long double b = simpson(mid, qr, vm, fy, vr);
        if (!depth || fabsl(a + b - whole) <= 15 * tol)
        {
            return a + b + (a + b - whole) / 15;
        }
        return self(self, ql, mid, vl, fx, vm, a, tol / 2, depth - 1) +
               self(self, mid, qr, vm, fy, vr, b, tol / 2, depth - 1);
    };
    return dfs(dfs, l, r, fl, fm, fr, simpson(l, r, fl, fm, fr), err, dep);
}
```

调用 `simpson(f,l,r,err,dep)`，要求 `err>0`、`dep>=0`，且 `f` 在采样点处处有定义；对 $f(x)=x^4$ 在 $[0,1]$ 积分结果应在 $10^{-10}$ 内逼近 $0.2$。间断、尖峰或强振荡函数要在奇点处主动分段，`dep` 只是递归上限、并非误差保证。若实际细分出 $K$ 个区间，则有 $O(K)$ 次函数求值，递归栈不超过 `dep` 层。被积函数很光滑、想要更高阶收敛时用 [9.3.002 Romberg](#93002) 的外推；本条的优势在于对局部剧烈变化的函数按需加密、实现简短。

### 9.3.002 Romberg 积分

复合梯形公式的误差按步长有 $h^2,h^4,h^6,\dots$ 的偶次幂展开（Euler–Maclaurin），这正好适合 Richardson 外推逐阶消去。Romberg 先用梯形法在步长 $h,h/2,h/4,\dots$ 上算出第一列 $R_{i,0}$——第 $i$ 层复用上一层的值、只补新加入的中点，不重算旧点；再按 $R_{i,j}=\dfrac{4^{j}R_{i,j-1}-R_{i-1,j-1}}{4^{j}-1}$ 组合相邻两层，每做一次 $j$ 就消掉一阶误差，对光滑函数收敛极快。本实现只滚动保存一维数组，原地从高层往低层外推。

```cpp
template <class F> long double romberg(F f, long double l, long double r, int levels = 15)
{
    // f 是被积函数，[l,r] 是区间，levels 是外推层数；返回 Romberg 积分近似值。
    // 调试检查，可删。
    assert(1 <= levels && levels <= 25);
    vector<long double> a(levels);
    a[0] = (f(l) + f(r)) * (r - l) / 2;
    for (int i = 1; i < levels; i++)
    {
        unsigned long long cnt = 1ULL << i;
        long double h = (r - l) / cnt;
        long double sum = 0;
        for (unsigned long long j = 1; j < cnt; j += 2)
        {
            sum += f(l + j * h);
        }
        a[i] = a[i - 1] / 2 + h * sum;
        long double p = 4;
        for (int j = i - 1; j >= 0; j--)
        {
            a[j] = (p * a[j + 1] - a[j]) / (p - 1);
            p *= 4;
        }
    }
    return a[0];
}
```

调用 `romberg(f,l,r,levels)`，`levels` 每加 $1$ 采样点约翻倍，限制为 `1<=levels<=25`；最高层约有 $2^{24}$ 个小区间，继续放宽通常已没有竞赛实用性。外推因子使用 `long double`，不会发生整数连乘溢出。对 $f(x)=x^4$ 在 $[0,1]$ 应收敛到 $0.2$。若外推层数为 $L$，函数求值次数为 $O(2^L)$，空间为 $O(L)$。端点有可积奇异时须先变量代换；函数有间断或尖峰时用 [9.3.001 自适应 Simpson](#93001) 更合适。

### 9.3.003 Newton 多项式求根

Newton 迭代用当前点的切线零点逼近函数零点：把 $f$ 在 $x$ 处一阶泰勒展开 $f(x)+f'(x)\Delta\approx0$ 解出步长，得 $x\leftarrow x-\dfrac{f(x)}{f'(x)}$。在单根附近它二次收敛（有效位数每步约翻倍），但对任意初值并不保证收敛：初值离根远、或迭代路过导数为零的驻点时步长会爆炸甚至发散。本实现把零导数与非有限数判为失败，用相对位移 $|y-x|\le\varepsilon\max(1,|y|)$ 判收敛；耗尽迭代上限仍未满足该条件就返回空。这里不对导数或函数值使用固定绝对阈值，因而把 $f$ 整体乘一个很小的非零常数不会改变判定。

```cpp
template <class F, class G> optional<long double> newton(F f, G df, long double x, int it = 80)
{
    // f、df 是函数及导数，x 是初值，it 是迭代上限；收敛时返回一个根，导数为零或发散返回空。
    // 调试检查，可删。
    assert(it >= 0);
    for (int i = 0; i < it; i++)
    {
        long double d = df(x);
        if (d == 0 || !isfinite(d))
        {
            return nullopt;
        }
        long double y = x - f(x) / d;
        if (!isfinite(y))
        {
            return nullopt;
        }
        if (fabsl(y - x) <= eps * max(1.L, fabsl(y)))
        {
            return y;
        }
        x = y;
    }
    return nullopt;
}
```

调用 `newton(f,df,x0)` 传函数、导数与初值，要求 `it>=0`；返回空表示当前初值未能可靠收敛。若实际迭代 $I$ 轮，就有 $O(I)$ 次函数与导数求值和 $O(1)$ 空间；二次收敛阶不等于最坏迭代次数保证。没有好初值但已知一个异号区间 $[l,r]$（$f(l)f(r)<0$）时，用有界二分更稳、必定收敛；专门求实系数三次方程全部实根时用 [9.3.004 一元三次方程](#93004)，它先用导数临界点定位单调段再二分，不依赖初值。

### 9.3.004 一元三次方程

求实系数三次方程 $ax^3+bx^2+cx+d=0$ 的全部互异实根，思路是“分段单调 $+$ 二分”。次数按归一化前的精确零判断；随后把系数除以最大绝对值，要求原来非零的最高有效系数归一化后仍非零，这一比例范围取决于平台的 `long double`。三次函数的导数至多两个实根，它们把实轴切成至多三段单调区间，再由 Cauchy 根界包住全部实根并逐段二分。

```cpp
optional<vector<long double>> cubicRoots(long double a, long double b, long double c, long double d)
{
    // a、b、c、d 是有限实系数；返回升序互异实根，恒等式返回空 optional，无实根返回存在的空向量。
    // 调试检查，可删。
    assert(isfinite(a) && isfinite(b) && isfinite(c) && isfinite(d));
    bool hasA = a != 0;
    bool hasB = b != 0;
    bool hasC = c != 0;
    long double sc = max({fabsl(a), fabsl(b), fabsl(c), fabsl(d)});
    if (sc == 0)
    {
        return nullopt;
    }
    a /= sc;
    b /= sc;
    c /= sc;
    d /= sc;
    assert((!hasA || a != 0) && (!hasB || b != 0) && (!hasC || c != 0)); // 调试检查，可删。
    if (!hasA)
    {
        if (!hasB)
        {
            if (!hasC)
            {
                return vector<long double>{};
            }
            return vector<long double>{-d / c};
        }
        long double det = c * c - 4 * b * d;
        if (det < -eps)
        {
            return vector<long double>{};
        }
        det = max(0.L, det);
        if (det == 0)
        {
            return vector<long double>{-c / (2 * b)};
        }
        long double s = sqrtl(det);
        long double q = -0.5L * (c + copysignl(s, c));
        vector<long double> ans{q / b, d / q};
        sort(ans.begin(), ans.end());
        if (ans[0] == ans[1])
        {
            ans.pop_back();
        }
        return ans;
    }
    auto f = [&](long double x)
    {
        return ((a * x + b) * x + c) * x + d;
    };
    long double lim = 1 + max({fabsl(b), fabsl(c), fabsl(d)}) / fabsl(a);
    vector<long double> key{-lim, lim};
    long double det = b * b - 3 * a * c;
    if (det >= -eps)
    {
        det = sqrtl(max(0.L, det));
        long double x = (-b - det) / (3 * a);
        long double y = (-b + det) / (3 * a);
        if (-lim < x && x < lim)
        {
            key.push_back(x);
        }
        if (-lim < y && y < lim)
        {
            key.push_back(y);
        }
    }
    sort(key.begin(), key.end());
    vector<long double> ans;
    for (auto x : key)
    {
        if (fabsl(f(x)) <= eps)
        {
            ans.push_back(x);
        }
    }
    for (int i = 0; i + 1 < (int)key.size(); i++)
    {
        long double l = key[i];
        long double r = key[i + 1];
        long double fl = f(l);
        long double fr = f(r);
        if (fl == 0 || fr == 0 || (fl < 0) == (fr < 0))
        {
            continue;
        }
        for (int t = 0; t < 100; t++)
        {
            long double m = (l + r) / 2;
            long double fm = f(m);
            if ((fl < 0) == (fm < 0))
            {
                l = m;
                fl = fm;
            }
            else
            {
                r = m;
                fr = fm;
            }
        }
        ans.push_back((l + r) / 2);
    }
    sort(ans.begin(), ans.end());
    ans.erase(unique(ans.begin(),
                     ans.end(),
                     [&](auto x, auto y)
                     {
                         return fabsl(x - y) <= 100 * eps * max({1.L, fabsl(x), fabsl(y)});
                     }),
              ans.end());
    return ans;
}
```

调用 `cubicRoots(a,b,c,d)`：四个系数必须有限，且原来非零的最高有效系数除以最大绝对值后仍可表示为非零；MSVC 的 `long double` 通常等同 `double`，范围小于 GCC x86 的扩展精度。返回 `nullopt` 表示恒等式，存在但为空的向量表示没有实根，否则返回升序互异实根。原始系数精确零只用于次数分流，归一化判别式用相对其各项量级的容差，根去重使用相对根大小的 `100*eps`，三者不是同一零判断。内部二分固定 $100$ 轮，时间、空间均为 $O(1)$。

下面检查三个实根、退化二次、无实根、重根与恒等式：

```cpp
signed main()
{
    auto a = cubicRoots(1, -6, 11, -6);
    assert(a && a->size() == 3);
    auto b = cubicRoots(0, -1, 3, -2);
    assert(b && b->size() == 2 && fabsl((*b)[0] - 1) < 1E-10L && fabsl((*b)[1] - 2) < 1E-10L);
    auto wide = cubicRoots(0, 1, 1E16L, 1);
    assert(wide && wide->size() == 2 && fabsl((*wide)[1] + 1E-16L) < 1E-20L);
    auto none = cubicRoots(0, 1, 0, 1);
    assert(none && none->empty());
    auto repeated = cubicRoots(1, 0, 0, 0);
    assert(repeated && repeated->size() == 1 && fabsl((*repeated)[0]) < 1E-10L);
    assert(!cubicRoots(0, 0, 0, 0));
    cout << llround((*a)[0]) << ' ' << llround((*a)[1]) << ' ' << llround((*a)[2]) << '\n'; // 1 2 3
    return 0;
}
```

### 9.3.005 线性插值与幂和

很多计数题的答案关于某个变量 $x$ 是次数已知的多项式（典型如幂和 $\sum_{i=1}^x i^k$ 是 $x$ 的 $k+1$ 次多项式），却只能算出前若干个点值。给定 $f(0),f(1),\dots,f(n-1)$ 这 $n$ 个等距样本，$f$ 的次数小于 $n$，就能用拉格朗日插值唯一恢复 $f(x)$。等距点让公式大幅简化：$f(x)=\sum_i y_i\prod_{j\ne i}\dfrac{x-j}{i-j}$ 中，分母 $\prod_{j\ne i}(i-j)=(-1)^{n-1-i}\,i!\,(n-1-i)!$ 可用阶乘逆元 $O(1)$ 得到，分子的连乘用前缀积 `pre` 与后缀积 `suf` 拆开，于是整体 $O(n)$。模数须为素数以保证阶乘可逆。`interpSeq` 里 $x<n$ 直接返回样本值。

```cpp
inline long long powerMod(long long a, long long b, long long mod)
{
    // a 是底数，b 是非负指数，mod 是正模数；返回模幂。
    // 调试检查，可删。
    assert(b >= 0 && mod > 0);
    a %= mod;
    if (a < 0)
    {
        a += mod;
    }
    long long ans = 1 % mod;
    while (b)
    {
        if (b & 1)
        {
            ans = linearMulMod(ans, a, mod);
        }
        a = linearMulMod(a, a, mod);
        b >>= 1;
    }
    return ans;
}

inline long long interpSeq(const vector<long long> &y, long long x, long long mod)
{
    // y[i]=f(i)，x 是查询点，mod 是素数；返回 f(x) 对 mod 的最小非负余数。
    int n = y.size();
    // 调试检查，可删。
    assert(n > 0 && n < mod);
    x %= mod;
    if (x < 0)
    {
        x += mod;
    }
    if (x < n)
    {
        long long ans = y[x] % mod;
        return ans < 0 ? ans + mod : ans;
    }
    vector<long long> fac(n, 1), ifac(n, 1), pre(n + 1, 1), suf(n + 1, 1);
    for (int i = 1; i < n; ++i)
    {
        fac[i] = linearMulMod(fac[i - 1], i, mod);
    }
    ifac[n - 1] = powerMod(fac[n - 1], mod - 2, mod);
    for (int i = n - 1; i; --i)
    {
        ifac[i - 1] = linearMulMod(ifac[i], i, mod);
    }
    for (int i = 0; i < n; ++i)
    {
        long long d = x >= i ? x - i : mod - (i - x);
        pre[i + 1] = linearMulMod(pre[i], d, mod);
    }
    for (int i = n - 1; i >= 0; --i)
    {
        long long d = x >= i ? x - i : mod - (i - x);
        suf[i] = linearMulMod(suf[i + 1], d, mod);
    }
    long long ans = 0;
    for (int i = 0; i < n; ++i)
    {
        long long v = y[i] % mod;
        if (v < 0)
        {
            v += mod;
        }
        long long term = linearMulMod(v, pre[i], mod);
        term = linearMulMod(term, suf[i + 1], mod);
        term = linearMulMod(term, ifac[i], mod);
        term = linearMulMod(term, ifac[n - 1 - i], mod);
        if ((n - 1 - i) & 1)
        {
            term = term ? mod - term : 0;
        }
        ans = ans >= mod - term ? ans - (mod - term) : ans + term;
    }
    return ans;
}
```

调用 `interpSeq(values,x,mod)`，要求样本非空、$f$ 次数小于样本数、`mod` 为素数且 `values.size()<mod`；空样本不能唯一确定多项式，属于不满足前提。`x` 和样本值可以为负，函数统一返回最小非负余数；模乘统一调用 `linearMulMod`，常用小模数只做 64 位乘法，大模数才局部提升，末尾模加用减法比较避免溢出。若模数为 $M$，预处理与单次求值为 $O(n+\log M)$ 时间、$O(n)$ 空间。求幂和 $\sum_{i=0}^{x}i^k$ 时先算 $k+2$ 个前缀样本再插值；非等距样本改用一般拉格朗日插值或第 10 章的多点求值。

下面用 $f(x)=x^2$ 检查区间外正查询和负查询，再检查负样本的直接返回：

```cpp
signed main()
{
    assert(interpSeq({0, 1, 4, 9}, 5, 998244353) == 25);
    constexpr long long mod = 9'223'372'036'854'775'783LL;
    assert(interpSeq({0, 1, 4}, mod - 1, mod) == 1);
    assert(interpSeq({0, 1, 4}, -1, 7) == 1);
    assert(interpSeq({0, -1}, 1, 5) == 4);
    cout << interpSeq({0, 1, 4, 9}, 5, 998244353) << '\n'; // 25
    return 0;
}
```

### 9.3.006 指数乘多项式有限和

给定次数不超过 $d$ 的多项式 $f$ 在 $0,1,\dots,d$ 的值，求 $\sum_{i=0}^{n-1}r^if(i)$。当 $r=1$ 时，前缀和本身是 $d+1$ 次多项式，算出前 $d+2$ 个值后等距插值即可；当 $r\ne0,1$ 时，先求形式无穷和常数 $C=\sum_{i\ge0}r^if(i)$，则 $r^{-n}(S_n-C)$ 是关于 $n$ 的 $d$ 次多项式，同样由前 $d+2$ 个值插值得到。`r=0` 必须单独处理，因为一般分支会请求它的逆元。

```cpp
template <class T> T fieldPow(T a, unsigned long long b)
{
    // a 是域元素，b 是非负指数；返回 a 的 b 次幂。
    T ans = 1;
    while (b)
    {
        if (b & 1)
        {
            ans *= a;
        }
        b >>= 1;
        if (b)
        {
            a *= a;
        }
    }
    return ans;
}

template <class T> T interpIota(const vector<T> &y, long long x)
{
    // y[i] 是多项式在 i 处的值，x 是非负查询点；返回该多项式在 x 处的值。
    int n = y.size();
    // 调试检查，可删。
    assert(n > 0 && x >= 0);
    if (x < n)
    {
        return y[x];
    }
    vector<T> fac(n, T(1)), ifac(n, T(1));
    vector<T> pre(n + 1, T(1)), suf(n + 1, T(1));
    for (int i = 1; i < n; i++)
    {
        fac[i] = fac[i - 1] * T(i);
    }
    ifac[n - 1] = T(1) / fac[n - 1];
    for (int i = n - 1; i; i--)
    {
        ifac[i - 1] = ifac[i] * T(i);
    }
    for (int i = 0; i < n; i++)
    {
        pre[i + 1] = pre[i] * T(x - i);
    }
    for (int i = n - 1; i >= 0; i--)
    {
        suf[i] = suf[i + 1] * T(x - i);
    }
    T ans = 0;
    for (int i = 0; i < n; i++)
    {
        T v = y[i] * pre[i] * suf[i + 1] * ifac[i] * ifac[n - 1 - i];
        ans += (n - 1 - i) & 1 ? -v : v;
    }
    return ans;
}

template <class T> T sumExpPLim(T r, vector<T> f)
{
    // f[i] 是多项式在 i 处的值且 r!=1；返回生成函数给出的形式无穷和。
    // 调试检查，可删。
    assert(!f.empty() && r != T(1));
    T q = r / (T(1) - r);
    T term = T(1) / (T(1) - r);
    T ans = 0;
    for (int k = 0; k < (int)f.size(); k++)
    {
        ans += f[0] * term;
        for (int i = 0; i + 1 < (int)f.size() - k; i++)
        {
            f[i] = f[i + 1] - f[i];
        }
        term *= q;
    }
    return ans;
}

template <class T> T sumExpPoly(T r, const vector<T> &f, long long n)
{
    // f[i] 是多项式在 i 处的值，n 是项数；返回前 n 项的指数加权和。
    // 调试检查，可删。
    assert(!f.empty() && n >= 0);
    if (n == 0)
    {
        return T(0);
    }
    if (r == T(0))
    {
        return f[0];
    }
    int d = f.size() - 1;
    vector<T> s(d + 2);
    T pw = 1;
    for (int i = 0; i <= d; i++)
    {
        s[i + 1] = s[i] + pw * f[i];
        pw *= r;
    }
    if (r == T(1))
    {
        return interpIota(s, n);
    }
    T limit = sumExpPLim(r, f);
    T inv = T(1) / r;
    T ipw = 1;
    for (int i = 0; i <= d + 1; i++)
    {
        s[i] = (s[i] - limit) * ipw;
        ipw *= inv;
    }
    return limit + fieldPow(r, (unsigned long long)n) * interpIota(s, n);
}
```

`sumExpPoly(r,f,n)` 要求 `f[i]` 是次数小于 `f.size()` 的多项式在连续整数点的值，`n>=0`。`T` 必须是域，且 $1,2,\dots,d+1$ 可逆；素数模下通常要求 `d+1<mod`。有限差分占 $O(d^2)$ 时间，其他步骤为 $O(d+\log n)$，空间为 $O(d)$。若已有快速卷积，可用二项式变换加速各阶差分的计算；普通竞赛次数下本版更短。

下面覆盖 `r=0`、`r=1` 与一般分支：

```cpp
signed main()
{
    assert(fabsl(sumExpPoly(2.L, vector<long double>{0, 1}, 4) - 34) < 1E-10L);
    assert(fabsl(sumExpPoly(1.L, vector<long double>{0, 1}, 5) - 10) < 1E-10L);
    assert(fabsl(sumExpPoly(0.L, vector<long double>{7, 8}, 100) - 7) < 1E-10L);
    cout << llround(sumExpPoly(2.L, vector<long double>{0, 1}, 4)) << '\n'; // 34
    return 0;
}
```

### 9.3.007 指数乘多项式无穷和

把 $f(i)$ 写成 Newton 展开 $f(i)=\sum_k\Delta^kf(0)\binom{i}{k}$，再用 $\sum_{i\ge0}r^i\binom{i}{k}=\dfrac{r^k}{(1-r)^{k+1}}$，即可得到上条 `sumExpPLim(r,f)`。调用时 `f[i]` 必须是次数不超过 $d$ 的多项式在 $i=0,1,\dots,d$ 处的值，`r=1` 非法，`r=0` 合法且结果为 `f(0)`。这里的“无穷和”是在域上按生成函数得到的有理式；当 `T` 是实数时，只有 $|r|<1$ 才同时等于通常意义下的收敛级数。时间为 $O(d^2)$，空间为 $O(d)$；实现与有限和共享，故不重复打印。

```cpp
signed main()
{
    long double a = sumExpPLim(0.5L, vector<long double>{1});
    long double b = sumExpPLim(0.5L, vector<long double>{0, 1});
    assert(fabsl(a - 2) < 1E-10L && fabsl(b - 2) < 1E-10L);
    assert(fabsl(sumExpPLim(0.L, vector<long double>{7, 8}) - 7) < 1E-10L);
    cout << fixed << setprecision(1) << a << ' ' << b << '\n'; // 2.0 2.0
    return 0;
}
```
