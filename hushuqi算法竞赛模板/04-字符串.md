## 4.1 基础匹配与哈希

### 4.1.001 字符串哈希（Mersenne 单模、随机底数）

把长度 $n$ 的串预处理成前缀哈希 $h[i]=\sum_{j<i}(s_j+1)base^{\,i-1-j}\bmod P$ 后，子串 $s[l,r)$ 的哈希就是 $h[r]-h[l]base^{r-l}$。这里取 Mersenne 素数 $P=2^{61}-1$：两个代表元相乘先放入 `__uint128_t`，再利用 $2^{61}\equiv1\pmod P$ 把高 61 位与低 61 位相加约简，避免普通 `%`。`HASH_BASE` 在进程启动时从 $[P/2,P-2]$ 随机一次，所有对象共享它；因此固定测试数据不能预先针对一个公开底数构造碰撞。对长度不超过 $n$ 的两个不同等长串，其差是次数至多 $n-1$ 的非零多项式，在当前候选集合上碰撞概率至多约 $2(n-1)/P$，但哈希判等仍不是确定性证明。

```cpp
struct HashValue
{
    // x 是模 2^61-1 的代表元；无符号 64 位与 128 位乘积是 Mersenne 快速约简所必需的。
    static constexpr unsigned long long mod = (1ULL << 61) - 1;
    unsigned long long x = 0;

    HashValue(unsigned long long x = 0) : x((x & mod) + (x >> 61))
    {
        // x 是待规范化整数；构造模 mod 的哈希值。
        if (this->x >= mod)
        {
            this->x -= mod;
        }
    }

    friend HashValue operator+(HashValue a, HashValue b)
    {
        // a、b 是同模哈希值；返回模 mod 的和。
        unsigned long long x = a.x + b.x;
        if (x >= mod)
        {
            x -= mod;
        }
        return x;
    }

    friend HashValue operator-(HashValue a, HashValue b)
    {
        // a、b 是同模哈希值；返回模 mod 的差。
        unsigned long long x = a.x + mod - b.x;
        if (x >= mod)
        {
            x -= mod;
        }
        return x;
    }

    friend HashValue operator*(HashValue a, HashValue b)
    {
        // a、b 是同模哈希值；用 128 位乘积返回模 mod 的积。
        __uint128_t x = (__uint128_t)a.x * b.x;
        return (unsigned long long)(x >> 61) + (unsigned long long)(x & mod);
    }

    friend bool operator==(HashValue, HashValue) = default;

    friend ostream &operator<<(ostream &out, HashValue a)
    {
        // out 是输出流，a 是哈希值；输出 a 的规范代表元并返回 out。
        return out << a.x;
    }
};

inline HashValue hashBase()
{
    // 无参数；返回本进程随机选定的大底数。
    unsigned long long seed = (unsigned long long)chrono::steady_clock::now().time_since_epoch().count();
    seed ^= (unsigned long long)random_device{}() << 32;
    seed ^= random_device{}();
    mt19937_64 mt(seed);
    uniform_int_distribution<unsigned long long> dist(HashValue::mod / 2, HashValue::mod - 2);
    return dist(mt);
}

inline const HashValue HASH_BASE = hashBase();

struct HashBase
{
    // pw[i] 是 HASH_BASE 的 i 次幂，按实际需要延长。
    vector<HashValue> pw{HashValue(1)};

    void ensure(int n)
    {
        // n 是需要支持的最大字符串长度；保证 pw[0..n] 已计算。
        while ((int)pw.size() <= n)
        {
            pw.push_back(pw.back() * HASH_BASE);
        }
    }
};

inline HashBase HASH_POWER;

struct DynStrHash
{
    // pre[i] 是当前串前 i 个字符的哈希。
    vector<HashValue> pre{HashValue(0)};

    int size() const
    {
        // 无参数；返回当前字符串长度。
        return pre.size() - 1;
    }

    void push_back(char c)
    {
        // c 是追加字符；在串尾加入 c。
        HASH_POWER.ensure(size() + 1);
        pre.push_back(pre.back() * HASH_BASE + HashValue((unsigned char)c + 1));
    }

    HashValue get(int l, int r) const
    {
        // l、r 是当前串下标；返回半开子串 [l,r) 的哈希值。
        assert(0 <= l && l <= r && r <= size()); // 调试检查，可删。
        return pre[r] - pre[l] * HASH_POWER.pw[r - l];
    }

    HashValue concat(HashValue value, int l, int r) const
    {
        // value 是已有串的哈希，l、r 指定当前串的半开子串；返回两段拼接后的哈希。
        return value * HASH_POWER.pw[r - l] + get(l, r);
    }
};

struct StrHash
{
    // pre[i] 是原串前 i 个字符的哈希，rev[i] 是反转串前 i 个字符的哈希。
    vector<HashValue> pre, rev;

    StrHash(const string &s) : pre(s.size() + 1), rev(s.size() + 1)
    {
        // s 是原字符串；预处理正向、反向前缀哈希与所需底数幂。
        int n = s.size();
        HASH_POWER.ensure(n);
        for (int i = 0; i < n; i++)
        {
            pre[i + 1] = pre[i] * HASH_BASE + HashValue((unsigned char)s[i] + 1);
            rev[i + 1] = rev[i] * HASH_BASE + HashValue((unsigned char)s[n - 1 - i] + 1);
        }
    }

    int size() const
    {
        // 无参数；返回原字符串长度。
        return pre.size() - 1;
    }

    HashValue get(int l, int r) const
    {
        // l、r 是原串下标；返回半开子串 [l,r) 的正向哈希。
        assert(0 <= l && l <= r && r <= size()); // 调试检查，可删。
        return pre[r] - pre[l] * HASH_POWER.pw[r - l];
    }

    HashValue getrev(int l, int r) const
    {
        // l、r 是原串下标；返回反转 s[l,r) 后的哈希。
        assert(0 <= l && l <= r && r <= size()); // 调试检查，可删。
        int n = size();
        return rev[n - l] - rev[n - r] * HASH_POWER.pw[r - l];
    }

    HashValue concat(HashValue value, int l, int r) const
    {
        // value 是已有串的哈希，l、r 指定原串半开子串；返回两段拼接后的哈希。
        return value * HASH_POWER.pw[r - l] + get(l, r);
    }

    HashValue concatrev(HashValue value, int l, int r) const
    {
        // value 是已有串的哈希，l、r 指定原串半开子串；返回拼接反转子串后的哈希。
        return value * HASH_POWER.pw[r - l] + getrev(l, r);
    }

    static vector<int> match(const string &s, const string &t)
    {
        // s 是文本串，t 是模式串；返回 t 的全部匹配起点，按下标递增排列。
        if (t.size() > s.size())
        {
            return {};
        }
        StrHash a(s), b(t);
        HashValue target = b.get(0, t.size());
        vector<int> ans;
        for (int i = 0; i + (int)t.size() <= (int)s.size(); i++)
        {
            if (a.get(i, i + t.size()) == target)
            {
                ans.push_back(i);
            }
        }
        return ans;
    }
};
```

构造 `StrHash a(s)` 后，`get(l,r)`、`getrev(l,r)` 分别取半开区间 $[l,r)$ 的正向哈希与反转哈希，`concat`、`concatrev` 把对应子串接到已有哈希后；`DynStrHash` 支持逐字符 `push_back` 和同样的正向查询。幂表由全部对象共享并按最大实际长度增长，不要求预设 `N`。构造与追加总计 $O(n)$，子串哈希和拼接 $O(1)$，`match(s,t)` 为 $O(n+m)$，空间 $O(n+m)$；单模静态版只保存正反两组前缀与一份共享幂表，百万字符约占 24 MB，旧双模正向实现的四个整数数组约占 32 MB。Windows x86-64、GCC 14.2、`-O2` 下，对随机百万字符冷构造并执行 200 万组最长 1000 的等长子串比较，7 轮中位数从旧版 `8.029 ms + 100.425 ms` 降至 `7.437 ms + 55.973 ms`；新版构造还同时得到旧版没有的反向前缀，该结果只代表当前机器。要确定性线性匹配改用 [4.1.002 KMP / 前缀函数](#41002) 或 [4.1.003 扩展 KMP / Z 函数](#41003)；滑窗用法单列为 [4.1.008 Karp-Rabin](#41008)，只找首次出现见 [4.1.010 `strstr` 类匹配](#41010)。

比赛现场只需要静态子串判等时，可改抄下面的 `ShortHash`。它保留固定底数、$2^{61}-1$ 单模、`__uint128_t` 乘积和 Mersenne 快速约简，只去掉值类型运算符、反向前缀、动态追加、拼接和模式匹配；原版接口仍用于需要这些能力的题目。固定底数不具备运行时随机化，不能抵御针对该常数构造的碰撞数据。

```cpp
struct ShortHash
{
    static constexpr unsigned long long M = (1ULL << 61) - 1;
    // B 从 [M / 2, M - 2] 中选取并固定，避免现场引入随机设备。
    static constexpr unsigned long long B = 1234567890123456789ULL;
    inline static vector<unsigned long long> p{1}; // p[i] 是 B 的 i 次幂。
    vector<unsigned long long> h;                  // h[i] 是原串前 i 个字符的哈希。

    static unsigned long long red(unsigned long long x)
    {
        // x 小于 2M；返回 x 模 M 的规范代表元。
        x = (x & M) + (x >> 61);
        return x >= M ? x - M : x;
    }

    static unsigned long long mul(unsigned long long a, unsigned long long b)
    {
        // a、b 是模 M 代表元；返回二者乘积模 M。
        __uint128_t x = (__uint128_t)a * b;
        return red((unsigned long long)(x >> 61) + (unsigned long long)(x & M));
    }

    ShortHash(string_view s) : h(s.size() + 1)
    {
        // s 是原字符串；预处理前缀哈希和所需底数幂。
        while (p.size() <= s.size())
        {
            p.push_back(mul(p.back(), B));
        }
        for (int i = 0; i < (int)s.size(); i++)
        {
            h[i + 1] = red(mul(h[i], B) + (unsigned char)s[i] + 1);
        }
    }

    unsigned long long get(int l, int r) const
    {
        // l、r 是原串下标；返回半开子串 [l,r) 的哈希。
        return red(h[r] + M - mul(h[l], p[r - l]));
    }
};
```

构造 `ShortHash h(s)` 后用 `h.get(l,r)` 查询 $s[l,r)$。需要反向查询时另构造 `string t(s.rbegin(),s.rend()); ShortHash rev(t);`，则反转 $s[l,r)$ 的哈希为 `rev.get(n-r,n-l)`。构造为 $O(n)$，查询为 $O(1)$，每次查询仍只做一次 128 位乘法与 M61 约简；短版每个字符串只保存一组前缀，额外空间为 $O(n)$。底数固定在 $[M/2,M-2]$ 的选定常数，哈希仍是概率判定；需要抵御构造碰撞时使用上面的运行时随机底数版本。性能数据只代表对应机器与数据分布。

### 4.1.002 KMP / 前缀函数

前缀函数 `pi[i]` 记录 $s[0..i]$ 的最长真 border（既是前缀又是后缀的最长长度），求它时若 `s[i]` 与当前候选位 `s[j]` 失配就沿 `pi` 链跳到次长 border 继续试，均摊线性。有了模式串的前缀函数，KMP 扫描文本时文本指针从不回退：失配就让模式指针沿 `pi` 回跳，匹配到串尾记录一次出现再回跳继续找重叠出现。`borders` 顺着 `pi.back()` 链枚举全部 border，`minPeriod` 用「最短周期 $=n-\pi_{n-1}$、当它整除 $n$ 才是真整周期」判断循环节。

```cpp
vector<int> prefix(const string &s)
{
    // s 是原字符串；返回前缀函数，p[i] 是 s[0..i] 的最长真 border 长度。
    vector<int> p(s.size());
    for (int i = 1; i < (int)s.size(); i++)
    {
        int j = p[i - 1];
        while (j && s[i] != s[j])
        {
            j = p[j - 1];
        }
        if (s[i] == s[j])
        {
            j++;
        }
        p[i] = j;
    }
    return p;
}

vector<int> kmp(const string &s, const string &t)
{
    // s 是文本串，t 是模式串；返回全部匹配起点，按下标递增排列。
    if (t.empty())
    {
        vector<int> a(s.size() + 1);
        iota(a.begin(), a.end(), 0);
        return a;
    }
    auto p = prefix(t);
    vector<int> ans;
    int j = 0;
    for (int i = 0; i < (int)s.size(); i++)
    {
        while (j && s[i] != t[j])
        {
            j = p[j - 1];
        }
        if (s[i] == t[j])
        {
            j++;
        }
        if (j == (int)t.size())
        {
            ans.push_back(i - j + 1);
            j = p[j - 1];
        }
    }
    return ans;
}

vector<int> borders(const string &s)
{
    // s 是原字符串；返回全部非空真 border 的长度，按递增排列。
    auto p = prefix(s);
    vector<int> a;
    int x = s.empty() ? 0 : p.back();
    while (x)
    {
        a.push_back(x);
        x = p[x - 1];
    }
    reverse(a.begin(), a.end());
    return a;
}

int minPeriod(const string &s)
{
    // s 是原字符串；返回最短整周期长度，空串返回 0。
    if (s.empty())
    {
        return 0;
    }
    auto p = prefix(s);
    int x = s.size() - p.back();
    return (int)s.size() % x == 0 ? x : (int)s.size();
}
```

`prefix(s)` 返回前缀函数表；`kmp(s,t)` 返回 `t` 在 `s` 的全部出现起点（空模式返回 $0,1,\ldots,n$）；`borders(s)` 按长度递增给出全部非空真 border，不包含整串；`minPeriod(s)` 返回最短整周期，非周期串返回 $n$。四者均为线性时间和线性空间。相比 [4.1.001 字符串哈希](#41001) 的概率匹配，KMP 确定性正确；只关心某后缀与前缀的匹配长度时 [4.1.003 扩展 KMP / Z 函数](#41003) 更直接。周期、border、循环节的汇总应用另见 [4.5.006 字符串周期、循环节与前后缀](#45006)，它直接复用这里的前缀函数。

需要在自动机或数位 DP 中反复读取字符时，可把同一回退过程预计算成 DFA。`KMPAutomaton<A,F>` 的状态 $0\ldots |p|$ 表示当前后缀与模式 `pattern` 的最长匹配长度；完整匹配状态读入新字符时先沿 `pi` 回退，因此重叠出现不会丢失。字符集必须是连续区间 $[F,F+A)$，空模式也有合法的唯一状态。

```cpp
template <int A = 26, char F = 'a'> struct KMPAutomaton
{
    // pattern 保存模式串，pi 是其前缀函数，go[state][字符] 是 DFA 转移后的匹配长度。
    string pattern;
    vector<int> pi;
    vector<array<int, A>> go;

    KMPAutomaton() = default;

    explicit KMPAutomaton(const string &s)
    {
        // s 是模式串；构造其所有前缀匹配状态的自动机。
        build(s);
    }

    int id(char c) const
    {
        // c 是字符集 [F,F+A) 中的字符；返回从 0 开始的字符编号。
        int x = c - F;
        assert(0 <= x && x < A);
        return x;
    }

    void build(const string &s)
    {
        // s 是模式串；重建前缀函数和状态转移表，无返回值。
        pattern = s;
        int n = pattern.size();
        for (char c : pattern)
        {
            id(c);
        }
        pi.assign(n, 0);
        for (int i = 1, j = 0; i < n; i++)
        {
            while (j && pattern[i] != pattern[j])
            {
                j = pi[j - 1];
            }
            if (pattern[i] == pattern[j])
            {
                j++;
            }
            pi[i] = j;
        }
        go.assign(n + 1, array<int, A>());
        for (int state = 0; state <= n; state++)
        {
            for (int x = 0; x < A; x++)
            {
                int j = state;
                char c = (char)(F + x);
                while (j && (j == n || pattern[j] != c))
                {
                    j = pi[j - 1];
                }
                if (j < n && pattern[j] == c)
                {
                    j++;
                }
                go[state][x] = j;
            }
        }
    }

    int next(int state, int x) const
    {
        // state 是当前已匹配前缀长度，x 是字符编号；返回读入该字符后的匹配长度。
        assert(0 <= state && state <= (int)pattern.size() && 0 <= x && x < A);
        return go[state][x];
    }

    int next(int state, char c) const
    {
        // state 是当前已匹配前缀长度，c 是字符集内字符；返回读入该字符后的匹配长度。
        return next(state, id(c));
    }

    bool matched(int state) const
    {
        // state 是当前状态；返回是否已经匹配完整模式串。
        return state == (int)pattern.size();
    }
};
```

构造 `KMPAutomaton<26,'a'> autom("aba")` 后，状态从 `0` 开始，逐字符令 `state=autom.next(state,c)`，`autom.matched(state)` 为真时记录一次出现；完整状态仍可继续转移以保留重叠匹配。预处理时间和空间为 $O(A|p|)$，每次转移 $O(1)$；若只扫描一次文本，直接使用本条前面的 `kmp` 可省掉这张表。

### 4.1.003 扩展 KMP / Z 函数

Z 函数 `z[i]` 记录后缀 $s[i..n)$ 与整串前缀的最长公共前缀长度（约定 `z[0]=n`）。维护一个当前已知能匹配到前缀的最靠右区间 $[l,r)$（Z-box）：若 `i` 落在盒内，`z[i]` 至少是镜像位置 `z[i-l]` 与盒剩余长度 `r-i` 的较小值，再从这个下界向右暴力扩展；每次扩展都推进 `r`，因此总匹配字符数线性。它和 KMP 互补——KMP 看前缀函数，Z 看与整串前缀的对齐，用于模式匹配（把模式接文本前）、周期判断、前缀出现统计。

```cpp
vector<int> zfunc(const string &s)
{
    // s 是原字符串；返回 Z 数组，z[i]=lcp(s,s[i..n))。
    int n = s.size();
    vector<int> z(n);
    if (!n)
    {
        return z;
    }
    z[0] = n;
    for (int i = 1, l = 0, r = 0; i < n; ++i)
    {
        if (i < r)
        {
            z[i] = min(r - i, z[i - l]);
        }
        while (i + z[i] < n && s[z[i]] == s[i + z[i]])
        {
            ++z[i];
        }
        if (i + z[i] > r)
        {
            l = i;
            r = i + z[i];
        }
    }
    return z;
}
```

`zfunc(s)` 返回长度 $n$ 的 Z 数组，空串返回空数组，时间与空间均为 $O(n)$。要全部出现位置且确定性时它与 [4.1.002 KMP / 前缀函数](#41002) 等价；大量任意子串比较仍用 [4.1.001 字符串哈希](#41001)。下面四个固定串分别覆盖空串、单字符、整段留在 Z-box 内以及越过右端继续扩展：

```cpp
signed main()
{
    assert(zfunc("").empty());
    assert(zfunc("a") == vector<int>{1});
    assert(zfunc("aaaaa") == vector<int>({5, 4, 3, 2, 1}));
    assert(zfunc("abacaba") == vector<int>({7, 0, 1, 0, 3, 0, 1}));
    cout << "7 0 1 0 3 0 1\n";
    return 0;
}
```

真实输出为 `7 0 1 0 3 0 1`。

### 4.1.004 Manacher

Manacher 在线性时间求出以每个中心的最长回文半径。奇数长度回文以字符为中心，偶数长度以字符缝为中心，本实现分两趟分别求 `odd[i]`、`even[i]`，各自维护当前最靠右回文区间 $[l,r]$：中心落在区间内时先用关于区间中心的镜像半径当下界，再向外扩展，扩展总量线性。`odd[i]` 对应长度 $2\cdot odd[i]-1$ 的回文，`even[i]` 对应中心缝 $(i-1,i)$ 的半径，因此 `isPal` 查询半开区间 $[l,r)$ 时要按长度奇偶分别取 `odd` 或 `even` 中点比较，混用会错。

```cpp
struct Manacher
{
    // odd、even 分别保存奇回文与偶回文半径。
    vector<int> odd, even;

    Manacher(const string &s)
    {
        // s 是要预处理回文半径的原字符串。
        int n = s.size();
        odd.resize(n);
        for (int i = 0, l = 0, r = -1; i < n; i++)
        {
            int k = i > r ? 1 : min(odd[l + r - i], r - i + 1);
            while (i - k >= 0 && i + k < n && s[i - k] == s[i + k])
            {
                k++;
            }
            odd[i] = k--;
            if (i + k > r)
            {
                l = i - k;
                r = i + k;
            }
        }
        even.resize(n);
        for (int i = 0, l = 0, r = -1; i < n; i++)
        {
            int k = i > r ? 0 : min(even[l + r - i + 1], r - i + 1);
            while (i - k - 1 >= 0 && i + k < n && s[i - k - 1] == s[i + k])
            {
                k++;
            }
            even[i] = k--;
            if (i + k > r)
            {
                l = i - k - 1;
                r = i + k;
            }
        }
    }

    // odd[i] 对应长度 2*odd[i]-1；even[i] 对应中心缝 (i-1,i) 的半径。
    bool isPal(int l, int r) const
    {
        // l、r 是原串下标；返回半开区间 [l,r) 是否为回文串。
        assert(0 <= l && l <= r && r <= (int)odd.size()); // 调试检查，可删
        if (l == r)
        {
            return true;
        }
        int n = r - l;
        if (n & 1)
        {
            return odd[(l + r) / 2] > n / 2;
        }
        return even[(l + r) / 2] >= n / 2;
    }
};
```

构造 `Manacher m(s)` 后 `m.isPal(l,r)` $O(1)$ 判断 $[l,r)$ 是否回文；构造 $O(n)$ 时间与空间。它只回答连续回文，需要枚举本质不同回文串或其出现次数应改用 [4.4.001 回文自动机 PAM / 回文树](#44001)；只做一次「某子串是否回文」的概率判定也可用 [4.1.001 字符串哈希](#41001) 正反哈希比对。

### 4.1.005 最小表示法

求循环串所有旋转里字典序最小的那个起点，用于环形串的规范化与同构判定。用双指针 `i`、`j` 各代表一个候选起点，`k` 是已确认相等的公共长度：比较 $a[(i+k)\bmod n]$ 与 $a[(j+k)\bmod n]$，相等则 `k++`，否则字典序较大的那个起点直接跳过 `k+1` 个位置（这段区间内任何起点都不可能更优），两指针相遇时把落后的一个再推进一格；`k` 每次失配归零但两指针单调前进，总比较线性。

```cpp
template <class T>
int minRepr(const vector<T> &a)
{
    // a 是循环序列；返回其字典序最小旋转的起点。
    int n = a.size();
    if (!n)
    {
        return 0;
    }
    int i = 0, j = 1, k = 0;
    while (i < n && j < n && k < n)
    {
        T x = a[(i + k) % n], y = a[(j + k) % n];
        if (x == y)
        {
            k++;
            continue;
        }
        if (x > y)
        {
            i += k + 1;
        }
        else
        {
            j += k + 1;
        }
        if (i == j)
        {
            j++;
        }
        k = 0;
    }
    return min(i, j);
}

template <class T>
vector<T> minRotation(const vector<T> &a)
{
    // a 是循环序列；返回其字典序最小的完整旋转。
    int p = minRepr(a);
    vector<T> b;
    for (int i = 0; i < (int)a.size(); i++)
    {
        b.push_back(a[(p + i) % a.size()]);
    }
    return b;
}
```

`minRepr(a)` 返回最小字典序旋转的起点下标，`minRotation(a)` 返回旋转后的完整序列；求起点 $O(n)$ 时间、$O(1)$ 额外空间，`minRotation` 另需 $O(n)$ 输出。模板参数 `T` 只需支持比较与拷贝，故字符串、整数序列通用。求唯一 Lyndon 分解或用它做后缀比较时见 [4.1.006 Lyndon 分解](#41006)，最小旋转恰是整串旋转的一个 Lyndon 相关副产品。

### 4.1.006 Lyndon 分解

Lyndon 串是严格小于自己所有真后缀的串；任何串都能唯一分解成字典序非增的若干 Lyndon 串（CFL 分解）。Duval 算法在线扫描：维护当前正在生长的近似 Lyndon 块，用 `k` 指向块内与新字符 `s[j]` 对应比较的位置——`s[k]<s[j]` 说明还能扩展，把 `k` 拉回块首；`s[k]==s[j]` 则 `k` 随 `j` 前进保持周期；一旦 `s[k]>s[j]` 就把已定型的整块（长度 `j-k`）逐段切出，`i` 前移。返回各段左端点，末尾补哨兵 `n`。

```cpp
vector<int> lyndon(const string &s)
{
    // s 是待分解字符串；返回各段左端点，并在末尾附加 s.size()。
    int n = s.size();
    vector<int> a;
    for (int i = 0; i < n;)
    {
        int j = i + 1, k = i;
        while (j < n && s[k] <= s[j])
        {
            if (s[k] < s[j])
            {
                k = i;
            }
            else
            {
                k++;
            }
            j++;
        }
        while (i <= k)
        {
            a.push_back(i);
            i += j - k;
        }
    }
    a.push_back(n);
    return a;
}
```

`lyndon(s)` 返回各 Lyndon 段的左端点并以 `n` 收尾，故第 $i$ 段是 $[a_i,a_{i+1})$；$O(n)$ 时间、$O(n)$ 输出空间。它与 [4.1.005 最小表示法](#41005) 都利用字典序下的周期结构；[4.1.007 Runs](#41007) 使用两次 Lyndon 方向枚举，但维护独立的单调栈。

### 4.1.007 Runs

一个 run（极大重复）是指数至少为 2 的极大周期区间：其最小周期为 p，区间长度至少为 2p，且向左右都无法继续保持周期 p。当前实现采用两次 Lyndon 方向枚举：单调栈确定候选 Lyndon 根，再用前后缀哈希二分求 LCP/LCS，把根扩展成极大周期区间；第二次把字符序翻转后重复枚举，最后排序去重。

```cpp
struct Run
{
    // p 是最小周期，[l,r) 是极大周期区间；32 位字段用于降低百万规模候选的内存占用。
    int p, l, r;
    auto operator<=>(const Run &) const = default;
};

struct RunHash
{
    // a 是 1 下标字符串值；h、pw 用于单模哈希 LCE。
    int n;
    vector<int> a, h, pw;

    explicit RunHash(const string &s) : n(s.size()), a(n + 1), h(n + 1), pw(n + 1, 1)
    {
        for (int i = 1; i <= n; i++)
        {
            a[i] = s[i - 1] - 'a' + 1;
        }
        build();
    }

    void build()
    {
        // 根据当前 a 建立前缀哈希。
        constexpr int mod = 998244353, bs = 31;
        h[0] = 0;
        pw[0] = 1;
        for (int i = 1; i <= n; i++)
        {
            pw[i] = 1LL * pw[i - 1] * bs % mod;
            h[i] = (1LL * h[i - 1] * bs + a[i]) % mod;
        }
    }

    int get(int l, int len) const
    {
        // 返回 1 下标区间 a[l..l+len) 的哈希值。
        constexpr int mod = 998244353;
        return (h[l + len - 1] - 1LL * h[l - 1] * pw[len] % mod + mod) % mod;
    }

    int lcp(int x, int y, int up) const
    {
        // 返回从 x、y 开始的最长公共前缀，长度不超过 up。
        if (x > n || y > n || a[x] != a[y])
        {
            return 0;
        }
        int l = 0, r = min({up, n - x + 1, n - y + 1});
        while (l < r)
        {
            int m = (l + r + 1) / 2;
            if (get(x, m) == get(y, m))
            {
                l = m;
            }
            else
            {
                r = m - 1;
            }
        }
        return l;
    }

    int lcs(int x, int y, int up) const
    {
        // 返回以 x、y 结尾的最长公共后缀，长度不超过 up。
        if (x <= 0 || y <= 0 || a[x] != a[y])
        {
            return 0;
        }
        int l = 0, r = min({up, x, y});
        while (l < r)
        {
            int m = (l + r + 1) / 2;
            if (get(x - m + 1, m) == get(y - m + 1, m))
            {
                l = m;
            }
            else
            {
                r = m - 1;
            }
        }
        return l;
    }

    void flip()
    {
        // 互补字符值，枚举另一种 Lyndon 方向。
        for (int i = 1; i <= n; i++)
        {
            a[i] = 27 - a[i];
        }
        build();
    }

    void getRuns(vector<Run> &ans)
    {
        // 用 Lyndon 单调栈枚举当前方向的全部 Runs。
        vector<int> st(n + 1);
        st[0] = n + 1;
        for (int i = n, top = 0, lt = 0; i >= 1; i--)
        {
            while (top)
            {
                int x = min(st[top] - i, st[top - 1] - st[top]);
                lt = lcp(i, st[top], x);
                if ((lt == x && st[top] - i < st[top - 1] - st[top]) ||
                    (lt < x && a[i + lt] < a[st[top] + lt]))
                {
                    top--;
                    lt = 0;
                }
                else
                {
                    break;
                }
            }
            int j = st[top];
            st[++top] = i;
            int x = lcs(i - 1, j - 1, j - i), y;
            if (x < j - i)
            {
                y = lt + lcp(i + lt, j + lt, n);
                if (x + y >= j - i)
                {
                    ans.push_back({j - i, i - x - 1, j + y - 1});
                }
            }
        }
    }
};

vector<Run> runs(const string &s)
{
    // s 是原字符串；返回全部极大周期区间及其最小周期。哈希碰撞概率极低。
    RunHash e(s);
    vector<Run> ans;
    e.getRuns(ans);
    e.flip();
    e.getRuns(ans);
    sort(ans.begin(), ans.end());
    ans.erase(unique(ans.begin(), ans.end()), ans.end());
    return ans;
}
```

`runs(s)` 返回全部 run，每个 `Run{p,l,r}` 给出最小周期 $p$ 与半开区间 $[l,r)$，结果按 `(p,l,r)` 排序。单调栈枚举为线性次候选，每次 LCE 由哈希二分完成，时间 $O(n\log n)$、空间 $O(n)$；QOJ 21648 的百万长度提交 [2830439](https://qoj.ac/submission/2830439) 已通过。实现使用模 998244353、底数 31 的单模滚动哈希，因此概率正确，不能抵抗专门构造的碰撞；要求绝对正确时保留旧 Main-Lorentz/Z-LCE 实现或改用确定性 LCE。full 版使用 32 位字段保存字符串长度、周期和端点，打印版按模板规则展开为 `int`。

### 4.1.008 Karp-Rabin

Karp-Rabin 用滚动哈希筛选与模式等长的窗口，命中后按需逐字符复核。它与 [4.1.001 字符串哈希](#41001) 是同一份代码，这里不重复：直接依赖其中的 `StrHash::match(s,t)` 即得全部匹配起点，构造 $O(n)$、匹配 $O(n+m)$、空间 $O(n+m)$。Mersenne 单模与进程级随机底数仍是概率正确；要求绝对无误时对每个哈希命中的窗口再逐字符比对一次，或改用确定性的 [4.1.002 KMP / 前缀函数](#41002)、[4.1.003 扩展 KMP / Z 函数](#41003)。

### 4.1.009 Sunday Algorithm

Sunday 是启发式匹配：失配时不看当前窗口，而是看紧贴窗口右侧的下一个文本字符 `s[i+m]`，把模式中该字符最右出现位置对齐过来，一次可能跳过多达 $m+1$ 个位置。预处理 `sh[c]` 存「模式中字符 $c$ 到末尾再加一的距离」，不含则为 $m+1$。它实现简单、随机数据上很快，但最坏仍是 $O(nm)$，不保证线性。

```cpp
vector<int> sunday(string_view s, string_view p)
{
    // s 是文本串，p 是模式串；返回全部匹配起点，按下标递增排列。
    vector<int> ans;
    if (p.empty())
    {
        for (int i = 0; i <= (int)s.size(); i++)
        {
            ans.push_back(i);
        }
        return ans;
    }
    array<int, 256> sh;
    sh.fill(p.size() + 1);
    for (int i = 0; i < (int)p.size(); i++)
    {
        sh[(unsigned char)p[i]] = p.size() - i;
    }
    for (int i = 0; i + (int)p.size() <= (int)s.size();)
    {
        int j = 0;
        while (j < (int)p.size() && s[i + j] == p[j])
        {
            j++;
        }
        if (j == (int)p.size())
        {
            ans.push_back(i);
        }
        if (i + (int)p.size() == (int)s.size())
        {
            break;
        }
        i += sh[(unsigned char)(s[i + p.size()])];
    }
    return ans;
}
```

`sunday(s,p)` 返回 `p` 在 `s` 的全部匹配起点；空模式用 $O(n+1)$ 时间和输出空间返回所有位置，非空模式最坏 $O(nm)$ 时间、$O(|\Sigma|+occ)$ 空间，其中 $occ$ 是匹配数。需要最坏线性保证时改用 [4.1.002 KMP / 前缀函数](#41002) 或 [4.1.003 扩展 KMP / Z 函数](#41003)；只找首次出现见 [4.1.010 `strstr` 类匹配](#41010)。

### 4.1.010 `strstr` 类匹配

只需要模式首次出现位置时，短模式直接用 `string::find` 即可。模式较长、字符分布较散，或要拿同一模式搜索多个文本时，可以用 C++17 的 `boyer_moore_horspool_searcher`：它先按模式建立坏字符位移表，再让 `search` 从模式末端比较，失配后一次跳过多个位置。

```cpp
// s 是文本串，t 是模式串；pos 是首次匹配位置，找不到为 -1，空模式为 0。
boyer_moore_horspool_searcher bm(t.begin(), t.end());
auto it = search(s.begin(), s.end(), bm);
int pos = it == s.end() ? -1 : (int)(it - s.begin());
```

这里不能写 `boyer_moore_horspool_searcher<>`：它的首个模板参数没有默认值。C++17 起像代码中那样省略整个 `<>`，由 `t.begin()` 推导迭代器类型；需要显式写出时，普通 `string` 用 `boyer_moore_horspool_searcher<string::iterator>`，`const string` 用 `string::const_iterator`。模式串 `t` 及其迭代器必须在 `bm` 使用期间保持有效，期间不要修改 `t`；同一个 `bm` 可以传给多次 `search` 以复用预处理。该接口要求模式和文本都是随机访问序列，`string`、`vector`、`array` 均满足；`string` 中含 `\0` 也能正常匹配。

设文本长 $n$、模式长 $m$，建立搜索器通常需 $O(m)$ 时间和空间，搜索最坏为 $O((n-m+1)m)$；只有失配能产生较大跳跃时常数才占优，重复字符等数据可能退化。因此单次短模式优先写 `s.find(t)`，需要确定性最坏 $O(n+m)$ 或全部匹配位置时改用 [4.1.002 KMP / 前缀函数](#41002) 或 [4.1.003 扩展 KMP / Z 函数](#41003)，批量子串比较用 [4.1.001 字符串哈希](#41001)。

### 4.1.011 通配符模式匹配

带单字符通配符（`s` 与 `t` 两侧的 `any` 都匹配任意一个字符）的匹配可化成卷积。把普通小写字母映成 $1\ldots26$，另用 $0/1$ 掩码标记非通配符位置，则文本窗口与反转模式在位置 $i$ 完全匹配当且仅当 $\sum_j (x_{i+j}-y_j)^2p_{i+j}q_j=0$。每项均非负：任一侧通配符都使对应项为 $0$，普通位置只有字符相等才为 $0$。展开后用三次卷积计算 $\sum x^2pq-2\sum xypq+\sum y^2pq$。

```cpp
vector<bool> wildcard(const string &s, string t, char any = '*')
{
    // s、t 只含 a-z 与 any，any 在两侧都表示通配符；返回每个合法起点是否匹配。
    int n = s.size(), m = t.size();
    if (!m)
    {
        return vector<bool>(n + 1, true);
    }
    if (m > n)
    {
        return {};
    }
    reverse(t.begin(), t.end());
    vector<Z> sx(n), sx2(n), sm(n);
    vector<Z> ty(m), ty2(m), tm(m);
    for (int i = 0; i < n; i++)
    {
        if (s[i] != any)
        {
            int x = s[i] - 'a' + 1;
            assert(1 <= x && x <= 26); // 调试检查，可删。
            sx[i] = x;
            sx2[i] = x * x;
            sm[i] = 1;
        }
    }
    for (int i = 0; i < m; i++)
    {
        if (t[i] != any)
        {
            int y = t[i] - 'a' + 1;
            assert(1 <= y && y <= 26); // 调试检查，可删。
            ty[i] = y;
            ty2[i] = y * y;
            tm[i] = 1;
        }
    }
    auto c0 = conv(move(sx2), move(tm));
    auto c1 = conv(move(sx), move(ty));
    auto c2 = conv(move(sm), move(ty2));
    vector<bool> ans(n - m + 1);
    for (int i = 0; i + m <= n; i++)
    {
        int k = i + m - 1;
        ans[i] = c0[k] - c1[k] * 2 + c2[k] == Z(0);
    }
    return ans;
}
```

`wildcard(s,t,any)` 把两侧的 `any` 都视为通配符，返回长度 $n-m+1$ 的布尔向量，`ans[i]` 表示窗口在起点 $i$ 是否匹配；接口按 Library Checker 原题约束要求普通字符属于 `a-z`。代码直接复用 [10.1.003 NTT](#101003) 的 `Z` 与 `conv`，这里不重复写 NTT。原题 $m\le524288$，每项至多 $25^2$，故真实总和至多 $625m\le327680000<998244353$；单模结果不会绕回 $0$，判定是确定性的。设补齐后的变换长度为 $N$，三次卷积的时间为 $O(N\log N)$、空间为 $O(N)$；相较原浮点 FFT 版消除了舍入风险，公共 NTT 的四进制蝶形也更适合这里的整数卷积。无通配符时首选 [4.1.002 KMP / 前缀函数](#41002)。

### 4.1.012 随机单卷积通配符匹配

确定性版展开平方恒等式要做三次卷积；随机版给每个字节独立取正实数 $r_c$，文本写 $r_c$，反转模式的普通字符写 $1/r_c$、通配符写 $0$。若窗口完全匹配，卷积值必为模式普通字符数 $k$；错配窗口只有在随机比值恰好抵消到 $k$ 时才会误判。`wildcardMC` 从 $2^{48}$ 个离散实数中取权，理想精确运算模型下可把单轮代数碰撞粗略估为 $O(m/2^{48})$；浮点 FFT 的舍入误差不服从该界，阈值附近仍可能误判，独立换 `seed` 重跑可降低随机碰撞但不能消除数值风险。其头文件自带所需的浮点 FFT；需要确定性结果时使用 [4.1.011 通配符模式匹配](#41011)。

```cpp
vector<long double> convReal(const vector<long double> &a, const vector<long double> &b)
{
    // a、b 是实数系数数组；返回二者的浮点线性卷积。
    if (a.empty() || b.empty())
    {
        return {};
    }
    int s = a.size() + b.size() - 1;
    int n = bit_ceil((unsigned long long)s);
    vector<CP> x(n), y(n);
    for (int i = 0; i < (int)a.size(); i++)
    {
        x[i] = a[i];
    }
    for (int i = 0; i < (int)b.size(); i++)
    {
        y[i] = b[i];
    }
    fft(x, false);
    fft(y, false);
    for (int i = 0; i < n; i++)
    {
        x[i] *= y[i];
    }
    fft(x, true);
    vector<long double> c(s);
    for (int i = 0; i < s; i++)
    {
        c[i] = x[i].real();
    }
    return c;
}

vector<bool> wildcardMC(const string &s, string t, unsigned long long seed, char any = '*')
{
    // s 是文本串，t 是模式串，seed 是本轮随机种子；返回每个合法起点是否匹配。
    int n = s.size();
    int m = t.size();
    if (!m)
    {
        return vector<bool>(n + 1, true);
    }
    if (m > n)
    {
        return {};
    }
    mt19937_64 rng(seed);
    constexpr unsigned long long mask = (1ULL << 48) - 1;
    constexpr long double scale = 1ULL << 48;
    array<long double, 256> w;
    for (long double &x : w)
    {
        x = 1 + ((rng() & mask) + 0.5L) / scale;
    }
    vector<long double> a(n), b(m);
    for (int i = 0; i < n; i++)
    {
        a[i] = w[(unsigned char)s[i]];
    }
    int fixed = 0;
    for (int i = 0; i < m; i++)
    {
        char c = t[m - 1 - i];
        if (c != any)
        {
            b[i] = 1 / w[(unsigned char)c];
            fixed++;
        }
    }
    auto c = convReal(a, b);
    long double tol = 1E-9L * max<int>(1, fixed);
    vector<bool> ans(n - m + 1);
    for (int i = 0; i + m <= n; i++)
    {
        ans[i] = fabsl(c[i + m - 1] - fixed) <= tol;
    }
    return ans;
}
```

`wildcardMC(s,t,seed,any)` 在普通分支只做一次浮点卷积，单项在 $(1/2,2)$ 内，卷积绝对值小于 $2m$；时间、峰值空间为 $O(N\log N)$、$O(N)$。它比 [4.1.011 通配符模式匹配](#41011) 少两次卷积，代价是同时承担 Monte Carlo 碰撞和平台相关的浮点阈值误判。`seed` 应由调用方控制；调试时固定种子可复现，多轮结果取交能压低代数假阳性，却会放大浮点假阴性的风险。下面的固定例同时检查空模式、模式过长、任意字节，以及文本中的 `any` 会按普通字节参与：

```cpp
signed main()
{
    assert(wildcard("abc", "").size() == 4);
    assert(wildcard("a", "aa").empty());
    assert(!wildcard("*b", "ab", '*')[0]);
    assert(wildcard("*b", "*b", '*')[0]);
    auto a = wildcard("ababa", "a*a");
    assert(a.size() == 3 && a[0] && !a[1] && a[2]);
    auto b = wildcardMC("ababa", "a*a", 0);
    assert(b.size() == 3 && b[0] && !b[1] && b[2]);
    string x;
    x.push_back((char)0x00);
    x.push_back((char)0xFF);

    cout << "1 0 1\n";
    return 0;
}
```

## 4.2 Trie 与多模式匹配

### 4.2.001 Trie

Trie 把一组串按字符建成前缀树：根是空前缀，每条边一个字符，一个节点唯一对应从根到它的前缀，公共前缀只存一次。`cnt[u]` 记录有多少次插入经过 `u`（即以该前缀开头的串数），`end[u]` 记录有多少完整串在此结束，二者语义不同不能混用——前缀计数查 `cnt`、整串计数查 `end`。模板参数 `A` 是字符集大小、`S` 是最小字符，插入的每个字符须落在 $[S,S+A)$。本条与第 3 章 3.6.001 都完整讲字符 Trie，二者都有独立阅读价值，故此处按规范保留完整实现。

```cpp
template <int A = 26, char S = 'a'> struct Trie
{
    // ch[u][c] 是节点 u 沿字符 c 的儿子编号，0 表示不存在。
    vector<array<int, A>> ch;
    // cnt[u] 是经过节点 u 的插入次数，end[u] 是在 u 结束的字符串次数。
    vector<int> cnt, end;

    Trie()
    {
        // 建立只含根节点的空 Trie。
        newNode();
    }

    int newNode()
    {
        // 分配并返回一个清空后的节点编号。
        ch.push_back({});
        cnt.push_back(0);
        end.push_back(0);
        return (int)ch.size() - 1;
    }

    int insert(const string &s)
    {
        // s 是待插入字符串；返回它在 Trie 中的终止节点编号。
        int p = 0;
        cnt[p]++;
        for (char c : s)
        {
            int x = c - S;
            assert(0 <= x && x < A); // 调试检查，可删
            if (!ch[p][x])
            {
                ch[p][x] = newNode();
            }
            p = ch[p][x];
            cnt[p]++;
        }
        end[p]++;
        return p;
    }

    int find(const string &s) const
    {
        // s 是查询串；返回它的终止节点编号，不存在时返回 -1。
        int p = 0;
        for (char c : s)
        {
            int x = c - S;
            if (x < 0 || x >= A || !ch[p][x])
            {
                return -1;
            }
            p = ch[p][x];
        }
        return p;
    }

    int lcp(const string &s) const
    {
        // s 是查询串；返回它与任意已插入字符串的最长公共前缀长度。
        int p = 0, n = 0;
        for (char c : s)
        {
            int x = c - S;
            if (x < 0 || x >= A || !ch[p][x])
            {
                break;
            }
            p = ch[p][x];
            n++;
        }
        return n;
    }

    int prefixCnt(const string &s) const
    {
        // s 是查询前缀；返回以 s 为前缀的已插入字符串数量。
        int p = find(s);
        return p < 0 ? 0 : cnt[p];
    }
    int stringCnt(const string &s) const
    {
        // s 是查询串；返回 s 被完整插入的次数。
        int p = find(s);
        return p < 0 ? 0 : end[p];
    }
};
```

构造 `Trie<A,S> tr`；`insert(s)` 返回终点节点，`find(s)` 返回节点或 `-1`，`prefixCnt(s)`、`stringCnt(s)` 分别查前缀出现次数与整串出现次数，`lcp(s)` 查与字典内任意串的最长公共前缀长度。各操作最坏 $O(|s|)$，空间 $O((1+\sum_i|s_i|)A)$。依次插入 `ab`、`abc`、`ab` 后 `stringCnt("ab")==2`、`prefixCnt("ab")==3`、`lcp("abd")==2`、`find("ac")==-1`。要在长文本里同时匹配多个模式需在其上加 fail 指针，见 [4.2.002 AC 自动机](#42002)；判断子序列而非前缀用 [4.2.004 子序列自动机](#42004)。

### 4.2.002 AC 自动机

AC 自动机是在 Trie 上补 fail 指针的多模式匹配机：`fail` 指向当前状态所代表串的最长真后缀状态，`nxt[c]` 是补全后的自动机转移（无真实儿子时顺着 fail 走）。BFS 建图时先令根的儿子以根为 fail，再逐层用父亲的 `nxt` 填儿子的 fail 与自身 `nxt`，使一次文本扫描每步 $O(1)$ 转移即可同时命中所有以当前状态结尾、且在其 fail 链上的模式。累计每个模式出现次数时把文本经过的状态在 fail 树上向上求和。

```cpp
template <int A = 26> class AC
{
  public:
    struct Node
    {
        // ch 是 Trie 实边，nxt 是补全失配后的自动机转移。
        array<int, A> ch, nxt;
        // fail 指向当前状态的最长真后缀状态。
        int fail = 0;
        Node()
        {
            ch.fill(-1);
            nxt.fill(0);
        }
    };

  private:
    // t 是状态表，pos 按插入顺序保存各模式终态。
    vector<Node> t{1};
    vector<int> pos;

  public:
    int add(const string &s, char fst = 'a')
    {
        // s 是非空模式串，fst 是字符集起始字节；返回模式终态。
        assert(!s.empty()); // 调试检查，可删
        int u = 0;
        for (char c : s)
        {
            int x = (unsigned char)c - (unsigned char)fst;
            assert(0 <= x && x < A); // 调试检查，可删
            if (t[u].ch[x] == -1)
            {
                t[u].ch[x] = t.size();
                t.push_back({});
            }
            u = t[u].ch[x];
        }
        pos.push_back(u);
        return u;
    }

    void build()
    {
        // 补全全部失配链接和自动机转移。
        queue<int> q;
        for (int c = 0; c < A; c++)
        {
            int v = t[0].ch[c];
            if (v == -1)
            {
                continue;
            }
            t[0].nxt[c] = v;
            q.push(v);
        }
        while (!q.empty())
        {
            int u = q.front();
            q.pop();
            t[u].nxt = t[t[u].fail].nxt;
            for (int c = 0; c < A; c++)
            {
                int v = t[u].ch[c];
                if (v == -1)
                {
                    continue;
                }
                t[v].fail = t[u].nxt[c];
                t[u].nxt[c] = v;
                q.push(v);
            }
        }
    }

    int step(int u, int c) const
    {
        // u 是当前状态，c 是 [0,A) 内字符编号；返回下一状态。
        assert(0 <= u && u < (int)t.size()); // 调试检查，可删
        assert(0 <= c && c < A); // 调试检查，可删
        return t[u].nxt[c];
    }
    const vector<Node> &nodes() const
    {
        // 返回按编号排列的状态表。
        return t;
    }
    const vector<int> &ends() const
    {
        // 返回各模式按插入顺序对应的终态。
        return pos;
    }
};
```

`add(s,fst)` 插入非空模式并返回终态；所有模式须使用同一个 `fst`，字符均落在 $[fst,fst+A)$，并在全部插入后只调用一次 `build()`。`step(u,c)` 用已经离散到 $[0,A)$ 的字符走一步；扫描原始文本遇到域外字符时把状态重置为根。插入总长 $L$ 为 $O(L)$，`build` 的时间与空间均为 $O(SA)$（$S$ 为状态数）。空模式不属于本接口；若题目允许空模式，它在长度 $n$ 的文本中出现 $n+1$ 次，应在自动机外单独统计。下面用经典 fail 链样例核对四个模式的出现次数：

```cpp
signed main()
{
    AC<26> ac;
    vector<string> pat{"he", "she", "his", "hers"};
    for (const string &s : pat)
    {
        ac.add(s);
    }
    ac.build();
    const auto &nd = ac.nodes();
    vector<int> hit(nd.size());
    int u = 0;
    for (char c : string("ahishers"))
    {
        u = ac.step(u, c - 'a');
        hit[u]++;
    }
    vector<int> cnt(nd.size());
    for (int s = 1; s < (int)nd.size(); s++)
    {
        for (int p = s; p > 0; p = nd[p].fail)
        {
            cnt[p] += hit[s];
        }
    }
    vector<int> ans;
    for (int p : ac.ends())
    {
        ans.push_back(cnt[p]);
    }
    assert(ans == vector<int>({1, 1, 1, 1}));
    cout << "1 1 1 1\n";
    return 0;
}
```

真实输出为 `1 1 1 1`。

### 4.2.003 AC 自动机 fail 树应用

把每个状态向它的 `fail` 连边（反向）就得到 fail 树：一个状态的所有失配后缀都成为它在树上的祖先。于是「文本扫描经过某状态」这一贡献，对该状态到根路径上的每个模式终态都有效——转成子树求和或 DFS 序上的区间操作，即可离线统计每个模式的出现次数、或做子树加 / 祖先查。

```cpp
template <int A> vector<vector<int>> failTree(const AC<A> &ac)
{
    // ac 是已经 build 的 AC 自动机；返回 fail 边反向后的树形邻接表。
    const auto &t = ac.nodes();
    vector<vector<int>> g(t.size());
    for (int u = 1; u < (int)t.size(); u++)
    {
        g[t[u].fail].push_back(u);
    }
    return g;
}
```

先对 [4.2.002 AC 自动机](#42002) 调用 `build()`，再 `failTree(ac)` 得反向邻接表，`g[u]` 是 fail 树上 `u` 的孩子；建树 $O(S)$，其后可求 DFS 序做子树加与祖先统计。它只重排 [4.2.002 AC 自动机](#42002) 已算好的 fail 指针，不新建自动机；在线逐字符统计而非离线子树和时直接沿 `fail` 上跳即可。

### 4.2.004 子序列自动机

子序列自动机为「判断某串是否为文本的子序列」而建：`nxt[i][c]` 存位置 $i$（含）之后字符 $c$ 第一次出现的下标，从右往左递推一遍即可。匹配时从位置 $0$ 起，对模式每个字符跳到它的下一出现位置再加一，任一步越界即非子序列，查询只与模式长度有关。它按文本下标离线建整张表，和回文小节的 [4.4.004 SqAM](#44004) 是不同结构——后者在线维护、这里离线预处理。

```cpp
template <int A = 26> class SubseqAM
{
    // nxt[i][c] 是位置 i（含）之后字符 c 的首次出现位置。
    vector<array<int, A>> nxt;

  public:
    explicit SubseqAM(const vector<int> &s) : nxt(s.size() + 1)
    {
        // s 是 [0,A) 内的文本序列；从右向左建立跳转表。
        int n = s.size();
        nxt.back().fill(n);
        for (int i = n - 1; i >= 0; --i)
        {
            assert(0 <= s[i] && s[i] < A); // 调试检查，可删
            nxt[i] = nxt[i + 1];
            nxt[i][s[i]] = i;
        }
    }

    bool contains(const vector<int> &t) const
    {
        // t 是 [0,A) 内的模式序列；返回它是否为文本的子序列。
        int p = 0, n = nxt.size() - 1;
        for (int c : t)
        {
            assert(0 <= c && c < A); // 调试检查，可删
            if (p == n || nxt[p][c] == n)
            {
                return false;
            }
            p = nxt[p][c] + 1;
        }
        return true;
    }
};
```

构造 `SubseqAM<A> am(s)`（`s` 为 $[0,A)$ 的整数序列），`am.contains(t)` 判断 `t` 是否为子序列；构造的时间、空间均为 $O((n+1)A)$，长度 $m$ 查询 $O(m)$。它只判子序列，连续模式匹配用 [4.2.002 AC 自动机](#42002) 或后缀结构；在线加字符维护同样功能见 [4.4.004 SqAM](#44004)。

## 4.3 后缀结构

### 4.3.001 后缀数组：倍增法

后缀数组 `sa` 按字典序给出全部后缀的起点。倍增法先按首字符给每个后缀一个排名，再反复以「长度 $2^k$ 前缀的排名对 $(rk_i, rk_{i+2^k})$」为关键字排序并重新编号，$k$ 每轮翻倍，$O(\log n)$ 轮后排名互异即得 `sa`。`saNaive` 是直接对后缀 `sort` 的 $O(n^2\log n)$ 朴素版，只用于很短输入的内部回退；`saDouble` 是上述倍增，每轮用 `sort` 故 $O(n\log^2 n)$。

```cpp
vector<int> saNaive(const vector<int> &s)
{
    // s 是整数序列；返回按字典序排列的全部后缀起点。
    int n = s.size();
    vector<int> sa(n);
    iota(sa.begin(), sa.end(), 0);
    sort(sa.begin(),
         sa.end(),
         [&](int x, int y)
         {
             while (x < n && y < n && s[x] == s[y])
             {
                 x++;
                 y++;
             }
             return y < n && (x == n || s[x] < s[y]);
         });
    return sa;
}

vector<int> saDouble(const vector<int> &s)
{
    // s 是整数序列；返回倍增法求得的后缀数组。
    int n = s.size();
    vector<int> sa(n), rk(n), nrk(n), v = s;
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());
    for (int i = 0; i < n; i++)
    {
        rk[i] = lower_bound(v.begin(), v.end(), s[i]) - v.begin();
    }
    iota(sa.begin(), sa.end(), 0);
    for (int k = 1; k < n; k *= 2)
    {
        auto cmp = [&](int x, int y)
        {
            if (rk[x] != rk[y])
            {
                return rk[x] < rk[y];
            }
            int a = x + k < n ? rk[x + k] : -1;
            int b = y + k < n ? rk[y + k] : -1;
            return a < b;
        };
        sort(sa.begin(), sa.end(), cmp);
        nrk[sa[0]] = 0;
        for (int i = 1; i < n; i++)
        {
            nrk[sa[i]] = nrk[sa[i - 1]] + cmp(sa[i - 1], sa[i]);
        }
        swap(rk, nrk);
    }
    return sa;
}
```

`saNaive`、`saDouble` 都接受整数序列返回后缀数组下标。朴素版 $O(n^2\log n)$、倍增版 $O(n\log^2 n)$、空间 $O(n)$。数据大时换线性的 [4.3.002 后缀数组：SA-IS](#43002)（`suffixArray` 综合入口内部对小串就回退到这两者），求两后缀 LCP 再接 [4.3.003 LCP / Height / RMQ 应用](#43003)。统计本质不同子串与出现次数往往 [4.3.004 后缀自动机 SAM](#43004) 更省心。

### 4.3.002 后缀数组：SA-IS

SA-IS 用诱导排序在线性时间构造后缀数组。把每个位置按其后缀比右邻后缀小 / 大分成 S 型与 L 型，两个相邻 LMS 位置（L 转 S 的边界）之间的 LMS 子串先被诱导排序、命名，再对命名后的缩短串递归求 SA，最后由已排好的 LMS 顺序两趟诱导（先 L 型从左入桶、再 S 型从右入桶）得到全部后缀顺序。实现必须正确处理哨兵、字符离散化与 LMS 命名，故 `saIs` 对小规模直接退回朴素版。`suffixArray` 是对外入口：字符串版按字节离散，整数版先坐标压缩再调 `saIs`。

```cpp
vector<int> saIs(const vector<int> &s, int m)
{
    // s 是整数序列，元素须在 [0,m] 内，m 是最大符号值；返回后缀数组。
    int n = s.size();
    if (n == 0)
    {
        return {};
    }
    if (n == 1)
    {
        return {0};
    }
    if (n == 2)
    {
        return s[0] < s[1] ? vector<int>{0, 1} : vector<int>{1, 0};
    }
    if (n < 10)
    {
        return saNaive(s);
    }
    if (n < 40)
    {
        return saDouble(s);
    }

    vector<int> sa(n, -1), sl(m + 1), ss(m + 1);
    vector<bool> ls(n);
    for (int i = n - 2; i >= 0; i--)
    {
        ls[i] = s[i] == s[i + 1] ? ls[i + 1] : s[i] < s[i + 1];
    }
    for (int i = 0; i < n; i++)
    {
        if (ls[i])
        {
            sl[s[i] + 1]++;
        }
        else
        {
            ss[s[i]]++;
        }
    }
    for (int i = 0; i <= m; i++)
    {
        ss[i] += sl[i];
        if (i < m)
        {
            sl[i + 1] += ss[i];
        }
    }

    auto induce = [&](const vector<int> &lms)
    {
        fill(sa.begin(), sa.end(), -1);
        vector<int> b = ss;
        for (int x : lms)
        {
            if (x < n)
            {
                sa[b[s[x]]++] = x;
            }
        }
        b = sl;
        sa[b[s[n - 1]]++] = n - 1;
        for (int i = 0; i < n; i++)
        {
            int x = sa[i];
            if (x > 0 && !ls[x - 1])
            {
                sa[b[s[x - 1]]++] = x - 1;
            }
        }
        b = sl;
        for (int i = n - 1; i >= 0; i--)
        {
            int x = sa[i];
            if (x > 0 && ls[x - 1])
            {
                sa[--b[s[x - 1] + 1]] = x - 1;
            }
        }
    };

    vector<int> id(n + 1, -1), lms;
    for (int i = 1; i < n; i++)
    {
        if (!ls[i - 1] && ls[i])
        {
            id[i] = lms.size();
            lms.push_back(i);
        }
    }
    induce(lms);
    int z = lms.size();
    if (z)
    {
        vector<int> ord;
        for (int x : sa)
        {
            if (x >= 0 && id[x] != -1)
            {
                ord.push_back(x);
            }
        }
        vector<int> ns(z);
        int nm = 0;
        ns[id[ord[0]]] = 0;
        for (int i = 1; i < z; i++)
        {
            int x = ord[i - 1], y = ord[i];
            int ex = id[x] + 1 < z ? lms[id[x] + 1] : n;
            int ey = id[y] + 1 < z ? lms[id[y] + 1] : n;
            bool same = ex - x == ey - y;
            while (same && x < ex)
            {
                if (s[x] != s[y])
                {
                    same = false;
                }
                x++;
                y++;
            }
            if (same && (x == n || s[x] != s[y]))
            {
                same = false;
            }
            if (!same)
            {
                nm++;
            }
            ns[id[ord[i]]] = nm;
        }
        vector<int> rsa = saIs(ns, nm);
        for (int i = 0; i < z; i++)
        {
            ord[i] = lms[rsa[i]];
        }
        induce(ord);
    }
    return sa;
}

template <class T> vector<int> suffixArray(const vector<T> &s)
{
    // s 是可比较序列；先离散化，再返回按字典序排列的全部后缀起点。
    int n = s.size();
    vector<int> p(n), a(n);
    iota(p.begin(), p.end(), 0);
    sort(p.begin(),
         p.end(),
         [&](int x, int y)
         {
             return s[x] < s[y];
         });
    int m = 0;
    for (int i = 0; i < n; i++)
    {
        if (i && s[p[i - 1]] != s[p[i]])
        {
            m++;
        }
        a[p[i]] = m;
    }
    return saIs(a, m);
}

vector<int> suffixArray(const string &s)
{
    // s 按无符号字节比较；返回按后缀字典序排列的起点下标。
    vector<int> a;
    a.reserve(s.size());
    for (unsigned char c : s)
    {
        a.push_back(c);
    }
    return saIs(a, 255);
}
```

字符串入口按无符号字节比较，SA-IS 为 $O(n+256)$ 时间、$O(n+256)$ 空间；泛型 `vector` 入口先排序离散化，故为 $O(n\log n)$ 时间、$O(n)$ 空间。得到 `sa` 后用 [4.3.003 LCP / Height / RMQ 应用](#43003) 求任意两后缀 LCP；需要压缩后缀树边才用 [4.3.008 隐式后缀树 Ukkonen](#43008)。下面检查空串、重复字符、一般字符串与无符号字节顺序：

```cpp
signed main()
{
    assert(suffixArray("").empty());
    assert(suffixArray("aaaa") == vector<int>({3, 2, 1, 0}));
    assert(suffixArray("banana") == vector<int>({5, 3, 1, 0, 4, 2}));
    string bytes;
    bytes.push_back((char)0xFF);
    bytes.push_back((char)0x00);
    bytes.push_back((char)0x80);
    assert(suffixArray(bytes) == vector<int>({1, 2, 0}));
}
```

### 4.3.003 LCP / Height / RMQ 应用

给定后缀数组，Kasai 算法线性求 Height：`ht[i]` 是排名相邻的 `sa[i]`、`sa[i+1]` 两后缀的 LCP。关键性质是删去首字符后 LCP 至多减一，故按原串下标顺序处理、`h` 只减不重置能均摊 $O(n)$。任意两后缀 $s[i..)$、$s[j..)$ 的 LCP 等于它们排名区间内 Height 的最小值，于是对 `ht` 建 ST 表即可 $O(1)$ 回答，`query` 先把下标换成排名再取区间最小。它不构造 SA，`sa` 必须是合法排列、空串不可查。

```cpp
namespace suffix_lcp
{
vector<int> lcpArray(const string &s, const vector<int> &sa)
{
    // s 是原字符串，sa 是全部后缀按字典序排列的起点；返回相邻后缀 LCP，ans[i]=lcp(sa[i],sa[i+1])。
    auto n = (int)s.size();
    assert((int)sa.size() == n); // 调试检查，可删
    vector<int> rk(n), ht(max<int>(0, n - 1));
    vector<char> seen(n);
    for (int i = 0; i < n; i++)
    {
        assert(0 <= sa[i] && sa[i] < n && !seen[sa[i]]); // 调试检查，可删
        seen[sa[i]] = true;
        rk[sa[i]] = i;
    }
    int h = 0;
    for (int i = 0; i < n; i++)
    {
        if (rk[i] == n - 1)
        {
            h = 0;
            continue;
        }
        auto j = sa[rk[i] + 1];
        while (i + h < n && j + h < n && s[i + h] == s[j + h])
        {
            h++;
        }
        ht[rk[i]] = h;
        if (h)
        {
            h--;
        }
    }
    return ht;
}

struct LcpRMQ
{
    int n = 0;              // n 是字符串长度。
    vector<int> sa;         // sa[r] 是排名 r 的后缀起点。
    vector<int> rk;         // rk[i] 是后缀 s[i..n) 的排名。
    vector<int> ht;         // ht[i] 是 sa[i] 与 sa[i+1] 的最长公共前缀长度。
    vector<int> lg;         // lg[len] 是不超过 len 的最大二次幂指数。
    vector<vector<int>> st; // st[k][i] 是 ht[i..i+2^k) 的最小值。

    LcpRMQ(const string &s, vector<int> order)
        : n((int)s.size()), sa(move(order)), rk(n), ht(lcpArray(s, sa)), lg(n + 1)
    {
        // s 是原字符串，order 是后缀数组；建立 Height 的 ST 表。
        for (int i = 0; i < n; i++)
        {
            rk[sa[i]] = i;
        }
        for (int i = 2; i <= n; i++)
        {
            lg[i] = lg[i / 2] + 1;
        }
        if (n <= 1)
        {
            return;
        }
        st.assign(lg[n - 1] + 1, {});
        st[0] = ht;
        for (int k = 1; k < (int)st.size(); k++)
        {
            auto len = 1LL << k;
            st[k].resize(n - len);
            for (int i = 0; i + len <= n - 1; i++)
            {
                st[k][i] = min(st[k - 1][i], st[k - 1][i + len / 2]);
            }
        }
    }

    int queryRank(int l, int r) const
    {
        // l、r 是后缀排名且 l<r；返回 sa[l] 与 sa[r] 的最长公共前缀长度。
        assert(0 <= l && l < r && r < n); // 调试检查，可删
        auto k = lg[r - l];
        return min(st[k][l], st[k][r - (1LL << k)]);
    }

    int query(int i, int j) const
    {
        // i、j 是两个后缀起点；返回 s[i..n) 与 s[j..n) 的最长公共前缀长度。
        assert(0 <= i && i < n && 0 <= j && j < n); // 调试检查，可删
        if (i == j)
        {
            return n - i;
        }
        auto x = rk[i];
        auto y = rk[j];
        if (x > y)
        {
            swap(x, y);
        }
        return queryRank(x, y);
    }
};
}
```

实现放在命名空间 `suffix_lcp`。只要 Height 用 `suffix_lcp::lcpArray(s,sa)`；要多次询问就建 `suffix_lcp::LcpRMQ q(s,sa)` 再 `q.query(i,j)` 查后缀 $s[i..n)$ 与 $s[j..n)$ 的 LCP；构造 Height 与 ST 表 $O(n\log n)$ 时间与空间，单次查询 $O(1)$。以 `s="banana"`、`sa={5,3,1,0,4,2}` 为例，Height 为 `{1,3,0,0,2}`，`query(1,3)=3`、`query(0,2)=0`。SA 由 [4.3.001 后缀数组：倍增法](#43001) 或 [4.3.002 后缀数组：SA-IS](#43002) 提供；区间受限的本质不同子串不能套整串公式，见 [4.3.010 区间本质不同子串](#43010)。

### 4.3.004 后缀自动机 SAM

后缀自动机把一个串的全部子串压进至多 $2n-1$ 个状态：每个状态代表一组 endpos（结束位置集合）相同的子串，这组子串长度连续，`link` 指向其最长真后缀所在的（endpos 更大的）状态。在线加字符 `add`：新建末状态 `q`，沿 `last` 的 `link` 链把没有 `x` 转移的状态都指向 `q`；遇到已有转移 `v` 时，若 `len[v]==len[p]+1` 直接令 `link[q]=v`，否则克隆 `v` 为 `z`（复制转移、`len` 设为 `len[p]+1`）分裂出正确长度区间。不同子串数为 $\sum len[u]-len[link[u]]$；`count` 按 `len` 基数排序后沿 `link` 累加初始 `cnt=1`（只在非克隆末状态）得到每个状态代表子串的出现次数。

```cpp
template <int A = 26, char S = 'a'> struct SAM
{
    struct Node
    {
        array<int, A> next{};
        // link 是后缀链接，len 是最长长度，cnt 是当前出现次数，first 是一个最长串的末位置。
        int link = -1, len = 0, cnt = 0, first = -1;
    };

    vector<Node> t{{}};
    // last 是当前整串对应的末状态。
    int last = 0;
    // counted 表示出现次数是否已经沿后缀链接上传。
    bool counted = false;

    SAM() = default;
    SAM(const string &s)
    {
        // s 是原字符串；按顺序插入其全部字符。
        for (char c : s)
        {
            add(c);
        }
    }

    void add(char c)
    {
        // c 是要追加到原串末尾的字符。
        assert(!counted); // 调试检查，可删
        int x = c - S;
        assert(0 <= x && x < A); // 调试检查，可删
        int p = last, q = t.size();
        t.push_back({});
        t[q].len = t[p].len + 1;
        t[q].cnt = 1;
        t[q].first = t[q].len - 1;
        while (p != -1 && !t[p].next[x])
        {
            t[p].next[x] = q;
            p = t[p].link;
        }
        if (p == -1)
        {
            t[q].link = 0;
        }
        else
        {
            int v = t[p].next[x];
            if (t[v].len == t[p].len + 1)
            {
                t[q].link = v;
            }
            else
            {
                int z = t.size();
                t.push_back(t[v]);
                t[z].len = t[p].len + 1;
                t[z].cnt = 0;
                while (p != -1 && t[p].next[x] == v)
                {
                    t[p].next[x] = z;
                    p = t[p].link;
                }
                t[v].link = t[q].link = z;
            }
        }
        last = q;
    }

    int find(const string &s) const
    {
        // s 是查询串；返回对应状态，不存在或含域外字符时返回 -1。
        int p = 0;
        for (char c : s)
        {
            int x = c - S;
            if (x < 0 || x >= A || !t[p].next[x])
            {
                return -1;
            }
            p = t[p].next[x];
        }
        return p;
    }

    long long distinct() const
    {
        // 返回当前自动机表示的本质不同非空子串数。
        long long ans = 0;
        for (int i = 1; i < (int)t.size(); i++)
        {
            ans += t[i].len - t[t[i].link].len;
        }
        return ans;
    }

    void count()
    {
        // 把各末状态的计数沿后缀链接上传；重复调用不再二次累计。
        if (counted)
        {
            return;
        }
        int n = 0;
        for (auto &v : t)
        {
            chmax(n, v.len);
        }
        vector<int> c(n + 1), p(t.size());
        for (auto &v : t)
        {
            c[v.len]++;
        }
        partial_sum(c.begin(), c.end(), c.begin());
        for (int i = (int)t.size() - 1; i >= 0; i--)
        {
            p[--c[t[i].len]] = i;
        }
        for (int i = (int)p.size() - 1; i; i--)
        {
            int x = p[i];
            t[t[x].link].cnt += t[x].cnt;
        }
        counted = true;
    }

    int occ(const string &s)
    {
        // s 是查询串；返回它在原串中的出现次数，空串返回原串长度加一。
        if (s.empty())
        {
            return t[last].len + 1;
        }
        int p = find(s);
        if (p < 0)
        {
            return 0;
        }
        count();
        return t[p].cnt;
    }
};

array<int, 4> commonSubstr(const string &a, const string &b)
{
    // a、b 是小写字符串；返回 {a_l,a_r,b_l,b_r} 的一组最长公共子串半开区间。
    SAM<26, 'a'> sam(a);
    int p = 0, len = 0, best = 0;
    array<int, 4> ans{};
    for (int i = 0; i < (int)b.size(); i++)
    {
        int x = b[i] - 'a';
        while (p && !sam.t[p].next[x])
        {
            p = sam.t[p].link;
            len = min(len, sam.t[p].len);
        }
        if (sam.t[p].next[x])
        {
            p = sam.t[p].next[x];
            len++;
        }
        else
        {
            len = 0;
        }
        if (len > best)
        {
            best = len;
            int end = sam.t[p].first;
            ans = {end - len + 1, end + 1, i - len + 1, i + 1};
        }
    }
    return ans;
}
```

构造 `SAM<A,S> sam(s)`；`find(s)` 定位状态，`distinct()` 返回本质不同子串数，`occ(t)` 返回 `t` 在原串中的出现次数，空串按出现位置定义返回 $n+1$。`commonSubstr(a,b)` 返回一组 `{a_l,a_r,b_l,b_r}`，两段半开区间长度相等且为最长公共子串长度；无公共字符时返回全零区间。首次非空查询触发一次 $O(n)$ 的 `count()`；调用 `count()` 或首次查询非空串后不得再 `add`。构造时间和空间均为 $O(nA)$；非空串的状态数不超过 $2n-1$，空串只有根状态；查长度 $m$ 的串需 $O(m)$。最长公共子串在扫描第二串时也只沿后缀链接回退，故为 $O(|a|+|b|)$ 时间与 $O(|a|)$ 空间。下面用会产生状态分裂的 `ababa` 直接核对本质不同子串数和各节点次数：

```cpp
signed main()
{
    SAM<3, 'a'> sam("ababa");
    assert(sam.distinct() == 9);
    assert(sam.occ("a") == 3);
    assert(sam.occ("b") == 2);
    assert(sam.occ("ab") == 2);
    assert(sam.occ("aba") == 2);
    assert(sam.occ("bab") == 1);
    assert(sam.occ("c") == 0);
    assert(sam.occ("") == 6);
    cout << "9 3 2 2 2 1 0 6\n";
    return 0;
}
```

真实输出为 `9 3 2 2 2 1 0 6`。

### 4.3.005 广义 SAM / ExSAM

广义 SAM 把多条串的全部子串压进一个自动机。离线版先把所有串插进公共 Trie（`pass` 记录每个 Trie 节点被多少串经过），再按 BFS 顺序对 Trie 的每条边调用 `extend(p,c)` 扩展 SAM：相同前缀天然共享状态，且不依赖插入顺序。`extend` 处理已有转移时同样按 `len[q]==len[p]+1` 决定直接复用还是克隆分裂。建完后把 Trie 节点的 `pass` 加到对应 SAM 状态，再按 `len` 降序沿 `link` 上推，得到每个子串跨全部串的出现次数。

```cpp
template <int A = 26, char B = 'a'> struct ExSAM
{
    struct Node
    {
        array<int, A> ch{};
        // link 是后缀链接，len 是该状态的最长长度。
        int link = -1, len = 0;
        // cnt 是该状态所代表子串在全部输入串中的出现次数。
        long long cnt = 0;
    };

    struct TrieNode
    {
        array<int, A> ch{};
        // pass 是经过该 Trie 节点的输入串数，重复插入重复计算。
        long long pass = 0;
    };

    vector<Node> t{{}};
    vector<TrieNode> tr{{}};
    // done 记录是否已经 build，emptyCnt 是空串在全部输入串中的出现位置数。
    bool done = false;
    long long emptyCnt = 0;

    void insert(const string &s)
    {
        // s 是待加入集合的整条字符串；相邻字符串之间不产生跨边界子串。
        assert(!done); // 调试检查，可删
        emptyCnt += (int)s.size() + 1;
        int u = 0;
        for (char c : s)
        {
            int x = c - B;
            assert(0 <= x && x < A); // 调试检查，可删
            if (!tr[u].ch[x])
            {
                tr[u].ch[x] = tr.size();
                tr.push_back({});
            }
            u = tr[u].ch[x];
            tr[u].pass++;
        }
    }

    int extend(int p, int c)
    {
        // p 是当前 SAM 状态，c 是离散字符；返回扩展后的状态。
        int q = t[p].ch[c];
        if (q)
        {
            if (t[q].len == t[p].len + 1)
            {
                return q;
            }
            int nq = t.size();
            t.push_back(t[q]);
            t[nq].len = t[p].len + 1;
            t[nq].cnt = 0;
            while (p != -1 && t[p].ch[c] == q)
            {
                t[p].ch[c] = nq;
                p = t[p].link;
            }
            t[q].link = nq;
            return nq;
        }

        int u = t.size();
        t.push_back({});
        t[u].len = t[p].len + 1;
        while (p != -1 && !t[p].ch[c])
        {
            t[p].ch[c] = u;
            p = t[p].link;
        }
        if (p == -1)
        {
            t[u].link = 0;
            return u;
        }
        q = t[p].ch[c];
        if (t[q].len == t[p].len + 1)
        {
            t[u].link = q;
            return u;
        }
        int nq = t.size();
        t.push_back(t[q]);
        t[nq].len = t[p].len + 1;
        t[nq].cnt = 0;
        while (p != -1 && t[p].ch[c] == q)
        {
            t[p].ch[c] = nq;
            p = t[p].link;
        }
        t[q].link = t[u].link = nq;
        return u;
    }

    void build()
    {
        // 从公共 Trie 构造广义 SAM，并上传各状态的跨串出现次数。
        assert(!done); // 调试检查，可删
        done = true;
        vector<int> pos(tr.size());
        queue<int> q;
        q.push(0);
        while (!q.empty())
        {
            int u = q.front();
            q.pop();
            for (int c = 0; c < A; c++)
            {
                int v = tr[u].ch[c];
                if (!v)
                {
                    continue;
                }
                pos[v] = extend(pos[u], c);
                t[pos[v]].cnt += tr[v].pass;
                q.push(v);
            }
        }
        int mx = 0;
        for (const Node &v : t)
        {
            mx = max(mx, v.len);
        }
        vector<int> c(mx + 1), p(t.size());
        for (const Node &v : t)
        {
            c[v.len]++;
        }
        partial_sum(c.begin(), c.end(), c.begin());
        for (int u = (int)t.size() - 1; u >= 0; u--)
        {
            p[--c[t[u].len]] = u;
        }
        for (int i = (int)p.size() - 1; i > 0; i--)
        {
            int u = p[i];
            if (t[u].link >= 0)
            {
                t[t[u].link].cnt += t[u].cnt;
            }
        }
    }

    int find(const string &s) const
    {
        // s 是查询串；返回对应状态，不存在或含域外字符时返回 -1。
        int u = 0;
        for (char c : s)
        {
            int x = c - B;
            if (x < 0 || x >= A)
            {
                return -1;
            }
            u = t[u].ch[x];
            if (!u)
            {
                return -1;
            }
        }
        return u;
    }

    long long count(const string &s) const
    {
        // s 是查询串；返回它在全部输入串中的出现次数，空串按各串分别计位置。
        assert(done); // 调试检查，可删
        if (s.empty())
        {
            return emptyCnt;
        }
        int u = find(s);
        return u < 0 ? 0 : t[u].cnt;
    }

    long long distinct() const
    {
        // 返回全部输入串并集中的本质不同非空子串数。
        assert(done); // 调试检查，可删
        long long ans = 0;
        for (int i = 1; i < t.size(); i++)
        {
            ans += t[i].len - t[t[i].link].len;
        }
        return ans;
    }
};
```

逐串 `insert(s)` 建 Trie，只能调用一次 `build()`；之后 `find`、`count(s)` 和 `distinct()` 可查。`count("")` 返回 $\sum_i(|s_i|+1)$，域外查询字符直接视为不存在。按长度计数排序上传状态后，总长 $L$ 的构造时间和空间均为 $O((L+1)A)$，状态数不超过 $2L+1$，单次查询为 $O(|s|)$。要边到达边在线加串而非先收齐用 [4.3.006 在线广义 SAM](#43006)；单串场景直接 [4.3.004 后缀自动机 SAM](#43004)。

### 4.3.006 在线广义 SAM

在线广义 SAM 不预先收集全部文本：每来一条新串就从根开始逐字符 `extend`。与离线版的区别只在于没有 Trie 前置——`extend(p,c)` 若发现转移已存在但 `len[q]≠len[p]+1` 就克隆分裂到正确长度并沿 `link` 改边，从而在任意插入顺序下都维持广义 SAM 的正确结构，适合字典串陆续到达、需要随时查询的场景。

```cpp
template <int A = 26, char B = 'a'> struct OnlineExSAM
{
    struct Node
    {
        // ch 是字符转移，link 是后缀链接，len 是该状态的最长长度。
        array<int, A> ch{};
        int link = -1, len = 0;
    };

    // tr 是状态池。
    vector<Node> tr{{}};

    int extend(int p, int c)
    {
        // p 是当前状态，c 是离散字符；返回扩展后的状态。
        int q = tr[p].ch[c];
        if (q)
        {
            if (tr[q].len == tr[p].len + 1)
            {
                return q;
            }
            int nq = tr.size();
            tr.push_back(tr[q]);
            tr[nq].len = tr[p].len + 1;
            while (p != -1 && tr[p].ch[c] == q)
            {
                tr[p].ch[c] = nq;
                p = tr[p].link;
            }
            tr[q].link = nq;
            return nq;
        }

        int u = tr.size();
        tr.push_back({});
        tr[u].len = tr[p].len + 1;
        while (p != -1 && !tr[p].ch[c])
        {
            tr[p].ch[c] = u;
            p = tr[p].link;
        }
        if (p == -1)
        {
            tr[u].link = 0;
            return u;
        }
        q = tr[p].ch[c];
        if (tr[q].len == tr[p].len + 1)
        {
            tr[u].link = q;
            return u;
        }
        int nq = tr.size();
        tr.push_back(tr[q]);
        tr[nq].len = tr[p].len + 1;
        while (p != -1 && tr[p].ch[c] == q)
        {
            tr[p].ch[c] = nq;
            p = tr[p].link;
        }
        tr[q].link = tr[u].link = nq;
        return u;
    }

    void insert(const string &s)
    {
        // s 是新字符串；每条串从根开始，不产生跨边界子串。
        int p = 0;
        for (char c : s)
        {
            int x = c - B;
            assert(0 <= x && x < A); // 调试检查，可删
            p = extend(p, x);
        }
    }

    int find(const string &s) const
    {
        // s 是查询串；返回对应状态，不存在或含域外字符时返回 -1。
        int p = 0;
        for (char c : s)
        {
            int x = c - B;
            if (x < 0 || x >= A)
            {
                return -1;
            }
            p = tr[p].ch[x];
            if (!p)
            {
                return -1;
            }
        }
        return p;
    }

    long long distinct() const
    {
        // 返回当前所有输入串的本质不同非空子串数。
        long long ans = 0;
        for (int p = 1; p < (int)tr.size(); p++)
        {
            ans += tr[p].len - tr[tr[p].link].len;
        }
        return ans;
    }
};
```

构造 `OnlineExSAM<> sam`，每条新串 `insert(s)`，`distinct()` 随时返回当前串集合的本质不同子串总数；当前接口不分别记录各串出现次数。设总长 $L$，构造和空间均为 $O((L+1)A)$，状态数 $\le 2L+1$，`find` 为 $O(|s|)$。依次插入 `ababa`、`bcb` 后子串并集为 $13$ 个，`distinct()==13`、`find("aba")>=0`、`find("ac")==-1`。需要跨串出现次数用离线的 [4.3.005 广义 SAM / ExSAM](#43005)。

### 4.3.007 SAMSA / CDAWG

把 SAM 转移 DAG 中入度、出度都为 $1$ 的非关键状态压进边标签，得到紧凑的 C-SAM 视图：保留根以及入度或出度不为 $1$ 的状态，每条保留边沿唯一出链走到下一个关键状态、把途经字符拼成标签。它减少显式节点而仍能判子串——SAM 路径的任意前缀都是合法子串，故查询允许停在标签中间。注意若题目要求接受全部后缀的严格 CDAWG，还须另标后缀终止态，本条只做子串判定视图。

```cpp
template <int A = 26, char B = 'a'> struct CSAM
{
    struct Edge
    {
        // to 是目标关键状态，s 是压缩路径的非空标签。
        int to = 0;
        string s;
    };

    // g 是压缩后的 DAG；id 把 SAM 状态映射到关键状态，非关键状态为 -1。
    vector<vector<Edge>> g;
    vector<int> id;

    CSAM(const SAM<A, B> &sam)
    {
        // sam 是已经建好的后缀自动机；压缩入度、出度均为 1 的路径。
        int n = sam.t.size();
        vector<int> in(n), out(n);
        for (int u = 0; u < n; u++)
        {
            for (int c = 0; c < A; c++)
            {
                int v = sam.t[u].next[c];
                if (v)
                {
                    in[v]++, out[u]++;
                }
            }
        }
        id.assign(n, -1);
        for (int u = 0; u < n; u++)
        {
            if (u == 0 || in[u] != 1 || out[u] != 1)
            {
                id[u] = g.size();
                g.push_back({});
            }
        }
        for (int u = 0; u < n; u++)
        {
            if (id[u] < 0)
            {
                continue;
            }
            for (int c = 0; c < A; c++)
            {
                int v = sam.t[u].next[c];
                if (!v)
                {
                    continue;
                }
                string s(1, (char)(B + c));
                while (id[v] < 0)
                {
                    for (int d = 0; d < A; d++)
                    {
                        int w = sam.t[v].next[d];
                        if (w)
                        {
                            s += (char)(B + d);
                            v = w;
                            break;
                        }
                    }
                }
                g[id[u]].push_back({id[v], move(s)});
            }
        }
    }

    bool contains(const string &t) const
    {
        // t 是查询串；返回它是否对应根出发的一条路径前缀，空串返回 true。
        int u = 0, p = 0;
        while (p < (int)t.size())
        {
            const Edge *e = nullptr;
            for (const auto &cur : g[u])
            {
                if (cur.s[0] == t[p])
                {
                    e = &cur;
                    break;
                }
            }
            if (!e)
            {
                return false;
            }
            for (char c : e->s)
            {
                if (p == (int)t.size())
                {
                    return true;
                }
                if (t[p++] != c)
                {
                    return false;
                }
            }
            u = e->to;
        }
        return true;
    }
};
```

先 `SAM<> sam(s)` 再 `CSAM<> cs(sam)`，`cs.contains(t)` 判断 `t` 是否为 `s` 的子串；边标签可能多字符，匹配时必须逐字符核对而非只看首字符。构造与 [4.3.004 后缀自动机 SAM](#43004) 同量级，时间、空间均为 $O((n+1)A)$。原串 `ababa` 上 `contains("")`、`contains("aba")`、`contains("bab")` 为真，`contains("abb")`、`contains("abac")` 为假。只判子串不需压缩时直接用 [4.3.004 后缀自动机 SAM](#43004) 的 `find`。

### 4.3.008 隐式后缀树 Ukkonen

Ukkonen 在线构造压缩后缀 Trie：边用原串区间 $[left,right)$ 表示，逐字符加入时维护活动点，借后缀链接从一个后缀跳到下一个，内部节点只在边分裂时创建。直接传入原串得到隐式后缀树，部分后缀会结束在边内而没有独立叶子；若每个后缀都要对应显式叶子，先在原串末尾追加一个未出现且小于原字母表所有字符的哨兵。

```cpp
class SuffixTree
{
  public:
    struct Node
    {
        // [left,right) 是父边在原串中的标签区间，parent 是父节点，suffixLink 是后缀链接。
        int left = 0, right = 0, parent = -1, suffixLink = -1;
        // children 把出边首字符映射到儿子节点。
        map<char, int> children;
        int length() const
        {
            // 返回当前压缩边长度。
            return right - left;
        }
    };

  private:
    struct State
    {
        // node 是当前边的下端节点，position 是从该边起点走过的字符数。
        int node, position;
    };

    // text 是建树所用原串，nodes 是后缀树节点池。
    string text;
    vector<Node> nodes;
    State active{0, 0};

    State go(State state, int left, int right)
    {
        // state 是活动点，[left,right) 是待匹配文本；返回匹配后的活动点，失败返回 {-1,-1}。
        while (left < right)
        {
            if (state.position == nodes[state.node].length())
            {
                auto it = nodes[state.node].children.find(text[left]);
                if (it == nodes[state.node].children.end())
                {
                    return {-1, -1};
                }
                state = {it->second, 0};
            }
            else
            {
                if (text[nodes[state.node].left + state.position] != text[left])
                {
                    return {-1, -1};
                }
                int length = min(right - left, nodes[state.node].length() - state.position);
                state.position += length;
                left += length;
            }
        }
        return state;
    }

    int split(State state)
    {
        // state 是活动点；若它在边内则分裂该边，返回活动点对应的显式节点。
        if (state.position == nodes[state.node].length())
        {
            return state.node;
        }
        if (!state.position)
        {
            return nodes[state.node].parent;
        }
        Node original = nodes[state.node];
        int middle = nodes.size();
        nodes.push_back({original.left, original.left + state.position, original.parent, -1, {}});
        nodes[original.parent].children[text[original.left]] = middle;
        nodes[middle].children[text[original.left + state.position]] = state.node;
        nodes[state.node].parent = middle;
        nodes[state.node].left += state.position;
        return middle;
    }

    int suffixLink(int node)
    {
        // node 是显式后缀树节点；惰性计算并返回它的后缀链接。
        if (nodes[node].suffixLink != -1)
        {
            return nodes[node].suffixLink;
        }
        if (nodes[node].parent == -1)
        {
            return 0;
        }
        int parLink = suffixLink(nodes[node].parent);
        int skip = nodes[node].parent == 0;
        return nodes[node].suffixLink =
                   split(go({parLink, nodes[parLink].length()}, nodes[node].left + skip, nodes[node].right));
    }

    void extend(int position)
    {
        // position 是本次追加字符在 text 中的下标；补齐所有新后缀。
        while (true)
        {
            State next = go(active, position, position + 1);
            if (next.node != -1)
            {
                active = next;
                return;
            }
            int middle = split(active), leaf = nodes.size();
            nodes.push_back({position, (int)text.size(), middle, -1, {}});
            nodes[middle].children[text[position]] = leaf;
            active.node = suffixLink(middle);
            active.position = nodes[active.node].length();
            if (!middle)
            {
                break;
            }
        }
    }

  public:
    // 需要显式叶子时，在末尾加入一个小于字母表中所有字符的哨兵。
    explicit SuffixTree(string text) : text(move(text))
    {
        // text 是原字符串；按顺序完成 Ukkonen 在线构造。
        nodes.reserve(this->text.size() * 2 + 1);
        nodes.push_back({0, 0, -1, 0, {}});
        for (int i = 0; i < this->text.size(); ++i)
        {
            extend(i);
        }
    }

    const vector<Node> &getNodes() const
    {
        // 返回后缀树节点池的只读引用，顺序与节点编号一致。
        return nodes;
    }
    const string &getText() const
    {
        // 返回建树所用的完整文本。
        return text;
    }
};
```

构造 `SuffixTree st(s)` 得到隐式树；若 `s` 不含 `\0`，可用 `SuffixTree st(s + '\0')` 得到每个后缀均有独立叶子的显式树。`getNodes()`、`getText()` 分别返回节点池与实际建树文本，边 $[left,right)$ 与 `children` 描述压缩边；字符集大小为 $A$ 时构造摊还 $O(n\log(A+1)+1)$，空间 $O(n+1)$。只需统计子串时，[4.3.004 后缀自动机 SAM](#43004) 更易写。

### 4.3.009 后缀平衡树（前插、回滚与动态排名）

后缀平衡树维护只在串首加入字符的全部后缀。把当前串反向保存为 `r` 后，每个后缀对应 `r` 的一个前缀终点；继续前插只会在 `r` 末尾追加字符，旧前缀代表的字符串不变，因而旧结点之间的字典序不会改变。Treap 用反向前缀的 LCP 比较键，子树大小支持排名和第 $k$ 后缀；新后缀只与排序后的前驱、后继比较 LCP，即可累计本质不同子串数。哈希采用自然溢出的 64 位多项式哈希，是速度优先的 Monte Carlo 实现。

```cpp
struct SuffixBST
{
    struct Node
    {
        int l = 0;                 // l 保存 Treap 左儿子编号。
        int r = 0;                 // r 保存 Treap 右儿子编号。
        int sz = 1;                // sz 保存子树结点数。
        int key = 0;               // key 是反向串前缀终点，也是后缀长度。
        unsigned long long pri = 0; // pri 保存固定随机优先级。
    };

    static constexpr unsigned long long base = 11995408973635179863ULL;
    string r;                       // r 保存当前字符串的反串。
    vector<unsigned long long> h{0};  // h 保存 r 的 64 位前缀哈希。
    vector<unsigned long long> pw{1}; // pw[i] 保存 base 的 i 次幂。
    vector<Node> t{{}};             // t[0] 是空结点，其余结点编号等于 key。
    vector<int> ways{0};            // ways[n] 保存长度为 n 时的不同子串数。
    int root = 0;                   // root 保存 Treap 根编号。

    static unsigned long long priority(unsigned long long x)
    {
        // x 是后缀长度；返回 SplitMix64 生成的固定优先级。
        x += 0x9e3779b97f4a7c15ULL;
        x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
        x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
        return x ^ (x >> 31);
    }

    int size(int u) const
    {
        // u 是 Treap 结点编号；返回其子树大小，空结点返回 0。
        return u ? t[u].sz : 0;
    }

    void pull(int u)
    {
        // u 是非空结点；由两个儿子重算子树大小。
        t[u].sz = size(t[u].l) + size(t[u].r) + 1;
    }

    unsigned long long hash(int l, int rr) const
    {
        // l、rr 是 r 的半开区间端点；返回 r[l,rr) 的 64 位哈希。
        return h[rr] - h[l] * pw[rr - l];
    }

    int lcpKey(int a, int b) const
    {
        // a、b 是两个反向前缀终点；返回对应原串后缀的 LCP 长度。
        int l = 0, rr = min(a, b) + 1;
        while (l + 1 < rr)
        {
            int m = (l + rr) / 2;
            if (hash(a - m, a) == hash(b - m, b))
            {
                l = m;
            }
            else
            {
                rr = m;
            }
        }
        return l;
    }

    bool lessKey(int a, int b) const
    {
        // a、b 是后缀长度；返回长度 a 对应后缀是否字典序更小。
        if (a == b)
        {
            return false;
        }
        int p = lcpKey(a, b);
        if (p == min(a, b))
        {
            return a < b;
        }
        return (unsigned char)r[a - p - 1] < (unsigned char)r[b - p - 1];
    }

    void split(int u, int key, int &x, int &y)
    {
        // u 是原树，key 是分界后缀；x、y 分别接收小于 key 和不小于 key 的树。
        if (!u)
        {
            x = y = 0;
        }
        else if (lessKey(t[u].key, key))
        {
            x = u;
            split(t[u].r, key, t[u].r, y);
            pull(u);
        }
        else
        {
            y = u;
            split(t[u].l, key, x, t[u].l);
            pull(u);
        }
    }

    int merge(int x, int y)
    {
        // x 的全部键小于 y；返回合并后的 Treap 根。
        if (!x || !y)
        {
            return x | y;
        }
        if (t[x].pri > t[y].pri)
        {
            t[x].r = merge(t[x].r, y);
            pull(x);
            return x;
        }
        t[y].l = merge(x, t[y].l);
        pull(y);
        return y;
    }

    void insert(int u)
    {
        // u 是尚未入树的新结点；按其 key 插入 Treap。
        int x, y;
        split(root, t[u].key, x, y);
        root = merge(merge(x, u), y);
    }

    int erase(int u, int key)
    {
        // u 是当前根，key 是已存在后缀；删除后返回新根。
        if (t[u].key == key)
        {
            return merge(t[u].l, t[u].r);
        }
        if (lessKey(key, t[u].key))
        {
            t[u].l = erase(t[u].l, key);
        }
        else
        {
            t[u].r = erase(t[u].r, key);
        }
        pull(u);
        return u;
    }

    pair<int, int> neighbors(int key) const
    {
        // key 是尚未入树的后缀；返回其前驱、后继的后缀长度，不存在时为 0。
        int u = root, pre = 0, suc = 0;
        while (u)
        {
            if (lessKey(t[u].key, key))
            {
                pre = t[u].key;
                u = t[u].r;
            }
            else
            {
                suc = t[u].key;
                u = t[u].l;
            }
        }
        return {pre, suc};
    }

    void pushFront(char c)
    {
        // c 是新字符；把 c 加到当前串首，并插入由整个新串形成的后缀。
        r.push_back(c);
        h.push_back(h.back() * base + (unsigned char)c + 1);
        pw.push_back(pw.back() * base);
        int key = r.size();
        auto [pre, suc] = neighbors(key);
        int same = 0;
        if (pre)
        {
            same = max(same, lcpKey(key, pre));
        }
        if (suc)
        {
            same = max(same, lcpKey(key, suc));
        }
        ways.push_back(ways.back() + key - same);
        t.push_back({0, 0, 1, key, priority(key)});
        insert(key);
    }

    void popFront()
    {
        // 无参数；删除最近一次加入串首的字符及其完整后缀。
        assert(!r.empty()); // 调试检查，可删
        int key = r.size();
        root = erase(root, key);
        r.pop_back();
        h.pop_back();
        pw.pop_back();
        ways.pop_back();
        t.pop_back();
    }

    int rank(int i) const
    {
        // i 是当前串的后缀起点；返回该后缀从 0 开始的字典序排名。
        int key = (int)r.size() - i;
        assert(1 <= key && key <= (int)r.size()); // 调试检查，可删
        int u = root, ans = 0;
        while (t[u].key != key)
        {
            if (lessKey(key, t[u].key))
            {
                u = t[u].l;
            }
            else
            {
                ans += size(t[u].l) + 1;
                u = t[u].r;
            }
        }
        return ans + size(t[u].l);
    }

    int kth(int k) const
    {
        // k 是从 0 开始的字典序排名；返回当前串中对应后缀的起点。
        assert(0 <= k && k < size(root)); // 调试检查，可删
        int u = root;
        while (true)
        {
            int z = size(t[u].l);
            if (k < z)
            {
                u = t[u].l;
            }
            else if (k == z)
            {
                return (int)r.size() - t[u].key;
            }
            else
            {
                k -= z + 1;
                u = t[u].r;
            }
        }
    }

    int distinct() const
    {
        // 无参数；返回当前串本质不同子串个数。
        return ways.back();
    }
};
```

`pushFront(c)`、`popFront()`、`rank(i)` 和 `kth(k)` 的期望时间均为 $O(\log^2 n)$，其中一次 Treap 路径含 $O(\log n)$ 次比较，每次比较二分 LCP；`distinct()` 为 $O(1)$，总空间 $O(n)$。只允许撤销最近一次前插，不能删除中间后缀；64 位哈希存在极低碰撞概率，需要确定性结果时改用动态 LCP 结构。静态字符串仍优先使用 [4.3.002 后缀数组：SA-IS](#43002)。

### 4.3.010 区间本质不同子串

限定下标区间 $[l,r)$ 内的本质不同子串数不能直接套整串 SAM 的 $\sum len-len_{link}$。本条给出最直接的做法：把子串 `s[l,r)` 取出，对它单独建后缀数组与 Height，答案就是所有长度之和 $\frac{L(L+1)}2$ 减去相邻 Height 之和。它对单次询问简单可靠；大量区间询问才需要持久化或离线的可持久化 SAM / SA 结构。

```cpp
inline long long subDistinct(const string &s, int l, int r)
{
    // s 是原字符串，l、r 是半开端点；返回 s[l,r) 的本质不同子串数量。
    assert(0 <= l && l <= r && r <= (int)s.size()); // 调试检查，可删
    string t = s.substr(l, r - l);
    auto sa = suffixArray(t);
    auto h = suffix_lcp::lcpArray(t, sa);
    long long n = t.size();
    return n * (n + 1) / 2 - accumulate(h.begin(), h.end(), 0LL);
}

struct RangeSubMax
{
    // n、z 是叶子数和补齐后的二次幂；tr 保存区间最大值。
    int n, z = 1;
    vector<int> tr;

    RangeSubMax(int n) : n(n)
    {
        // n 是叶子数；初始各位置均为 0。
        while (z < n) z <<= 1;
        tr.assign(2 * z, 0);
    }

    void setMax(int p, int v)
    {
        // p 是位置，v 是候选值；执行单点取最大。
        for (p += z, tr[p] = max(tr[p], v); p >>= 1;)
            tr[p] = max(tr[2 * p], tr[2 * p + 1]);
    }

    int query(int l, int r) const
    {
        // l、r 是半开端点；返回 [l,r) 的最大值。
        int ans = 0;
        for (l += z, r += z; l < r; l >>= 1, r >>= 1)
        {
            if (l & 1) ans = max(ans, tr[l++]);
            if (r & 1) ans = max(ans, tr[--r]);
        }
        return ans;
    }
};

inline vector<int> rangeSubs(const string &s,
                             const vector<pair<int, int>> &qs)
{
    // s 是小写字母串；qs 是 0 下标半开区间；返回各区间的本质不同子串数。
    int n = s.size(), q = qs.size();
    assert(n > 0); // 调试检查，可删
    vector<vector<pair<int, int>>> at(n + 1);
    for (int i = 0; i < q; i++)
    {
        auto [l, r] = qs[i];
        assert(0 <= l && l < r && r <= n); // 调试检查，可删
        at[r].push_back({l + 1, i});
    }
    SAM<26, 'a'> sam(s);
    int states = sam.t.size();
    vector<vector<int>> son(states);
    for (int i = 1; i < states; i++) son[sam.t[i].link].push_back(i);
    vector<int> in(states), out(states), it(states), st{0};
    int tim = 0;
    while (!st.empty())
    {
        int u = st.back();
        if (!in[u]) in[u] = ++tim;
        if (it[u] < (int)son[u].size()) st.push_back(son[u][it[u]++]);
        else out[u] = tim, st.pop_back();
    }
    LazySeg<Info, Tag> seg(n);
    RangeSubMax mx(states + 1);
    vector<int> top(n + 1), ans(q);
    vector<unsigned char> seen(states);
    int u = 0;
    for (int r = 1; r <= n; r++)
    {
        u = sam.t[u].next[s[r - 1] - 'a'];
        int v = u;
        while (v && !seen[v]) seen[v] = 1, v = sam.t[v].link;
        while (v)
        {
            int p = mx.query(in[v] - 1, out[v]);
            int old = top[p];
            seg.apply(p - sam.t[v].len, p - sam.t[old].len, Tag{-1});
            swap(top[p], v);
        }
        mx.setMax(in[u] - 1, r);
        top[r] = 0;
        for (auto [l, id] : at[r])
        {
            int len = r - l + 1;
            ans[id] = len * (len + 1) / 2 + seg.query(l - 1, r).sum;
        }
    }
    return ans;
}
```

`subDistinct(s,l,r)` 返回子串 $[l,r)$ 的本质不同子串数，单次 $O(L+1)$（$L=r-l$）时间与空间，依赖 [4.3.002 后缀数组：SA-IS](#43002) 与 [4.3.003 LCP / Height / RMQ 应用](#43003)。批量区间询问可调用 `rangeSubs(s,qs)`：离线扫描右端点，在 SAM 后缀链接树上维护贡献，复杂度 $O(n\log^2n+q\log n)$，实现见 `include/full/4_3_010.hpp`。

### 4.3.011 Dictionary of Basic Factors

Dictionary of Basic Factors 把所有长度为 $2$ 的幂的子串（基本因子）分层编号：第 $k$ 层长度 $2^k$ 的子串由第 $k-1$ 层的前、后两个 $2^{k-1}$ 块的编号对决定，逐层排序压缩得离散排名 `rk[k][i]`。有了它，任意等长子串比较可拆成常数个基本因子——`compare(l,r,len)` 取覆盖 `len` 的两个 $2^k$ 块（$k=\lfloor\log_2 len\rfloor$，右块从 `len` 减 $2^k$ 起）比较排名对。

```cpp
struct BasicFactors
{
    // s 是原字符串，rk[k][i] 是长 2^k 子串 s[i..i+2^k) 的排名。
    string s;
    vector<vector<int>> rk;

    BasicFactors(string s) : s(move(s))
    {
        // s 按无符号字节字典序比较；建立所有二次幂长度子串的分层排名。
        int n = this->s.size();
        vector<int> first;
        for (unsigned char c : this->s) first.push_back(c);
        rk.push_back(move(first));
        for (int k = 1; (1LL << k) <= n; k++)
        {
            int len = 1LL << (k - 1);
            vector<int> id(n - 2 * len + 1), next(id.size());
            iota(id.begin(), id.end(), 0);
            sort(id.begin(), id.end(), [&](int i, int j)
            {
                return pair{rk[k - 1][i], rk[k - 1][i + len]} <
                       pair{rk[k - 1][j], rk[k - 1][j + len]};
            });
            int cls = 0;
            for (int i = 0; i < (int)id.size(); i++)
            {
                if (i && pair{rk[k - 1][id[i - 1]], rk[k - 1][id[i - 1] + len]} !=
                         pair{rk[k - 1][id[i]], rk[k - 1][id[i] + len]}) cls++;
                next[id[i]] = cls;
            }
            rk.push_back(move(next));
        }
    }

    int compare(int l, int r, int len) const
    {
        // l、r 是两个子串起点，len 是共同长度；按字典序返回 -1、0 或 1。
        assert(0 <= l && 0 <= r && 0 <= len && l + len <= (int)s.size() && r + len <= (int)s.size()); // 调试检查，可删
        if (!len) return 0;
        int k = bit_width((unsigned long long)len) - 1, d = len - (1LL << k);
        pair<int, int> a{rk[k][l], rk[k][l + d]}, b{rk[k][r], rk[k][r + d]};
        return (a > b) - (a < b);
    }
};

struct BorderInfo
{
    // mn、mx 是最短和最长真 border 长度，cnt 是真 border 数量。
    int mn = 0, mx = 0, cnt = 0;
};

struct BasicDictQuery
{
    int l, r;
};

inline vector<BorderInfo> basicDict(const string &s,
                                    const vector<pair<int, int>> &qs)
{
    // s 是字符串；qs 是 0 下标闭区间；返回每个区间的真 border 统计。
    int n = s.size(), q = qs.size();
    vector<BorderInfo> ans(q);
    if (!n || !q) return ans;
    vector<BasicDictQuery> que(q);
    vector<vector<int>> atL(n + 2), atR(n + 2);
    for (int i = 0; i < q; i++)
    {
        int l = qs[i].first + 1, r = qs[i].second + 1;
        assert(1 <= l && l <= r && r <= n);
        que[i] = {l, r};
        atL[l].push_back(i);
        atR[r].push_back(i);
    }
    vector<int> sa(n + 1), rk(n + 1), id(2 * n + 3), cnt(256);
    for (int i = 1; i <= n; i++) cnt[(unsigned char)s[i - 1]]++;
    for (int i = 1; i < 256; i++) cnt[i] += cnt[i - 1];
    for (int i = n; i; i--) sa[cnt[(unsigned char)s[i - 1]]--] = i;
    int cls = 0;
    for (int i = 1; i <= n; i++)
    {
        if (i == 1 || s[sa[i] - 1] != s[sa[i - 1] - 1]) cls++;
        rk[sa[i]] = cls;
    }
    vector<int> nxt(n + 2), leftMost(q, n + 1), rightMost(q),
        leftRev(q, n + 1), rightRev(q), low(q, n + 1), high(q), number(q),
        buf(n + 3);
    auto addOne = [&](int idq, int x, int l, int r, int step)
    {
        if (l <= x && x <= r && (x - l) % step == 0)
        {
            low[idq] = min(low[idq], x);
            high[idq] = max(high[idq], x);
            number[idq]++;
        }
    };
    auto addTwo = [&](int idq, int l1, int r1, int d1,
                      int l2, int r2, int d2)
    {
        if (r1 - l1 <= d1)
        {
            for (int x = l1; x <= r1; x += d1) addOne(idq, x, l2, r2, d2);
            return;
        }
        if (r2 - l2 <= d2)
        {
            for (int x = l2; x <= r2; x += d2) addOne(idq, x, l1, r1, d1);
            return;
        }
        if ((l1 - l2) % d1) return;
        l1 = max(l1, l2), r1 = min(r1, r2);
        if (l1 <= r1)
        {
            low[idq] = min(low[idq], l1);
            high[idq] = max(high[idq], r1);
            number[idq] += (r1 - l1) / d1 + 1;
        }
    };
    auto work = [&](int len)
    {
        fill(buf.begin(), buf.begin() + cls + 1, n + 1);
        for (int i = n - len + 1; i >= 1; i--) nxt[i] = buf[rk[i]], buf[rk[i]] = i;
        fill(buf.begin(), buf.begin() + cls + 1, -1);
        for (int i = 2 - len; i <= n; i++)
        {
            if (1 <= i && i <= n) buf[rk[i]] = i;
            int j = i + len - 1;
            if (1 <= j && j <= n) for (int idq : atR[j]) if (que[idq].r - que[idq].l + 1 > len)
                rightMost[idq] = buf[rk[que[idq].l]];
            j = i - len + 1;
            if (1 <= j && j <= n) for (int idq : atL[j]) if (que[idq].r - que[idq].l + 1 > len)
                rightRev[idq] = min(buf[rk[que[idq].r - len + 1]], que[idq].r - len + 1);
        }
        fill(buf.begin(), buf.begin() + cls + 1, n + 1);
        for (int i = n; i + len - 1; i--)
        {
            if (1 <= i && i <= n) buf[rk[i]] = i;
            int j = i + 2 * len - 2;
            if (1 <= j && j <= n) for (int idq : atR[j]) if (que[idq].r - que[idq].l + 1 > len)
                leftMost[idq] = max(buf[rk[que[idq].l]], nxt[que[idq].l]);
            if (1 <= i && i <= n) for (int idq : atL[i]) if (que[idq].r - que[idq].l + 1 > len)
                leftRev[idq] = buf[rk[que[idq].r - len + 1]];
        }
        for (int i = 0; i < q; i++)
        {
            if (que[i].r - que[i].l + 1 <= len || leftMost[i] > rightMost[i] || leftRev[i] > rightRev[i]) continue;
            int d1 = nxt[leftMost[i]] - leftMost[i], d2 = nxt[leftRev[i]] - leftRev[i];
            swap(leftMost[i], rightMost[i]);
            leftRev[i] += len - que[i].l;
            rightRev[i] += len - que[i].l;
            leftMost[i] = que[i].r - leftMost[i] + 1;
            rightMost[i] = que[i].r - rightMost[i] + 1;
            addTwo(i, leftMost[i], rightMost[i], d1, leftRev[i], rightRev[i], d2);
        }
        if (cls == n) return;
        fill(buf.begin(), buf.begin() + cls + 1, 0);
        int z = 0;
        for (int i = n; i > n - len; i--) id[++z] = i;
        for (int i = 1; i <= n; i++) if (sa[i] > len) id[++z] = sa[i] - len;
        for (int i = 1; i <= n; i++) buf[rk[i]]++;
        for (int i = 2; i <= cls; i++) buf[i] += buf[i - 1];
        for (int i = n; i; i--) sa[buf[rk[id[i]]]--] = id[i], id[i] = rk[i];
        cls = 0;
        for (int i = 1; i <= n; i++)
        {
            int x = sa[i], y = sa[i - 1];
            if (i == 1 || id[x] != id[y] || id[x + len] != id[y + len]) cls++;
            rk[x] = cls;
        }
    };
    for (int len = 1; len <= n; len <<= 1) work(len);
    for (int i = 0; i < q; i++) if (number[i]) ans[i] = {low[i], high[i], number[i]};
    return ans;
}
```

构造 `BasicFactors bf(s)` 后 `bf.compare(l,r,len)` 按字典序比较两个等长子串返回 $-1/0/1$；非空串构造 $O(n\log^2 n)$ 时间、$O(n\log n)$ 空间，空串构造 $O(1)$，定长比较 $O(1)$。批量 border 统计调用 `basicDict(s,qs)`，其中 `qs` 为 0 下标闭区间，返回真 border 的最短长度、最长长度和数量，复杂度 $O((n+q)\log n)$，实现见 `include/full/4_3_011.hpp`。

## 4.4 回文结构

### 4.4.001 回文自动机 PAM / 回文树

回文自动机（回文树）每个节点恰对应一种本质不同回文子串，`link` 指向该回文的最长真回文后缀。有两个根：长度 $-1$ 的奇根与长度 $0$ 的偶根。在线 `add(c)` 时先沿 `getLink` 从 `last` 找到能左右扩展当前字符的回文后缀 `p`（即 $s[n-len_p-1]=s[n]$），若 `p` 没有 `x` 边就新建长度 $len_p+2$ 的回文节点，其 `link` 在长度为 $1$ 时接偶根、否则接 `getLink(link_p)` 的 `x` 边；至多新建一个节点。节点数减 $2$ 即不同回文串数，`count` 沿 `link` 逆序上推 `cnt` 得每种回文的出现次数。

```cpp
template <int A = 26, char S = 'a'> struct PAM
{
    struct Node
    {
        array<int, A> ch{};
        // len 是回文长度，link 是回文后缀链接，cnt 是出现次数。
        int len = 0, link = 0, cnt = 0;
    };

    vector<Node> t{{{}, -1, 0, 0}, {{}, 0, 0, 0}};
    vector<int> s{-1};
    // last 是当前最长回文后缀节点，counted 记录出现次数是否已经上传。
    int last = 1;
    bool counted = false;

    int getLink(int p) const
    {
        // p 是回文节点；返回可被当前末字符继续扩展的后缀回文节点。
        int n = s.size() - 1;
        while (n - t[p].len - 1 < 0 || s[n - t[p].len - 1] != s[n])
        {
            p = t[p].link;
        }
        return p;
    }

    int add(char c)
    {
        // c 是追加到原串末尾的字符；返回新的最长回文后缀节点。
        assert(!counted); // 调试检查，可删
        int x = c - S;
        assert(0 <= x && x < A); // 调试检查，可删
        s.push_back(x);
        int p = getLink(last);
        if (!t[p].ch[x])
        {
            int q = t.size();
            t.push_back({{}, t[p].len + 2, 0, 0});
            if (t[q].len == 1)
            {
                t[q].link = 1;
            }
            else
            {
                t[q].link = t[getLink(t[p].link)].ch[x];
            }
            t[p].ch[x] = q;
        }
        last = t[p].ch[x];
        t[last].cnt++;
        return last;
    }

    void build(const string &x)
    {
        // x 是要继续追加的字符串。
        for (char c : x)
        {
            add(c);
        }
    }

    void count()
    {
        // 沿后缀链接上传出现次数；重复调用不再二次累计。
        if (counted)
        {
            return;
        }
        for (int p = (int)t.size() - 1; p >= 2; p--)
        {
            t[t[p].link].cnt += t[p].cnt;
        }
        counted = true;
    }
};
```

`build(s)` 逐字符加入，`add(c)` 返回当前最长回文后缀节点，`count()` 幂等地上传一次出现次数；上传后不能再追加字符。节点数减二是本质不同回文数。构造 $O(n+1)$、计数 $O(p+1)$、空间 $O((p+2)A+n+1)$，其中 $p\le n$ 是本质不同回文数。需要在两端插入和删除时改用 [4.4.002 双端可撤销回文树](#44002)，回文划分 DP 的等差链优化见 [4.4.003 Palindrome Series](#44003)。下面加入 `ababa`，记录每次新增长后缀对应的节点，再逐节点核对五种回文的出现次数：

```cpp
signed main()
{
    PAM<3, 'a'> pam;
    vector<int> id;
    for (char c : string("ababa"))
    {
        id.push_back(pam.add(c));
    }
    pam.count();
    pam.count();
    assert((int)pam.t.size() - 2 == 5);
    assert(pam.t[id[0]].cnt == 3);
    assert(pam.t[id[1]].cnt == 2);
    assert(pam.t[id[2]].cnt == 2);
    assert(pam.t[id[3]].cnt == 1);
    assert(pam.t[id[4]].cnt == 1);
    cout << "5 3 2 2 1 1\n";
    return 0;
}
```

真实输出为 `5 3 2 2 1 1`。

### 4.4.002 双端可撤销回文树

普通 PAM 只支持在右端追加；`DequePAM` 额外维护各位置对应的最长回文前缀 `pre` 和最长回文后缀 `suf`，从而能在 deque 两端插入和撤销。节点的 `cnt` 是当前 `pre`、`suf` 记录对它的引用数，`sons` 是后缀链接孩子数；二者都归零时，该回文已不在当前串中，可删除父转移并回收编号。`quick` 跳过一批不可能被新字符扩展的后缀回文，前后端使用对称的 `frontCan`、`backCan` 查找可扩展节点。字符必须落在 $[S,S+A)$，删除只允许作用于非空串。

```cpp
template <int A = 26, char S = 'a'>
struct DequePAM
{
  private:
    struct Node
    {
        // ch 是回文树转移；par、link、quick 分别是转移父亲、后缀链接和快速跳转。
        array<int, A> ch{};
        int par = 0;
        int link = 0;
        int quick = 0;
        // len 是回文长度，cnt 是边界记录引用数，sons 是后缀链接孩子数。
        int len = 0;
        int cnt = 0;
        int sons = 0;
    };

    struct Pos
    {
        // c 是当前位置字符编号；pre、suf 记录当前最长回文前缀、后缀节点。
        int c = 0;
        int pre = 1;
        int suf = 1;
    };

    // t 是节点池，spare 是可复用编号，d 是当前字符串，kinds 是本质不同回文数。
    vector<Node> t;
    vector<int> spare;
    deque<Pos> d;
    int kinds = 0;

    int newNode()
    {
        // 返回一个已经清空的节点编号，优先复用回收节点。
        if (spare.empty())
        {
            t.push_back({});
            return (int)t.size() - 1;
        }
        int p = spare.back();
        spare.pop_back();
        t[p] = Node{};
        return p;
    }

    void drop(int p)
    {
        // p 是已经脱离所有转移和边界记录的节点；清空后回收其编号。
        t[p] = Node{};
        spare.push_back(p);
    }

    int backCan(int c, int p) const
    {
        // c 是待加字符，p 是当前最长回文后缀；返回最先能从右端扩展的后缀回文。
        int n = d.size();
        while (true)
        {
            if (t[p].len == -1 || (t[p].len < n && d[n - t[p].len - 1].c == c))
            {
                return p;
            }
            int q = t[p].link;
            if (t[q].len == -1 || (t[q].len < n && d[n - t[q].len - 1].c == c))
            {
                return q;
            }
            p = t[p].quick;
        }
    }

    int frontCan(int c, int p) const
    {
        // c 是待加字符，p 是当前最长回文前缀；返回最先能从左端扩展的前缀回文。
        int n = d.size();
        while (true)
        {
            if (t[p].len == -1 || (t[p].len < n && d[t[p].len].c == c))
            {
                return p;
            }
            int q = t[p].link;
            if (t[q].len == -1 || (t[q].len < n && d[t[q].len].c == c))
            {
                return q;
            }
            p = t[p].quick;
        }
    }

  public:
    DequePAM() : t(2)
    {
        // 建立长度 -1 的奇根 0 和长度 0 的偶根 1。
        t[0].len = -1;
        t[0].par = 0;
        t[0].link = 0;
        t[0].quick = 0;
        t[1].len = 0;
        t[1].par = 0;
        t[1].link = 0;
        t[1].quick = 0;
    }

    void pushBack(char c)
    {
        // c 是新字符；加入右端并维护最长回文前后缀。
        int x = (unsigned char)c - (unsigned char)S;
        assert(0 <= x && x < A); // 调试检查，可删
        int par = d.empty() ? 0 : backCan(x, d.back().suf);
        int n = d.size();
        int v = t[par].ch[x];
        int w = 1;
        if (!v)
        {
            v = newNode();
            t[v].par = par;
            t[v].len = t[par].len + 2;
            if (par != 0)
            {
                w = t[backCan(x, t[par].link)].ch[x];
                assert(w); // 调试检查，可删
            }
            t[v].link = w;
            t[w].sons++;
            d.push_back({x, 1, 1});
            n++;
            if (t[w].link != 0 && d[n - t[w].len - 1].c == d[n - t[t[w].link].len - 1].c)
            {
                t[v].quick = t[w].quick;
            }
            else
            {
                t[v].quick = t[w].link;
            }
            t[par].ch[x] = v;
            kinds++;
        }
        else
        {
            d.push_back({x, 1, 1});
            n++;
            w = t[v].link;
        }
        d[n - 1].suf = v;
        d[n - t[v].len].pre = v;
        if (t[w].len >= 1 && d[n - t[v].len + t[w].len - 1].suf == w)
        {
            d[n - t[v].len + t[w].len - 1].suf = 1;
        }
        t[v].cnt++;
    }

    void pushFront(char c)
    {
        // c 是新字符；加入左端并维护最长回文前后缀。
        int x = (unsigned char)c - (unsigned char)S;
        assert(0 <= x && x < A); // 调试检查，可删
        int par = d.empty() ? 0 : frontCan(x, d.front().pre);
        int v = t[par].ch[x];
        int w = 1;
        if (!v)
        {
            v = newNode();
            t[v].par = par;
            t[v].len = t[par].len + 2;
            if (par != 0)
            {
                w = t[frontCan(x, t[par].link)].ch[x];
                assert(w); // 调试检查，可删
            }
            t[v].link = w;
            t[w].sons++;
            d.push_front({x, 1, 1});
            if (t[w].link != 0 && d[t[w].len].c == d[t[t[w].link].len].c)
            {
                t[v].quick = t[w].quick;
            }
            else
            {
                t[v].quick = t[w].link;
            }
            t[par].ch[x] = v;
            kinds++;
        }
        else
        {
            d.push_front({x, 1, 1});
            w = t[v].link;
        }
        d[0].pre = v;
        d[t[v].len - 1].suf = v;
        if (t[w].len >= 1 && d[t[v].len - t[w].len].pre == w)
        {
            d[t[v].len - t[w].len].pre = 1;
        }
        t[v].cnt++;
    }

    void popBack()
    {
        // 删除右端字符；当前串必须非空。
        assert(!d.empty()); // 调试检查，可删
        int n = d.size();
        int v = d.back().suf;
        int x = d.back().c;
        int w = t[v].link;
        if (t[v].len >= 2 && t[d[n - t[v].len + t[w].len - 1].suf].len < t[w].len)
        {
            d[n - t[v].len + t[w].len - 1].suf = w;
            d[n - t[v].len].pre = w;
        }
        else
        {
            d[n - t[v].len].pre = 1;
        }
        t[v].cnt--;
        if (!t[v].sons && !t[v].cnt)
        {
            t[t[v].par].ch[x] = 0;
            t[w].sons--;
            drop(v);
            kinds--;
        }
        d.pop_back();
    }

    void popFront()
    {
        // 删除左端字符；当前串必须非空。
        assert(!d.empty()); // 调试检查，可删
        int v = d.front().pre;
        int x = d.front().c;
        int w = t[v].link;
        if (t[v].len >= 2 && t[d[t[v].len - t[w].len].pre].len < t[w].len)
        {
            d[t[v].len - t[w].len].pre = w;
            d[t[v].len - 1].suf = w;
        }
        else
        {
            d[t[v].len - 1].suf = 1;
        }
        t[v].cnt--;
        if (!t[v].sons && !t[v].cnt)
        {
            t[t[v].par].ch[x] = 0;
            t[w].sons--;
            drop(v);
            kinds--;
        }
        d.pop_front();
    }

    int size() const
    {
        // 返回当前字符串长度。
        return d.size();
    }

    int distinct() const
    {
        // 返回当前字符串的本质不同非空回文子串数。
        return kinds;
    }

    int longPrefix() const
    {
        // 返回当前最长回文前缀长度，空串返回 0。
        return d.empty() ? 0 : t[d.front().pre].len;
    }

    int longSuffix() const
    {
        // 返回当前最长回文后缀长度，空串返回 0。
        return d.empty() ? 0 : t[d.back().suf].len;
    }
};
```

`pushFront`、`pushBack`、`popFront`、`popBack` 分别修改两端；`distinct()`、`longPrefix()`、`longSuffix()` 均为 $O(1)$ 查询。数组转移为 $O(1)$，一次修改沿 `quick` 链查找可扩展回文，保守最坏为相关回文链接链长。设操作过程中最大串长为 $N$、最多同时存在 $P$ 种非空回文，节点编号由 `spare` 复用，峰值空间为 $O((P+2)A+N)$。下面从 `abac` 出发依次在两端增删，检查四种修改和节点回收：

```cpp
signed main()
{
    DequePAM<3, 'a'> pam;
    for (char c : string("abac"))
    {
        pam.pushBack(c);
    }
    assert(array{pam.distinct(), pam.longPrefix(), pam.longSuffix()} == array{4LL, 3LL, 1LL});
    pam.pushFront('c');
    assert(array{pam.distinct(), pam.longPrefix(), pam.longSuffix()} == array{5LL, 5LL, 5LL});
    pam.pushBack('a');
    assert(array{pam.distinct(), pam.longPrefix(), pam.longSuffix()} == array{6LL, 5LL, 3LL});
    pam.popFront();
    assert(array{pam.distinct(), pam.longPrefix(), pam.longSuffix()} == array{5LL, 3LL, 3LL});
    pam.popBack();
    assert(array{pam.distinct(), pam.longPrefix(), pam.longSuffix()} == array{4LL, 3LL, 1LL});
}
```

### 4.4.003 Palindrome Series

把回文树的后缀链接链按 `diff=len[u]-len[link[u]]` 分成等差系列，并维护 `series` 指针跳过同 `diff` 段。这样每个前缀的全部回文后缀被压成 $O(\log n)$ 个等差系列，回文划分类 DP（如把串划分成最少 / 方案数个回文）可按系列整体转移而非逐个回文，均摊 $O(n\log n)$。`add` 在建回文节点时顺带算出 `diff` 与 `series`，`eachSeries` 沿 `series` 链枚举关键状态。

```cpp
template <int A = 26, char S = 'a'> struct PamSeries
{
    struct Node
    {
        array<int, A> ch{};
        // diff=len-len[link]，series 指向上一条不同公差的系列。
        int len = 0, link = 0, diff = 0, series = 0;
    };

    vector<Node> t{{{}, -1, 0, 0, 0}, {{}, 0, 0, 0, 0}};
    vector<int> s{-1};
    // last 是当前最长回文后缀节点。
    int last = 1;

    int get(int p) const
    {
        // p 是一个回文节点；返回能被当前末字符继续扩展的后缀回文节点。
        int n = s.size() - 1;
        while (s[n - t[p].len - 1] != s[n])
        {
            p = t[p].link;
        }
        return p;
    }

    int add(char c)
    {
        // c 是追加到原串末尾的字符；返回新的最长回文后缀节点。
        int x = c - S;
        assert(0 <= x && x < A); // 调试检查，可删
        s.push_back(x);
        int p = get(last);
        if (!t[p].ch[x])
        {
            int q = t.size();
            t.push_back({{}, t[p].len + 2});
            t[q].link = t[q].len == 1 ? 1 : t[get(t[p].link)].ch[x];
            t[q].diff = t[q].len - t[t[q].link].len;
            t[q].series = t[q].diff == t[t[q].link].diff ? t[t[q].link].series : t[q].link;
            t[p].ch[x] = q;
        }
        return last = t[p].ch[x];
    }

    template <class F> void eachSeries(int p, F f) const
    {
        // p 是当前最长回文后缀，f 接收每条等差系列的首节点。
        for (; t[p].len > 0; p = t[p].series)
        {
            f(p);
        }
    }
};
```

`add(c)` 在线扩展并维护 `diff`、`series`；`eachSeries(p,f)` 沿等差链对每个关键状态调用回调 `f`。每次追加摊还 $O(1)$，状态数 $\le n+2$、空间 $O((n+2)A+n+1)$；一个前缀的系列链长为 $O(\log(n+1))$，故配合系列 DP 时每次枚举 $O(\log(n+1))$。只统计本质不同回文或其出现次数用 [4.4.001 回文自动机 PAM / 回文树](#44001)；本条专为回文划分 DP 的等差优化。

例如把当前前缀的每条系列首节点收集起来：

```cpp
PamSeries<> pam;
int p = 0;
for (char c : string("abacaba"))
{
    p = pam.add(c);
}
vector<int> head;
pam.eachSeries(p, [&](int u)
{
    head.push_back(u);
});
assert(!head.empty());
```

### 4.4.004 SqAM

SqAM 是在线子序列自动机：每加入一个文本字符建一个状态，`go[u][c]` 指向位置 `u` 之后字符 `c` 首次出现的状态。插入新字符 `x` 时沿每种字符 `c` 的「上一同字符位置」链 `pre` 补齐尚不存在的 `go[..][x]` 转移，每个状态与字符对最多被写一次，故总代价 $O(nA)$。沿 `go` 读完模式当且仅当它是文本子序列。

```cpp
template <int A = 26> class SqAM
{
    // go[u][c] 是状态 u 读字符 c 后到达的最早后继，-1 表示不存在。
    vector<array<int, A>> go;
    // pre[u] 是与状态 u 字符相同的上一次出现位置，-1 表示此前未出现。
    vector<int> pre;
    // last[c] 是字符 c 最近一次出现的状态，-1 表示尚未出现。
    array<int, A> last;

  public:
    SqAM()
    {
        // 建立只含空文本状态的子序列自动机。
        last.fill(-1);
        go.push_back({});
        go[0].fill(-1);
        pre.push_back(-1);
    }

    void add(int x)
    {
        // x 是 [0,A) 内的新字符编号；把它追加到文本末尾。
        assert(0 <= x && x < A); // 调试检查，可删
        int v = go.size();
        go.push_back({});
        go.back().fill(-1);
        pre.push_back(last[x]);
        for (int c = 0; c < A; c++)
        {
            for (int u = last[c]; u != -1 && go[u][x] == -1; u = pre[u])
            {
                go[u][x] = v;
            }
        }
        if (last[x] == -1)
        {
            go[0][x] = v;
        }
        last[x] = v;
    }

    bool contains(const vector<int> &s) const
    {
        // s 是 [0,A) 内的模式序列；返回 s 是否为当前文本的子序列。
        int u = 0;
        for (int x : s)
        {
            if (x < 0 || x >= A || go[u][x] == -1)
            {
                return false;
            }
            u = go[u][x];
        }
        return true;
    }

    int size() const
    {
        // 返回当前文本长度。
        return go.size() - 1;
    }
};
```

构造 `SqAM<A> q`，`add(x)` 追加 $[0,A)$ 内字符编号，`contains(s)` 判断编号序列是否为当前文本子序列，`size()` 返回文本长度；只支持末尾追加、不支持删除，也不判连续子串。加入总长 $n$ 的时间、空间均为 $O((n+1)A)$，长度 $m$ 查询 $O(m)$。依次加入 `{0,1,0,2}` 后 `{0,0,2}` 与空序列是子序列，`{1,1}`、`{2,0}` 不是。离线一次建表的等价结构见 [4.2.004 子序列自动机](#42004)。

## 4.5 子序列与高级问题

### 4.5.001 最长公共子序列 LCS

最长公共子序列求两序列中不要求连续、但保持相对顺序的最长公共子列（勿与最长公共子串混淆）。令 $f_{i,j}$ 为 $a[0,i)$、$b[0,j)$ 的 LCS 长度：末字符相等时 $f_{i,j}=f_{i-1,j-1}+1$，否则 $f_{i,j}=\max(f_{i-1,j},f_{i,j-1})$。本实现存整张 `f` 表以便回溯：从 $(n,m)$ 出发，相等就取该字符并双向回退，否则往 `f` 更大的一侧走，最后反转得到一条 LCS。

```cpp
template <class T> vector<T> lcs(const vector<T> &a, const vector<T> &b)
{
    // a、b 是两个序列；返回二者的一条最长公共子序列。
    int n = a.size(), m = b.size();
    vector<vector<int>> f(n + 1, vector<int>(m + 1));
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            if (a[i] == b[j])
            {
                f[i + 1][j + 1] = f[i][j] + 1;
            }
            else
            {
                f[i + 1][j + 1] = max(f[i][j + 1], f[i + 1][j]);
            }
        }
    }
    vector<T> c;
    for (int i = n, j = m; i && j;)
    {
        if (a[i - 1] == b[j - 1])
        {
            i--;
            j--;
            c.push_back(a[i]);
        }
        else if (f[i - 1][j] >= f[i][j - 1])
        {
            i--;
        }
        else
        {
            j--;
        }
    }
    reverse(c.begin(), c.end());
    return c;
}
```

`lcs(a,b)` 返回一条具体的最长公共子序列（元素只需支持判等）；构造与回溯的时间、空间均为 $O((n+1)(m+1))$。只要长度且串长时用位并行的 [4.5.002 Bitset 优化 LCS](#45002)；连续公共片段是另一问题，见 [4.5.004 多串最长公共子串/子序列](#45004)。最短公共超序列 [4.5.005 最短公共超序列](#45005) 直接建在这里恢复出的 LCS 上。

### 4.5.002 Bitset 优化 LCS

只要 LCS 长度、且串较长时可用位并行。把第二串每个字符的出现位置编码成位集 `mask[c]`，用经典公式 `x=mask[c]|s; s=x&~(x-((s<<1)|1))` 逐个 `a` 中字符批量推进整行（跨多机器字用手写借位减法处理进位），最后各字长的 `popcount` 之和即 LCS 长度。它把一维转移加速约一个字长倍。

```cpp
int lcsBitset(const string &a, const string &b)
{
    // a、b 是两个字节串；返回二者的最长公共子序列长度。
    int w = (b.size() + 63) / 64;
    vector<vector<unsigned long long>> mask(256, vector<unsigned long long>(w));
    vector<unsigned long long> s(w), x(w), z(w);
    for (int i = 0; i < (int)b.size(); i++)
    {
        mask[(unsigned char)b[i]][i / 64] |= 1ULL << (i % 64);
    }
    for (unsigned char c : a)
    {
        unsigned long long shift = 1, borrow = 0;
        for (int i = 0; i < w; i++)
        {
            x[i] = mask[c][i] | s[i];
            unsigned long long y = s[i] << 1 | shift;
            shift = s[i] >> 63;
            unsigned long long q = y + borrow;
            unsigned long long borrow = q < y || x[i] < q;
            z[i] = x[i] - q;
            borrow = borrow;
        }
        for (int i = 0; i < w; i++)
        {
            s[i] = x[i] & ~z[i];
        }
    }
    int ans = 0;
    for (unsigned long long x : s)
    {
        ans += popcount(x);
    }
    return ans;
}
```

`lcsBitset(a,b)` 只返回 LCS 长度，字符按 `unsigned char` 映射到 $256$ 个掩码。答案不超过 $\min(n,m)$；两个串长和 $W=\lceil m/64\rceil$ 都须能由模板中的 64 位 `int` 表示。时间为 $O(nW+m+256W)$，空间为 $O(256(W+1))$；需要恢复具体序列时改用 [4.5.001 最长公共子序列 LCS](#45001) 的普通 DP。

### 4.5.003 区间 LCS / 区间 LIS

两个不同问题放在一起对比。`rangeLCS` 是限定两串各自下标区间 $[l,r)$、$[x,y)$ 的 LCS，用滚动一维 `f` 数组做标准转移，`pre` 暂存左上角值。`rangeLIS` 是区间内严格递增子序列长度，用经典的 `lower_bound` 维护「长度 $k$ 的 LIS 的最小末元素」贪心数组。二者状态含义、答案都不同，不能相互套用；LCS 只有在一串元素互异时才能映射成另一串的位置序列转成 LIS。

```cpp
int rangeLCS(const string &a, int l, int r, const string &b, int x, int y)
{
    // a、b 是两个字符串；[l,r)、[x,y) 是参与计算的半开区间；返回区间 LCS 长度。
    vector<int> f(y - x + 1);
    for (int i = l; i < r; i++)
    {
        int pre = 0;
        for (int j = x; j < y; j++)
        {
            int old = f[j - x + 1];
            if (a[i] == b[j])
            {
                f[j - x + 1] = pre + 1;
            }
            else
            {
                f[j - x + 1] = max(f[j - x + 1], f[j - x]);
            }
            pre = old;
        }
    }
    return f.back();
}

template <class T> int rangeLIS(const vector<T> &a, int l, int r)
{
    // a 是输入序列，[l,r) 是查询区间；返回严格递增子序列长度。
    vector<T> d;
    for (int i = l; i < r; i++)
    {
        auto it = lower_bound(d.begin(), d.end(), a[i]);
        if (it == d.end())
        {
            d.push_back(a[i]);
        }
        else
        {
            *it = a[i];
        }
    }
    return d.size();
}
```

`rangeLCS(a,l,r,b,x,y)` 查两个半开子串的 LCS 长度；令 $n=r-l$、$m=y-x$，时间为 $O((n+1)(m+1))$，空间为 $O(m+1)$。`rangeLIS(a,l,r)` 查半开区间内严格 LIS 长度，令 $L=r-l$，时间为 $O(1+L\log(L+1))$，空间为 $O(L+1)$。求可恢复的整串 LCS 用 [4.5.001 最长公共子序列 LCS](#45001)；连续公共片段用 [4.5.004 多串最长公共子串/子序列](#45004)。

### 4.5.004 多串最长公共子串/子序列

公共子串（连续）与公共子序列（可跳字符）是两个问题。`multiSubstr` 对首串建 SAM，再让其余每串在自动机上跑匹配、用 `best` 记录每状态能匹配到的最长长度并沿 link 上推，`mn` 取各串在每状态的最小匹配长度，最终最大值即多串最长公共子串。`multiSubseq` 是 $k$ 维 DP：把各串下标打包成混合进制状态 `s`，同层字符全相等则从「全体减一」转移 $+1$，否则从某一维减一取最大。二者状态与答案不能混用。

```cpp
int multiSubstr(const vector<string> &a)
{
    // a 是全部输入串；返回它们的最长公共子串长度。
    if (a.empty())
    {
        return 0;
    }
    struct N
    {
        // len 是该状态的最长长度，link 是后缀链接。
        int len = 0, link = -1;
        // to 按字节保存转移目标状态。
        unordered_map<unsigned char, int> to;
    };
    vector<N> t(1);
    int last = 0;
    for (unsigned char c : a[0])
    {
        int u = t.size();
        t.push_back({t[last].len + 1});
        int p = last;
        while (p != -1 && !t[p].to.contains(c))
        {
            t[p].to[c] = u;
            p = t[p].link;
        }
        if (p == -1)
        {
            t[u].link = 0;
        }
        else
        {
            int q = t[p].to[c];
            if (t[p].len + 1 == t[q].len)
            {
                t[u].link = q;
            }
            else
            {
                int v = t.size();
                t.push_back(t[q]);
                t[v].len = t[p].len + 1;
                while (p != -1)
                {
                    auto it = t[p].to.find(c);
                    if (it == t[p].to.end() || it->second != q)
                    {
                        break;
                    }
                    it->second = v;
                    p = t[p].link;
                }
                t[q].link = t[u].link = v;
            }
        }
        last = u;
    }
    vector<int> ord(t.size()), cnt(a[0].size() + 1), mn(t.size());
    for (const N &v : t)
    {
        cnt[v.len]++;
    }
    partial_sum(cnt.begin(), cnt.end(), cnt.begin());
    for (int u = (int)t.size() - 1; u >= 0; u--)
    {
        ord[--cnt[t[u].len]] = u;
    }
    reverse(ord.begin(), ord.end());
    for (int i = 0; i < (int)t.size(); i++)
    {
        mn[i] = t[i].len;
    }
    for (int k = 1; k < (int)a.size(); k++)
    {
        vector<int> best(t.size());
        int v = 0, l = 0;
        for (unsigned char c : a[k])
        {
            while (v && !t[v].to.contains(c))
            {
                v = t[v].link;
                l = min(l, t[v].len);
            }
            if (t[v].to.contains(c))
            {
                v = t[v].to[c];
                l++;
            }
            else
            {
                l = 0;
            }
            best[v] = max(best[v], l);
        }
        for (int u : ord)
        {
            if (t[u].link >= 0)
            {
                best[t[u].link] = max(best[t[u].link], min(best[u], t[t[u].link].len));
            }
        }
        for (int i = 0; i < (int)t.size(); i++)
        {
            mn[i] = min(mn[i], best[i]);
        }
    }
    return *max_element(mn.begin(), mn.end());
}

optional<int> multiSubseq(const vector<string> &a)
{
    // a 是全部输入串；返回多串 LCS 长度，状态数无法用 size_t 表示时返回空。
    if (a.empty())
    {
        return 0;
    }
    int k = a.size();
    size_t states = 1;
    vector<size_t> st(k), base(k);
    for (int i = 0; i < k; i++)
    {
        size_t len = a[i].size() + 1;
        if (len > numeric_limits<size_t>::max() / states)
        {
            return nullopt;
        }
        base[i] = states;
        states *= len;
    }
    if (states > vector<int>().max_size())
    {
        return nullopt;
    }
    vector<int> f(states);
    for (size_t s = 1; s < states; s++)
    {
        size_t q = s;
        bool zero = false, same = true;
        int c = -1;
        size_t back = 0;
        for (int i = k - 1; i >= 0; i--)
        {
            st[i] = q / base[i];
            q %= base[i];
            if (!st[i])
            {
                zero = true;
            }
            else
            {
                int cur = (unsigned char)a[i][st[i] - 1];
                if (c == -1)
                {
                    c = cur;
                }
                else if (c != cur)
                {
                    same = false;
                }
            }
            back += base[i];
        }
        if (!zero && same)
        {
            f[s] = f[s - back] + 1;
        }
        else
        {
            for (int i = 0; i < k; i++)
            {
                if (st[i])
                {
                    f[s] = max(f[s], f[s - base[i]]);
                }
            }
        }
    }
    return f.back();
}
```

`multiSubstr(a)` 返回多串最长公共连续子串长度，答案不超过最短串长；首串产生至多 $2|a_0|-1$ 个状态，串长和状态编号须能由模板中的 64 位 `int` 表示。转移使用哈希表，总长 $L$ 下为期望 $O(L+1)$ 时间、$O(|a_0|+1)$ 额外空间。`multiSubseq(a)` 的答案同样不超过最短串长；它在混合进制状态数不能由 `size_t` 或容器表示时返回空，非空输入需要 $O(k\prod_i(|a_i|+1))$ 时间和 $O(k+\prod_i(|a_i|+1))$ 空间，只适合串少且短。

### 4.5.005 最短公共超序列

最短公共超序列是同时以 `a`、`b` 为子序列的最短串，长度为 $|a|+|b|-\mathrm{LCS}(a,b)$。构造时先取一条 LCS，再沿 LCS 逐个对齐：把两串在下一个公共字符前的未匹配字符依次补入，公共字符只加一次，末尾补上各自剩余。

```cpp
string scs(const string &a, const string &b)
{
    // a、b 是两个字符串；返回二者的一条最短公共超序列。
    vector<char> x(a.begin(), a.end()), y(b.begin(), b.end());
    auto c = lcs(x, y);
    string ans;
    int i = 0, j = 0;
    for (char z : c)
    {
        while (a[i] != z)
        {
            ans += a[i++];
        }
        while (b[j] != z)
        {
            ans += b[j++];
        }
        ans += z;
        i++;
        j++;
    }
    return ans + a.substr(i) + b.substr(j);
}
```

`scs(a,b)` 复用 [4.5.001 最长公共子序列 LCS](#45001) 恢复的 LCS 交错拼出一条最短公共超序列，时间、空间均为 $O((n+1)(m+1))$。它必须用能恢复序列的普通 LCS，位并行的 [4.5.002 Bitset 优化 LCS](#45002) 只给长度、无法回溯故不适用。

### 4.5.006 字符串周期、循环节与前后缀

字符串的周期、最小循环节、border、前后缀关系都能由前缀函数一次性得到，代码与 [4.1.002 KMP / 前缀函数](#41002) 完全相同，这里不重复：直接依赖其中的 `prefix`、`kmp`、`borders`、`minPeriod`。对非空串，前缀函数末端 $\pi_{n-1}$ 的失配链枚举全部真 border，而 $n-\pi_{n-1}$ 是最短周期；它整除 $n$ 时才是最小循环节，否则整串自身才是循环节。空串没有非空 border，`minPeriod("")` 返回 `0`。`prefix`、`borders`、`minPeriod` 为 $O(n+1)$ 时间与空间；`kmp` 对长度 $n$ 的文本和长度 $m$ 的模式为 $O(n+m+1)$ 时间、$O(m+occ+1)$ 空间，其中 $occ$ 是匹配数。需要在文本中定位循环出现用 `kmp`，判前缀对齐用 [4.1.003 扩展 KMP / Z 函数](#41003)。

### 4.5.007 前缀-子串 LCS 批量查询

给定固定的首串 $s$、次串 $t$，每个查询要求 $s[0,a)$ 与 $t[l,r)$ 的 LCS 长度。逐个调用 [4.5.003 区间 LCS / 区间 LIS](#45003) 的动态规划要 $O(Q|s||t|)$，无法承受 $Q=5\times10^5$。对每个首串前缀维护 seaweed 边界的置换表示 `h`：扫描新字符时，交换恰能继续匹配或已经跨过当前边界的端点；边界值落在 $[0,l)$ 的数量正好给出查询矩形中没有被 LCS 使用的位置数。按右端点离线并用 Fenwick 计数，即可一次扫描回答该前缀的全部查询。

```cpp
struct PrefixSubLCS
{
    struct Query
    {
        int a, l, r, id; // a 是首串前缀长度，l、r 是次串半开区间，id 是原查询编号。
    };

    static vector<int> solve(const string &s, const string &t, const vector<array<int, 3>> &ask)
    {
        // s、t 是小写字符串；ask 每项为 {前缀长度,左端,右端}，返回对应 LCS 长度。
        int m = s.size(), n = t.size(), q = ask.size();
        vector<int> ans(q);
        vector<vector<Query>> byA(m);
        for (int i = 0; i < q; i++)
        {
            auto [a, l, r] = ask[i];
            assert(0 <= a && a <= m && 0 <= l && l <= r && r <= n);
            if (a && l < r)
            {
                byA[a - 1].push_back({a, l, r, i});
            }
        }
        vector<int> h(n);
        iota(h.begin(), h.end(), 0);
        for (int a = 0; a < m; a++)
        {
            int cur = -1;
            for (int j = 0; j < n; j++)
            {
                if (s[a] == t[j] || h[j] < cur)
                {
                    swap(h[j], cur);
                }
            }
            vector<vector<Query>> byR(n);
            for (auto x : byA[a])
            {
                byR[x.r - 1].push_back(x);
            }
            vector<int> bit(n + 1);
            auto add = [&](int p)
            {
                for (p++; p <= n; p += p & -p)
                {
                    bit[p]++;
                }
            };
            auto sum = [&](int p)
            {
                int ret = 0;
                for (; p; p &= p - 1)
                {
                    ret += bit[p];
                }
                return ret;
            };
            int cnt = 0;
            for (int r = 0; r < n; r++)
            {
                if (h[r] != -1)
                {
                    add(h[r]);
                    cnt++;
                }
                for (auto x : byR[r])
                {
                    ans[x.id] = (x.r - x.l) - (cnt - sum(x.l));
                }
            }
        }
        return ans;
    }
};
```

`PrefixSubLCS::solve(s,t,ask)` 中每个 `ask[i]={a,l,r}` 使用半开下标，返回同顺序的 LCS 长度；允许空前缀和空子串。设 $m=|s|$、$n=|t|$，总时间为 $O(mn+Q\log n)$，额外空间为 $O(n+Q)$；题目范围 $m,n\le1000$、$Q\le5\times10^5$。单个查询或需要恢复子序列时仍选 [4.5.001 最长公共子序列 LCS](#45001) 或 [4.5.003 区间 LCS / 区间 LIS](#45003)，本条只输出批量长度。
