```cpp
template<class T> bool chmin(T &a, const T &b)
{
    // a 是当前值，b 是候选值；若 b 更小则更新 a 并返回 true。
    if (b < a)
    {

}

    // a 是当前值，b 是候选值；若 b 更大则更新 a 并返回 true。

    {
        a = b;
        return true;
    }
    return false;
}
```

### 14.1.001 高精度整数（BigInt）

带符号任意精度整数，内部按 $10^9$ 一块、低位块在前存放绝对值，符号单独记在 `s` 里（取 $-1,0,1$）。加减法在同号时走 `addAbs`、异号时先比较绝对值再走 `subAbs`；乘法是 $O(nm)$ 的分块卷积，单项乘积加旧块与进位严格小于 $10^{18}+2\times10^9$，用 `unsigned long long` 即可。除法 `divmod` 从最高块到最低块逐位定商——每步把当前余数左移一块，再二分出最大的 $q_i\in[0,B)$ 使 `mulAbs(d,q_i)` 不超过余数，商向零取整、余数与被除数同号。每次运算后都要 `norm` 删掉最高端的零块并把空值的符号归零，这是比较与 `str` 输出正确的前提；忘记归一化会让 $0$ 带上残留符号、进而破坏 `operator<`。

```cpp
struct BigInt
{
    // B 是每个低位块使用的进制 10^9。
    static constexpr int B = 1000000000;
    // a 按低位在前保存绝对值的 10^9 进制块。
    vector<int> a;
    // s 保存符号：-1、0、1 分别表示负数、零、正数。
    int s = 0;

    BigInt(long long x = 0)
    {
        // x 是待转换的 64 位整数；构造同值高精度整数。
        *this = x;
    }
    BigInt(const string &x)
    {
        // x 是带可选单个正负号的十进制整数串；构造同值高精度整数。
        assign(x);
    }

    BigInt &operator=(long long x)
    {
        // x 是待转换的 64 位整数；返回赋值后的当前对象。
        a.clear();
        s = (x > 0) - (x < 0);
        unsigned long long y = x < 0 ? (unsigned long long)(-(x + 1)) + 1 : (unsigned long long)x;
        while (y)
        {
            a.push_back(y % B);
            y /= B;
        }
        return *this;
    }

    BigInt &assign(const string &x)
    {
        // x 是带可选单个正负号且至少含一位数字的十进制整数串；返回赋值后的当前对象。
        a.clear();
        int p = 0;
        s = 1;
        if (!x.empty() && (x[0] == '-' || x[0] == '+'))
        {
            s = x[0] == '-' ? -1 : 1;
            p = 1;
        }
        assert(p < (int)x.size()); // 调试检查，可删。
        for (int i = p; i < (int)x.size(); i++)
        {
            assert('0' <= x[i] && x[i] <= '9'); // 调试检查，可删。
        }
        for (int r = (int)x.size(); r > p; r -= 9)
        {
            int l = max(p, r - 9);
            int v = 0;
            for (int i = l; i < r; i++)
            {
                v = v * 10 + x[i] - '0';
            }
            a.push_back(v);
        }
        norm();
        return *this;
    }

    string str() const
    {
        // 无参数；返回当前高精度整数的十进制字符串。
        if (!s)
        {
            return "0";
        }
        string x = s < 0 ? "-" : "";
        x += to_string(a.back());
        for (int i = (int)a.size() - 2; i >= 0; i--)
        {
            string t = to_string(a[i]);
            x += string(9 - t.size(), '0') + t;
        }
        return x;
    }

    void norm()
    {
        // 删除最高端零块并同步零值符号，无返回值。
        while (!a.empty() && !a.back())
        {
            a.pop_back();
        }
        if (a.empty())
        {
            s = 0;
        }
    }

    static int cmpAbs(const BigInt &x, const BigInt &y)
    {
        // x、y 是待比较整数；按 |x| 与 |y| 的大小返回 -1、0 或 1。
        if (x.a.size() != y.a.size())
        {
            return x.a.size() < y.a.size() ? -1 : 1;
        }
        for (int i = (int)(x.a.size()) - 1; i >= 0; i--)
        {
            if (x.a[i] != y.a[i])
            {
                return x.a[i] < y.a[i] ? -1 : 1;
            }
        }
        return 0;
    }

    friend bool operator==(const BigInt &x, const BigInt &y)
    {
        // x、y 是待比较整数；返回二者是否相等。
        return x.s == y.s && x.a == y.a;
    }
    friend bool operator<(const BigInt &x, const BigInt &y)
    {
        // x、y 是待比较整数；返回 x 是否小于 y。
        if (x.s != y.s)
        {
            return x.s < y.s;
        }
        if (!x.s)
        {
            return false;
        }
        int c = cmpAbs(x, y);
        return x.s > 0 ? c < 0 : c > 0;
    }
    friend bool operator!=(const BigInt &x, const BigInt &y)
    {
        // x、y 是待比较整数；返回二者是否不相等。
        return !(x == y);
    }
    friend bool operator>(const BigInt &x, const BigInt &y)
    {
        // x、y 是待比较整数；返回 x 是否大于 y。
        return y < x;
    }
    friend bool operator<=(const BigInt &x, const BigInt &y)
    {
        // x、y 是待比较整数；返回 x 是否不大于 y。
        return !(y < x);
    }
    friend bool operator>=(const BigInt &x, const BigInt &y)
    {
        // x、y 是待比较整数；返回 x 是否不小于 y。
        return !(x < y);
    }

    BigInt operator-() const
    {
        // 无参数；返回当前整数的相反数。
        BigInt x = *this;
        x.s = -x.s;
        return x;
    }

    static BigInt addAbs(const BigInt &x, const BigInt &y)
    {
        // x、y 是两个整数；返回 |x|+|y|。
        BigInt z;
        z.s = 1;
        int n = max(x.a.size(), y.a.size());
        z.a.resize(n);
        long long c = 0;
        for (int i = 0; i < n; i++)
        {
            c += (i < (int)(x.a.size()) ? x.a[i] : 0);
            c += (i < (int)(y.a.size()) ? y.a[i] : 0);
            z.a[i] = c % B;
            c /= B;
        }
        if (c)
        {
            z.a.push_back(c);
        }
        z.norm();
        return z;
    }

    // 要求 |x| >= |y|。
    static BigInt subAbs(const BigInt &x, const BigInt &y)
    {
        // x、y 满足 |x|>=|y|；返回 |x|-|y|。
        BigInt z;
        z.s = 1;
        z.a.resize(x.a.size());
        long long c = 0;
        for (int i = 0; i < (int)(x.a.size()); i++)
        {
            c = (long long)(x.a[i]) - c - (i < (int)(y.a.size()) ? y.a[i] : 0);
            if (c < 0)
            {
                c += B;
                z.a[i] = c;
                c = 1;
            }
            else
            {
                z.a[i] = c;
                c = 0;
            }
        }
        z.norm();
        return z;
    }

    BigInt &operator+=(const BigInt &y)
    {
        // y 是加数；把 y 加到当前整数并返回当前对象。
        if (!y.s)
        {
            return *this;
        }
        if (!s)
        {
            return *this = y;
        }
        if (s == y.s)
        {
            int t = s;
            *this = addAbs(*this, y);
            s = t;
        }
        else if (cmpAbs(*this, y) >= 0)
        {
            int t = s;
            *this = subAbs(*this, y);
            if (a.size())
            {
                s = t;
            }
        }
        else
        {
            int t = y.s;
            *this = subAbs(y, *this);
            if (a.size())
            {
                s = t;
            }
        }
        return *this;
    }
    BigInt &operator-=(const BigInt &y)
    {
        // y 是减数；从当前整数减去 y 并返回当前对象。
        return *this += -y;
    }

    BigInt &operator*=(const BigInt &y)
    {
        // y 是乘数；把当前整数乘以 y 并返回当前对象。
        if (!s || !y.s)
        {
            return *this = 0;
        }
        vector<int> c(a.size() + y.a.size());
        for (int i = 0; i < (int)a.size(); i++)
        {
            unsigned long long v = 0;
            for (int j = 0; j < (int)(y.a.size()) || v; j++)
            {
                unsigned long long w = c[i + j] + v;
                if (j < (int)(y.a.size()))
                {
                    w += (unsigned long long)a[i] * y.a[j];
                }
                c[i + j] = w % B;
                v = w / B;
            }
        }
        a = move(c);
        s *= y.s;
        norm();
        return *this;
    }

    static BigInt mulAbs(const BigInt &x, int y)
    {
        // x 是整数，y 满足 0<=y<B；返回 |x|*y。
        BigInt z;
        if (!y || !x.s)
        {
            return z;
        }
        z.s = 1;
        z.a.resize(x.a.size());
        unsigned long long c = 0;
        for (int i = 0; i < (int)x.a.size(); i++)
        {
            unsigned long long w = c + (unsigned long long)x.a[i] * y;
            z.a[i] = w % B;
            c = w / B;
        }
        if (c)
        {
            z.a.push_back(c);
        }
        return z;
    }

    // 商向 0 取整，余数与被除数同号；除数不能为 0。
    static pair<BigInt, BigInt> divmod(const BigInt &x, const BigInt &y)
    {
        // x 是被除数，y 是非零除数；返回向零取整的商和与 x 同号的余数。
        assert(y.s); // 调试检查，可删。
        BigInt q, r, d = y;
        d.s = 1;
        q.a.resize(x.a.size());
        q.s = x.s * y.s;
        for (int i = (int)(x.a.size()) - 1; i >= 0; i--)
        {
            r.a.insert(r.a.begin(), x.a[i]);
            r.s = 1;
            r.norm();
            int l = 0, h = B - 1;
            while (l < h)
            {
                int m = l + (h - l + 1) / 2;
                if (cmpAbs(mulAbs(d, m), r) <= 0)
                {
                    l = m;
                }
                else
                {
                    h = m - 1;
                }
            }
            q.a[i] = l;
            r = subAbs(r, mulAbs(d, l));
        }
        q.norm();
        if (r.a.size())
        {
            r.s = x.s;
        }
        return {q, r};
    }

    BigInt &operator/=(const BigInt &y)
    {
        // y 是非零除数；把当前整数改成向零取整的商并返回当前对象。
        return *this = divmod(*this, y).first;
    }
    BigInt &operator%=(const BigInt &y)
    {
        // y 是非零除数；把当前整数改成与原值同号的余数并返回当前对象。
        return *this = divmod(*this, y).second;
    }

    friend BigInt operator+(BigInt x, const BigInt &y)
    {
        // x、y 是加数；返回 x+y。
        return x += y;
    }
    friend BigInt operator-(BigInt x, const BigInt &y)
    {
        // x 是被减数，y 是减数；返回 x-y。
        return x -= y;
    }
    friend BigInt operator*(BigInt x, const BigInt &y)
    {
        // x、y 是乘数；返回 x*y。
        return x *= y;
    }
    friend BigInt operator/(BigInt x, const BigInt &y)
    {
        // x 是被除数，y 是非零除数；返回向零取整的商。
        return x /= y;
    }
    friend BigInt operator%(BigInt x, const BigInt &y)
    {
        // x 是被除数，y 是非零除数；返回与 x 同号的余数。
        return x %= y;
    }
};
```

用 `BigInt x(十进制字符串)` 或 `BigInt(long long)` 构造；字符串必须是可选单个正负号后接至少一位十进制数字，合法性用 `assert` 检查。四则与取模直接用重载的 `+ - * / %`，除数必须非零；输出调用 `str()`，本类不重载流运算符。该类适合常规十进制中等位数；十进制长乘长除使用 [14.1.019 百万位大整数](#141019) 的 `FastBigInt`；需要与二进制 limb 互转时使用同条的 `HexBigInt::fromDecimal` / `toDecimal`。设两数块数为 $n,m$：比较与加减为 $O(\max(n,m))$，乘法为 $O(nm)$，当前逐块二分除法为 $O(n(n+m)\log B)$，额外空间为 $O(n+m)$。需要幂或进制转换见 [14.1.006 高精度进制转换与快速幂](#141006)，只求向下取整平方根见 [14.1.016 高精度整数平方根](#141016)。

除法逐位定商和跨块进位最易写错。下面检查 $10^9$ 分块边界和一组固定四则答案；长位数压力测试放在本地测试文件：

```cpp

    assert((BigInt("1000000000") - 1).str() == "999999999");

    BigInt x("1234567890123"), y("987654321");
    assert((x + y).str() == "1235555544444");
    assert((x - y).str() == "1233580235802");
    assert((x * y).str() == "1219326311247834171483");

    assert((x % y).str() == "987643194");
```

### 14.1.002 分数类与有理数还原

`Frac` 用分子和正分母保存唯一既约表示，乘除先交叉约分，比较用 `__int128` 交叉相乘。模有理数还原则给定剩余 $0\le a<m$ 与界 $b\ge1$，求满足

$$
p\equiv aq\pmod m,\qquad |p|\le b,\quad 1\le q\le b,\quad \gcd(p,q)=\gcd(q,m)=1
$$

的分数 $p/q$，要求 $2b^2<m$。两解之差 $p_1q_2-p_2q_1$ 是 $m$ 的倍数，且绝对值小于 $m$，所以解至多一个。截断扩展欧几里得始终保持 $r_i\equiv at_i\pmod m$，在首个 $r_i\le b$ 处检查分母界与互素性；小分数必出现在对应的连分数收敛项中。普通 `exgcd` 只返回最终系数，不能代替这里的提前停止。

```cpp
// 依赖 14.1.001 高精度整数（BigInt）
struct Frac
{
    int x, y; // x、y 是既约分子和正分母。

    Frac(int x = 0, int y = 1) : x(x), y(y)
    {
        // x、y 是分子和非零分母；构造既约分数 x/y。
        assert(y); // 调试检查，可删。
        if (y < 0)
        {
            x = -x;
            y = -y;
        }
        int g = gcd(abs(x), y);
        this->x = x / g;
        this->y = y / g;
    }

    friend Frac operator+(const Frac &a, const Frac &b)
    {
        // a、b 是两个分数；返回 a+b。
        int g = gcd(a.y, b.y);
        return {a.x * (b.y / g) + b.x * (a.y / g), a.y / g * b.y};
    }
    friend Frac operator-(const Frac &a, const Frac &b)
    {
        // a 是被减数，b 是减数；返回 a-b。
        int g = gcd(a.y, b.y);
        return {a.x * (b.y / g) - b.x * (a.y / g), a.y / g * b.y};
    }
    friend Frac operator*(const Frac &a, const Frac &b)
    {
        // a、b 是两个分数；返回 a*b。
        int g = gcd(abs(a.x), b.y);
        int h = gcd(abs(b.x), a.y);
        return {a.x / g * (b.x / h), a.y / h * (b.y / g)};
    }
    friend Frac operator/(const Frac &a, const Frac &b)
    {
        // a 是被除数，b 是非零除数；返回 a/b。
        assert(b.x); // 调试检查，可删。
        int g = gcd(abs(a.x), abs(b.x));
        int h = gcd(a.y, b.y);
        return {a.x / g * (b.y / h), a.y / h * (b.x / g)};
    }
    friend bool operator==(const Frac &a, const Frac &b)
    {
        // a、b 是两个分数；返回二者是否相等。
        return a.x == b.x && a.y == b.y;
    }
    friend bool operator<(const Frac &a, const Frac &b)
    {
        // a、b 是两个分数；返回 a 是否小于 b。
        return (__int128)a.x * b.y < (__int128)b.x * a.y;
    }
};

optional<Frac> ratRecover(int a, int m, int b)
{
    // 0<=a<m，b>=1，2b^2<m；返回满足 p=a*q (mod m)、|p|<=b、1<=q<=b
    // 且 gcd(q,m)=1 的唯一既约分数 p/q；不存在时返回空。
    assert(0 <= a && a < m && b >= 1 && 2 * (__int128)b * b < m);
    int r0 = m, r1 = a, t0 = 0, t1 = 1; // 始终有 ri=a*ti (mod m)。
    while (r1 > b)
    {
        int k = r0 / r1;
        int r = r0 - k * r1;
        // 系数幅值不超过 m；乘减局部加宽，避免中间乘积溢出。
        int t = (int)((__int128)t0 - (__int128)k * t1);
        r0 = r1;
        r1 = r;
        t0 = t1;
        t1 = t;
    }
    if (t1 < 0)
    {
        r1 = -r1;
        t1 = -t1;
    }
    if (t1 > b || gcd(abs(r1), t1) != 1 || gcd(t1, m) != 1) return nullopt;
    return Frac(r1, t1);
}
```

用 `Frac(x,y)` 构造，其中 $y\ne0$；使用 `+ - * / < ==`，除数必须非零。调用者须保证绝对值、变号及四则中间量均可由当前 `int` 表示；默认整数宏下为有符号 64 位，关闭后为 32 位。构造与四则各用 $O(\log M)$ 时间、$O(1)$ 额外空间，$M$ 是输入绝对值的最大值；比较为 $O(1)$。超出范围时使用 [14.1.001 高精度整数](#141001)。

`ratRecover(a,m,b)` 返回唯一的小分数，无解返回 `nullopt`，允许合数模数；时间为 $O(\log m)$，额外空间为 $O(1)$，按机器字运算计。它从模剩余恢复精确分数；有理数上下界逼近使用 [7.5.005 Stern–Brocot 树与 Farey 序列](07-数论.md#75005)，Python 精确分数见 [14.1.014 Python 常用库](#141014)。下面检查固定四则答案与还原的有解、无解情况：

```cpp
signed main()
{
    assert(Frac(2, -4) == Frac(-1, 2));
    assert(Frac(1, 6) + Frac(1, 3) == Frac(1, 2));
    assert(Frac(5, 6) - Frac(1, 4) == Frac(7, 12));
    assert(Frac(-7, 3) < Frac(-2, 1));
    assert(Frac(2, 3) * Frac(9, 4) == Frac(3, 2));
    assert(Frac(2, 3) / Frac(4, 9) == Frac(3, 2));
    assert(ratRecover(11323, 100000, 200) == Frac(119, 53));
    assert(ratRecover(9, 17, 2) == Frac(1, 2));
    assert(ratRecover(8, 17, 2) == Frac(-1, 2));
    assert(!ratRecover(400, 1000, 20));
}
```

### 14.1.003 表达式解析：中缀转后缀、表达式树

递归下降先构建语法树：`expr` 处理 $+,-$，`term` 处理 $*,/$，`atom` 处理一元符号、括号和十进制非负数字串。层级实现优先级，二元运算用循环实现左结合；连续一元正负号按负号个数的奇偶折叠为一个 `u+` 或 `u-` 节点。括号不产生节点，所有节点按后序追加，因此孩子编号小于父节点编号，节点记号依次排列就是后缀序列。

`parse` 只检查语法，允许先建树再选择求值类型；语法错误会清空树。`value` 按后序计算当前 `int` 范围内的整数值，除法向零取整；除零或任一运算结果越界返回 `nullopt`。数字串幅值允许恰为最大正数加一，但必须直接被一元负号消费成最小负数，不能作为二元操作数或最终正数。故默认整数宏下 `-(9223372036854775808)` 合法，`9223372036854775808-1` 不合法；连续符号先折叠，`--9223372036854775808` 仍越界。

```cpp
// 依赖 14.1.001 高精度整数（BigInt）
struct Expr
{
    struct Node
    {
        string tok;               // 数字串、二元运算符，或折叠连续符号后的一元 u+、u-。
        array<int, 2> ch{-1, -1}; // [0] 左或唯一孩子，[1] 右孩子；-1 表示没有。
    };
    vector<Node> a; // 后序排列的语法树节点；孩子编号小于父节点编号，仅读取。
    int rt = -1;    // 根编号；未成功解析时为 -1。

private:
    string s;  // 当前中缀表达式。
    int p = 0; // 下一个待读字符的位置。

    void skip()
    {
        // 无参数；把 p 移过连续空白，无返回值。
        while (p < (int)s.size() && isspace((unsigned char)s[p]))
        {
            p++;
        }
    }

    int expr()
    {
        // 无参数；解析当前位置的加减表达式，返回根编号；语法错误返回 -1。
        int x = term();
        skip();
        while (x >= 0 && p < (int)s.size() && (s[p] == '+' || s[p] == '-'))
        {
            string op(1, s[p++]);
            int y = term();
            if (y < 0) return -1;
            a.push_back({op, {x, y}});
            x = (int)a.size() - 1;
            skip();
        }
        return x;
    }

    int term()
    {
        // 无参数；解析当前位置的乘除项，返回根编号；语法错误返回 -1。
        int x = atom();
        skip();
        while (x >= 0 && p < (int)s.size() && (s[p] == '*' || s[p] == '/'))
        {
            string op(1, s[p++]);
            int y = atom();
            if (y < 0) return -1;
            a.push_back({op, {x, y}});
            x = (int)a.size() - 1;
            skip();
        }
        return x;
    }

    int atom()
    {
        // 无参数；解析一元符号及括号或数字，返回根编号；语法错误返回 -1。
        skip();
        bool sign = false, neg = false;
        while (p < (int)s.size() && (s[p] == '+' || s[p] == '-'))
        {
            sign = true;
            neg ^= s[p++] == '-';
            skip();
        }
        if (p == (int)s.size()) return -1;
        int x;
        if (s[p] == '(')
        {
            p++;
            x = expr();
            if (x < 0 || p == (int)s.size() || s[p] != ')') return -1;
            p++;
        }
        else
        {
            int l = p;
            while (p < (int)s.size() && s[p] >= '0' && s[p] <= '9')
            {
                p++;
            }
            if (l == p) return -1;
            a.push_back({s.substr(l, p - l), {-1, -1}});
            x = (int)a.size() - 1;
        }
        if (sign)
        {
            a.push_back({neg ? "u-" : "u+", {x, -1}});
            x = (int)a.size() - 1;
        }
        return x;
    }

public:
    bool parse(const string &t)
    {
        // t 是中缀表达式；成功时保存 a、rt 并返回 true，语法错误时清空树并返回 false。
        s = t;
        p = 0;
        a.clear();
        rt = expr();
        skip();
        if (rt < 0 || p != (int)s.size())
        {
            a.clear();
            rt = -1;
            return false;
        }
        return true;
    }

    vector<string> postfix() const
    {
        // 无参数；返回当前树的后缀记号序列，一元运算记作 u+、u-；无树时返回空序列。
        vector<string> res;
        for (const auto &v : a)
        {
            res.push_back(v.tok);
        }
        return res;
    }

    optional<int> value() const
    {
        // 无参数；返回当前树在 int 范围内的整数值；无树、除零或运算结果越界时返回空。
        // __int128 承载字面量幅值和乘法中间量，关闭整数宏后仍显式加宽。
        if (rt < 0) return nullopt;
        vector<__int128> val(a.size());
        for (int i = 0; i < (int)a.size(); i++)
        {
            const auto &[tok, ch] = a[i];
            auto &v = val[i];
            if (ch[0] < 0)
            {
                for (char c : tok)
                {
                    v = v * 10 + c - '0';
                    if (v > (__int128)numeric_limits<int>::max() + 1) return nullopt;
                }
                continue;
            }
            __int128 x = val[ch[0]];
            if (ch[1] < 0)
            {
                v = tok == "u-" ? -x : x;
            }
            else
            {
                __int128 y = val[ch[1]];
                if (x > numeric_limits<int>::max() || y > numeric_limits<int>::max())
                    return nullopt;
                if (tok == "+") v = x + y;
                if (tok == "-") v = x - y;
                if (tok == "*") v = x * y;
                if (tok == "/")
                {
                    if (!y) return nullopt;
                    v = x / y;
                }
            }
            if (v < numeric_limits<int>::min() || v > numeric_limits<int>::max()) return nullopt;
        }
        if (val[rt] > numeric_limits<int>::max()) return nullopt;
        return (int)val[rt];
    }

    optional<int> eval(const string &t)
    {
        // t 是中缀表达式；解析后返回当前树的值，语法、除零或数值错误时返回空。
        if (!parse(t)) return nullopt;
        return value();
    }
};
```

`e.parse(t)` 成功后可读取 `e.a`、`e.rt`，调用 `e.postfix()` 或 `e.value()`；不要修改节点和根编号。`e.eval(t)` 合并解析与求值。语法正确但数值失败时仍保留树；空串、残缺括号、非法字符和多余记号均为语法错误。只支持整数、`+ - * /`、括号与空白，不含变量、函数或乘方。解析、生成后缀及求值各为 $O(n)$ 时间、$O(n)$ 额外空间，$n$ 是字符串长度；递归深度为括号嵌套深度 $d$，调用者须保证栈足够，连续符号与同层二元链不增加递归深度。精确有理数求值可用 [14.1.002 分数类](#141002)，任意精度整数用 [14.1.001 高精度整数](#141001) 替换求值部分，并重新约定除法与失败语义。

下面检查树、后缀序列及固定求值结果：

```cpp
signed main()
{
    Expr e;
    assert(e.parse("-(2+3)*4"));
    assert((e.postfix() == vector<string>{"2", "3", "+", "u-", "4", "*"}));
    assert(e.a[e.rt].tok == "*" && e.value() == -20);
    assert(e.eval("2+3*4") == 14);
    assert(e.eval("8/3*3") == 6);
    assert(e.eval("2*-3") == -6);
    assert(e.eval("--5") == 5);
    assert(e.eval("((2+3)*(4-1))") == 15);
    assert(e.parse("1/0") && !e.value());
    assert(!e.parse("1+") && e.a.empty() && e.rt == -1);
    assert(e.postfix().empty() && !e.value());
}
```

### 14.1.004 德州扑克牌型判断（五张牌与多手牌最佳五张）

`pokerScore` 把五张牌压成一个可字典序比较的评分数组 `z`：`z[0]` 是牌型等级（$8$ 同花顺、$7$ 四条、$6$ 葫芦、$5$ 同花、$4$ 顺子、$3$ 三条、$2$ 两对、$1$ 一对、$0$ 高牌），其后各维依次填决胜的关键点数，于是两手牌直接按 `array` 字典序比大小即可。点数用 $2\!-\!14$（A 记 $14$），先降序排序，并特判把 A-5-4-3-2 的“轮子顺”改写成 $5,4,3,2,1$ 使最小顺子也能被检出；`flush` 看五张花色是否全同，`straight` 看相邻差是否恒为 $1$，再用按出现次数降序的 `(cnt,rank)` 对区分四条 / 葫芦 / 三条 / 两对 / 一对。判定顺序必须“同花顺→四条→葫芦→……”从强到弱，否则同花顺会被同花或顺子截胡。七张或更多牌取最佳五张时，在外层枚举五张子集分别调用本函数取字典序最大评分即可，本条只实现五张的核心比较。

```cpp
array<int, 6> pokerScore(vector<int> r, vector<int> suit)
{
    // r、suit 依次给出五张牌的点数与花色；返回可直接字典序比较的六元评分。
    assert(r.size() == 5 && suit.size() == 5); // 调试检查，可删。
    set<pair<int, int>> card;
    for (int i = 0; i < 5; i++)
    {
        assert(2 <= r[i] && r[i] <= 14); // 调试检查，可删。
        assert(0 <= suit[i] && suit[i] < 4); // 调试检查，可删。
        card.insert({r[i], suit[i]});
    }
    assert(card.size() == 5); // 调试检查，可删。
    sort(r.begin(), r.end(), greater<>());
    if (r == vector<int>{14, 5, 4, 3, 2})
    {
        r = {5, 4, 3, 2, 1};
    }
    bool flush = count(suit.begin(), suit.end(), suit[0]) == 5;
    bool straight = true;
    for (int i = 1; i < 5; i++)
    {
        straight &= r[i - 1] == r[i] + 1;
    }
    map<int, int, greater<>> f;
    for (int x : r)
    {
        f[x]++;
    }
    vector<pair<int, int>> a;
    for (auto [x, c] : f)
    {
        a.push_back({c, x});
    }
    sort(a.begin(), a.end(), greater<>());
    array<int, 6> z{};
    if (straight && flush)
    {
        z[0] = 8;
        z[1] = r[0];
    }
    else if (a[0].first == 4)
    {
        z[0] = 7;
        z[1] = a[0].second;
        z[2] = a[1].second;
    }
    else if (a[0].first == 3 && a[1].first == 2)
    {
        z[0] = 6;
        z[1] = a[0].second;
        z[2] = a[1].second;
    }
    else if (flush)
    {
        z[0] = 5;
        copy(r.begin(), r.end(), z.begin() + 1);
    }
    else if (straight)
    {
        z[0] = 4;
        z[1] = r[0];
    }
    else if (a[0].first == 3)
    {
        z[0] = 3;
        z[1] = a[0].second;
        vector<int> q;
        for (auto [c, x] : a)
        {
            if (c == 1)
            {
                q.push_back(x);
            }
        }
        sort(q.begin(), q.end(), greater<>());
        copy(q.begin(), q.end(), z.begin() + 2);
    }
    else if (a[0].first == 2 && a[1].first == 2)
    {
        z[0] = 2;
        z[1] = max(a[0].second, a[1].second);
        z[2] = min(a[0].second, a[1].second);
        z[3] = a[2].second;
    }
    else if (a[0].first == 2)
    {
        z[0] = 1;
        z[1] = a[0].second;
        vector<int> q;
        for (auto [c, x] : a)
        {
            if (c == 1)
            {
                q.push_back(x);
            }
        }
        sort(q.begin(), q.end(), greater<>());
        copy(q.begin(), q.end(), z.begin() + 2);
    }
    else
    {
        copy(r.begin(), r.end(), z.begin() + 1);
    }
    return z;
}
```

调用 `pokerScore(r,suit)`，两向量必须按相同次序描述五张互不重复的实体牌，点数在 $[2,14]$，花色在 $[0,4)$；返回数组可直接用 `<`、`>` 比较。单手判定为 $O(1)$ 时间和空间；七张牌枚举 $\binom{7}{5}=21$ 个子集仍为 $O(1)$。胜率估计需在外层采样，不属于牌型判定。

下面固定检查顺子同花、A-5 轮子顺、葫芦、两对踢脚和普通高牌；完整牌型覆盖放在本地测试文件：

```cpp

{
    assert((pokerScore({14, 13, 12, 11, 10}, {0, 0, 0, 0, 0})
            == array<int, 6>{8, 14, 0, 0, 0, 0}));
    assert((pokerScore({4, 4, 4, 2, 2}, {0, 1, 2, 0, 1})
            == array<int, 6>{6, 4, 2, 0, 0, 0}));
    assert((pokerScore({14, 5, 4, 3, 2}, {0, 1, 2, 3, 0})
            == array<int, 6>{4, 5, 0, 0, 0, 0}));
    assert((pokerScore({14, 14, 2, 2, 9}, {0, 1, 0, 1, 2})
            == array<int, 6>{2, 14, 2, 9, 0, 0}));
    assert((pokerScore({14, 11, 9, 5, 2}, {0, 1, 2, 3, 0})
```

### 14.1.005 日期与时间换算

公历日期与“距 1970-01-01 的天数”互转，核心是 Howard Hinnant 的纪元–年–日算法：`days()` 先把 $\le2$ 月的月份并入上一年（`yy=y-(m<=2)`），以 $400$ 年为一个纪元（恰 $146097$ 天）算出纪元编号 `era`、纪元内年 `yo` 与内日 `dy`，合成后减 $719468$ 把原点对齐到 1970-01-01；`fromDays` 是它逐步的逆变换。闰年按公历“四年一闰、百年不闰、四百年再闰”判定，月天数查表并对二月补闰日——所以任何日期加减都必须先转成连续日序，绝不能按固定 $365$ 天硬算。星期由 `(days()%7+11)%7` 给出（$0$ 为周日），$+11$ 是把可能为负的取模结果掰回 $[0,7)$ 的偏移。

```cpp
// 依赖 14.1.001 高精度整数（BigInt）
struct Date
{
    // y 保存公历年份；完整 64 位日序对应的年份可能超过 32 位。
    long long y;
    // m 保存公历月份。
    int m;
    // d 保存公历月内日期。
    int d;

    static bool leap(long long y)
    {
        // y 是公历年份；返回该年是否为闰年。
        return y % 4 == 0 && (y % 100 != 0 || y % 400 == 0);
    }

    static int monthDays(long long y, int m)
    {
        // y、m 是公历年份和月份；返回该月的天数。
        static constexpr array<int, 13> a = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
        assert(1 <= m && m <= 12); // 调试检查，可删。
        return a[m] + (m == 2 && leap(y));
    }

    bool valid() const
    {
        // 无参数；返回当前年月日是否为合法公历日期。
        return 1 <= m && m <= 12 && 1 <= d && d <= monthDays(y, m);
    }

    // 与 1970-01-01 相差的天数，可为负。
    long long days() const
    {
        // 无参数；返回当前日期与 1970-01-01 相差的天数。
        assert(valid()); // 调试检查，可删。
        __int128 yy = (__int128)y - (m <= 2);
        __int128 era = (yy >= 0 ? yy : yy - 399) / 400;
        __int128 yo = yy - era * 400;
        __int128 mp = m + (m > 2 ? -3 : 9);
        __int128 dy = (153 * mp + 2) / 5 + d - 1;
        __int128 de = yo * 365 + yo / 4 - yo / 100 + dy;
        __int128 z = era * 146097 + de - 719468;
        assert(LLONG_MIN <= z && z <= LLONG_MAX); // 调试检查，可删。
        return (long long)z;
    }

    // days() 的逆变换。
    static Date fromDays(long long z)
    {
        // z 是相对 1970-01-01 的天数；返回对应公历日期。
        __int128 zz = (__int128)z + 719468;
        __int128 era = (zz >= 0 ? zz : zz - 146096) / 146097;
        __int128 de = zz - era * 146097;
        __int128 yo = (de - de / 1460 + de / 36524 - de / 146096) / 365;
        __int128 yy = yo + era * 400;
        __int128 dy = de - (365 * yo + yo / 4 - yo / 100);
        __int128 mp = (5 * dy + 2) / 153;
        __int128 dd = dy - (153 * mp + 2) / 5 + 1;
        __int128 mm = mp + (mp < 10 ? 3 : -9);
        yy += mm <= 2;
        return {(long long)yy, (int)mm, (int)dd};
    }

    // 周日为 0，周一为 1，...，周六为 6。
    int weekday() const
    {
        // 无参数；返回当前日期的星期编号，周日为 0。
        return (int)((days() % 7 + 11) % 7);
    }
};
```

构造 `Date{y,m,d}` 后可用 `valid()` 判断日期是否合法；`days()` 要求日期合法，并要求相对 1970-01-01 的日序可由 `long long` 表示。年份显式使用 `long long`，关闭整数宏后也能承载任意 64 位日序对应的年份；`Date::fromDays(z)` 接受任意 64 位日序，`weekday()` 返回 $[0,7)$，周日为 $0$。所有接口均为 $O(1)$ 时间和空间。Python 的 `date`、`timedelta` 见 [14.1.014 Python：`datetime`、`Fraction`、递归深度与文件 IO](#141014)。

下面固定检查纪元、世纪闰年和日期往返：

```cpp

    assert((Date{1970, 1, 1}.weekday() == 4));
    assert(!Date::leap(1900));
    assert(Date::leap(2000));
    long long z = Date{2000, 2, 29}.days();

}
```

### 14.1.006 高精度进制转换与快速幂

两个只依赖字符串或大整数的小工具。`convertBase` 在 $2\!-\!62$ 进制间转换任意长带符号整数，数码严格按 `0-9A-Za-z` 映射到 $0\!-\!61$，因此大小写在 $37$ 进制以上含义不同；它反复对目标进制做整型短除，每次余数逆序拼成答案，零统一输出为 `0`。`bigPow` 是 `BigInt` 上的二进制快速幂——按指数 `e` 的二进制位决定是否把当前底数乘入答案、同时底数自乘，乘法次数从 $e$ 降到 $O(\log e)$，底层乘法复用 [14.1.001 高精度整数（BigInt）](#141001)；`convertBase` 本身不用 `BigInt`。

```cpp
// 依赖 14.1.001 高精度整数（BigInt）
string convertBase(string s, int from, int to)
{
    // s 是字符序 0-9A-Za-z 下带可选正负号的 from 进制整数；返回同字符序、无前导零的 to 进制字符串。
    assert(!s.empty()); // 调试检查，可删。
    assert(2 <= from && from <= 62 && 2 <= to && to <= 62); // 调试检查，可删。
    const string digit = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";
    bool neg = s[0] == '-';
    int p = neg || s[0] == '+';
    assert(p < (int)s.size()); // 调试检查，可删。
    vector<int> a;
    for (; p < (int)s.size(); p++)
    {
        size_t pos = digit.find(s[p]);
        assert(pos != string::npos && pos < (size_t)from); // 调试检查，可删。
        int x = pos;
        a.push_back(x);
    }
    while (a.size() > 1 && a[0] == 0)
    {
        a.erase(a.begin());
    }
    string ans;
    while (!(a.size() == 1 && a[0] == 0))
    {
        vector<int> b;
        int rem = 0;
        for (int x : a)
        {
            int cur = rem * from + x;
            if (!b.empty() || cur / to)
            {
                b.push_back(cur / to);
            }
            rem = cur % to;
        }
        ans.push_back(digit[rem]);
        a = b.empty() ? vector<int>{0} : move(b);
    }
    if (ans.empty())
    {
        return "0";
    }
    reverse(ans.begin(), ans.end());
    return neg ? "-" + ans : ans;
}

BigInt bigPow(BigInt a, int e)
{
    // a 是任意精度整数，e 是非负整数；返回 a^e，依赖 14.1.001 的 BigInt 乘法。
    assert(e >= 0); // 调试检查，可删。
    BigInt ans = 1;
    while (e)
    {
        if (e & 1)
        {
            ans *= a;
        }
        e >>= 1;
        if (e)
        {
            a *= a;
        }
    }
    return ans;
}
```

`convertBase(s,from,to)` 要求 `s` 除可选首字符 `+`、`-` 外至少有一位合法数码；`from`、`to` 均在 $[2,62]$，输出沿用同一字符序并去掉前导零。先引入 [14.1.001 高精度整数（BigInt）](#141001) 后可调用 `bigPow(a,e)`，其中 $e\ge0$。设转换途中数值最多 $D$ 位，反复短除用 $O(D^2)$ 时间、$O(D)$ 空间；快速幂做 $O(\log e)$ 次高精度乘法。`convertBase("-z",62,10)` 返回 `-61`，`convertBase("61",10,62)` 返回 `z`，`bigPow(BigInt("12"),10).str()` 返回 `61917364224`。

### 14.1.007 9×9 数独字典序最小解

标准 $9\times9$ 数独求解，用三个位掩码 `row`/`col`/`box` 各记一行、一列、一宫已用的数字（第 $v$ 位表示数字 $v$，常量 `1022` 即二进制 `1111111110`、圈定候选 $1\!-\!9$）。要得到标题承诺的字典序最小解，`dfs` 必须严格按行优先顺序逐格填、每格从最小候选试起、找到首解即停：先固定的靠前格因此拿到尽可能小的值，整盘按行读出才最小。原实现此处改用了最少候选数（MRV）启发式挑格，能快速求出某个合法解却不保证字典序最小——空盘下它给出的第 $4$ 行是 `231674895`，而真正字典序最小解的第 $4$ 行为 `214365897`；下面已改回行优先扫描，参数 `z` 也回归“下一个待检查格编号”的本义。`solve` 先把初盘已填数字灌入三个掩码并顺带检测冲突，再调 `dfs`。

```cpp
struct Sudoku
{
    // a 保存 9x9 棋盘，0 表示空格，1..9 表示已填数字。
    array<array<int, 9>, 9> a{};
    // row[i] 的第 v 位表示第 i 行是否已使用数字 v。
    array<int, 9> row{};
    // col[j] 的第 v 位表示第 j 列是否已使用数字 v。
    array<int, 9> col{};
    // box[b] 的第 v 位表示第 b 个 3x3 宫是否已使用数字 v。
    array<int, 9> box{};

    bool dfs(int z = 0)
    {
        // 参数 z 表示 按行优先扫描时下一个待检查的格编号；从该格起继续深度优先搜索，成功返回 true。
        while (z < 81 && a[z / 9][z % 9])
        {
            z++;
        }
        if (z == 81)
        {
            return true;
        }
        int x = z / 9;
        int y = z % 9;
        int b = x / 3 * 3 + y / 3;
        int m = ~(row[x] | col[y] | box[b]) & 1022;
        while (m)
        {
            int v = countr_zero((unsigned)m);
            m &= m - 1;
            a[x][y] = v;
            row[x] |= 1 << v;
            col[y] |= 1 << v;
            box[b] |= 1 << v;
            if (dfs(z + 1))
            {
                return true;
            }
            row[x] ^= 1 << v;
            col[y] ^= 1 << v;
            box[b] ^= 1 << v;
            a[x][y] = 0;
        }
        return false;
    }

    bool solve()
    {
        // 无参数；验证初盘后求字典序最小完整数独，成功返回 true，矛盾时返回 false。
        row.fill(0);
        col.fill(0);
        box.fill(0);
        for (int i = 0; i < 9; i++)
        {
            for (int j = 0; j < 9; j++)
            {
                assert(0 <= a[i][j] && a[i][j] <= 9); // 调试检查，可删。
                if (a[i][j])
                {
                    int b = i / 3 * 3 + j / 3;
                    int q = 1 << a[i][j];
                    if ((row[i] | col[j] | box[b]) & q)
                    {
                        return false;
                    }
                    row[i] |= q;
                    col[j] |= q;
                    box[b] |= q;
                }
            }
        }
        return dfs();
    }
};
```

填好标准 $9\times9$ 初盘后调用 `solve()`，其中 $0$ 表示空格，非零格必须在 $[1,9]$。成功时 `a` 就地改成按行展开后字典序最小的完整解；取值合法但已有行、列或宫冲突时返回 `false`。设空格数为 $k$，最坏 $O(9^k)$ 时间、$O(k)$ 递归栈。需要更快地求任意一组解时可改用 Dancing Links；同类构造见 [14.1.018 N 皇后构造解](#141018)。

### 14.1.008 装箱问题

装箱是 NP-hard，这里给三档实现。`firstFitDec` 是首次适应下降近似：物品按体积降序，逐个放进第一个装得下的箱（`rem` 按开箱顺序存各箱剩余容量），放不下就开新箱，简单但当前线性找箱为 $O(n^2)$。`firstFit` 保留原输入顺序，复用 [3.3.001 单点修改线段树与摘要二分](03-数据结构.md#33001) 的 `Seg<MaxInfo>`：每个叶子是一个箱子的剩余容量，区间摘要取最大值，`findFirst` 找最早还能容纳当前物品的箱子；尚未使用的叶子初值也是箱容量，所以第一次走到它就等价于开新箱。`exactBins` 是 $O(n2^n)$ 的子集 DP 求精确最少箱数：`dp[s]` 记“已装物品集合为 $s$”时按字典序最优的二元组 $(\text{已用箱数},\text{末箱已用容量})$，转移把某个未装物品 $i$ 塞进当前末箱（装得下）或新开一箱、取字典序最小。以 $(\text{箱数},\text{末箱负载})$ 为最小化目标是这套 DP 正确的关键：箱数相同再压低末箱负载，才能给后续物品留最大空隙。

```cpp
// 依赖 3.3.001 单点修改线段树与摘要二分
// 知识点 14.1.008：装箱问题的近似与小规模精确解。
namespace bin_packing
{
int firstFitDec(vector<int> a, int cap)
{
    // a 是物品体积，cap 是箱容量；返回降序首次适应使用的箱数。
    assert(cap >= 0); // 调试检查，可删。
    for (int x : a)
    {
        assert(0 <= x && x <= cap); // 调试检查，可删。
    }
    sort(a.begin(), a.end(), greater<>());
    vector<int> rem;
    for (int x : a)
    {
        int p = 0;
        while (p < (int)rem.size() && rem[p] < x)
        {
            p++;
        }
        if (p == (int)rem.size())
        {
            rem.push_back(cap);
        }
        rem[p] -= x;
    }
    return rem.size();
}

pair<int, vector<int>> firstFit(const vector<int> &a, int cap)
{
    // 按输入顺序放置，返回使用箱数与每件物品所在的 0 下标箱号。
    assert(cap >= 0); // 调试检查，可删
    int n = (int)a.size();
    vector<int> rem(n, cap), box;
    vector<MaxInfo> init(n, MaxInfo(cap));
    Seg<MaxInfo> st(init);
    box.reserve(n);
    int used = 0;
    for (int x : a)
    {
        assert(0 <= x && x <= cap); // 调试检查，可删
        int p = st.findFirst(0, n, [&](MaxInfo z) { return z.val >= x; });
        assert(p != -1); // 前 i 件物品至多占用 i 个箱子
        rem[p] -= x;
        st.set(p, MaxInfo(rem[p]));
        used = max(used, p + 1);
        box.push_back(p);
    }
    return {used, box};
}

int exactBins(const vector<int> &a, int cap)
{
    // a 是至多 24 个物品体积，cap 是箱容量；返回最少箱数。
    int n = (int)a.size();
    assert(0 <= n && n <= 24); // 调试检查，可删。
    assert(cap >= 0); // 调试检查，可删。
    for (int x : a)
    {
        assert(0 <= x && x <= cap); // 调试检查，可删。
    }
    // State[0..1] 依次是已用箱数与最后一个箱的已用容量。
    using State = array<int, 2>;
    State inf{n + 1, 0};
    vector<State> dp(1LL << n, inf);
    dp[0] = {1, 0};
    for (int s = 0; s < (1LL << n); s++)
    {
        for (int i = 0; i < n; i++)
        {
            if (s >> i & 1)
            {
                continue;
            }
            State cur = dp[s];
            if (a[i] <= cap - cur[1])
            {
                cur[1] += a[i];
            }
            else
            {
                cur[0]++;
                cur[1] = a[i];
            }
            dp[s | 1LL << i] = min(dp[s | 1LL << i], cur);
        }
    }
    return n == 0 ? 0 : dp.back()[0];
}
}
```

`firstFitDec(a,cap)` 返回 FFD 的近似箱数；`firstFit(a,cap)` 返回按输入顺序 First-Fit 使用的箱数及每件物品的 $0$ 下标箱号；二者都要求物品体积落在 $[0,cap]$，但排序会改变放置顺序，结果不能混称。当前 FFD 为 $O(n^2)$，线段树 First-Fit 为 $O(n\log n)$ 时间、$O(n)$ 空间。`exactBins(a,cap)` 要求物品数 $\le24$，以 $O(n2^n)$ 时间、$O(2^n)$ 空间返回精确最少箱数。物品多且必须保持到达顺序时用 `firstFit`，允许重排且只需简单近似时用 FFD，$n\le24$ 要精确最优用 `exactBins`；若还要在容量约束下优化物品价值则退化为背包，属第 12 章动态规划范畴。

### 14.1.009 选数与网格路径计数

只向下或向右走的网格路径计数。无障碍时从 $(0,0)$ 到 $(h-1,w-1)$ 有 $\binom{h+w-2}{h-1}$ 条路径。有障碍时把障碍与终点按坐标排序，令 `dp[i]` 为到第 $i$ 个点且不经过任何更早障碍的路径数；先计全部路径，再减去每个能到达它的障碍 $j$ 所截获的 $dp_j\binom{(x_i-x_j)+(y_i-y_j)}{x_i-x_j}$。组合数复用第 8 章代表实现，本条只保留容斥转移。

```cpp
// 知识点 14.1.009：障碍点网格最短路径计数。
namespace grid_count
{
template<class T, class C>
T gridPaths(int h, int w, vector<array<int, 2>> ban, C &comb)
{
    // h、w 是网格尺寸，ban[i] 是障碍坐标，comb 提供 C(n,k)；返回模 T::mod() 的路径数。
    assert(h > 0 && w > 0); // 调试检查，可删。
    assert(h - 1 < T::mod() && w - 1 < T::mod() - (h - 1)); // 调试检查，可删。
    for (auto p : ban)
    {
        assert(0 <= p[0] && p[0] < h && 0 <= p[1] && p[1] < w); // 调试检查，可删。
    }
    sort(ban.begin(), ban.end());
    ban.erase(unique(ban.begin(), ban.end()), ban.end());
    array<int, 2> goal{h - 1, w - 1};
    if (binary_search(ban.begin(), ban.end(), goal))
    {
        return T(0);
    }
    ban.push_back(goal);
    sort(ban.begin(), ban.end());
    int goalPos = lower_bound(ban.begin(), ban.end(), goal) - ban.begin();
    vector<T> dp(ban.size());
    for (int i = 0; i < (int)ban.size(); i++)
    {
        auto [x, y] = ban[i];
        dp[i] = comb.C(x + y, x);
        for (int j = 0; j < i; j++)
        {
            auto [u, v] = ban[j];
            if (u <= x && v <= y)
            {
                dp[i] -= dp[j] * comb.C(x - u + y - v, x - u);
            }
        }
    }
    return dp[goalPos];
}
}
```

先引入 [1.1.008 静态取模类](01-基础与工程.md#11008) 的 `MInt<P>` 与 [8.1.001 阶乘、逆阶乘与组合数](08-组合数学.md#81001) 的 `Comb<T>`，再以 `Comb<Z> c; grid_count::gridPaths<Z>(h,w,ban,c)` 调用。要求 $h,w>0$、障碍均在网格内，且 $h+w-2<P$，从而所有阶乘在素模 $P$ 下可逆；重复障碍会自动去重，终点是障碍时返回 $0$。设去重后有 $k$ 个障碍，预处理与 DP 共用 $O(h+w+k^2)$ 时间、$O(h+w+k)$ 空间。网格较小而障碍很多时改用 $O(hw)$ 的逐格递推。

### 14.1.012 Python：并查集、线段树、最大流、费用流、FFT、NTT

Python 版的并查集、点修改区间和线段树、Dinic 最大流、SPFA 增广的最小费用最大流、FFT 与 NTT，供允许 Python 且规模适中的题直接取用。`SegTree` 把内部叶数补到 $N=\operatorname{bit\_ceil}(n)$，使用长度 $2N$ 的数组；Dinic 的阻塞流搜索使用显式路径栈，不受 Python 递归深度限制；流量、距离与费用直接使用任意精度整数，不设置有限“无穷”哨兵。

```python
import cmath
from collections import deque


class DSU:
    def __init__(self, n):
        # n 是点数；建立 n 个单点集合，无返回值。
        self.fa = list(range(n))
        self.siz = [1] * n

    def find(self, x):
        # x 是点编号；返回 x 所在集合的根并做路径压缩。
        while x != self.fa[x]:
            self.fa[x] = self.fa[self.fa[x]]
            x = self.fa[x]
        return x

    def merge(self, x, y):
        # x、y 是点编号；合并成功返回 True，原本连通返回 False。
        x, y = self.find(x), self.find(y)
        if x == y:
            return False
        if self.siz[x] < self.siz[y]:
            x, y = y, x
        self.fa[y] = x
        self.siz[x] += self.siz[y]
        return True


class SegTree:
    def __init__(self, a):
        # a 是初始数列；建立点修改、区间和线段树，无返回值。
        self.n = 1
        while self.n < len(a):
            self.n <<= 1
        self.tr = [0] * (2 * self.n)
        self.tr[self.n:self.n + len(a)] = a
        for p in range(self.n - 1, 0, -1):
            self.tr[p] = self.tr[p << 1] + self.tr[p << 1 | 1]

    def set(self, p, x):
        # p 是 0 下标位置，x 是新值；完成单点赋值，无返回值。
        p += self.n
        self.tr[p] = x
        while p > 1:
            p >>= 1
            self.tr[p] = self.tr[p << 1] + self.tr[p << 1 | 1]

    def prod(self, l, r):
        # l、r 定义半开区间；返回该区间元素和。
        l += self.n
        r += self.n
        x = y = 0
        while l < r:
            if l & 1:
                x += self.tr[l]
                l += 1
            if r & 1:
                r -= 1
                y += self.tr[r]
            l >>= 1
            r >>= 1
        return x + y


class Dinic:
    def __init__(self, n):
        # n 是点数；g 保存边的终点、反向边编号和剩余容量。
        self.n = n
        self.g = [[] for _ in range(n)]

    def add(self, u, v, c):
        # u、v 是端点，c 是非负容量；加入一条有向边，无返回值。
        assert 0 <= u < self.n and 0 <= v < self.n
        assert c >= 0
        # 自环的正反边写入同一表，正向边须指向随后追加的反向边。
        a = [v, len(self.g[v]) + (u == v), c]
        b = [u, len(self.g[u]), 0]
        self.g[u].append(a)
        self.g[v].append(b)

    def flow(self, s, t):
        # s、t 是源汇；返回最大流并修改残量网络。
        assert 0 <= s < self.n and 0 <= t < self.n and s != t
        ans = 0
        while True:
            dep = [-1] * self.n
            dep[s] = 0
            q = deque([s])
            while q:
                u = q.popleft()
                for v, _, c in self.g[u]:
                    if c and dep[v] < 0:
                        dep[v] = dep[u] + 1
                        q.append(v)
            if dep[t] < 0:
                return ans
            cur = [0] * self.n

            def send():
                # 无参数；在当前层次图中用显式栈找到一条增广路并返回流量。
                path_v = [s]
                path_e = []
                while path_v:
                    u = path_v[-1]
                    if u == t:
                        f = min(self.g[x][i][2] for x, i in path_e)
                        for x, i in path_e:
                            e = self.g[x][i]
                            e[2] -= f
                            self.g[e[0]][e[1]][2] += f
                        return f
                    while cur[u] < len(self.g[u]):
                        e = self.g[u][cur[u]]
                        if e[2] and dep[e[0]] == dep[u] + 1:
                            break
                        cur[u] += 1
                    if cur[u] == len(self.g[u]):
                        path_v.pop()
                        if path_e:
                            x, _ = path_e.pop()
                            cur[x] += 1
                        continue
                    path_e.append((u, cur[u]))
                    path_v.append(self.g[u][cur[u]][0])
                return 0

            while True:
                f = send()
                if not f:
                    break
                ans += f


class MinCostFlow:
    def __init__(self, n):
        # n 是点数；g 保存终点、反向边编号、容量和单位费用。
        self.n = n
        self.g = [[] for _ in range(n)]

    def add(self, u, v, c, w):
        # u、v 是端点，c 是容量，w 是单位费用；加入有向边，无返回值。
        assert 0 <= u < self.n and 0 <= v < self.n
        assert c >= 0
        # 自环的反向边位于同一邻接表的下一格。
        a = [v, len(self.g[v]) + (u == v), c, w]
        b = [u, len(self.g[u]), 0, -w]
        self.g[u].append(a)
        self.g[v].append(b)

    def flow(self, s, t, lim=None):
        # s、t 是源汇，lim 是可选流量上限；返回实际流量与最小费用。
        n = self.n
        assert 0 <= s < n and 0 <= t < n and s != t
        assert lim is None or lim >= 0
        total = 0
        cost = 0
        while lim is None or total < lim:
            dis = [None] * n
            pre = [None] * n
            inq = [False] * n
            depth = [0] * n
            dis[s] = 0
            q = deque([s])
            inq[s] = True
            while q:
                u = q.popleft()
                inq[u] = False
                for i, e in enumerate(self.g[u]):
                    v, _, c, w = e
                    nd = dis[u] + w
                    if c and (dis[v] is None or nd < dis[v]):
                        dis[v] = nd
                        pre[v] = (u, i)
                        depth[v] = depth[u] + 1
                        if depth[v] >= n:
                            raise ValueError("source-reachable negative cycle")
                        if not inq[v]:
                            inq[v] = True
                            q.append(v)
            if pre[t] is None:
                break
            f = None if lim is None else lim - total
            v = t
            while v != s:
                u, i = pre[v]
                c = self.g[u][i][2]
                f = c if f is None else min(f, c)
                v = u
            v = t
            while v != s:
                u, i = pre[v]
                e = self.g[u][i]
                e[2] -= f
                self.g[v][e[1]][2] += f
                cost += f * e[3]
                v = u
            total += f
        return total, cost


def fft(a, inv=False):
    # a 是长度为二次幂的复数表，inv 表示逆变换；原地修改 a，无返回值。
    n = len(a)
    if n == 0 or n & (n - 1):
        raise ValueError("FFT length must be a nonzero power of two")
    j = 0
    for i in range(1, n):
        bit = n >> 1
        while j & bit:
            j ^= bit
            bit >>= 1
        j ^= bit
        if i < j:
            a[i], a[j] = a[j], a[i]
    length = 2
    while length <= n:
        wn = cmath.exp((2j if inv else -2j) * cmath.pi / length)
        for l in range(0, n, length):
            w = 1
            half = length >> 1
            for i in range(l, l + half):
                x = a[i]
                y = a[i + half] * w
                a[i] = x + y
                a[i + half] = x - y
                w *= wn
        length <<= 1
    if inv:
        for i in range(n):
            a[i] /= n


def conv_fft(a, b):
    # a、b 是实数系数表；返回普通卷积的浮点近似值。
    if not a or not b:
        return []
    n = 1
    while n < len(a) + len(b) - 1:
        n <<= 1
    x = list(map(complex, a)) + [0j] * (n - len(a))
    y = list(map(complex, b)) + [0j] * (n - len(b))
    fft(x)
    fft(y)
    for i in range(n):
        x[i] *= y[i]
    fft(x, True)
    return [x[i].real for i in range(len(a) + len(b) - 1)]


def ntt(a, inv=False, mod=998244353, root=3):
    # a 是模 mod 系数表，inv 表示逆变换，root 是原根；原地修改 a。
    n = len(a)
    supported = {(998244353, 3), (1004535809, 3), (469762049, 3)}
    if (mod, root) not in supported:
        raise ValueError("unsupported NTT modulus or primitive root")
    if n == 0 or n & (n - 1) or (mod - 1) % n:
        raise ValueError("invalid NTT length")
    j = 0
    for i in range(1, n):
        bit = n >> 1
        while j & bit:
            j ^= bit
            bit >>= 1
        j ^= bit
        if i < j:
            a[i], a[j] = a[j], a[i]
    length = 2
    while length <= n:
        wn = pow(root, (mod - 1) // length, mod)
        if inv:
            wn = pow(wn, mod - 2, mod)
        for l in range(0, n, length):
            w = 1
            half = length >> 1
            for i in range(l, l + half):
                x = a[i]
                y = a[i + half] * w % mod
                a[i] = (x + y) % mod
                a[i + half] = (x - y) % mod
                w = w * wn % mod
        length <<= 1
    if inv:
        ni = pow(n, mod - 2, mod)
        for i in range(n):
            a[i] = a[i] * ni % mod


def conv_ntt(a, b, mod=998244353, root=3):
    # a、b 是模 mod 系数表；返回精确模卷积，模数与原根必须在受支持表中。
    if not a or not b:
        return []
    n = 1
    while n < len(a) + len(b) - 1:
        n <<= 1
    x = [v % mod for v in a] + [0] * (n - len(a))
    y = [v % mod for v in b] + [0] * (n - len(b))
    ntt(x, mod=mod, root=root)
    ntt(y, mod=mod, root=root)
    for i in range(n):
        x[i] = x[i] * y[i] % mod
    ntt(x, True, mod, root)
    return x[:len(a) + len(b) - 1]
```

所有下标从 $0$ 起。`Dinic.flow(s,t)` 要求端点合法、$s\ne t$、容量非负；显式栈实现不受递归深度限制。`MinCostFlow.flow(s,t,lim)` 中 `lim=None` 表示尽量增广，要求每轮残量图不存在源可达负费用环，检测到时抛出 `ValueError`。`conv_fft` 返回浮点近似，只有调用者另行证明每个系数误差小于 $\frac12$ 时才能舍入成精确整数；无条件精确结果应使用 `conv_ntt` 或多模 NTT 加 CRT。当前 NTT 支持 $(998244353,3)$、$(1004535809,3)$、$(469762049,3)$，并要求补零长度为非零二次幂且整除 $mod-1$。并查集单次均摊 $O(\alpha(n))$，线段树单次 $O(\log n)$、空间 $O(N)$，其中 $N=\operatorname{bit\_ceil}(n)$；Dinic 为 $O(V^2E)$，SPFA 费用流最坏 $O(FVE)$，FFT/NTT 为 $O(n\log n)$。

### 14.1.013 Python：Splay

维护可重复整数的伸展树（Python 版），支持插入、删一个、按排名取值 `kth` 与查询严格小于给定值的元素数 `rank`。用平行数组 `ch`/`fa`/`val`/`cnt`/`siz` 存节点、第 $0$ 项当空哨兵；每次查找命中或触底后都把相应节点 `_splay` 旋到根，摊还掉最坏深度。`_rotate` 单旋后必须先 `_pull` 儿子再 `_pull` 自己以自底向上重算子树大小，`_splay` 按 zig-zig / zig-zag 分类决定先旋父还是先旋子——次序错会破坏平衡性的摊还保证。

```python
class Splay:
    def __init__(self):
        # 无参数；建立空伸展树，各列表第 0 项表示空节点。
        self.ch = [[0, 0]]
        self.fa = [0]
        self.val = [0]
        self.cnt = [0]
        self.siz = [0]
        self.rt = 0
        self.free = []

    def _new(self, x, f):
        # x 是键值，f 是父节点；创建节点并返回编号。
        if self.free:
            u = self.free.pop()
            self.ch[u] = [0, 0]
            self.fa[u] = f
            self.val[u] = x
            self.cnt[u] = 1
            self.siz[u] = 1
            return u
        self.ch.append([0, 0])
        self.fa.append(f)
        self.val.append(x)
        self.cnt.append(1)
        self.siz.append(1)
        return len(self.val) - 1

    def _recycle(self, u):
        # u 是已从树中断开的节点；清空并回收其编号，无返回值。
        self.ch[u] = [0, 0]
        self.fa[u] = 0
        self.cnt[u] = 0
        self.siz[u] = 0
        self.free.append(u)

    def _pull(self, u):
        # u 是节点编号；根据儿子和重复计数重算子树大小，无返回值。
        self.siz[u] = self.siz[self.ch[u][0]] + self.siz[self.ch[u][1]] + self.cnt[u]

    def _rotate(self, x):
        # x 是待上旋节点；完成一次旋转并维护父子关系，无返回值。
        y = self.fa[x]
        z = self.fa[y]
        d = int(self.ch[y][1] == x)
        b = self.ch[x][d ^ 1]
        if z:
            self.ch[z][int(self.ch[z][1] == y)] = x
        else:
            self.rt = x
        self.fa[x] = z
        self.ch[x][d ^ 1] = y
        self.fa[y] = x
        self.ch[y][d] = b
        if b:
            self.fa[b] = y
        self._pull(y)
        self._pull(x)

    def _splay(self, x, goal=0):
        # x 是目标节点，goal 是目标父亲；把 x 旋到 goal 下，无返回值。
        while self.fa[x] != goal:
            y = self.fa[x]
            z = self.fa[y]
            if z != goal:
                if (self.ch[y][1] == x) == (self.ch[z][1] == y):
                    self._rotate(y)
                else:
                    self._rotate(x)
            self._rotate(x)

    def _find(self, x):
        # x 是键值；若存在则伸展到根并返回节点，否则伸展最后访问点并返回 0。
        u = self.rt
        last = 0
        while u and self.val[u] != x:
            last = u
            u = self.ch[u][x > self.val[u]]
        if u:
            self._splay(u)
        elif last:
            self._splay(last)
        return u

    def insert(self, x):
        # x 是键值；插入一个 x，无返回值。
        if not self.rt:
            self.rt = self._new(x, 0)
            return
        u = self.rt
        while True:
            if self.val[u] == x:
                self.cnt[u] += 1
                self._pull(u)
                self._splay(u)
                return
            d = int(x > self.val[u])
            if not self.ch[u][d]:
                v = self._new(x, u)
                self.ch[u][d] = v
                self._pull(u)
                self._splay(v)
                return
            u = self.ch[u][d]

    def erase(self, x):
        # x 是键值；删除一个 x，成功返回 True，不存在返回 False。
        u = self._find(x)
        if not u:
            return False
        if self.cnt[u] > 1:
            self.cnt[u] -= 1
            self._pull(u)
            return True
        l, r = self.ch[u]
        if not l:
            self.rt = r
            if r:
                self.fa[r] = 0
            self._recycle(u)
            return True
        self.rt = l
        self.fa[l] = 0
        v = l
        while self.ch[v][1]:
            v = self.ch[v][1]
        self._splay(v)
        self.ch[v][1] = r
        if r:
            self.fa[r] = v
        self._pull(v)
        self._recycle(u)
        return True

    def kth(self, k):
        # k 是从 0 开始的排名；返回对应键值，越界返回 None。
        if k < 0 or k >= self.siz[self.rt]:
            return None
        u = self.rt
        while True:
            left = self.siz[self.ch[u][0]]
            if k < left:
                u = self.ch[u][0]
            elif k < left + self.cnt[u]:
                self._splay(u)
                return self.val[u]
            else:
                k -= left + self.cnt[u]
                u = self.ch[u][1]

    def rank(self, x):
        # x 是查询值；返回严格小于 x 的元素个数，并伸展最后访问节点。
        u = self.rt
        last = 0
        ans = 0
        while u:
            last = u
            if x <= self.val[u]:
                u = self.ch[u][0]
            else:
                ans += self.siz[self.ch[u][0]] + self.cnt[u]
                u = self.ch[u][1]
        if last:
            self._splay(last)
        return ans
```

先构造 `Splay()`：`insert(x)` 插一个，`erase(x)` 删一个，`kth(k)` 取第 $k$ 小，`rank(x)` 返回严格小于 $x$ 的元素数。各操作均摊 $O(\log n)$、单次最坏 $O(n)$；空闲编号会复用，数组空间为 $O(M)$，其中 $M$ 是历史上同时存在的不同键数峰值，不随累计插入次数增长。`sortedcontainers.SortedList` 是第三方库，只有评测环境已安装时才能替代本实现；C++ 平衡树见第 3 章数据结构。

### 14.1.014 Python：`datetime`、`Fraction`、递归深度与文件 IO

Python 的日期差、精确分数、递归深度与文本文件 IO 汇总。`day_diff` 用 `datetime.date` 相减取天数；`Fraction` 自动约分、保持精确有理数；`read_text`/`write_text` 显式按 UTF-8 读写。`sys.setrecursionlimit` 调整解释器的递归阈值，不能保证栈或内存足够；CPython 版本、PyPy 与扩展调用的栈行为不同，深搜优先使用显式栈。

### 14.1.015 完全图部分边染色的 1-因子扩展尝试

本条尝试在不改已有颜色的前提下，把 $K_n$ 的合法 $m$-边染色补成 $K_{m+1}$ 的 1-因子分解。最终每种颜色都须是大小 $\frac{m+1}{2}$ 的完美匹配，故 $m$ 必为奇数。代码逐点加入新顶点，并用一次二分图匹配为该点的全部新边选色；已经确定的选择不回溯到先前顶点，因此成功时一定给出合法扩展，失败只表示这套逐点策略没有找到，不能证明不存在扩展。

```cpp
// 知识点 14.1.015：把部分完全图边染色扩展成完整 1-因子分解。
namespace factor_ext
{
optional<vector<vector<int>>> extendFactor(vector<vector<int>> a, int m)
{
    // a 是部分染色矩阵，m 是颜色数；成功时返回完整染色，本策略失败时返回空。
    auto n = (int)a.size();
    if (m <= 0 || m % 2 == 0 || n > m + 1)
    {
        return nullopt;
    }
    for (auto &r : a)
    {
        if ((int)r.size() != n)
        {
            return nullopt;
        }
    }
    vector<vector<char>> used(n, vector<char>(m));
    vector<int> ec(m);
    for (int i = 0; i < n; i++)
    {
        if (a[i][i] != -1)
        {
            return nullopt;
        }
        for (int j = i + 1; j < n; j++)
        {
            auto c = a[i][j];
            if (c < 0 || c >= m || a[j][i] != c)
            {
                return nullopt;
            }
            if (used[i][c] || used[j][c])
            {
                return nullopt;
            }
            used[i][c] = true;
            used[j][c] = true;
            ec[c]++;
        }
    }

    auto cap = (m + 1) / 2;
    while (n < m + 1)
    {
        for (int c = 0; c < m; c++)
        {
            if (n - ec[c] > cap)
            {
                return nullopt;
            }
        }

        vector<vector<int>> g(m);
        for (int c = 0; c < m; c++)
        {
            for (int v = 0; v < n; v++)
            {
                if (!used[v][c])
                {
                    g[c].push_back(v);
                }
            }
            if (n - ec[c] < cap)
            {
                for (int v = n; v < m; v++)
                {
                    g[c].push_back(v);
                }
            }
        }

        vector<int> mt(m, -1);
        vector<int> ord(m);
        iota(ord.begin(), ord.end(), 0);
        stable_sort(ord.begin(),
                    ord.end(),
                    [&](int x, int y)
                    {
                        return g[x].size() < g[y].size();
                    });
        auto dfs = [&](auto &&self, int c, vector<char> &vis) -> bool
        {
            for (auto v : g[c])
            {
                if (vis[v])
                {
                    continue;
                }
                vis[v] = true;
                if (mt[v] == -1 || self(self, mt[v], vis))
                {
                    mt[v] = c;
                    return true;
                }
            }
            return false;
        };
        for (auto c : ord)
        {
            vector<char> vis(m);
            if (!dfs(dfs, c, vis))
            {
                return nullopt;
            }
        }

        for (auto &r : a)
        {
            r.push_back(-1);
        }
        a.push_back(vector<int>(n + 1, -1));
        used.push_back(vector<char>(m));
        for (int v = 0; v < n; v++)
        {
            auto c = mt[v];
            a[v][n] = c;
            a[n][v] = c;
            used[v][c] = true;
            used[n][c] = true;
            ec[c]++;
        }
        n++;
    }
    return a;
}
} // 命名空间 factor_ext
```

`extendFactor(a,m)` 的输入是对称颜色矩阵，主对角线为 $-1$，已有边色在 $[0,m)$ 内且同色边不共端点。成功时返回完整染色；若输入非法或当前贪心顺序失败则返回空，后一种情况不代表数学上无解。逐点匈牙利增广最坏 $O(m^4)$ 时间，空间 $O(m^2)$。

### 14.1.016 高精度整数平方根

任意长非负十进制整数的向下取整平方根，全程走字符串高精度、不依赖机器整数宽度。用的是竖式逐位开方（不是二分）：把数字从高位起每两位一组“落下”，维护当前已得根 `root` 与余数 `rem`；每组令 `base = 20*root`，从 $9$ 往下试最大的数字 $d$ 使 $(base+d)\cdot d\le rem$——这正是 $(10\,root+d)^2-100\,root^2$ 的展开，把它从 `rem` 减掉并令 `root\gets 10\,root+d`。配套的 `trimNum`/`cmpNum`/`mulSmall`/`addSmall`/`subNum` 是够用的最小字符串算术（乘、加的第二操作数限定在小常数范围内）。结束时 `root` 满足 $root^2\le n<(root+1)^2$。

```cpp
string trimNum(string s)
{
    // 参数 s 是非空十进制数字串；返回删除多余前导零后的规范表示。
    auto p = s.find_first_not_of('0');
    return p == string::npos ? "0" : s.substr(p);
}

int cmpNum(const string &a, const string &b)
{
    // 参数 a、b 是规范非负十进制数；返回 -1、0、1 表示 a 与 b 的大小关系。
    if (a.size() != b.size())
    {
        return a.size() < b.size() ? -1 : 1;
    }
    if (a == b)
    {
        return 0;
    }
    return a < b ? -1 : 1;
}

string mulSmall(const string &a, int b)
{
    // 参数 a 是规范非负十进制数，b 在 [0,100]；返回精确乘积。
    if (b == 0 || a == "0")
    {
        return "0";
    }
    string c;
    int carry = 0;
    for (int i = (int)a.size() - 1; i >= 0; i--)
    {
        auto v = (a[i] - '0') * b + carry;
        c.push_back(char('0' + v % 10));
        carry = v / 10;
    }
    while (carry)
    {
        c.push_back(char('0' + carry % 10));
        carry /= 10;
    }
    reverse(c.begin(), c.end());
    return c;
}

string addSmall(string a, int b)
{
    // 参数 a 是规范非负十进制数，b 在 [0,99]；返回精确和。
    for (int i = (int)a.size() - 1; i >= 0 && b; i--)
    {
        auto v = a[i] - '0' + b;
        a[i] = char('0' + v % 10);
        b = v / 10;
    }
    while (b)
    {
        a.insert(a.begin(), char('0' + b % 10));
        b /= 10;
    }
    return trimNum(a);
}

string subNum(string a, const string &b)
{
    // 参数 a、b 是规范非负十进制数且 a>=b；返回精确差。
    assert(cmpNum(a, b) >= 0); // 调试检查，可删。
    int borrow = 0;
    int j = (int)b.size() - 1;
    for (int i = (int)a.size() - 1; i >= 0; i--)
    {
        auto v = a[i] - '0' - borrow - (j >= 0 ? b[j] - '0' : 0);
        borrow = v < 0;
        if (borrow)
        {
            v += 10;
        }
        a[i] = char('0' + v);
        j--;
    }
    return trimNum(a);
}

string bigSqrt(string s)
{
    // 参数 s 是无前导符号的非负十进制整数；返回 floor(sqrt(s)) 的十进制表示。
    assert(!s.empty()); // 调试检查，可删。
    for (auto c : s)
    {
        assert('0' <= c && c <= '9'); // 调试检查，可删。
    }
    s = trimNum(s);
    if (s == "0")
    {
        return "0";
    }
    if (s.size() & 1)
    {
        s = "0" + s;
    }
    string root = "0";
    string rem = "0";
    for (int i = 0; i < (int)s.size(); i += 2)
    {
        rem = trimNum(rem + s.substr(i, 2));
        auto base = mulSmall(root, 20);
        int d = 9;
        while (cmpNum(mulSmall(addSmall(base, d), d), rem) > 0)
        {
            d--;
        }
        rem = subNum(rem, mulSmall(addSmall(base, d), d));
        root = addSmall(mulSmall(root, 10), d);
    }
    return root;
}
```

`bigSqrt(s)` 的 `s` 是不带符号的非负十进制串（允许前导零），返回规范化（无前导零）的 $\lfloor\sqrt{s}\rfloor$。设十进制位数为 $D$，当前实现 $O(D^2)$ 时间、$O(D)$ 空间。数值落在 `long long` 内时直接用整型 `sqrtl` 加一两步修正更快；要完整大整数四则见 [14.1.001 高精度整数（BigInt）](#141001)，本条刻意用轻量字符串算术，免得为一次开方引入整套 `BigInt`。

竖式试商和落下 `00` 分组最易写错。下面覆盖零、平方边界和一组长数：

```cpp

    assert(bigSqrt("0") == "0");
    assert(bigSqrt("9999") == "99");
    assert(bigSqrt("15241578753238836750495351562536198787501905199875019052100")
           == "123456789012345678901234567890");
}
```

### 14.1.017 一元二次方程实根

实系数一元二次方程 $ax^2+bx+c=0$ 的全部实根，兼容退化成一次方程。次数按归一化前的精确零判断；随后把系数除以最大绝对值以控制判别式量级，要求原来非零的最高有效系数归一化后仍非零，这一可表示比例取决于评测机的 `long double`。二次情形用数值稳定的 $q=-\tfrac12\big(b+\operatorname{sgn}(b)\sqrt\Delta\big)$，两根取 $q/a$ 与 $c/q$，避免直接公式中的相消。

```cpp
vector<long double> quadRoots(long double a, long double b, long double c)
{
    // 参数 a、b、c 是方程 ax^2+bx+c=0 的系数；返回升序实根，重根只返回一次，无实根返回空表。
    assert(isfinite(a) && isfinite(b) && isfinite(c)); // 调试检查，可删。
    bool hasA = a != 0, hasB = b != 0;
    long double scale = max({fabsl(a), fabsl(b), fabsl(c)});
    if (scale == 0)
    {
        return {};
    }
    a /= scale;
    b /= scale;
    c /= scale;
    assert((!hasA || a != 0) && (!hasB || b != 0)); // 调试检查，可删。
    if (!hasA)
    {
        return !hasB ? vector<long double>{} : vector<long double>{-c / b};
    }
    long double bb = b * b;
    long double ac4 = 4 * a * c;
    long double d = fmal(-4 * a, c, bb);
    long double tol = 64 * numeric_limits<long double>::epsilon()
                      * (fabsl(bb) + fabsl(ac4));
    if (d < -tol)
    {
        return {};
    }
    if (fabsl(d) <= tol)
    {
        return {-b / (2 * a)};
    }
    long double q = -0.5L * (b + copysignl(sqrtl(d), b));
    vector<long double> ans{q / a, c / q};
    sort(ans.begin(), ans.end());
    return ans;
}
```

`quadRoots(a,b,c)` 要求三个系数有限，且原来非零的最高有效系数除以最大绝对值后仍可表示为非零；MSVC 的 `long double` 通常等同 `double`，可用比例范围会小于 GCC x86 的扩展精度。函数返回升序实根，重根只保留一次；恒等式与无解的一次退化都返回空。时间、空间均为 $O(1)$。

下面检查一次退化、重根、无实根、相消明显的两根，以及同一方程在不同数量级缩放下的结果：

```cpp
bool near(long double x, long double y)
{
    // x、y 是待比较浮点数；返回二者是否在相对误差 1e-15 内相等。
    return fabsl(x - y) <= 1E-15L * max({1.0L, fabsl(x), fabsl(y)});
}

signed main()
{
    assert(quadRoots(0, 2, -4) == vector<long double>{2});
    assert(quadRoots(0, 0, 1).empty());
    assert(quadRoots(1, 0, 1).empty());
    auto one = quadRoots(1, -2, 1);
    assert(one.size() == 1 && near(one[0], 1));
    for (long double k : {1E-200L, 1.0L, 1E200L})
    {
        auto r = quadRoots(k, -3 * k, 2 * k);
        assert(r.size() == 2 && near(r[0], 1) && near(r[1], 2));
    }
    auto r = quadRoots(1, -1E20L, 1);
    assert(r.size() == 2 && near(r[0], 1E-20L) && near(r[1], 1E20L));
    cout << "OK\n";
    return 0;
}
```

### 14.1.018 N 皇后构造解

$O(n)$ 直接构造一个 $n$ 皇后方案（不是回溯枚举全部解），返回每行皇后所在列。$n=1$ 平凡，$n=2,3$ 无解返回空表，$n\ge4$ 按 $n\bmod6$ 分三类摆放：奇数 $n$ 先解 $n-1$ 再把最后一行放到最后一列；偶数且 $n\bmod6\ne2$ 用一套先偶列后奇列的等差排布；$n\bmod6=2$ 用另一套带环移位的排布。三类构造都保证任意两皇后不同列、且行差绝对值不等于列差绝对值（即不同对角线），因此天然合法。

```cpp
vector<int> nQueens(int n)
{
    // 参数 n 是棋盘边长；返回每行皇后的 0 下标列号，n=2 或 3 时返回空表。
    assert(n >= 1); // 调试检查，可删。
    if (n == 1)
    {
        return {0};
    }
    if (n < 4)
    {
        return {};
    }
    vector<int> p(n);
    auto gen = [&](auto &&self, int m, vector<int> &q) -> void
    {
        if (m & 1)
        {
            vector<int> t(m - 1);
            self(self, m - 1, t);
            copy(t.begin(), t.end(), q.begin());
            q[m - 1] = m - 1;
        }
        else if (m % 6 != 2)
        {
            for (int i = 1; i <= m / 2; i++)
            {
                q[i - 1] = 2 * i - 1;
            }
            for (int i = m / 2 + 1; i <= m; i++)
            {
                q[i - 1] = 2 * i - m - 2;
            }
        }
        else
        {
            for (int i = 1; i <= m / 2; i++)
            {
                q[i - 1] = (2 * i + m / 2 - 3) % m;
            }
            for (int i = m / 2 + 1; i <= m; i++)
            {
                q[i - 1] = m - 1 - (2 * (m - i + 1) + m / 2 - 3) % m;
            }
        }
    };
    gen(gen, n, p);
    return p;
}
```

`nQueens(n)` 要求 $n\ge1$，返回第 $i$ 行皇后的 $0$ 下标列号；无解的 $n=2,3$ 返回空向量。构造用 $O(n)$ 时间和空间。只需一个可行解时使用本条；枚举全部方案或附加约束时改用 DFS 与位运算剪枝。同类约束满足问题见 [14.1.007 9×9 数独字典序最小解](#141007)。

下面检查最小有解、无解边界和一个常用规模：

```cpp

    assert(nQueens(1) == vector<int>({0}));
    assert(nQueens(2).empty() && nQueens(3).empty());
    assert(nQueens(4) == vector<int>({1, 3, 0, 2}));
    assert(nQueens(8) == vector<int>({3, 5, 7, 1, 6, 0, 2, 4}));
}
```

### 14.1.019 百万位大整数：十六进制与十进制

`HexBigInt` 用固定 32 位小端 limb 保存绝对值，适合十六进制长整数；`FastBigInt` 用 $B=10^9$ 的小端块直接读写十进制长整数。两者支持带符号比较、加减乘除与取模，删除高位零，零恒为非负。除法满足 $x=qy+r$、$|r|<|y|$，商向零截断，非零余数与被除数同号；除数必须非零。

十六进制乘法按 $x=x_0+x_1\beta^k$、$y=y_0+y_1\beta^k$ 分块，其中 $\beta=2^{32}$；Karatsuba 用三次子乘法恢复

$$
xy=x_0y_0+((x_0+x_1)(y_0+y_1)-x_0y_0-x_1y_1)\beta^k+x_1y_1\beta^{2k}.
$$

Burnikel--Ziegler 除法先把除数规格化，再把 $2n/n$ 除法拆成两次 $3n/2n$，用高位估商并修正余数。以下类也提供按 $10^9$ 幂分治合并、拆分的十进制转换；只做十进制运算时直接用 `FastBigInt`。

```cpp
// limb、进位和移位槽固定为 32 位，依赖截断与二进制位布局；64 位中间量承载双字运算。
struct HexBigInt
{
    using It = vector<uint32_t>::iterator;
    using CIt = vector<uint32_t>::const_iterator;

    bool neg = false;   // 是否为负数，零恒为非负。
    vector<uint32_t> a; // 绝对值的固定 32 位小端 limb，无高位零，零为空。

    HexBigInt()
    {
        // 无参数；构造整数零。
    }

    HexBigInt(long long x)
    {
        // x 是任意 64 位有符号整数（含 LLONG_MIN）；按符号与绝对值的 32 位小端 limb 构造。
        neg = x < 0;
        unsigned long long y = neg ? 0 - (unsigned long long)x : (unsigned long long)x;
        while (y)
        {
            a.push_back((uint32_t)y);
            y >>= 32;
        }
    }

    HexBigInt(const string &s)
    {
        // s 是带可选 +/- 号、至少一位数字的十六进制串（大小写均可）；解析为对应大整数。
        assign(s);
    }

    HexBigInt(bool sgn, vector<uint32_t> v) : neg(sgn), a(move(v))
    {
        // sgn 为真表示负数，v 是绝对值的 32 位小端 limb；构造后去高零，值为零时强制非负。
        trim(this->a);
        if (this->a.empty())
        {
            this->neg = false;
        }
    }

    HexBigInt &assign(const string &s)
    {
        // s 是带可选 +/- 号、至少一位数字的十六进制串（0-9A-Fa-f）；覆盖当前值并返回 *this，零非负。
        assert(!s.empty());
        int p = 0;
        neg = s[0] == '-';
        if (s[0] == '-' || s[0] == '+')
        {
            p++;
        }
        assert(p < (int)s.size());
        a.clear();
        for (int r = s.size(); r > p; r -= 8)
        {
            int l = max(p, r - 8);
            uint32_t x = 0;
            for (int i = l; i < r; i++)
            {
                x <<= 4;
                if ('0' <= s[i] && s[i] <= '9')
                {
                    x |= s[i] - '0';
                }
                else if ('A' <= s[i] && s[i] <= 'F')
                {
                    x |= s[i] - 'A' + 10;
                }
                else
                {
                    assert('a' <= s[i] && s[i] <= 'f');
                    x |= s[i] - 'a' + 10;
                }
            }
            a.push_back(x);
        }
        trim(a);
        if (a.empty())
        {
            neg = false;
        }
        return *this;
    }

    string toHex() const
    {
        // 无参数；返回大写十六进制串，负数带前导 -，无前导零，零返回 "0"。
        if (a.empty())
        {
            return "0";
        }
        const char *digit = "0123456789ABCDEF";
        string s = neg ? "-" : "";
        for (int i = a.size() - 1; i >= 0; i--)
        {
            string t(8, '0');
            uint32_t x = a[i];
            for (int j = 7; j >= 0; j--)
            {
                t[j] = digit[x & 15];
                x >>= 4;
            }
            if (i == (int)a.size() - 1)
            {
                t.erase(0, t.find_first_not_of('0'));
            }
            s += t;
        }
        return s;
    }

    bool isZero() const
    {
        // 无参数；返回当前值是否为零（limb 数组为空）。
        return a.empty();
    }

    HexBigInt operator-() const
    {
        // 无参数；返回相反数，零仍保持非负。
        HexBigInt x = *this;
        if (!x.a.empty())
        {
            x.neg = !x.neg;
        }
        return x;
    }

    friend bool operator==(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否相等 y。
        return x.neg == y.neg && x.a == y.a;
    }

    friend bool operator<(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范大整数；返回按有符号数值比较 x<y 是否成立。
        if (x.neg != y.neg)
        {
            return x.neg;
        }
        int c = cmp(x.a, y.a);
        return x.neg ? c > 0 : c < 0;
    }

    friend bool operator!=(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否不等 y。
        return !(x == y);
    }

    friend bool operator>(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范大整数；返回 x>y 是否成立（即 y<x）。
        return y < x;
    }

    friend bool operator<=(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否小于等于 y。
        return !(y < x);
    }

    friend bool operator>=(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否大于等于 y。
        return !(x < y);
    }

    friend HexBigInt operator+(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范大整数；返回 x+y：同号相加绝对值，异号用较大绝对值减较小绝对值。
        if (x.neg == y.neg)
        {
            return {x.neg, addVec(x.a, y.a)};
        }
        int c = cmp(x.a, y.a);
        if (c <= 0)
        {
            return {c ? y.neg : false, subVec(y.a, x.a)};
        }
        return {x.neg, subVec(x.a, y.a)};
    }

    friend HexBigInt operator-(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范大整数；返回 x-y，即 x+(-y)。
        return x + (-y);
    }

    friend HexBigInt operator*(const HexBigInt &x, const HexBigInt &y)
    {
        // x、y 是规范大整数；返回 x*y，积为零时非负。
        auto z = mulVec(x.a, y.a);
        return {z.empty() ? false : x.neg != y.neg, move(z)};
    }

    friend pair<HexBigInt, HexBigInt> divmod(const HexBigInt &x, const HexBigInt &y)
    {
        // x 是被除数，y 是非零除数；返回 {商,余数}，商向零截断，余数与 x 同号（零非负）。
        assert(!y.a.empty());
        auto [q, r] = divVec(x.a, y.a);
        return {{q.empty() ? false : x.neg != y.neg, move(q)}, {r.empty() ? false : x.neg, move(r)}};
    }

    friend HexBigInt operator/(const HexBigInt &x, const HexBigInt &y)
    {
        // x 是被除数，y 是非零除数；返回向零截断的商。
        return divmod(x, y).first;
    }

    friend HexBigInt operator%(const HexBigInt &x, const HexBigInt &y)
    {
        // x 是被除数，y 是非零除数；返回与 x 同号的余数 x-(x/y)*y。
        return divmod(x, y).second;
    }

    HexBigInt &operator+=(const HexBigInt &x)
    {
        // x 是右操作数；原地加上 x，返回 *this。
        return *this = *this + x;
    }

    HexBigInt &operator-=(const HexBigInt &x)
    {
        // x 是右操作数；原地减去 x，返回 *this。
        return *this = *this - x;
    }

    HexBigInt &operator*=(const HexBigInt &x)
    {
        // x 是右操作数；原地乘以 x，返回 *this。
        return *this = *this * x;
    }

    HexBigInt &operator/=(const HexBigInt &x)
    {
        // x 是右操作数；原地除以非零 x，向零截断，返回 *this。
        return *this = *this / x;
    }

    HexBigInt &operator%=(const HexBigInt &x)
    {
        // x 是右操作数；原地对非零 x 取余，余数与原值同号，返回 *this。
        return *this = *this % x;
    }

    friend ostream &operator<<(ostream &out, const HexBigInt &x)
    {
        // out 是输出流，x 是大整数；写入 x.toHex() 并返回 out。
        return out << x.toHex();
    }

    static HexBigInt fromDecimal(const string &s)
    {
        // s 是带可选正负号的非空十进制整数串；返回其精确值，以二进制 limb 保存。
        assert(!s.empty());
        int p = 0;
        bool sign = s[0] == '-';
        if (s[0] == '-' || s[0] == '+')
        {
            p++;
        }
        assert(p < (int)s.size());
        vector<int> group;
        for (int r = s.size(); r > p; r -= 9)
        {
            int l = max<int>(p, r - 9);
            int x = 0;
            for (int i = l; i < r; i++)
            {
                assert('0' <= s[i] && s[i] <= '9');
                x = x * 10 + s[i] - '0';
            }
            group.push_back(x);
        }
        int z = bit_ceil((unsigned long long)max<int>(1, group.size()));
        group.resize(z);
        vector<HexBigInt> power;
        power.emplace_back(1000000000LL);
        for (int n = 2; n < z; n <<= 1)
        {
            power.push_back(power.back() * power.back());
        }
        auto build = [&](auto &&self, int l, int len, int level) -> HexBigInt
        {
            // self 是递归自身；合并 group 的 [l,l+len)，len=2^level，返回对应非负整数。
            if (len == 1)
            {
                return HexBigInt((long long)group[l]);
            }
            int half = len / 2;
            HexBigInt low = self(self, l, half, level - 1);
            HexBigInt high = self(self, l + half, half, level - 1);
            return low + high * power[level - 1];
        };
        HexBigInt ans = build(build, 0, z, countr_zero((unsigned long long)z));
        if (!ans.a.empty())
        {
            ans.neg = sign;
        }
        return ans;
    }

    string toDecimal() const
    {
        // 无参数；按 10^9 幂递归分裂为十进制分组，返回规范字符串。
        if (a.empty())
        {
            return "0";
        }
        HexBigInt x = *this;
        x.neg = false;
        vector<HexBigInt> power{HexBigInt(1000000000LL)};
        int level = 0;
        while (!(x < power[level]))
        {
            power.push_back(power.back() * power.back());
            level++;
        }
        vector<int> group(1LL << level);
        auto split = [&](auto &&self, const HexBigInt &v, int l, int lv) -> void
        {
            // self 是递归自身；把非负 v 的 2^lv 个十进制块写入 group[l..l+2^lv)，无返回值。
            if (lv == 0)
            {
                group[l] = v.a.empty() ? 0 : v.a[0];
                return;
            }
            auto [high, low] = divmod(v, power[lv - 1]);
            self(self, low, l, lv - 1);
            self(self, high, l + (1LL << (lv - 1)), lv - 1);
        };
        split(split, x, 0, level);
        while (group.size() > 1 && group.back() == 0)
        {
            group.pop_back();
        }
        string s = neg ? "-" : "";
        s += to_string(group.back());
        for (int i = group.size() - 2; i >= 0; i--)
        {
            string t = to_string(group[i]);
            s += string(9 - t.size(), '0') + t;
        }
        return s;
    }

  private:
    static constexpr int naiveDiv = 64; // 改用竖式除法的 limb 数阈值。

    static void trim(vector<uint32_t> &x)
    {
        // x 是小端 limb 数组；原地弹掉最高位的零 limb，零变为空数组。
        while (!x.empty() && x.back() == 0)
        {
            x.pop_back();
        }
    }

    static bool less(CIt x, CIt xe, CIt y, CIt ye)
    {
        // [x,xe)、[y,ye) 是小端 limb 区间，长度可不同且允许高位零；返回无符号值 x<y 是否成立。
        int nx = xe - x, ny = ye - y;
        while (nx && x[nx - 1] == 0)
        {
            nx--;
        }
        while (ny && y[ny - 1] == 0)
        {
            ny--;
        }
        if (nx != ny)
        {
            return nx < ny;
        }
        for (int i = nx - 1; i >= 0; i--)
        {
            if (x[i] != y[i])
            {
                return x[i] < y[i];
            }
        }
        return false;
    }

    static int cmp(const vector<uint32_t> &x, const vector<uint32_t> &y)
    {
        // x、y 是已去高零的小端 limb 数组；返回无符号比较结果 -1（x<y）、0（相等）、1（x>y）。
        if (x == y)
        {
            return 0;
        }
        return less(x.begin(), x.end(), y.begin(), y.end()) ? -1 : 1;
    }

    static void add(CIt x, CIt xe, CIt y, CIt ye, It z, It ze)
    {
        // [x,xe)、[y,ye) 是小端 limb 区间；把无符号和 x+y 写入 z 的低 max(|x|,|y|) 个 limb，有进位时再写一位。
        // z 至少长 max(|x|,|y|)，产生最高进位时须再多一位；z 中更高的位不被改动。
        if (xe - x < ye - y)
        {
            swap(x, y);
            swap(xe, ye);
        }
        int nx = xe - x, ny = ye - y;
        assert(nx <= ze - z);
        uint32_t carry = 0;
        for (int i = 0; i < ny; i++)
        {
            unsigned long long v = (unsigned long long)x[i] + y[i] + carry;
            z[i] = v;
            carry = v >> 32;
        }
        for (int i = ny; i < nx; i++)
        {
            unsigned long long v = (unsigned long long)x[i] + carry;
            z[i] = v;
            carry = v >> 32;
        }
        if (carry)
        {
            assert(z + nx < ze);
            z[nx] = carry;
        }
    }

    static void addTo(It x, It xe, CIt y, CIt ye)
    {
        // [x,xe)、[y,ye) 是小端 limb 区间且 |x|≥|y|；原地令 x+=y，要求最终进位能在 x 的长度内吸收。
        int nx = xe - x, ny = ye - y;
        uint32_t carry = 0;
        for (int i = 0; i < ny; i++)
        {
            unsigned long long v = (unsigned long long)x[i] + y[i] + carry;
            x[i] = v;
            carry = v >> 32;
        }
        for (int i = ny; carry && i < nx; i++)
        {
            unsigned long long v = (unsigned long long)x[i] + carry;
            x[i] = v;
            carry = v >> 32;
        }
        assert(!carry);
    }

    static void subFrom(It x, It xe, CIt y, CIt ye)
    {
        // [x,xe)、[y,ye) 是小端 limb 区间且 |x|≥|y|；原地令 x-=y，要求 x 的值不小于 y（无最终借位）。
        int nx = xe - x, ny = ye - y;
        int borrow = 0;
        for (int i = 0; i < ny; i++)
        {
            long long v = (long long)x[i] - y[i] + borrow;
            x[i] = v;
            borrow = v >> 32;
        }
        for (int i = ny; borrow && i < nx; i++)
        {
            long long v = (long long)x[i] + borrow;
            x[i] = v;
            borrow = v >> 32;
        }
        assert(!borrow);
    }

    static vector<uint32_t> addVec(const vector<uint32_t> &x, const vector<uint32_t> &y)
    {
        // x、y 是小端 limb 数组；返回去高零后的无符号和 x+y。
        vector<uint32_t> z(max(x.size(), y.size()) + 1);
        add(x.begin(), x.end(), y.begin(), y.end(), z.begin(), z.end());
        trim(z);
        return z;
    }

    static vector<uint32_t> subVec(const vector<uint32_t> &x, const vector<uint32_t> &y)
    {
        // x、y 是去高零的小端 limb 数组且 x≥y；返回去高零后的无符号差 x-y。
        assert(cmp(x, y) >= 0);
        vector<uint32_t> z = x;
        subFrom(z.begin(), z.end(), y.begin(), y.end());
        trim(z);
        return z;
    }

    static void mulNaive(CIt x, CIt xe, CIt y, CIt ye, It z, It ze)
    {
        // [x,xe)、[y,ye) 是小端 limb 区间；竖式乘法把无符号积 x*y 写入 z。
        // z 长度恰为 |x|+|y|，且低 |y| 个 limb 须为零（通常整体清零）。
        int nx = xe - x, ny = ye - y;
        assert(ze - z == nx + ny);
        for (int i = 0; i < nx; i++)
        {
            uint32_t carry = 0;
            for (int j = 0; j < ny; j++)
            {
                unsigned long long v = (unsigned long long)x[i] * y[j] + carry + z[i + j];
                z[i + j] = v;
                carry = v >> 32;
            }
            z[i + ny] = carry;
        }
    }

    static void mul(CIt x, CIt xe, CIt y, CIt ye, It z, It ze)
    {
        // [x,xe)、[y,ye) 是小端 limb 区间，z 长度恰为 |x|+|y| 且须预先清零；把无符号积 x*y 写入 z。
        // 较短因子不超过 128 个 limb 时用竖式，否则用 Karatsuba（长短悬殊时先把长因子对半拆开）。
        if (xe - x < ye - y)
        {
            swap(x, y);
            swap(xe, ye);
        }
        int nx = xe - x, ny = ye - y;
        assert(ze - z == nx + ny);
        if (ny <= 128)
        {
            mulNaive(x, xe, y, ye, z, ze);
            return;
        }
        int n = (nx + 1) / 2;
        if (ny <= n)
        {
            mul(x, x + n, y, ye, z, z + n + ny);
            vector<uint32_t> carry(z + n, z + n + ny);
            fill(z + n, z + n + ny, 0);
            mul(x + n, xe, y, ye, z + n, ze);
            addTo(z + n, ze, carry.begin(), carry.end());
            return;
        }
        mul(x, x + n, y, y + n, z, z + 2 * n);
        mul(x + n, xe, y + n, ye, z + 2 * n, ze);
        vector<uint32_t> x1(n + 1), y1(n + 1), mid(2 * n + 2);
        add(x, x + n, x + n, xe, x1.begin(), x1.end());
        add(y, y + n, y + n, ye, y1.begin(), y1.end());
        mul(x1.begin(), x1.end(), y1.begin(), y1.end(), mid.begin(), mid.end());
        subFrom(mid.begin(), mid.end(), z, z + 2 * n);
        subFrom(mid.begin(), mid.end(), z + 2 * n, ze);
        trim(mid);
        addTo(z + n, ze, mid.begin(), mid.end());
    }

    static vector<uint32_t> mulVec(const vector<uint32_t> &x, const vector<uint32_t> &y)
    {
        // x、y 是去高零的小端 limb 数组；返回去高零后的无符号积，任一为空（零）时返回空。
        if (x.empty() || y.empty())
        {
            return {};
        }
        vector<uint32_t> z(x.size() + y.size());
        mul(x.begin(), x.end(), y.begin(), y.end(), z.begin(), z.end());
        trim(z);
        return z;
    }

    static void leftShift(It x, It xe, int s)
    {
        // [x,xe) 是小端 limb 区间，0<=s<32；原地整体左移 s 位，要求最高 limb 被移出的位全为零。
        assert(0 <= s && s < 32);
        if (!s)
        {
            return;
        }
        int back = 32 - s;
        uint32_t carry = 0;
        for (auto it = x; it != xe; ++it)
        {
            uint32_t v = (*it << s) | carry;
            carry = *it >> back;
            *it = v;
        }
        assert(!carry);
    }

    static void shiftR(It x, It xe, int s)
    {
        // [x,xe) 是小端 limb 区间，0<=s<32；原地整体右移 s 位，要求最低 limb 的低 s 位为零（值能被 2^s 整除）。
        assert(0 <= s && s < 32);
        if (!s)
        {
            return;
        }
        int back = 32 - s;
        uint32_t carry = 0;
        for (auto it = xe; it != x;)
        {
            --it;
            uint32_t v = (*it >> s) | carry;
            carry = *it << back;
            *it = v;
        }
        assert(!carry);
    }

    static void divNaive(CIt x, CIt xe, CIt y, CIt ye, It q, It qe, It r, It re)
    {
        // [x,xe) 是被除数、[y,ye) 是除数（最高 limb 非零）的小端 limb 区间；规格化长除法把商写入 q、余数写入 r。
        // 只写商与余数的有效低位，q 须能容纳商、r 至少 |y| 个 limb，二者须预先清零。
        int nx = xe - x, ny = ye - y;
        if (ny == 1)
        {
            unsigned long long carry = 0;
            for (int i = nx - 1; i >= 0; i--)
            {
                unsigned long long v = (carry << 32) | x[i];
                q[i] = v / y[0];
                carry = v - (unsigned long long)q[i] * y[0];
            }
            r[0] = carry;
            return;
        }
        if (max(nx, ny) <= 2)
        {
            unsigned long long xv = x[0] | (nx == 2 ? (unsigned long long)x[1] << 32 : 0);
            unsigned long long yv = y[0] | (ny == 2 ? (unsigned long long)y[1] << 32 : 0);
            unsigned long long qv = xv / yv, rv = xv % yv;
            if (q != qe)
            {
                q[0] = qv;
                if (q + 1 != qe)
                {
                    q[1] = qv >> 32;
                }
            }
            r[0] = rv;
            if (r + 1 != re)
            {
                r[1] = rv >> 32;
            }
            return;
        }
        if (less(x, xe, y, ye))
        {
            copy(x, min(xe, x + (ye - y)), r);
            return;
        }
        int shift = countl_zero(y[ny - 1]);
        vector<uint32_t> a(nx + (countl_zero(x[nx - 1]) < shift));
        vector<uint32_t> b(y, ye);
        copy(x, xe, a.begin());
        leftShift(a.begin(), a.end(), shift);
        leftShift(b.begin(), b.end(), shift);
        uint32_t top = b.back();
        vector<uint32_t> qv(a.size() - b.size() + 1);
        vector<uint32_t> rv(a.end() - b.size(), a.end());
        for (int i = qv.size() - 1; i >= 0; i--)
        {
            if (rv.size() == b.size())
            {
                if (!less(rv.begin(), rv.end(), b.begin(), b.end()))
                {
                    qv[i] = 1;
                    rv = subVec(rv, b);
                }
            }
            else if (rv.size() > b.size())
            {
                unsigned long long top2 = ((unsigned long long)rv.back() << 32) | rv[rv.size() - 2];
                unsigned long long z = min<unsigned long long>(UINT32_MAX, top2 / top);
                vector<uint32_t> bz = mulVec(b, vector<uint32_t>{(uint32_t)z});
                while (less(rv.begin(), rv.end(), bz.begin(), bz.end()))
                {
                    z--;
                    bz = subVec(bz, b);
                }
                rv = subVec(rv, bz);
                while (!less(rv.begin(), rv.end(), b.begin(), b.end()))
                {
                    z++;
                    rv = subVec(rv, b);
                }
                qv[i] = z;
            }
            if (i)
            {
                rv.insert(rv.begin(), a[i - 1]);
            }
            trim(rv);
        }
        shiftR(rv.begin(), rv.end(), shift);
        trim(qv);
        trim(rv);
        copy(qv.begin(), qv.end(), q);
        copy(rv.begin(), rv.end(), r);
    }

    static void div2n1n(CIt x, CIt xe, CIt y, CIt ye, It q, It qe, It r, It re)
    {
        // x 长 2n、y 长 n（divide 规格化后的除数），要求 x 的高 n 个 limb 小于 y；q 长 n、r 长 n+1，均须预先清零。
        // 把 x/y 的商写入 q、余数写入 r；n 为奇数或不超过 naiveDiv 时用长除法，否则拆成两次 3n/2n 除法。
        int n = ye - y;
        assert(xe - x == 2 * n && qe - q == n && re - r == n + 1);
        assert(less(x + n, xe, y, ye));
        if (n % 2 || n <= naiveDiv)
        {
            divNaive(x, xe, y, ye, q, qe, r, re);
            return;
        }
        int h = n / 2;
        vector<uint32_t> r1(n + h + 1);
        copy(x, x + h, r1.begin());
        div3n2n(x + h, xe, y, ye, q + h, qe, r1.begin() + h, r1.end());
        div3n2n(r1.begin(), r1.end() - 1, y, ye, q, q + h, r, re);
    }

    static void div3n2n(CIt x, CIt xe, CIt y, CIt ye, It q, It qe, It r, It re)
    {
        // x 长 3n、y 长 2n，要求 x 的高 2n 个 limb 小于 y；q 长 n、r 长 2n+1，均须预先清零。
        // 用高位 2n/n 除法（或全 1）估商，再反复 q--、r+=y 修正，把 x/y 的商写入 q、余数写入 r。
        int n = (ye - y) / 2;
        assert(xe - x == 3 * n && qe - q == n && re - r == 2 * n + 1);
        assert(less(x + n, xe, y, ye));
        vector<uint32_t> d(2 * n);
        if (less(x + 2 * n, xe, y + n, ye))
        {
            div2n1n(x + n, xe, y + n, ye, q, qe, r + n, re);
            mul(q, qe, y, y + n, d.begin(), d.end());
        }
        else
        {
            fill(q, qe, UINT32_MAX);
            add(x + n, x + 2 * n, y + n, ye, r + n, re);
            copy(y, y + n, d.begin() + n);
            subFrom(d.begin(), d.end(), y, y + n);
        }
        copy(x, x + n, r);
        while (less(r, re, d.begin(), d.end()))
        {
            addTo(r, re, y, ye);
            vector<uint32_t> one{1};
            subFrom(q, qe, one.begin(), one.end());
        }
        subFrom(r, re, d.begin(), d.end());
    }

    static void divide(CIt x, CIt xe, CIt y, CIt ye, It q, It qe, It r, It re)
    {
        // [x,xe)、[y,ye) 是小端 limb 区间（y 非空且最高 limb 非零），q 长 |x|-|y|+1、r 长 |y|，均须预先清零。
        // 把 x/y 的商写入 q、余数写入 r；较短时用长除法，否则补零规格化后逐块调用 2n/n 除法。
        int nx = xe - x, ny = ye - y;
        assert(ny > 0 && qe - q == nx - ny + 1 && re - r == ny);
        if (min(nx - ny, ny) <= naiveDiv)
        {
            divNaive(x, xe, y, ye, q, qe, r, re);
            return;
        }
        int blocks = (ny + naiveDiv - 1) / naiveDiv;
        if (blocks > 1)
        {
            blocks = bit_ceil((unsigned long long)blocks);
        }
        int n = (ny + blocks - 1) / blocks * blocks;
        int digit = n - ny;
        int shift = countl_zero(y[ny - 1]);
        vector<uint32_t> a(nx + digit + (countl_zero(x[nx - 1]) <= shift));
        vector<uint32_t> b(n), rem(n + 1), z(2 * n);
        copy(x, xe, a.begin() + digit);
        copy(y, ye, b.begin() + digit);
        leftShift(a.begin() + digit, a.end(), shift);
        leftShift(b.begin() + digit, b.end(), shift);
        int t = max<long long>(2, (a.size() + n - 1) / n);
        copy(a.begin() + (t - 2) * n, a.end(), z.begin());
        int upper = qe - (q + (t - 2) * n);
        if (upper < n)
        {
            vector<uint32_t> qq(n);
            div2n1n(z.begin(), z.end(), b.begin(), b.end(), qq.begin(), qq.end(), rem.begin(), rem.end());
            copy(qq.begin(), qq.begin() + upper, q + (t - 2) * n);
        }
        else
        {
            div2n1n(z.begin(), z.end(), b.begin(), b.end(), q + (t - 2) * n, q + (t - 1) * n, rem.begin(), rem.end());
        }
        for (int i = t - 3; i >= 0; i--)
        {
            copy(a.begin() + i * n, a.begin() + (i + 1) * n, z.begin());
            copy(rem.begin(), rem.begin() + n, z.begin() + n);
            fill(rem.begin(), rem.end(), 0);
            div2n1n(z.begin(), z.end(), b.begin(), b.end(), q + i * n, q + (i + 1) * n, rem.begin(), rem.end());
        }
        copy(rem.begin() + digit, rem.begin() + digit + (re - r), r);
        shiftR(r, re, shift);
    }

    static pair<vector<uint32_t>, vector<uint32_t>> divVec(const vector<uint32_t> &x, const vector<uint32_t> &y)
    {
        // x、y 是去高零的小端 limb 数组且 y 非空；返回去高零后的无符号 {商,余数}，x<y 时为 {空,x}。
        assert(!y.empty());
        if (cmp(x, y) < 0)
        {
            return {{}, x};
        }
        vector<uint32_t> q(x.size() - y.size() + 1), r(y.size());
        divide(x.begin(), x.end(), y.begin(), y.end(), q.begin(), q.end(), r.begin(), r.end());
        trim(q);
        trim(r);
        return {q, r};
    }
};
```

快速十进制版复用以下静态 Montgomery NTT 内核；固定 32 位约减要求素数 $p<2^{30}$，原根与变换长度必须合法。内核在本条完整打印，普通模逆元仍依赖 [10.1.004 任意模数卷积](10-多项式、变换与线性递推.md#101004) 的 `arbModConv::inv`。

```cpp
#pragma push_macro("int")
#undef int
namespace arbModConv
{
// 固定 32 位 Montgomery 模运算内核；约减、移位和槽布局依赖该位宽。
template <uint32_t mod, uint32_t primitive> struct StaticNTT
{
    static constexpr int level = countr_zero(mod - 1); // 最大二进制变换层数。
    static constexpr uint32_t montInv()
    {
        // 无参数；返回 mod 在模 2^32 下的逆元，供 Montgomery 约减使用。
        uint32_t x = mod;
        for (int i = 0; i < 4; i++)
        {
            x *= 2 - mod * x;
        }
        return x;
    }
    static constexpr uint32_t inv32 = montInv(); // 模 2^32 的逆元。
    static constexpr uint32_t r2 = -(unsigned long long)(mod) % mod; // 2^64 mod mod。
    // 正逆单位根、radix-2/4 旋转步长，以及连续布局的正逆旋转因子。
    static inline array<uint32_t, 32> root, iroot, rate2, irate2, rate3, irate3, dw, dy;
    static inline bool ready = false; // 是否已建立根表。

    static uint32_t reduce(unsigned long long x)
    {
        // 0<=x<mod*2^32；返回 [0,2mod) 内的 Montgomery 约减结果。
        unsigned long long t = (uint32_t)x * (uint32_t)(-inv32);
        return (x + t * mod) >> 32;
    }
    static uint32_t toMont(uint32_t x)
    {
        // 0<=x<mod；返回 x 的 Montgomery 表示。
        return reduce((unsigned long long)x * r2);
    }
    static uint32_t fromMont(uint32_t x)
    {
        // x 是小于 2mod 的 Montgomery 表示；返回 [0,mod) 内的普通剩余。
        uint32_t v = reduce(x);
        return v >= mod ? v - mod : v;
    }
    static uint32_t mul(uint32_t x, uint32_t y)
    {
        // x、y 是小于 2mod 的 Montgomery 表示；返回乘积的 Montgomery 表示。
        return reduce((unsigned long long)x * y);
    }
    static uint32_t add(uint32_t x, uint32_t y)
    {
        // x、y 是 [0,2mod) 内的表示；返回同区间内的和。
        uint32_t v = x + y - 2 * mod;
        return (int32_t)v < 0 ? v + 2 * mod : v;
    }
    static uint32_t sub(uint32_t x, uint32_t y)
    {
        // x、y 是 [0,2mod) 内的表示；返回同区间内的差。
        uint32_t v = x - y;
        return (int32_t)v < 0 ? v + 2 * mod : v;
    }

    static uint32_t power(uint32_t a, uint32_t e)
    {
        // a 是模 mod 剩余，e 是非负指数；返回 a^e mod mod。
        unsigned long long r = 1;
        while (e)
        {
            if (e & 1) r = r * a % mod;
            a = (unsigned long long)a * a % mod;
            e >>= 1;
        }
        return r;
    }

    static void init()
    {
        // 无参数；首次调用时建立 radix-2/radix-4 跨块旋转因子。
        if (ready) return;
        ready = true;
        root[level] = toMont(power(primitive, (mod - 1) >> level));
        iroot[level] = toMont(power(fromMont(root[level]), mod - 2));
        for (int i = level - 1; i >= 0; i--)
        {
            root[i] = mul(root[i + 1], root[i + 1]);
            iroot[i] = mul(iroot[i + 1], iroot[i + 1]);
        }
        uint32_t x = toMont(1), y = toMont(1);
        for (int i = 0; i + 2 <= level; i++)
        {
            rate2[i] = mul(root[i + 2], x);
            irate2[i] = mul(iroot[i + 2], y);
            x = mul(x, iroot[i + 2]);
            y = mul(y, root[i + 2]);
        }
        x = y = toMont(1);
        for (int i = 0; i + 3 <= level; i++)
        {
            rate3[i] = mul(root[i + 3], x);
            irate3[i] = mul(iroot[i + 3], y);
            x = mul(x, iroot[i + 3]);
            y = mul(y, root[i + 3]);
        }
        dw[1] = root[2], dy[1] = iroot[2];
        dw[2] = root[3], dy[2] = iroot[3];
        for (int i = 3; i < level; i++)
        {
            dw[i] = mul(mul(dw[i - 1], iroot[i]), root[i + 1]);
            dy[i] = mul(mul(dy[i - 1], root[i]), iroot[i + 1]);
        }
    }

    static void dft(vector<uint32_t> &a)
    {
        // a 的长度是 mod-1 所支持的 2 的幂；原地完成 radix-4 正变换。
        init();
        int n = a.size(), h = countr_zero((uint32_t)n), len = 0;
        assert(n > 0 && has_single_bit((unsigned long long)n) && h <= level);
        while (len < h)
        {
            if (h - len == 1)
            {
                int p = 1 << (h - len - 1);
                uint32_t rot = toMont(1);
                for (int s = 0; s < (1 << len); s++)
                {
                    int off = s << (h - len);
                    for (int i = 0; i < p; i++)
                    {
                        uint32_t l = a[off + i], r = mul(a[off + i + p], rot);
                        a[off + i] = add(l, r);
                        a[off + i + p] = sub(l, r);
                    }
                    if (s + 1 != (1 << len)) rot = mul(rot, rate2[countr_zero((uint32_t)~s)]);
                }
                len++;
            }
            else
            {
                int p = 1 << (h - len - 2);
                uint32_t rot = toMont(1), imag = root[2];
                for (int s = 0; s < (1 << len); s++)
                {
                    uint32_t rot2 = mul(rot, rot);
                    uint32_t rot3 = mul(rot2, rot);
                    int off = s << (h - len);
                    for (int i = 0; i < p; i++)
                    {
                        uint32_t a0 = a[off + i];
                        uint32_t a1 = mul(a[off + i + p], rot);
                        uint32_t a2 = mul(a[off + i + 2 * p], rot2);
                        uint32_t a3 = mul(a[off + i + 3 * p], rot3);
                        uint32_t t0 = add(a0, a2), t1 = add(a1, a3);
                        uint32_t x = mul(sub(a1, a3), imag), y = sub(a0, a2);
                        a[off + i] = add(t0, t1);
                        a[off + i + p] = sub(t0, t1);
                        a[off + i + 2 * p] = add(y, x);
                        a[off + i + 3 * p] = sub(y, x);
                    }
                    if (s + 1 != (1 << len)) rot = mul(rot, rate3[countr_zero((uint32_t)~s)]);
                }
                len += 2;
            }
        }
    }

    static void idft(vector<uint32_t> &a)
    {
        // a 来自 dft；原地完成逆变换并除以序列长度。
        init();
        int n = a.size(), h = countr_zero((uint32_t)n), len = h;
        assert(n > 0 && has_single_bit((unsigned long long)n) && h <= level);
        while (len)
        {
            if (len == 1)
            {
                int p = 1 << (h - len);
                uint32_t rot = toMont(1);
                for (int s = 0; s < (1 << (len - 1)); s++)
                {
                    int off = s << (h - len + 1);
                    for (int i = 0; i < p; i++)
                    {
                        uint32_t l = a[off + i], r = a[off + i + p];
                        a[off + i] = add(l, r);
                        a[off + i + p] = mul(sub(l, r), rot);
                    }
                    if (s + 1 != (1 << (len - 1))) rot = mul(rot, irate2[countr_zero((uint32_t)~s)]);
                }
                len--;
            }
            else
            {
                int p = 1 << (h - len);
                uint32_t rot = toMont(1), imag = iroot[2];
                for (int s = 0; s < (1 << (len - 2)); s++)
                {
                    uint32_t rot2 = mul(rot, rot);
                    uint32_t rot3 = mul(rot2, rot);
                    int off = s << (h - len + 2);
                    for (int i = 0; i < p; i++)
                    {
                        uint32_t a0 = a[off + i], a1 = a[off + i + p];
                        uint32_t a2 = a[off + i + 2 * p], a3 = a[off + i + 3 * p];
                        uint32_t t0 = add(a0, a1), t1 = add(a2, a3);
                        uint32_t x = mul(sub(a2, a3), imag), y = sub(a0, a1);
                        a[off + i] = add(t0, t1);
                        a[off + i + p] = mul(add(y, x), rot);
                        a[off + i + 2 * p] = mul(sub(t0, t1), rot2);
                        a[off + i + 3 * p] = mul(sub(y, x), rot3);
                    }
                    if (s + 1 != (1 << (len - 2))) rot = mul(rot, irate3[countr_zero((uint32_t)~s)]);
                }
                len -= 2;
            }
        }
        uint32_t iv = toMont(power(n, mod - 2));
        for (auto &x : a)
        {
            x = fromMont(mul(x, iv));
        }
    }

    static void dftCompact(vector<uint32_t> &a)
    {
        // a 以 Montgomery 形式保存；按连续 radix-4 块原地完成正变换。
        init();
        int n = a.size(), k = countr_zero((uint32_t)n);
        if (n <= 1) return;
        if (k == 1)
        {
            uint32_t x = a[1];
            a[1] = sub(a[0], x);
            a[0] = add(a[0], x);
            return;
        }
        if (k & 1)
        {
            int v = 1 << (k - 1);
            for (int i = 0; i < v; i++)
            {
                uint32_t x = a[i + v];
                a[i + v] = sub(a[i], x);
                a[i] = add(a[i], x);
            }
        }
        int u = 1 << (2 + (k & 1)), v = 1 << (k - 2 - (k & 1));
        uint32_t one = toMont(1), imag = dw[1];
        while (v)
        {
            for (int i = 0; i < v; i++)
            {
                uint32_t a0 = a[i], a1 = a[i + v], a2 = a[i + 2 * v], a3 = a[i + 3 * v];
                uint32_t t0 = add(a0, a2), t1 = add(a1, a3);
                uint32_t x = mul(sub(a1, a3), imag), y = sub(a0, a2);
                a[i] = add(t0, t1);
                a[i + v] = sub(t0, t1);
                a[i + 2 * v] = add(y, x);
                a[i + 3 * v] = sub(y, x);
            }
            uint32_t xx = mul(one, dw[2]);
            for (int jh = 4; jh < u;)
            {
                uint32_t ww = mul(xx, xx), wx = mul(ww, xx);
                int l = jh * v, r = l + v;
                for (int i = l; i < r; i++)
                {
                    uint32_t a0 = a[i], a1 = mul(a[i + v], xx);
                    uint32_t a2 = mul(a[i + 2 * v], ww), a3 = mul(a[i + 3 * v], wx);
                    uint32_t t0 = add(a0, a2), t1 = add(a1, a3);
                    uint32_t x = mul(sub(a1, a3), imag), y = sub(a0, a2);
                    a[i] = add(t0, t1);
                    a[i + v] = sub(t0, t1);
                    a[i + 2 * v] = add(y, x);
                    a[i + 3 * v] = sub(y, x);
                }
                xx = mul(xx, dw[countr_zero((uint32_t)(jh += 4))]);
            }
            u <<= 2;
            v >>= 2;
        }
    }

    static void idftCompact(vector<uint32_t> &a)
    {
        // a 来自 dftCompact；原地逆变换、除以长度并转回普通剩余。
        init();
        int n = a.size(), k = countr_zero((uint32_t)n);
        if (n <= 1)
        {
            if (n == 1) a[0] = fromMont(a[0]);
            return;
        }
        if (k == 1)
        {
            uint32_t x = a[1];
            a[1] = sub(a[0], x);
            a[0] = add(a[0], x);
        }
        else
        {
            int u = 1 << (k - 2), v = 1;
            uint32_t one = toMont(1), imag = dy[1];
            while (u)
            {
                for (int i = 0; i < v; i++)
                {
                    uint32_t a0 = a[i], a1 = a[i + v], a2 = a[i + 2 * v], a3 = a[i + 3 * v];
                    uint32_t t0 = add(a0, a1), t1 = add(a2, a3);
                    uint32_t x = mul(sub(a2, a3), imag), y = sub(a0, a1);
                    a[i] = add(t0, t1);
                    a[i + v] = add(y, x);
                    a[i + 2 * v] = sub(t0, t1);
                    a[i + 3 * v] = sub(y, x);
                }
                u <<= 2;
                uint32_t xx = mul(one, dy[2]);
                for (int jh = 4; jh < u;)
                {
                    uint32_t ww = mul(xx, xx), yy = mul(xx, imag);
                    int l = jh * v, r = l + v;
                    for (int i = l; i < r; i++)
                    {
                        uint32_t a0 = a[i], a1 = a[i + v], a2 = a[i + 2 * v], a3 = a[i + 3 * v];
                        uint32_t t0 = add(a0, a1), t1 = add(a2, a3);
                        uint32_t x = mul(sub(a2, a3), yy), y = mul(sub(a0, a1), xx);
                        a[i] = add(t0, t1);
                        a[i + v] = add(y, x);
                        a[i + 2 * v] = mul(sub(t0, t1), ww);
                        a[i + 3 * v] = mul(sub(y, x), ww);
                    }
                    xx = mul(xx, dy[countr_zero((uint32_t)(jh += 4))]);
                }
                u >>= 4;
                v <<= 2;
            }
            if (k & 1)
            {
                int u2 = 1 << (k - 1);
                for (int i = 0; i < u2; i++)
                {
                    uint32_t x = a[i + u2];
                    a[i + u2] = sub(a[i], x);
                    a[i] = add(a[i], x);
                }
            }
        }
        uint32_t iv = toMont(power(n, mod - 2));
        for (auto &x : a)
        {
            x = fromMont(mul(x, iv));
        }
    }

    static vector<uint32_t> conv(const vector<unsigned long long> &x,
                                 const vector<unsigned long long> &y)
    {
        // x、y 是普通整数系数；返回二者在 mod 下的线性卷积。
        if (x.empty() || y.empty()) return {};
        int need = x.size() + y.size() - 1;
        int n = bit_ceil((unsigned long long)need);
        vector<uint32_t> a(n), b(n);
        for (int i = 0; i < (int)x.size(); i++)
        {
            a[i] = toMont(x[i] % mod);
        }
        for (int i = 0; i < (int)y.size(); i++)
        {
            b[i] = toMont(y[i] % mod);
        }
        dft(a);
        dft(b);
        for (int i = 0; i < n; i++)
        {
            a[i] = mul(a[i], b[i]);
        }
        idft(a);
        a.resize(need);
        return a;
    }
};
}
```

`FastBigInt` 直接读写十进制分组；与普通 `BigInt`、`HexBigInt` 可以同时使用。

```cpp
// 依赖 10.1.004 的 arbModConv::inv 及上面的 StaticNTT。
// 十进制百万位整数：以 10^9 为基数，长乘法复用公共三模 NTT，除法使用 Newton 逆。
struct FastBigInt
{
    static constexpr int B = 1000000000; // 十进制分组基数。
    bool neg = false;                         // 是否为负数，零恒为非负。
    vector<int> a;                       // 绝对值的 B 进制小端块，无高位零；性能版用紧凑存储。

    FastBigInt()
    {
        // 无参数；构造整数零。
    }
    FastBigInt(long long x)
    {
        // x 是任意有符号 64 位整数；构造其精确值，包含 LLONG_MIN。
        neg = x < 0;
        // 无符号取绝对值，避免对 LLONG_MIN 做有符号取反。
        unsigned long long y = neg ? 0 - (unsigned long long)x : (unsigned long long)x;
        while (y)
        {
            a.push_back(y % B);
            y /= B;
        }
    }
    FastBigInt(const string &s)
    {
        // s 是带可选正负号的非空十进制串；构造其精确值。
        assign(s);
    }
    FastBigInt(bool s, vector<int> v) : neg(s), a(move(v))
    {
        // s 指定负号，v 是小端 B 进制块且各块小于 B；去高零并消除负零。
        trim(a);
        if (a.empty())
        {
            neg = false;
        }
    }

    FastBigInt &assign(const string &s)
    {
        // s 是带可选正负号的非空十进制串；覆盖当前值并返回 *this。
        assert(!s.empty());
        int p = (s[0] == '-' || s[0] == '+');
        neg = s[0] == '-';
        assert(p < (int)s.size());
        a.clear();
        for (int r = s.size(); r > p; r -= 9)
        {
            int l = max(p, r - 9);
            int v = 0;
            for (int i = l; i < r; i++)
            {
                assert('0' <= s[i] && s[i] <= '9');
                v = v * 10 + s[i] - '0';
            }
            a.push_back(v);
        }
        trim(a);
        if (a.empty())
        {
            neg = false;
        }
        return *this;
    }

    string toDecimal() const
    {
        // 无参数；返回无前导零的十进制串，负数带负号，零返回 "0"。
        if (a.empty())
        {
            return "0";
        }
        string s = neg ? "-" : "";
        s += to_string(a.back());
        for (int i = (int)a.size() - 2; i >= 0; i--)
        {
            string t = to_string(a[i]);
            s += string(9 - t.size(), '0') + t;
        }
        return s;
    }

    friend bool operator==(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否相等 y。
        return x.neg == y.neg && x.a == y.a;
    }
    friend bool operator<(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否小于 y。
        if (x.neg != y.neg)
        {
            return x.neg;
        }
        int c = cmp(x.a, y.a);
        return x.neg ? c > 0 : c < 0;
    }
    friend bool operator!=(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否不等 y。
        return !(x == y);
    }
    friend bool operator>(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否大于 y。
        return y < x;
    }
    friend bool operator<=(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否小于等于 y。
        return !(y < x);
    }
    friend bool operator>=(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回 x 是否大于等于 y。
        return !(x < y);
    }

    friend FastBigInt operator-(const FastBigInt &x)
    {
        // x 是规范整数；返回相反数，零仍为非负。
        FastBigInt y = x;
        if (!y.a.empty())
        {
            y.neg = !y.neg;
        }
        return y;
    }
    friend FastBigInt operator+(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回和。
        if (x.neg == y.neg)
        {
            return {x.neg, addVec(x.a, y.a)};
        }
        int c = cmp(x.a, y.a);
        if (c <= 0)
        {
            return {c ? y.neg : false, subVec(y.a, x.a)};
        }
        return {x.neg, subVec(x.a, y.a)};
    }
    friend FastBigInt operator-(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回差。
        return x + (-y);
    }
    friend FastBigInt operator*(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数；返回积。
        auto z = mulVec(x.a, y.a);
        return {z.empty() ? false : x.neg != y.neg, move(z)};
    }
    friend pair<FastBigInt, FastBigInt> divmod(const FastBigInt &x, const FastBigInt &y)
    {
        // x 是被除数，y 是非零除数；返回 {向零截断的商,与 x 同号的余数}，零非负。
        assert(!y.a.empty());
        auto z = divVec(x.a, y.a);
        return {{z.first.empty() ? false : x.neg != y.neg, move(z.first)},
                {z.second.empty() ? false : x.neg, move(z.second)}};
    }
    friend FastBigInt operator/(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数且 y 非零；返回向零截断的商。
        return divmod(x, y).first;
    }
    friend FastBigInt operator%(const FastBigInt &x, const FastBigInt &y)
    {
        // x、y 是规范整数且 y 非零；返回与 x 同号的余数。
        return divmod(x, y).second;
    }

  private:
    static void trim(vector<int> &x)
    {
        // x 是小端 B 进制块；原地删除高位零，零变为空，无返回值。
        while (!x.empty() && x.back() == 0)
        {
            x.pop_back();
        }
    }
    static int cmp(const vector<int> &x, const vector<int> &y)
    {
        // x、y 是无高零的小端块；按绝对值比较，返回 -1、0 或 1。
        if (x.size() != y.size())
        {
            return x.size() < y.size() ? -1 : 1;
        }
        for (int i = (int)x.size() - 1; i >= 0; i--)
        {
            if (x[i] != y[i])
            {
                return x[i] < y[i] ? -1 : 1;
            }
        }
        return 0;
    }
    static vector<int> addVec(const vector<int> &x, const vector<int> &y)
    {
        // x、y 是规范非负小端块；返回去高零的和。
        vector<int> z(max(x.size(), y.size()) + 1);
        int carry = 0;
        for (int i = 0; i < (int)z.size() - 1; i++)
        {
            int v = carry + (i < (int)x.size() ? x[i] : 0) + (i < (int)y.size() ? y[i] : 0);
            z[i] = v % B;
            carry = v / B;
        }
        z.back() = carry;
        trim(z);
        return z;
    }
    static vector<int> subVec(const vector<int> &x, const vector<int> &y)
    {
        // x、y 是规范非负小端块且 x>=y；返回去高零的差。
        assert(cmp(x, y) >= 0);
        vector<int> z = x;
        int borrow = 0;
        for (int i = 0; i < (int)z.size(); i++)
        {
            int v = (int)z[i] - (i < (int)y.size() ? y[i] : 0) - borrow;
            if (v < 0)
            {
                v += B, borrow = 1;
            }
            else
            {
                borrow = 0;
            }
            z[i] = v;
        }
        trim(z);
        return z;
    }
    static vector<int> mulSmall(const vector<int> &x, int m)
    {
        // x 是规范非负小端块，0<=m<B；返回去高零的 x*m。
        vector<int> z(x.size() + 1);
        int carry = 0;
        for (int i = 0; i < (int)x.size(); i++)
        {
            unsigned long long v = (unsigned long long)x[i] * m + carry;
            z[i] = v % B;
            carry = v / B;
        }
        z.back() = carry;
        trim(z);
        return z;
    }
    static pair<vector<int>, int> divSmall(const vector<int> &x, int d)
    {
        // x 是规范非负小端块，0<d<B；返回 {去高零的商,整型余数}。
        vector<int> q(x.size());
        int rem = 0;
        for (int i = (int)x.size() - 1; i >= 0; i--)
        {
            unsigned long long v = (unsigned long long)rem * B + x[i];
            q[i] = v / d;
            rem = v % d;
        }
        trim(q);
        return {q, (int)rem};
    }
    static vector<int> mulVec(const vector<int> &x, const vector<int> &y)
    {
        // x、y 是规范非负小端块；返回精确积，长乘法的卷积长度不得超过 2^24。
        if (x.empty() || y.empty())
        {
            return {};
        }
        if (min(x.size(), y.size()) <= 64)
        {
            // 竖式系数累计与 CRT 重建需要超过 64 位的非负中间量。
            vector<__uint128_t> c(x.size() + y.size());
            for (int i = 0; i < (int)x.size(); i++)
            {
                for (int j = 0; j < (int)y.size(); j++)
                {
                    c[i + j] += (__uint128_t)x[i] * y[j];
                }
            }
            vector<int> z(c.size());
            __uint128_t carry = 0; // 保存跨十进制块进位，不能缩成 64 位。
            for (int i = 0; i < (int)c.size(); i++)
            {
                __uint128_t v = c[i] + carry;
                z[i] = v % B;
                carry = v / B;
            }
            trim(z);
            return z;
        }
        // 三素数共同支持长度 2^24；该范围内整数卷积系数严格小于三模乘积。
        assert(x.size() + y.size() - 1 <= (1ULL << 24));
        constexpr int p0 = 167772161, p1 = 469762049, p2 = 754974721;
        constexpr int r0 = 3, r1 = 3, r2 = 11;
        vector<unsigned long long> a(x.begin(), x.end()), b(y.begin(), y.end());
        auto c0 = arbModConv::StaticNTT<p0, r0>::conv(a, b);
        auto c1 = arbModConv::StaticNTT<p1, r1>::conv(a, b);
        auto c2 = arbModConv::StaticNTT<p2, r2>::conv(a, b);
        int i01 = arbModConv::inv(p0 % p1, p1);
        int i012 = arbModConv::inv((unsigned long long)p0 * p1 % p2, p2);
        vector<int> z(c0.size());
        __uint128_t carry = 0; // 保存跨十进制块进位，不能缩成 64 位。
        for (int i = 0; i < (int)c0.size(); i++)
        {
            int u = c0[i];
            int v = ((c1[i] + p1 - u % p1) % p1) * (unsigned long long)i01 % p1;
            int seen = (u + (unsigned long long)p0 * v) % p2;
            int w = ((c2[i] + p2 - seen) % p2) * (unsigned long long)i012 % p2;
            __uint128_t cur = carry + u + (__uint128_t)p0 * v + (__uint128_t)p0 * p1 * w;
            z[i] = cur % B;
            carry = cur / B;
        }
        while (carry)
        {
            z.push_back(carry % B);
            carry /= B;
        }
        trim(z);
        return z;
    }
    static pair<vector<int>, vector<int>> divNaive(const vector<int> &x, const vector<int> &y)
    {
        // x、y 是规范非负小端块且 y 非零；竖式除法返回 {商,余数}。
        if (x.empty() || cmp(x, y) < 0)
        {
            return {{}, x};
        }
        if (y.size() == 1)
        {
            auto [q, r] = divSmall(x, y[0]);
            return {q, r ? vector<int>{r} : vector<int>{}};
        }
        int norm = B / (y.back() + 1);
        vector<int> a = mulSmall(x, norm), b = mulSmall(y, norm);
        vector<int> q(a.size() - b.size() + 1), r(a.end() - b.size(), a.end());
        for (int i = (int)q.size() - 1; i >= 0; i--)
        {
            unsigned long long top = r.size() > b.size() ? (unsigned long long)r.back() * B + r[r.size() - 2] : 0;
            int d = top ? min<unsigned long long>(B - 1, top / b.back()) : 1;
            vector<int> z = mulSmall(b, d);
            while (cmp(r, z) < 0)
            {
                d--, z = subVec(z, b);
            }
            r = subVec(r, z);
            while (cmp(r, b) >= 0)
            {
                d++, r = subVec(r, b);
            }
            q[i] = d;
            if (i)
            {
                r.insert(r.begin(), a[i - 1]);
            }
            trim(r);
        }
        auto rr = divSmall(r, norm);
        assert(rr.second == 0);
        trim(q);
        return {q, rr.first};
    }
    static vector<int> calcInv(const vector<int> &a, int deg)
    {
        // a 是最高块至少 B/2 的规范正整数，deg>0；返回 B^(|a|+deg)/a 的整数近似逆。
        int k = deg, c = a.size();
        while (k > 64)
        {
            k = (k + 1) / 2;
        }
        vector<int> z(c + k + 1);
        z.back() = 1;
        z = divNaive(z, a).first;
        while (k < deg)
        {
            vector<int> s = mulVec(z, z);
            s.insert(s.begin(), 0);
            vector<int> t(2 * k + 1);
            int take = min<int>(a.size(), t.size());
            copy(a.end() - take, a.end(), t.end() - take);
            vector<int> u = mulVec(s, t);
            if ((int)u.size() > 2 * k + 1)
            {
                u.erase(u.begin(), u.begin() + 2 * k + 1);
            }
            else
            {
                u.clear();
            }
            vector<int> w(k + 1), w2 = addVec(z, z);
            copy(w2.begin(), w2.end(), back_inserter(w));
            z = subVec(w, u);
            if (!z.empty())
            {
                z.erase(z.begin());
            }
            k *= 2;
        }
        if (k > deg)
        {
            z.erase(z.begin(), z.begin() + k - deg);
        }
        return z;
    }
    static pair<vector<int>, vector<int>> divVec(const vector<int> &x, const vector<int> &y)
    {
        // x、y 是规范非负小端块且 y 非零；返回精确 {商,余数}，长除用 Newton 逆估商后修正。
        assert(!y.empty());
        if (cmp(x, y) < 0)
        {
            return {{}, x};
        }
        if (y.size() <= 64 || x.size() - y.size() <= 64)
        {
            return divNaive(x, y);
        }
        int norm = B / (y.back() + 1);
        vector<int> a = mulSmall(x, norm), b = mulSmall(y, norm);
        int deg = a.size() - b.size() + 2;
        vector<int> inv = calcInv(b, deg);
        vector<int> q = mulVec(a, inv);
        if ((int)q.size() > (int)b.size() + deg)
        {
            q.erase(q.begin(), q.begin() + b.size() + deg);
        }
        else
        {
            q.clear();
        }
        vector<int> prod = mulVec(b, q);
        while (cmp(a, prod) < 0)
        {
            q = subVec(q, {1}), prod = subVec(prod, b);
        }
        vector<int> r = subVec(a, prod);
        while (cmp(r, b) >= 0)
        {
            q = addVec(q, {1}), r = subVec(r, b);
        }
        auto rr = divSmall(r, norm);
        assert(rr.second == 0);
        trim(q);
        return {q, rr.first};
    }
};
```

`HexBigInt x(s)` 的 `s` 是可选单个正负号后接十六进制数字，大小写均可；`toHex()` 和流输出返回大写规范串。`fromDecimal(s)`、`toDecimal()` 精确转换十进制串。`FastBigInt x(s)` 直接接收十进制串，`toDecimal()` 返回规范十进制串。两类均可从任意 `long long`（含最小负数）构造；构造符号与分组数组时，各块须落在相应基数范围内。不得直接修改成员而破坏规范表示。

十进制长乘法把块作为多项式系数，依次模 $p_0=167772161$、$p_1=469762049$、$p_2=754974721$ 卷积，再用 CRT 恢复整数系数并按 $B$ 进位。共同变换上限为 $2^{24}$，要求两数块数之和减一不超过该值；此时每个系数小于 $\min(n,m)(B-1)^2<p_0p_1p_2$，恢复唯一。除法先把最高块规格化到至少 $B/2$，以 $z\leftarrow z(2-az)$ 的定点 Newton 迭代逐次加倍逆的有效块数，再由乘积截位估商，修正到 $0\le r<y$。这里的逆是普通整数的缩放倒数，与模逆元不同。

设较长操作数有 $n$ 块，较短有 $m$ 块。两类读写各自原生进制、比较与加减为 $O(n)$ 时间、$O(n)$ 空间。`HexBigInt` 的平衡乘法为 $M(n)=O(n^{\log_2 3})$，除法上界为 $O(M(n)\log n)$；进制转换上界为 $O(M(n)\log^2 n)$，递归临时空间上界为 $O(n\log n)$。`FastBigInt` 的长乘为 $M(n)=O(n\log n)$，Newton 除法保守上界为 $O(M(n)\log n)$ 时间，当前根表与临时数组空间为 $O(n)$；短乘走 $O(nm)$ 竖式，短除数或短商走 $O(nm)$ 长除。这里 $n,m$ 分别以对应类型的基数计块；百万位是目标输入规模，并非固定容量保证，仍受变换长度与内存限制。常规中等位数四则选 [14.1.001 高精度整数](#141001)，带小数和精度控制选 [14.1.020 任意精度定点与十进制数](#141020)。

以下固定答案检查符号、精确余数与十进制转换：

```cpp
signed main()
{
    assert((HexBigInt("FFFFFFFF") + HexBigInt(1)).toHex() == "100000000");
    assert((HexBigInt("1234") * HexBigInt("10")).toHex() == "12340");
    auto [q0, r0] = divmod(HexBigInt("123456789ABCDEF"), HexBigInt("1000"));
    assert(q0.toHex() == "123456789ABC" && r0.toHex() == "DEF");

    assert(HexBigInt::fromDecimal("-18446744073709551616").toHex() == "-10000000000000000");
    auto [q1, r1] = divmod(FastBigInt("-10000000000000000001"), FastBigInt("1000000000"));
    assert(q1.toDecimal() == "-10000000000" && r1.toDecimal() == "-1");
}
```

### 14.1.020 任意精度定点与十进制数

把十进制数表示为 $x=\mathtt{man}\times10^{-\mathtt{scale}}$，其中 `man` 是带符号 `BigInt`，`scale` 是非负小数位数；`normalize` 会删去尾数的十进制末尾零，使相等比较可以直接检查规范表示。这一个类型统一覆盖动态精度定点数与常规任意精度十进制数，`str(fixed)` 还能固定输出小数位。全局精度 $p$ 约束解析与四则运算保留的小数位数，舍弃部分按绝对值四舍五入；加减先对齐小数位，乘法相乘尾数，除法先补 $p+4$ 位保护位再做整数除法，平方根则把被开方数缩放成整数后调用 [14.1.016 高精度整数平方根](#141016)。实现依赖 [14.1.001 高精度整数（BigInt）](#141001)，支持普通十进制串、比较、`+ - * / %`、`trunc`、`pow` 与 `sqrt`，不解析科学计数法。

```cpp
// 依赖 14.1.016 高精度整数平方根
struct Decimal
{
    // man 是带符号十进制尾数，scale 是小数点后的十进制位数。
    BigInt man = 0;
    int scale = 0;
    // prec 是构造及加减乘除、幂和平方根保留的小数位数；取模保留精确余数。
    inline static int prec = 50;

    Decimal() = default;

    Decimal(long long x) : man(x)
    {
        // x 是 64 位整数；构造同值十进制数。
    }

    explicit Decimal(const BigInt &x) : man(x)
    {
        // x 是任意精度整数；构造同值十进制数。
    }

    explicit Decimal(const string &s)
    {
        // s 是带可选正负号和小数点的十进制串；构造按当前精度舍入后的十进制数。
        assign(s);
    }

    static void setPrecision(int digits)
    {
        // digits 是后续运算保留的小数位数；修改全局动态精度，无返回值。
        assert(digits >= 0);
        prec = digits;
    }

    static int getPrecision()
    {
        // 无参数；返回当前动态精度。
        return prec;
    }

    Decimal &assign(const string &s)
    {
        // s 是带可选正负号和至多一个小数点的十进制串；解析、舍入并返回当前对象。
        assert(!s.empty());
        int p = 0, sign = 1;
        if (s[p] == '+' || s[p] == '-')
        {
            sign = s[p] == '-' ? -1 : 1;
            p++;
        }
        assert(p < (int)s.size());
        string digit;
        bool dot = false;
        scale = 0;
        for (; p < (int)s.size(); p++)
        {
            if (s[p] == '.')
            {
                assert(!dot);
                dot = true;
                continue;
            }
            assert('0' <= s[p] && s[p] <= '9');
            digit.push_back(s[p]);
            if (dot)
            {
                scale++;
            }
        }
        assert(!digit.empty());
        man = BigInt((sign < 0 ? "-" : "") + digit);
        round(prec);
        normalize();
        return *this;
    }

    string str(int fixed = -1) const
    {
        // fixed 小于 0 时删除末尾零，否则固定输出 fixed 位小数；返回十进制字符串。
        assert(fixed >= -1);
        Decimal x = *this;
        if (fixed >= 0)
        {
            x.round(fixed);
        }
        if (!x.man.s)
        {
            return fixed > 0 ? "0." + string(fixed, '0') : "0";
        }
        bool neg = x.man.s < 0;
        string s = x.man.str();
        if (neg)
        {
            s.erase(s.begin());
        }
        if ((int)s.size() <= x.scale)
        {
            s = string(x.scale + 1 - s.size(), '0') + s;
        }
        string whole = s.substr(0, s.size() - x.scale);
        string fraction = x.scale ? s.substr(s.size() - x.scale) : "";
        if (fixed >= 0)
        {
            fraction += string(fixed - fraction.size(), '0');
        }
        string ans = (neg ? "-" : "") + whole;
        if (!fraction.empty())
        {
            ans += "." + fraction;
        }
        return ans;
    }

    void round(int digits)
    {
        // digits 是要保留的小数位数；按绝对值五入舍去其余位，无返回值。
        assert(digits >= 0);
        if (scale > digits)
        {
            man = drop10(man, scale - digits, true);
            scale = digits;
        }
        normalize();
    }

    Decimal trunc() const
    {
        // 无参数；返回向零取整后的整数值。
        Decimal ans;
        ans.man = drop10(man, scale, false);
        ans.normalize();
        return ans;
    }

    Decimal abs() const
    {
        // 无参数；返回当前数的绝对值。
        Decimal ans = *this;
        if (ans.man.s < 0)
        {
            ans.man.s = 1;
        }
        return ans;
    }

    bool isZero() const
    {
        // 无参数；返回当前数是否为零。
        return man.s == 0;
    }

    static Decimal pow(Decimal base, int exponent)
    {
        // base 是底数，exponent 是非负整数指数；返回按当前精度计算的幂。
        assert(exponent >= 0);
        Decimal ans(1);
        while (exponent)
        {
            if (exponent & 1)
            {
                ans *= base;
            }
            exponent >>= 1;
            if (exponent)
            {
                base *= base;
            }
        }
        return ans;
    }

    Decimal sqrt() const
    {
        // 无参数；当前数须非负，返回按当前精度四舍五入的平方根。
        assert(man.s >= 0);
        if (!man.s)
        {
            return Decimal(0);
        }
        int work = max(prec + 4, (scale + 1) / 2 + 2);
        int exponent = 2 * work - scale;
        assert(exponent >= 0);
        Decimal ans;
        ans.man = BigInt(bigSqrt(shift10(man, exponent).str()));
        ans.scale = work;
        ans.round(prec);
        return ans;
    }

    Decimal operator-() const
    {
        // 无参数；返回当前数的相反数。
        Decimal ans = *this;
        ans.man = -ans.man;
        return ans;
    }

    Decimal &operator+=(const Decimal &other)
    {
        // other 是加数；把它加到当前数并返回当前对象。
        return *this = *this + other;
    }

    Decimal &operator-=(const Decimal &other)
    {
        // other 是减数；从当前数减去它并返回当前对象。
        return *this = *this - other;
    }

    Decimal &operator*=(const Decimal &other)
    {
        // other 是乘数；把当前数乘以它并返回当前对象。
        return *this = *this * other;
    }

    Decimal &operator/=(const Decimal &other)
    {
        // other 是非零除数；把当前数除以它并返回当前对象。
        return *this = *this / other;
    }

    Decimal &operator%=(const Decimal &other)
    {
        // other 是非零除数；把当前数改为向零取商对应的余数并返回当前对象。
        return *this = *this % other;
    }

    friend Decimal operator+(const Decimal &a, const Decimal &b)
    {
        // a、b 是加数；返回按当前精度舍入的和。
        int scale = max(a.scale, b.scale);
        Decimal ans;
        ans.man = shift10(a.man, scale - a.scale) + shift10(b.man, scale - b.scale);
        ans.scale = scale;
        ans.round(prec);
        return ans;
    }

    friend Decimal operator-(const Decimal &a, const Decimal &b)
    {
        // a 是被减数，b 是减数；返回按当前精度舍入的差。
        return a + (-b);
    }

    friend Decimal operator*(const Decimal &a, const Decimal &b)
    {
        // a、b 是乘数；返回按当前精度舍入的积。
        Decimal ans;
        ans.man = a.man * b.man;
        ans.scale = a.scale + b.scale;
        ans.round(prec);
        return ans;
    }

    friend Decimal operator/(const Decimal &a, const Decimal &b)
    {
        // a 是被除数，b 是非零除数；返回按当前精度舍入的商。
        assert(b.man.s);
        if (!a.man.s)
        {
            return Decimal(0);
        }
        int sign = a.man.s * b.man.s;
        BigInt num = absInt(a.man), den = absInt(b.man);
        int work = prec + 4;
        int exponent = work + b.scale - a.scale;
        if (exponent >= 0)
        {
            num = shift10(num, exponent);
        }
        else
        {
            den = shift10(den, -exponent);
        }
        Decimal ans;
        ans.man = num / den;
        ans.man.s *= sign;
        ans.scale = work;
        ans.round(prec);
        return ans;
    }

    friend Decimal operator%(const Decimal &a, const Decimal &b)
    {
        // a 是被除数，b 是非零除数；返回向零取商的精确余数，不按 prec 舍入。
        assert(b.man.s);
        BigInt num = a.man, den = b.man;
        if (b.scale >= a.scale)
        {
            num = shift10(num, b.scale - a.scale);
        }
        else
        {
            den = shift10(den, a.scale - b.scale);
        }
        Decimal ans;
        ans.man = num % den;
        ans.scale = max(a.scale, b.scale);
        ans.normalize();
        return ans;
    }

    friend bool operator==(const Decimal &a, const Decimal &b)
    {
        // a、b 是待比较十进制数；返回二者是否相等。
        return a.scale == b.scale && a.man == b.man;
    }

    friend bool operator!=(const Decimal &a, const Decimal &b)
    {
        // a、b 是待比较十进制数；返回二者是否不等。
        return !(a == b);
    }

    friend bool operator<(const Decimal &a, const Decimal &b)
    {
        // a、b 是待比较十进制数；返回 a 是否小于 b。
        int scale = max(a.scale, b.scale);
        return shift10(a.man, scale - a.scale) < shift10(b.man, scale - b.scale);
    }

    friend bool operator>(const Decimal &a, const Decimal &b)
    {
        // a、b 是待比较十进制数；返回 a 是否大于 b。
        return b < a;
    }

    friend bool operator<=(const Decimal &a, const Decimal &b)
    {
        // a、b 是待比较十进制数；返回 a 是否不大于 b。
        return !(b < a);
    }

    friend bool operator>=(const Decimal &a, const Decimal &b)
    {
        // a、b 是待比较十进制数；返回 a 是否不小于 b。
        return !(a < b);
    }

    friend ostream &operator<<(ostream &out, const Decimal &x)
    {
        // out 是输出流，x 是待输出十进制数；写入去掉末尾零的表示并返回输出流。
        return out << x.str();
    }

    friend istream &operator>>(istream &in, Decimal &x)
    {
        // in 是输入流，x 接收十进制数；读取一个字符串并返回输入流。
        string s;
        in >> s;
        if (in)
        {
            x.assign(s);
        }
        return in;
    }

  private:
    static BigInt absInt(BigInt x)
    {
        // x 是任意精度整数；返回其绝对值。
        if (x.s < 0)
        {
            x.s = 1;
        }
        return x;
    }

    static BigInt shift10(BigInt x, int digits)
    {
        // x 是任意精度整数，digits 是非负十进制位数；返回 x*10^digits。
        assert(digits >= 0);
        if (!x.s || !digits)
        {
            return x;
        }
        int sign = x.s;
        x.s = 1;
        int blocks = digits / 9, rest = digits % 9;
        if (rest)
        {
            int factor = 1;
            for (int i = 0; i < rest; i++)
            {
                factor *= 10;
            }
            x = BigInt::mulAbs(x, factor);
        }
        x.a.insert(x.a.begin(), blocks, 0);
        x.s = sign;
        return x;
    }

    static BigInt drop10(BigInt x, int digits, bool nearest)
    {
        // x 是任意精度整数，digits 是舍去位数；nearest 表示是否绝对值五入，返回向零截断或舍入后的整数。
        assert(digits >= 0);
        if (!x.s || !digits)
        {
            return x;
        }
        int sign = x.s;
        x.s = 1;
        int blocks = digits / 9, rest = digits % 9;
        int roundDig = 0;
        if (rest && blocks < (int)x.a.size())
        {
            int factor = 1;
            for (int i = 1; i < rest; i++)
            {
                factor *= 10;
            }
            roundDig = x.a[blocks] / factor % 10;
        }
        else if (!rest && blocks && blocks - 1 < (int)x.a.size())
        {
            roundDig = x.a[blocks - 1] / 100000000;
        }
        if (blocks >= (int)x.a.size())
        {
            x = 0;
        }
        else
        {
            x.a.erase(x.a.begin(), x.a.begin() + blocks);
            if (rest)
            {
                int divisor = 1;
                for (int i = 0; i < rest; i++)
                {
                    divisor *= 10;
                }
                unsigned long long carry = 0;
                for (int i = (int)x.a.size() - 1; i >= 0; i--)
                {
                    unsigned long long cur = carry * BigInt::B + x.a[i];
                    x.a[i] = cur / divisor;
                    carry = cur % divisor;
                }
            }
            x.norm();
            if (x.s)
            {
                x.s = sign;
            }
        }
        if (nearest && roundDig >= 5)
        {
            x += BigInt(sign);
        }
        return x;
    }

    void normalize()
    {
        // 无参数；删除尾数与小数位中的十进制末尾零，无返回值。
        man.norm();
        if (!man.s)
        {
            scale = 0;
            return;
        }
        while (scale >= 9 && !man.a.empty() && man.a[0] == 0)
        {
            man.a.erase(man.a.begin());
            scale -= 9;
        }
        man.norm();
        while (scale && man.a[0] % 10 == 0)
        {
            man = drop10(man, 1, false);
            scale--;
        }
    }
};
```

先用 `Decimal::setPrecision(p)` 设置 $p\ge0$，再用 `Decimal(s)` 构造；`str()` 删除小数末尾零，`str(fixed)` 固定输出 `fixed` 位并按同一规则舍入。`pow(base,e)` 要求 $e\ge0$，`sqrt()` 要求当前值非负，除法和取模要求除数非零；`%` 对齐尾数后直接做整数取模，保留精确余数，不受当前 `prec` 影响，满足 $a=\operatorname{trunc}(a/b)b+(a\bmod b)$。精度是全局状态，临时修改时应保存并恢复旧值。设参与运算的整数位与保留小数位合计至多 $D$，当前 `BigInt` 下加减、比较为 $O(D)$，乘除、取模与平方根最坏为 $O(D^2)$，`pow` 为 $O(D^2\log e)$，额外空间为 $O(D)$；加减乘除按当前精度舍入，连续运算不保证实数域中的结合律。

下面用固定答案检查符号、舍入、取模、幂和平方根：

```cpp
signed main()
{
    Decimal::setPrecision(6);
    assert((Decimal("1.25") + Decimal("-0.5")).str() == "0.75");
    assert((Decimal("1.2") * Decimal("3.4")).str() == "4.08");
    assert((Decimal(1) / Decimal(8)).str(6) == "0.125000");
    assert((Decimal(2) / Decimal(3)).str(6) == "0.666667");
    assert((Decimal("-5.5") % Decimal(2)).str() == "-1.5");
    assert(Decimal::pow(Decimal("1.5"), 3).str() == "3.375");

    Decimal::setPrecision(30);
    assert(Decimal(2).sqrt().str(30) == "1.414213562373095048801688724210");
}
```

### 14.1.021 高精度圆周率：Machin 与 Chudnovsky

Machin 公式把圆周率化为 $\pi=16\arctan\frac15-4\arctan\frac1{239}$，其中 $\arctan x=\sum_{k\ge0}(-1)^k\frac{x^{2k+1}}{2k+1}$ 在本条的 $|x|<1$ 下收敛；`atanRecip(q)` 逐项递推，直到新项在当前精度下舍入为零。Chudnovsky 版本使用 $\pi=\frac{426880\sqrt{10005}}{S}$，其中 $S=\sum_{k\ge0}\frac{(6k)!(13591409+545140134k)}{(3k)!(k!)^3(-262537412640768000)^k}$，相邻项的阶乘比可约成代码中的整数递推，每项约增加十四位正确数字。两个接口都临时把 [14.1.020 任意精度定点与十进制数](#141020) 的精度提高八位，求和结束后再舍入到目标位数。

```cpp
// 依赖 14.1.020 任意精度定点与十进制数
Decimal atanRecip(int q)
{
    // q 是大于 1 的正整数；返回按当前 Decimal 精度计算的 arctan(1/q)。
    assert(q > 1);
    Decimal x = Decimal(1) / Decimal(q);
    Decimal square = x * x;
    Decimal term = x;
    Decimal ans(0);
    for (int k = 0;; k++)
    {
        Decimal add = term / Decimal(2 * k + 1);
        if (add.isZero())
        {
            break;
        }
        ans += add;
        term = -term * square;
    }
    return ans;
}

Decimal piMachin(int digits)
{
    // digits 是结果保留的小数位数；返回 Machin 公式计算并四舍五入的圆周率。
    assert(0 <= digits && digits <= numeric_limits<int>::max() - 8);
    int old = Decimal::getPrecision();
    Decimal::setPrecision(digits + 8);
    Decimal ans = Decimal(16) * atanRecip(5) - Decimal(4) * atanRecip(239);
    ans.round(digits);
    Decimal::setPrecision(old);
    return ans;
}

Decimal piChudnovsky(int digits)
{
    // digits 是结果保留的小数位数；返回 Chudnovsky 级数计算并四舍五入的圆周率。
    assert(0 <= digits && digits <= numeric_limits<int>::max() - 8);
    int old = Decimal::getPrecision();
    Decimal::setPrecision(digits + 8);
    BigInt mul = 1;
    BigInt power = 1;
    BigInt linear = 13591409;
    Decimal sum(linear);
    for (int k = 1;; k++)
    {
        // kk、x 保存项编号及 12k-6；立方在大整数中计算，关闭整数宏也不溢出。
        BigInt kk(k), x = BigInt(12) * kk - BigInt(6);
        mul *= x * x * x - BigInt(16) * x;
        mul /= kk * kk * kk;
        linear += 545140134;
        power *= BigInt("-262537412640768000");
        Decimal term(mul * linear);
        term /= Decimal(power);
        if (term.isZero())
        {
            break;
        }
        sum += term;
    }
    Decimal ans = Decimal(426880) * Decimal(10005).sqrt() / sum;
    ans.round(digits);
    Decimal::setPrecision(old);
    return ans;
}
```

`piMachin(digits)` 与 `piChudnovsky(digits)` 要求 $0\le digits\le\mathtt{numeric\_limits<int>::max()}-8$，返回保留 `digits` 位小数的圆周率并恢复原精度。八位保护位适合竞赛常见位数，但不构成任意规模下的严格误差证明；需要正确舍入时应增加保护位并验证误差界。令 $P=digits+8$，当前二次复杂度大整数下二者最坏均为 $O(P^3)$ 时间、$O(P)$ 空间；Chudnovsky 约用 $P/14$ 项，通常更快，极高精度改用二分拆分。

下面用已知的四十位小数分别核对两种公式：

```cpp
signed main()
{
    const string expected = "3.1415926535897932384626433832795028841972";
    assert(piMachin(40).str(40) == expected);
    assert(piChudnovsky(40).str(40) == expected);
}
```
